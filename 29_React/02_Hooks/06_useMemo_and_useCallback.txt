━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. useMemo
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

useMemo memoizes the RESULT of a function call. It only recomputes when
dependencies change — skips recomputation on unrelated renders.

  const memoizedValue = useMemo(() => expensiveComputation(a, b), [a, b]);

  // Without useMemo: runs on EVERY render
  function ProductList({ products, filter }) {
    const filtered = products.filter(p => p.category === filter); // expensive if huge list
    return filtered.map(p => <Product key={p.id} {...p} />);
  }

  // With useMemo: only recomputes when products or filter changes
  function ProductList({ products, filter }) {
    const filtered = useMemo(
      () => products.filter(p => p.category === filter),
      [products, filter]
    );
    return filtered.map(p => <Product key={p.id} {...p} />);
  }

2. useCallback
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

useCallback memoizes the FUNCTION REFERENCE itself (not the return value).
It returns the same function instance across renders unless deps change.

  const memoizedFn = useCallback(() => doSomething(a, b), [a, b]);

  // Without useCallback: new function reference on every render
  function Parent({ userId }) {
    const handleClick = () => fetchUserData(userId);
    // handleClick is a NEW function on every render (new reference)
    return <Child onClick={handleClick} />;
  }

  // Child wrapped in React.memo would STILL re-render because onClick is always new
  const Child = React.memo(({ onClick }) => (
    <button onClick={onClick}>Load</button>
  ));

  // With useCallback: same function reference unless userId changes
  function Parent({ userId }) {
    const handleClick = useCallback(() => fetchUserData(userId), [userId]);
    return <Child onClick={handleClick} />; // Child.memo comparison: stable ✓
  }

3. useMemo vs useCallback
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  useMemo    = memoize a VALUE (computed result)
  useCallback = memoize a FUNCTION (reference)

  // They are equivalent:
  useCallback(fn, deps)
  // is the same as:
  useMemo(() => fn, deps)

  Use useMemo for:
    - Expensive calculations (filtering, sorting, transforming large datasets)
    - Creating stable object/array references for context values, props

  Use useCallback for:
    - Event handlers passed to memoized child components (React.memo)
    - Functions in useEffect dependency arrays
    - Functions passed as props to avoid child re-renders

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
4. WHEN TO USE THEM (AND WHEN NOT TO)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

DO use useMemo/useCallback when:
  ✅ Computation is genuinely expensive (sorting 10K items, heavy math)
  ✅ You're passing a function/object to a React.memo'd child
  ✅ A function is a dependency in useEffect deps array
  ✅ Creating stable references for context values

DO NOT over-use them — memoization has a COST:
  ❌ Adds memory overhead (stores cached value + dependency list)
  ❌ More complex code, harder to read
  ❌ Incorrect deps = stale values (bugs are subtle and hard to find)

  // PREMATURE OPTIMIZATION — don't useMemo for trivial operations:
  const sum = useMemo(() => a + b, [a, b]); // pointless, addition is instant
  const fullName = useMemo(() => `${first} ${last}`, [first, last]); // unnecessary

  // RULE OF THUMB: Only memoize when you've measured a performance problem,
  // or when referential stability is genuinely needed (deps, memoized children)

5. PRACTICAL EXAMPLE: Memoized List with Callback
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  function TodoApp() {
    const [todos, setTodos] = useState([...]);
    const [filter, setFilter] = useState('all');
    const [newTodo, setNewTodo] = useState('');

    // Memoize expensive filtering operation
    const filteredTodos = useMemo(
      () => todos.filter(t => {
        if (filter === 'done') return t.done;
        if (filter === 'pending') return !t.done;
        return true;
      }),
      [todos, filter]  // only recompute when todos or filter changes
    );

    // Memoize callback so TodoItem doesn't re-render unnecessarily
    const handleToggle = useCallback((id) => {
      setTodos(prev => prev.map(t => t.id === id ? { ...t, done: !t.done } : t));
    }, []); // stable — setTodos setter is stable

    const handleDelete = useCallback((id) => {
      setTodos(prev => prev.filter(t => t.id !== id));
    }, []);

    return (
      <div>
        <input value={newTodo} onChange={e => setNewTodo(e.target.value)} />
        {filteredTodos.map(todo => (
          <TodoItem
            key={todo.id}
            todo={todo}
            onToggle={handleToggle}
            onDelete={handleDelete}
          />
        ))}
      </div>
    );
  }

  // React.memo ensures TodoItem only re-renders when its props change
  const TodoItem = React.memo(function TodoItem({ todo, onToggle, onDelete }) {
    return (
      <div>
        <input type="checkbox" checked={todo.done} onChange={() => onToggle(todo.id)} />
        <span>{todo.text}</span>
        <button onClick={() => onDelete(todo.id)}>×</button>
      </div>
    );
  });

6. useMemo FOR STABLE CONTEXT VALUE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Without useMemo: new object on every render → all consumers re-render
  function AuthProvider({ children }) {
    const [user, setUser] = useState(null);
    return (
      <AuthContext.Provider value={{ user, setUser }}>
        {children}
      </AuthContext.Provider>
    );
  }

  // With useMemo: only new reference when user changes
  function AuthProvider({ children }) {
    const [user, setUser] = useState(null);
    const value = useMemo(() => ({ user, setUser }), [user]);
    return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
  }

7. REACT COMPILER (React 19) — The End of Manual Memoization?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

React 19 ships with the React Compiler (previously "React Forget"):
  - Automatically memoizes components and values at compile time
  - Eliminates the need to manually write useMemo and useCallback
  - Code is cleaner; compiler handles optimization
  - Still in rollout — not all projects use it yet (2024+)

  // Before compiler:
  const sorted = useMemo(() => [...items].sort(), [items]);
  const handleClick = useCallback(() => doSomething(id), [id]);

  // After compiler (no manual memoization needed — compiler adds it automatically):
  const sorted = [...items].sort();
  const handleClick = () => doSomething(id);

8. COMMON INTERVIEW QUESTIONS ON useMemo AND useCallback
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What is the difference between useMemo and useCallback?
A: useMemo memoizes a computed VALUE (result of a function).
   useCallback memoizes a FUNCTION REFERENCE.
   useCallback(fn, deps) is equivalent to useMemo(() => fn, deps).

Q: When should you use useMemo?
A: For computationally expensive operations (heavy filtering, sorting, complex
   calculations), and for creating stable references for context values or props
   passed to memoized children.

Q: Why does React.memo not prevent re-renders without useCallback?
A: React.memo does a shallow props comparison. Functions created in render are
   new references on every render. Without useCallback, the prop is always
   "new" → React.memo always triggers re-render.

Q: What are the downsides of over-using useMemo/useCallback?
A: Memory overhead (caches values), added complexity, subtle staleness bugs from
   missing deps, and false security (memoizing cheap operations wastes time).

Q: Should you always use useCallback for event handlers?
A: No. Only when passing to memoized children (React.memo) or as useEffect deps.
   Otherwise it adds overhead with no benefit.

Q: What is the React Compiler (React Forget)?
A: A build-time compiler introduced in React 19 that automatically memoizes
   components and values. Eliminates the need for manual useMemo and useCallback.
