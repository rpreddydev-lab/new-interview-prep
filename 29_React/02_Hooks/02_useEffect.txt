1. WHAT IS useEffect?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

useEffect lets you perform SIDE EFFECTS in functional components.

Side effects are operations that reach outside the component:
  - Data fetching (API calls)
  - DOM manipulation (document.title, event listeners)
  - Subscriptions (WebSockets, observables)
  - Timers (setTimeout, setInterval)
  - Local storage reads/writes
  - Analytics/logging

Signature:
  useEffect(setupFunction, dependencyArray?)

  // setup:   function to run the effect; can return a cleanup function
  // deps:    array of values — effect re-runs when any value changes

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
2. THREE FORMS OF useEffect
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

FORM 1: No dependency array — runs AFTER EVERY render
  useEffect(() => {
    console.log('Runs after every render');
  });
  // Equivalent to: componentDidMount + componentDidUpdate
  // CAUTION: Can cause infinite loop if you setState inside without care

FORM 2: Empty dependency array — runs ONCE on mount
  useEffect(() => {
    console.log('Runs once after initial render');
    // Fetch initial data, set up subscriptions, etc.
  }, []);
  // Equivalent to: componentDidMount

FORM 3: With dependencies — runs when deps change
  useEffect(() => {
    console.log(`userId changed to: ${userId}`);
    fetchUser(userId);
  }, [userId]);
  // Runs on MOUNT and whenever userId changes
  // Equivalent to: componentDidMount + componentDidUpdate (with dep check)

3. CLEANUP FUNCTION
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

The setup function CAN return a cleanup function. Cleanup runs:
  - Before the next time the effect runs (before re-run on dep change)
  - When the component unmounts

  // Event listener with cleanup
  useEffect(() => {
    function handleResize() {
      setWindowSize({ width: window.innerWidth, height: window.innerHeight });
    }
    window.addEventListener('resize', handleResize);
    return () => window.removeEventListener('resize', handleResize);  // cleanup
  }, []);

  // Timer with cleanup
  useEffect(() => {
    const id = setInterval(() => setSeconds(s => s + 1), 1000);
    return () => clearInterval(id);  // cleanup
  }, []);

  // Subscription with cleanup
  useEffect(() => {
    const subscription = dataStream.subscribe(setData);
    return () => subscription.unsubscribe();  // cleanup
  }, []);

  // WebSocket with cleanup
  useEffect(() => {
    const ws = new WebSocket('wss://api.example.com');
    ws.onmessage = (e) => setMessages(prev => [...prev, e.data]);
    return () => ws.close();  // cleanup
  }, []);

Without cleanup → memory leaks, phantom state updates on unmounted components

4. DATA FETCHING WITH useEffect
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  function UserProfile({ userId }) {
    const [user, setUser] = useState(null);
    const [loading, setLoading] = useState(true);
    const [error, setError] = useState(null);

    useEffect(() => {
      let cancelled = false;  // flag to prevent stale updates
      setLoading(true);
      setError(null);

      async function fetchUser() {
        try {
          const response = await fetch(`/api/users/${userId}`);
          if (!response.ok) throw new Error('Failed to fetch');
          const data = await response.json();
          if (!cancelled) {          // only update if effect is still valid
            setUser(data);
            setLoading(false);
          }
        } catch (err) {
          if (!cancelled) {
            setError(err.message);
            setLoading(false);
          }
        }
      }

      fetchUser();
      return () => { cancelled = true; };  // cleanup: prevent stale setState
    }, [userId]);  // re-fetch when userId changes

    if (loading) return <Spinner />;
    if (error) return <p>Error: {error}</p>;
    return <div>{user?.name}</div>;
  }

  // Better: Use AbortController for proper cancellation
  useEffect(() => {
    const controller = new AbortController();

    fetch(`/api/users/${userId}`, { signal: controller.signal })
      .then(r => r.json())
      .then(setUser)
      .catch(err => { if (err.name !== 'AbortError') setError(err.message); });

    return () => controller.abort();  // cancels the fetch on cleanup
  }, [userId]);

5. DEPENDENCY ARRAY DEEP DIVE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

The dependency array tells React WHEN to re-run the effect.
React uses Object.is() to compare deps.

COMMON MISTAKES:

  Mistake 1: Omitting required dependencies (stale closure)
    const [count, setCount] = useState(0);
    useEffect(() => {
      const id = setInterval(() => {
        console.log(count);  // stale! count is always 0 (captured at mount)
      }, 1000);
      return () => clearInterval(id);
    }, []);  // count is missing from deps!

    Fix: add count to deps (or use functional update)

  Mistake 2: Object/function as dependency (new reference every render)
    const options = { method: 'GET' };  // new object every render
    useEffect(() => {
      fetch('/api/data', options); // effect runs EVERY render!
    }, [options]);  // options has new reference each time

    Fix: move stable values inside effect, or use useMemo/useCallback
    useEffect(() => {
      fetch('/api/data', { method: 'GET' }); // defined inside → no dep needed
    }, []);

  Mistake 3: Missing cleanup (memory leak / stale state)
    useEffect(() => {
      subscribe(setData); // subscribe without unsubscribe → memory leak
    }, []);
    // Add: return () => unsubscribe();

6. INFINITE LOOP PITFALLS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // INFINITE LOOP: setState inside effect without proper deps
  const [count, setCount] = useState(0);
  useEffect(() => {
    setCount(count + 1); // triggers re-render → effect runs again → infinite loop!
  }); // no deps = runs every render

  // Fix 1: Add deps that prevent infinite looping
  useEffect(() => {
    if (count < 5) setCount(count + 1); // stops at 5
  }, [count]);

  // Fix 2: Use functional update with empty deps (if appropriate)
  useEffect(() => {
    const id = setInterval(() => setCount(c => c + 1), 1000);
    return () => clearInterval(id);
  }, []);

  // ANOTHER INFINITE LOOP: object dependency
  const [data, setData] = useState({});
  useEffect(() => {
    setData({ value: 'fetched' }); // creates new object → rerenders → re-runs → ∞
  }, [data]); // data is new object after setData

  // Fix: Only put specific primitive values in deps:
  useEffect(() => {
    setData({ value: 'fetched' });
  }, []); // run once

7. useEffect TIMING & EXECUTION ORDER
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Timeline for a component that mounts:
  1. React renders component (calls function)
  2. DOM is updated (committed)
  3. Browser PAINTS the screen
  4. useEffect runs (asynchronously, after paint)

Timeline when deps change (update):
  1. React renders component
  2. DOM is updated
  3. Cleanup of PREVIOUS effect runs
  4. Browser paints
  5. New effect runs

On unmount:
  1. Cleanup of the last effect runs

Key point: useEffect is ALWAYS asynchronous — it NEVER blocks painting.
For synchronous post-DOM work (before paint), use useLayoutEffect.

8. REACT 18 STRICT MODE DOUBLE INVOCATION
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

In React 18 development + Strict Mode, every effect runs TWICE:
  mount → (effect runs) → (cleanup runs) → (effect runs again)

This simulates the future behavior of React's Offscreen feature.
If your app breaks with double-invocation, you have a missing cleanup.

  // BROKEN: counter increments twice on mount (dev mode)
  useEffect(() => {
    setCount(c => c + 1); // runs twice in StrictMode dev
  }, []);

  // This is intentional — it exposes bugs

9. COMMON INTERVIEW QUESTIONS ON useEffect
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What is useEffect used for?
A: For side effects in functional components: data fetching, subscriptions,
   DOM manipulation, timers. It runs after render and can optionally clean up.

Q: What does the dependency array do?
A: Controls when the effect re-runs. No array = every render. [] = once on mount.
   [dep] = when dep changes. React compares deps using Object.is().

Q: How do you prevent memory leaks in useEffect?
A: Return a cleanup function that undoes the effect (unsubscribe, clearInterval,
   abort fetch, removeEventListener, close WebSocket).

Q: Why shouldn't you use async function directly as useEffect callback?
A: useEffect must return either nothing or a cleanup function. An async function
   returns a Promise, not undefined or a function.
   FIX: Define async function inside, call it:
   useEffect(() => {
     async function load() { ... }
     load();
   }, []);

Q: What is a stale closure in useEffect?
A: When the effect captures a state/prop value from an old render. The closure
   "closes over" the variable at the time the function was created. Fix: include
   the variable in the dependency array, or use functional state updates.

Q: What happens to useEffect in React 18 Strict Mode?
A: Effects run twice (mount, cleanup, mount) in development. This is intentional
   to expose bugs caused by missing cleanup. Production behavior is unchanged.

Q: How do you run code only on update (skip first render)?
A: Use a ref as a flag:
   const isMounted = useRef(false);
   useEffect(() => {
     if (!isMounted.current) { isMounted.current = true; return; }
     // runs only on updates
   }, [dep]);

Q: What is the difference between useEffect and useLayoutEffect?
A: useEffect fires asynchronously AFTER the browser paints. useLayoutEffect
   fires synchronously BEFORE the browser paints (after DOM mutation).
   Use useLayoutEffect for DOM measurements to avoid visual flicker.
