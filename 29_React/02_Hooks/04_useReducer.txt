1. WHAT IS useReducer?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

useReducer is a state management hook for complex state logic. It follows the
Redux pattern: state is updated via "actions" processed by a "reducer" function.

  const [state, dispatch] = useReducer(reducer, initialState, initFn?)

  - reducer: pure function (state, action) => newState
  - initialState: starting state
  - initFn (optional): lazy initialization function
  - state: current state
  - dispatch: function to send actions to the reducer

2. useState vs useReducer
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Use useState when:
    - State is a simple value (boolean, string, number)
    - State transitions are straightforward (toggle, set)
    - Few state variables

  Use useReducer when:
    - State is a complex object with multiple sub-values
    - State transitions have complex logic
    - Next state depends on previous state
    - Multiple pieces of state change together
    - You need to describe "what happened" rather than "what the new value is"
    - You want testable, pure state logic

3. BASIC EXAMPLE: Counter
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Reducer function (pure function, outside component)
  function counterReducer(state, action) {
    switch (action.type) {
      case 'INCREMENT': return { count: state.count + 1 };
      case 'DECREMENT': return { count: state.count - 1 };
      case 'RESET':     return { count: 0 };
      case 'SET':       return { count: action.payload };
      default:          return state; // always return state for unknown actions
    }
  }

  function Counter() {
    const [state, dispatch] = useReducer(counterReducer, { count: 0 });

    return (
      <div>
        <p>Count: {state.count}</p>
        <button onClick={() => dispatch({ type: 'INCREMENT' })}>+</button>
        <button onClick={() => dispatch({ type: 'DECREMENT' })}>-</button>
        <button onClick={() => dispatch({ type: 'RESET' })}>Reset</button>
        <button onClick={() => dispatch({ type: 'SET', payload: 10 })}>Set 10</button>
      </div>
    );
  }

4. COMPLEX EXAMPLE: Shopping Cart
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  const initialState = {
    items: [],
    total: 0,
    discount: 0,
    isCheckingOut: false,
  };

  function cartReducer(state, action) {
    switch (action.type) {
      case 'ADD_ITEM': {
        const existing = state.items.find(i => i.id === action.payload.id);
        const updatedItems = existing
          ? state.items.map(i =>
              i.id === action.payload.id
                ? { ...i, quantity: i.quantity + 1 }
                : i
            )
          : [...state.items, { ...action.payload, quantity: 1 }];
        return {
          ...state,
          items: updatedItems,
          total: calculateTotal(updatedItems, state.discount),
        };
      }
      case 'REMOVE_ITEM': {
        const updatedItems = state.items.filter(i => i.id !== action.payload);
        return {
          ...state,
          items: updatedItems,
          total: calculateTotal(updatedItems, state.discount),
        };
      }
      case 'UPDATE_QUANTITY': {
        const updatedItems = state.items.map(i =>
          i.id === action.payload.id
            ? { ...i, quantity: action.payload.quantity }
            : i
        ).filter(i => i.quantity > 0);
        return { ...state, items: updatedItems, total: calculateTotal(updatedItems, state.discount) };
      }
      case 'APPLY_DISCOUNT':
        return { ...state, discount: action.payload, total: calculateTotal(state.items, action.payload) };
      case 'START_CHECKOUT':
        return { ...state, isCheckingOut: true };
      case 'CLEAR_CART':
        return initialState;
      default:
        return state;
    }
  }

  function Cart() {
    const [cart, dispatch] = useReducer(cartReducer, initialState);
    return (
      <div>
        <CartItems items={cart.items} dispatch={dispatch} />
        <CartTotal total={cart.total} />
        <button onClick={() => dispatch({ type: 'CLEAR_CART' })}>Clear</button>
        <button onClick={() => dispatch({ type: 'START_CHECKOUT' })}>Checkout</button>
      </div>
    );
  }

  const formInitialState = {
    values: { name: '', email: '', password: '' },
    errors: {},
    isSubmitting: false,
    isSuccess: false,
  };

  function formReducer(state, action) {
    switch (action.type) {
      case 'SET_FIELD':
        return {
          ...state,
          values: { ...state.values, [action.field]: action.value },
          errors: { ...state.errors, [action.field]: '' }, // clear error on change
        };
      case 'SET_ERRORS':
        return { ...state, errors: action.errors, isSubmitting: false };
      case 'SUBMIT_START':
        return { ...state, isSubmitting: true };
      case 'SUBMIT_SUCCESS':
        return { ...formInitialState, isSuccess: true };
      default:
        return state;
    }
  }

  function RegistrationForm() {
    const [form, dispatch] = useReducer(formReducer, formInitialState);

    function handleChange(e) {
      dispatch({ type: 'SET_FIELD', field: e.target.name, value: e.target.value });
    }

    async function handleSubmit(e) {
      e.preventDefault();

        dispatch({ type: 'SET_ERRORS', errors });
        return;
      }
      await registerUser(form.values);
      dispatch({ type: 'SUBMIT_SUCCESS' });

  const DispatchContext = createContext(null);

  function AppProvider({ children }) {
    const [state, dispatch] = useReducer(appReducer, initialAppState);
    return (
      <StateContext.Provider value={state}>
        <DispatchContext.Provider value={dispatch}>
          {children}
        </DispatchContext.Provider>
      </StateContext.Provider>
    );
  }

  // Consumers only re-render when their specific slice changes
  function Counter() {
    const { count } = useContext(StateContext);
    const dispatch = useContext(DispatchContext);
    // dispatch is stable (same reference always), so consuming only dispatch
    // won't re-render when state changes
    return <button onClick={() => dispatch({ type: 'INCREMENT' })}>{count}</button>;
  }

8. COMMON INTERVIEW QUESTIONS ON useReducer
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What is useReducer and when do you use it over useState?
A: useReducer manages state via a reducer function (state, action) => newState.
   Use it for: complex state objects, multiple related state updates, complex
   transition logic, or when you want testable, declarative state logic.
   useState is simpler for single values or simple updates.

Q: What is a reducer function?
A: A pure function that takes current state and an action, and returns next state.
   It must be pure (no side effects), and must always return a value (even if
   the action is unknown — return state).

Q: What is the difference between useState and useReducer in terms of batching?
A: Both batch updates in React 18. useReducer dispatches can be called multiple
   times and React batches all into one re-render.

Q: How is useReducer similar to Redux?
A: Same mental model: state is a single object, updates happen via dispatching
   actions to a reducer. The main differences: useReducer is local to a component
   (not global), has no middleware, and no Redux DevTools (unless you add custom logging).

Q: Can you dispatch async actions with useReducer?
A: Not directly — reducers must be pure/synchronous. Handle async in event
   handlers: dispatch a 'LOADING' action, await the fetch, then dispatch 'SUCCESS'
   or 'ERROR'. Or use a custom hook/middleware layer.

Q: What happens if dispatch is called inside useEffect?
A: dispatch is a stable function (doesn't change between renders, unlike setState).
   You do NOT need to include dispatch in the useEffect dependency array.
