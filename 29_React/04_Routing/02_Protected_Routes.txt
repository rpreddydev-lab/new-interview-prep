1. WHAT ARE PROTECTED ROUTES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Protected routes guard certain pages so only authorized users can access them.
Unauthenticated users are redirected to a login page.

  Common patterns:
  - Authentication guard (must be logged in)
  - Role-based guard (must be admin)
  - Permission guard (must have specific permission)

2. BASIC PROTECTED ROUTE (React Router v6)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // components/ProtectedRoute.jsx
  import { Navigate, Outlet, useLocation } from 'react-router-dom';
  import { useAuth } from '../context/AuthContext';

  function ProtectedRoute() {
    const { isAuthenticated, loading } = useAuth();
    const location = useLocation();

    if (loading) {
      return <Spinner />;  // wait for auth check before deciding
    }

    if (!isAuthenticated) {
      // Redirect to login, but save where user was trying to go
      return <Navigate to="/login" state={{ from: location }} replace />;
    }

    // User is authenticated — render the child route
    return <Outlet />;
  }

  // Usage in App.jsx:
  function App() {
    return (
      <Routes>
        {/* Public routes */}
        <Route path="/login" element={<Login />} />
        <Route path="/register" element={<Register />} />

        {/* Protected routes — all children require authentication */}
        <Route element={<ProtectedRoute />}>
          <Route path="/dashboard" element={<Dashboard />} />
          <Route path="/profile" element={<Profile />} />
          <Route path="/settings" element={<Settings />} />
        </Route>

        <Route path="*" element={<NotFound />} />
      </Routes>
    );
  }

3. REDIRECT BACK TO ORIGINAL LOCATION AFTER LOGIN
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // ProtectedRoute passes location state to login page
  <Navigate to="/login" state={{ from: location }} replace />

  // Login.jsx — redirects to original page after successful login
  function Login() {
    const navigate = useNavigate();
    const location = useLocation();
    const { login } = useAuth();

    // Where the user was trying to go before being redirected to login
    const from = location.state?.from?.pathname || '/dashboard';

    async function handleSubmit(credentials) {
      await login(credentials);
      navigate(from, { replace: true });  // go back to original destination
    }

    return <LoginForm onSubmit={handleSubmit} />;
  }

4. ROLE-BASED PROTECTED ROUTE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  function RequireRole({ allowedRoles }) {
    const { user } = useAuth();
    const location = useLocation();

    if (!user) {
      return <Navigate to="/login" state={{ from: location }} replace />;
    }

    if (!allowedRoles.includes(user.role)) {
      return <Navigate to="/unauthorized" replace />;
    }

    return <Outlet />;
  }

  // Usage:
  <Routes>
    <Route element={<ProtectedRoute />}>
      {/* All authenticated users */}
      <Route path="/dashboard" element={<Dashboard />} />

      {/* Admin only */}
      <Route element={<RequireRole allowedRoles={['admin']} />}>
        <Route path="/admin" element={<AdminPanel />} />
        <Route path="/admin/users" element={<ManageUsers />} />
      </Route>

      {/* Admin or manager */}
      <Route element={<RequireRole allowedRoles={['admin', 'manager']} />}>
        <Route path="/reports" element={<Reports />} />
      </Route>
    </Route>
  </Routes>

5. COMMON INTERVIEW QUESTIONS

Q: How do you implement protected routes in React Router v6?
A: Create a wrapper component that reads authentication state. If not
   authenticated, render <Navigate to="/login" replace />. If authenticated,
   render <Outlet /> to show the child route. Use this wrapper as a layout
   route in the route tree.

Q: Why use replace={true} on the redirect to login?
A: Using replace prevents the login page from being added to browser history.
   If the user presses back after logging in, they won't be sent back to the
   login page (which would be a confusing UX).

Q: How do you redirect back to the original page after login?
A: Pass location as state to the login redirect:
   <Navigate to="/login" state={{ from: location }} replace />
   Then in the Login component, read location.state?.from?.pathname and
   navigate there after a successful login.

Q: What is the loading state check in ProtectedRoute for?
A: On app startup, auth is typically initialized asynchronously (checking
   localStorage, verifying a token with the server). Without the loading check,
   the protected route would redirect to login before auth has a chance to
   confirm the user is actually logged in.
