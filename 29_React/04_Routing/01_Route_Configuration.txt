1. NESTED ROUTES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Nested routes allow parent routes to render persistent layout (sidebar, nav)
while child routes render in an <Outlet> placeholder.

  // App.jsx
  import { Routes, Route } from 'react-router-dom';

  function App() {
    return (
      <Routes>
        {/* Layout route — renders Layout wrapper with Outlet */}
        <Route path="/" element={<Layout />}>
          <Route index element={<Home />} />          {/* exact "/" */}
          <Route path="about" element={<About />} />  {/* "/about" */}

          {/* Nested: /dashboard, /dashboard/stats, /dashboard/settings */}
          <Route path="dashboard" element={<DashboardLayout />}>
            <Route index element={<DashboardHome />} />
            <Route path="stats" element={<Stats />} />
            <Route path="settings" element={<Settings />} />
          </Route>

          {/* Dynamic route: /users/:userId */}
          <Route path="users">
            <Route index element={<UserList />} />
            <Route path=":userId" element={<UserDetail />} />
            <Route path=":userId/edit" element={<UserEdit />} />
          </Route>
        </Route>

        <Route path="*" element={<NotFound />} />
      </Routes>
    );
  }

2. OUTLET
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

<Outlet> is where child route components render inside a parent route component.

  // Layout.jsx — persistent wrapper, child routes render in Outlet
  import { Outlet, NavLink } from 'react-router-dom';

  function Layout() {
    return (
      <div className="app">
        <header>
          <NavLink to="/">Home</NavLink>
          <NavLink to="/about">About</NavLink>
          <NavLink to="/dashboard">Dashboard</NavLink>
        </header>

        <main>
          <Outlet />   {/* Child route components render here */}
        </main>

        <footer>© 2024</footer>
      </div>
    );
  }

  // DashboardLayout.jsx — nested layout with its own sidebar
  function DashboardLayout() {
    return (
      <div className="dashboard">
        <aside>
          <NavLink to="/dashboard">Overview</NavLink>
          <NavLink to="/dashboard/stats">Stats</NavLink>
          <NavLink to="/dashboard/settings">Settings</NavLink>
        </aside>
        <section>
          <Outlet />   {/* /dashboard/stats, /dashboard/settings render here */}
        </section>
      </div>
    );
  }

3. INDEX ROUTES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

An index route renders in the parent's Outlet when the parent's path is
exactly matched (no additional path segment).

  <Route path="dashboard" element={<DashboardLayout />}>
    <Route index element={<DashboardHome />} />  {/* renders at /dashboard */}
    <Route path="stats" element={<Stats />} />   {/* renders at /dashboard/stats */}
  </Route>

  // Without index route, nothing renders in the Outlet at /dashboard exactly.
  // index is the "default child" of a parent route.

4. createBrowserRouter — DATA API (v6.4+)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

React Router v6.4+ introduced a data API (loaders, actions, errorElement).
This allows routes to fetch data before rendering and handle form submissions.

  import { createBrowserRouter, RouterProvider, useLoaderData } from 'react-router-dom';

  // Define routes with loader functions
  const router = createBrowserRouter([
    {
      path: '/',
      element: <Layout />,
      errorElement: <ErrorPage />,
      children: [
        { index: true, element: <Home /> },
        {
          path: 'users',
          element: <UserList />,
          loader: async () => {
            const res = await fetch('/api/users');
            return res.json();  // available via useLoaderData()
          },
        },
        {
          path: 'users/:id',
          element: <UserDetail />,
          loader: async ({ params }) => {
            const res = await fetch(`/api/users/${params.id}`);
            if (!res.ok) throw new Response('Not Found', { status: 404 });
            return res.json();
          },
        },
      ],
    },
  ]);

  // Component reads loader data:
  function UserList() {
    const users = useLoaderData();
    return <ul>{users.map(u => <li key={u.id}>{u.name}</li>)}</ul>;
  }

  // Mount:
  function App() {
    return <RouterProvider router={router} />;
  }

5. RELATIVE vs ABSOLUTE PATHS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  In React Router v6, nested route paths are RELATIVE by default:

  <Route path="dashboard" element={<DashboardLayout />}>
    <Route path="stats" element={<Stats />} />
    {/* Full path is /dashboard/stats (not /stats) */}
  </Route>

  <Link to="stats">Stats</Link>      {/* Relative: resolves to /dashboard/stats */}
  <Link to="/stats">Stats</Link>     {/* Absolute: always /stats */}

Q: What is <Outlet> in React Router v6?
A: Outlet is a placeholder inside a parent route component where child route
   components render. It enables layout routes — a parent renders persistent
   UI (navbar, sidebar) and child routes change only the content area.

Q: What is an index route?
A: An index route renders in the parent's Outlet when the parent's URL is
   matched exactly (no additional path segment). It's the "default view"
   inside a layout, like /dashboard when there are sub-routes like
   /dashboard/stats.

Q: What is the difference between <Routes> in v6 vs <Switch> in v5?
A: <Routes> always renders the best match (exclusive). <Switch> rendered the
   first match and stopped. In v6, route paths are relative by default, exact
   matching is the default, and the route config is more declarative.

Q: What is the createBrowserRouter data API?
A: Introduced in v6.4, it lets you colocate data-loading logic (loaders) with
   route definitions. Loaders run before the component renders, making data
   available via useLoaderData without useEffect. Enables route-level loading
   states with <Suspense> and error boundaries via errorElement.
