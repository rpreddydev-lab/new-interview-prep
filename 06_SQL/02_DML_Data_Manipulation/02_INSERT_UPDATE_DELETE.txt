================================================================
INSERT
================================================================

  SINGLE ROW:
  INSERT INTO customers (email, name, is_active)
  VALUES ('alice@example.com', 'Alice Smith', TRUE);

  MULTIPLE ROWS (batch insert — preferred for performance):
  INSERT INTO products (sku, name, price, stock_qty)
  VALUES
      ('SKU-001', 'Widget A', 9.99, 100),
      ('SKU-002', 'Widget B', 14.99, 50),
      ('SKU-003', 'Widget C', 24.99, 25);

  INSERT FROM SELECT (copy data from another table or query):
  INSERT INTO orders_archive (id, customer_id, total, status, created_at)
  SELECT id, customer_id, total, status, created_at
  FROM orders
  WHERE created_at < '2023-01-01'
    AND status = 'DELIVERED';

  INSERT with RETURNING (PostgreSQL) — get generated values back:
  INSERT INTO customers (email, name)
  VALUES ('bob@example.com', 'Bob Jones')
  RETURNING id, created_at;

  UPSERT (PostgreSQL ON CONFLICT):
  INSERT INTO user_preferences (user_id, theme, language)
  VALUES (42, 'dark', 'en')
  ON CONFLICT (user_id) DO UPDATE
      SET theme = EXCLUDED.theme,
          language = EXCLUDED.language,
          updated_at = CURRENT_TIMESTAMP;

  UPSERT ignore duplicates:
  INSERT INTO event_log (event_id, event_type)
  VALUES ('abc-123', 'CLICK')
  ON CONFLICT (event_id) DO NOTHING;

================================================================
UPDATE
================================================================

  UPDATE ALL MATCHING ROWS (always use WHERE in production):
  UPDATE orders
  SET status = 'SHIPPED',
      shipped_at = CURRENT_TIMESTAMP
  WHERE id = 42;

  UPDATE MULTIPLE COLUMNS:
  UPDATE products
  SET price = price * 1.1,    -- 10% price increase
      updated_at = CURRENT_TIMESTAMP
  WHERE category_id = 5;

  UPDATE WITH CASE (conditional update):
  UPDATE orders
  SET status = CASE
      WHEN shipped_at IS NOT NULL  THEN 'SHIPPED'
      WHEN paid_at IS NOT NULL     THEN 'CONFIRMED'
      ELSE 'PENDING'
  END;

  UPDATE WITH SUBQUERY:
  UPDATE products
  SET price = price * 0.9
  WHERE id IN (
      SELECT product_id FROM order_items
      GROUP BY product_id
      HAVING COUNT(*) < 10     -- items rarely ordered get discount
  );

  UPDATE WITH JOIN (PostgreSQL):
  UPDATE orders o
  SET discount = c.member_discount
  FROM customers c
  WHERE o.customer_id = c.id
    AND c.membership_tier = 'GOLD';

  UPDATE WITH JOIN (MySQL):
  UPDATE orders o
  JOIN customers c ON o.customer_id = c.id
  SET o.discount = c.member_discount
  WHERE c.membership_tier = 'GOLD';

  RETURNING updated values (PostgreSQL):
  UPDATE orders SET status = 'SHIPPED' WHERE id = 42 RETURNING id, status;

================================================================
DELETE
================================================================

  DELETE specific rows (always use WHERE unless deliberately clearing all):
  DELETE FROM orders WHERE id = 42;
  DELETE FROM order_items WHERE order_id = 42;  -- delete children first if no CASCADE

  DELETE with subquery:
  DELETE FROM sessions
  WHERE user_id IN (
      SELECT id FROM users WHERE is_banned = TRUE
  );

  DELETE with JOIN (PostgreSQL):
  DELETE FROM order_items oi
  USING products p
  WHERE oi.product_id = p.id
    AND p.is_discontinued = TRUE;

  DELETE with RETURNING (PostgreSQL) — know what was deleted:
  DELETE FROM sessions
  WHERE expires_at < NOW()
  RETURNING id, user_id;

  SOFT DELETE pattern (preferred in many applications):
  -- Instead of physically deleting, mark as deleted:
  UPDATE customers SET deleted_at = CURRENT_TIMESTAMP WHERE id = 42;
  -- Query active records:
  SELECT * FROM customers WHERE deleted_at IS NULL;

================================================================
BATCH OPERATIONS FOR PERFORMANCE
================================================================

  Bulk insert: INSERT with multiple VALUES tuples in one statement.
  Batch delete: DELETE with LIMIT + loop (for very large deletions to avoid lock).
  Batch update: UPDATE with WHERE IN (ids) — gather IDs first, then update.

  Large DELETE example (chunk-based, avoids long lock):
  -- In application code (loop until count = 0):
  DELETE FROM audit_log
  WHERE created_at < NOW() - INTERVAL '1 year'
  LIMIT 1000;
  -- Repeat until 0 rows deleted

================================================================
COMMON PITFALLS
================================================================

  1. UPDATE or DELETE without WHERE — modifies/deletes ALL rows.
     Always double-check before running.
     Test: run as SELECT first to see what would be affected.

  2. Deadlocks: updating rows in different order from two transactions.
     Fix: always update rows in consistent order.

  3. INSERT without column list: INSERT INTO t VALUES (...) — breaks if columns added.
     Always name columns explicitly.

  4. Forgetting NOT NULL in INSERT: column has no default and is NOT NULL → error.

================================================================
INTERVIEW Q&A
================================================================

  Q: How do you insert multiple rows efficiently?
  A: INSERT with multiple VALUES tuples in a single statement.
     Much faster than individual INSERTs (reduces round trips, fewer transactions).

  Q: What is a soft delete and why use it?
  A: Instead of physically deleting a row, set a deleted_at timestamp.
     Preserves history, supports recovery, audit trail, no FK violations.
     Requires all queries to add WHERE deleted_at IS NULL.

  Q: How do you update rows based on values from another table?
  A: UPDATE ... FROM ... WHERE (PostgreSQL) or
     UPDATE ... JOIN ... SET (MySQL) for join-based updates.

  Q: What is RETURNING in PostgreSQL?
  A: An extension to INSERT/UPDATE/DELETE that returns the affected rows.
     Useful for getting generated IDs, confirming updated values, or
     processing deleted rows without a second SELECT.

  Q: What is ON CONFLICT DO UPDATE?
  A: PostgreSQL upsert syntax. If the INSERT would violate a unique constraint,
     update the existing row instead. Atomic — no race condition.
     EXCLUDED refers to the values that were attempted to be inserted.
