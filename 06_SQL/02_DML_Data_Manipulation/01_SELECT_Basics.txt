================================================================
  SELECT — BASICS
================================================================

  SELECT is the most important SQL statement.
  Used to retrieve data from one or more tables.

================================================================
BASIC SYNTAX
================================================================

  SELECT column1, column2, ...
  FROM   table_name
  WHERE  condition
  ORDER BY column [ASC|DESC]
  LIMIT  n;

================================================================
SELECT ALL COLUMNS
================================================================

  SELECT * FROM customers;    -- all columns
  SELECT * FROM customers WHERE is_active = TRUE;

  Avoid SELECT * in production:
  - Returns unnecessary data (network overhead)
  - Breaks if columns are added/reordered
  - Prevents covering index optimization
  Always name the columns you need.

================================================================
SELECT SPECIFIC COLUMNS
================================================================

  SELECT id, email, name FROM customers;
  SELECT order_id, total, status FROM orders WHERE customer_id = 42;

================================================================
COLUMN ALIASES
================================================================

  SELECT
      id            AS customer_id,
      email         AS customer_email,
      name          AS full_name,
      UPPER(name)   AS name_uppercase,
      total * 0.9   AS discounted_total
  FROM customers;

  Aliases appear in result set column headers.
  Can be referenced in ORDER BY (evaluated after SELECT).
  CANNOT be referenced in WHERE (evaluated before SELECT).

================================================================
ARITHMETIC AND EXPRESSIONS
================================================================

  SELECT
      price,
      quantity,
      price * quantity            AS line_total,
      price * quantity * 0.08     AS tax_amount,
      price * quantity * 1.08     AS total_with_tax
  FROM order_items;

================================================================
STRING FUNCTIONS
================================================================

  SELECT
      UPPER(name)                     AS upper_name,
      LOWER(email)                    AS lower_email,
      LENGTH(phone)                   AS phone_length,
      SUBSTRING(email, 1, 5)          AS email_prefix,
      CONCAT(first_name, ' ', last_name) AS full_name,
      TRIM('  hello  ')               AS trimmed,
      REPLACE(phone, '-', '')         AS clean_phone,
      POSITION('@' IN email)          AS at_position
  FROM customers;

  -- String concatenation:
  -- SQL Standard: name || ' ' || email
  -- MySQL: CONCAT(name, ' ', email)
  -- PostgreSQL: both work

================================================================
DATE FUNCTIONS
================================================================

  SELECT
      id,
      created_at,
      CURRENT_DATE                                AS today,
      CURRENT_TIMESTAMP                           AS now,
      DATE(created_at)                            AS date_only,
      EXTRACT(YEAR FROM created_at)               AS order_year,
      EXTRACT(MONTH FROM created_at)              AS order_month,
      AGE(CURRENT_TIMESTAMP, created_at)          AS age,             -- PostgreSQL
      created_at + INTERVAL '7 days'              AS week_later,
      CURRENT_TIMESTAMP - created_at              AS time_since_order
  FROM orders;

================================================================
SELECT DISTINCT
================================================================

  Removes duplicate rows from results:
  SELECT DISTINCT status FROM orders;
  SELECT DISTINCT customer_id, status FROM orders;   -- distinct combination

================================================================
SELECTING FROM MULTIPLE TABLES (BASIC JOIN)
================================================================

  SELECT
      c.name,
      o.id AS order_id,
      o.total,
      o.status
  FROM customers c
  JOIN orders o ON c.id = o.customer_id
  WHERE c.is_active = TRUE;

  Full JOIN coverage in 05_Joins_and_Subqueries.

================================================================
SUBQUERIES IN SELECT
================================================================

  Scalar subquery (returns single value):
  SELECT
      id,
      name,
      (SELECT COUNT(*) FROM orders WHERE customer_id = c.id) AS order_count
  FROM customers c;

  Note: Scalar subqueries in SELECT can be expensive (one query per row).
  Prefer JOIN + aggregation for performance.

================================================================
CONDITIONAL VALUES — CASE
================================================================

  SELECT
      id,
      total,
      CASE
          WHEN total >= 1000 THEN 'HIGH'
          WHEN total >= 500  THEN 'MEDIUM'
          ELSE 'LOW'
      END AS order_tier
  FROM orders;

LIMIT AND OFFSET — PAGINATION
================================================================

  SELECT * FROM orders ORDER BY created_at DESC LIMIT 10;           -- first 10
  SELECT * FROM orders ORDER BY created_at DESC LIMIT 10 OFFSET 20; -- page 3

================================================================
INTERVIEW Q&A
================================================================

  Q: Why should you avoid SELECT * in production?
  A: Returns unnecessary columns (network overhead, memory), prevents
     covering index optimization, breaks if columns are added or reordered,
     makes code harder to understand (unclear what's actually needed).

  Q: Can you use a column alias in WHERE?
  A: No. WHERE is evaluated before SELECT in logical processing order.
     Aliases defined in SELECT don't exist at WHERE evaluation time.
     Use the expression directly or wrap in a subquery/CTE.

  Q: What is the difference between WHERE and HAVING?
  A: WHERE filters individual rows before grouping.
     HAVING filters groups after GROUP BY.
     Cannot use aggregate functions (SUM, COUNT) in WHERE.

  Q: What is a scalar subquery?
  A: A subquery that returns exactly one row and one column.
     Can be used anywhere a single value is expected (SELECT list, WHERE, etc.)
