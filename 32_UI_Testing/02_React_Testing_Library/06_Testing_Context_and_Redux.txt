1. TESTING CONTEXT CONSUMERS
================================================================================

Components that consume React Context need the Provider wrapper to render correctly.
Without the Provider, context returns the default value (or throws, depending on implementation).

SIMPLE APPROACH — Wrap in render's wrapper option:
  const ThemeContext = createContext({ theme: 'light', toggle: () => {} });

  it('renders with dark theme', () => {
    render(
      <ThemeContext.Provider value={{ theme: 'dark', toggle: jest.fn() }}>
        <ThemeToggleButton />
      </ThemeContext.Provider>
    );

    expect(screen.getByRole('button')).toHaveTextContent('dark');
  });

USING wrapper IN RENDER OPTIONS:
  render(<ThemeToggleButton />, {
    wrapper: ({ children }) => (
      <ThemeContext.Provider value={{ theme: 'dark', toggle: jest.fn() }}>
        {children}
      </ThemeContext.Provider>
    ),
  });

USING DEFAULT CONTEXT VALUES:
  // If your context has sensible defaults, components may render without a provider:
  const AuthContext = createContext({
    user: null,
    isAuthenticated: false,
    login: () => {},
    logout: () => {},
  });

  it('shows login link when not authenticated (uses default context)', () => {
    render(<NavBar />); // Uses default context: isAuthenticated = false
    expect(screen.getByRole('link', { name: /log in/i })).toBeInTheDocument();
  });

WHEN CONTEXT HAS NO DEFAULT (throws on missing Provider):
  // Hook:
  export function useAuth() {
    const ctx = useContext(AuthContext);
    if (!ctx) throw new Error('useAuth must be used within AuthProvider');
    return ctx;
  }

  // Always provide the context in tests:
  render(
    <AuthContext.Provider value={mockAuthContext}>
      <ProtectedRoute />
    </AuthContext.Provider>
  );

2. CUSTOM RENDER() WITH PROVIDERS
================================================================================

For large apps with many providers (Theme, Auth, Router, I18n, etc.),
wrapping each render() call is verbose. Create a custom render() utility.

CUSTOM RENDER SETUP:
  // src/test-utils.tsx
  import React from 'react';
  import { render, RenderOptions } from '@testing-library/react';
  import { MemoryRouter } from 'react-router-dom';
  import { ThemeProvider } from './context/ThemeContext';
  import { AuthProvider } from './context/AuthContext';

  // Props you can pass to override default context values:
  interface AllProvidersProps {
    children: React.ReactNode;
    initialRoute?: string;
  }

  function AllProviders({ children, initialRoute = '/' }: AllProvidersProps) {
    return (
      <MemoryRouter initialEntries={[initialRoute]}>
        <AuthProvider>
          <ThemeProvider>
            {children}
          </ThemeProvider>
        </AuthProvider>
      </MemoryRouter>
    );
  }

  // Custom render function:
  const customRender = (
    ui: React.ReactElement,
    options?: Omit<RenderOptions, 'wrapper'> & { initialRoute?: string }
  ) => {
    const { initialRoute, ...renderOptions } = options ?? {};
    return render(ui, {
      wrapper: ({ children }) => (
        <AllProviders initialRoute={initialRoute}>{children}</AllProviders>
      ),
      ...renderOptions,
    });
  };

  // Re-export everything from RTL, with our custom render overriding:
  export * from '@testing-library/react';
  export { customRender as render };

USING CUSTOM RENDER IN TESTS:
  // Import from test-utils, not from @testing-library/react:
  import { render, screen } from '../test-utils';

  it('shows user dashboard when authenticated', () => {
    render(<Dashboard />); // All providers already included
    expect(screen.getByRole('heading', { name: /dashboard/i })).toBeInTheDocument();
  });

  it('redirects to login when not authenticated', () => {
    render(<ProtectedRoute />, { initialRoute: '/protected' });
    expect(screen.getByRole('heading', { name: /login/i })).toBeInTheDocument();
  });

ADVANCED: INJECTABLE CONTEXT VALUES
  interface CustomRenderOptions extends Omit<RenderOptions, 'wrapper'> {
    authValue?: Partial<AuthContextType>;
    themeValue?: Partial<ThemeContextType>;
  }

  const customRender = (ui: React.ReactElement, options: CustomRenderOptions = {}) => {
    const {
      authValue = defaultAuthContext,
      themeValue = defaultThemeContext,
      ...renderOptions
    } = options;

    return render(ui, {
      wrapper: ({ children }) => (
        <AuthContext.Provider value={{ ...defaultAuthContext, ...authValue }}>
          <ThemeContext.Provider value={{ ...defaultThemeContext, ...themeValue }}>
            {children}
          </ThemeContext.Provider>
        </AuthContext.Provider>
      ),
      ...renderOptions,
    });
  };

  // Usage:
  render(<ProfilePage />, { authValue: { user: mockUser, isAuthenticated: true } });

3. TESTING CONTEXT VALUE CHANGES
================================================================================

Components that re-render when context changes need to test state transitions.

APPROACH 1: Render the Provider and update state through it:
  function TestWrapper() {
    const [theme, setTheme] = useState('light');
    return (
      <ThemeContext.Provider value={{ theme, toggle: () => setTheme(t => t === 'light' ? 'dark' : 'light') }}>
        <ThemeToggle />
        <button onClick={() => setTheme('dark')}>Force Dark</button>
      </ThemeContext.Provider>
    );
  }

  it('updates when context changes', async () => {
    const user = userEvent.setup();
    render(<TestWrapper />);

    expect(screen.getByTestId('theme-label')).toHaveTextContent('light');
    await user.click(screen.getByRole('button', { name: /force dark/i }));
    expect(screen.getByTestId('theme-label')).toHaveTextContent('dark');
  });

APPROACH 2: Use rerender to change context:
  it('re-renders when auth state changes', () => {
    const { rerender } = render(
      <AuthContext.Provider value={{ user: null, isAuthenticated: false }}>
        <NavBar />
      </AuthContext.Provider>
    );

    expect(screen.getByRole('link', { name: /login/i })).toBeInTheDocument();

    rerender(
      <AuthContext.Provider value={{ user: mockUser, isAuthenticated: true }}>
        <NavBar />
      </AuthContext.Provider>
    );

    expect(screen.getByRole('button', { name: /logout/i })).toBeInTheDocument();
    expect(screen.queryByRole('link', { name: /login/i })).not.toBeInTheDocument();
  });

4. TESTING REDUX-CONNECTED COMPONENTS
================================================================================

Components connected to Redux store via useSelector / useDispatch need a
Redux store Provider (<Provider store={...}>) in the render wrapper.

APPROACH A: Render with a Real Store (recommended for integration tests)
  Create a store with the real reducers but initial state set for the test.

APPROACH B: Mock useSelector / useDispatch (unit tests, but discouraged)
  Bypasses Redux logic entirely — not recommended.

APPROACH C: Custom render with store factory (best approach)

5. SETTING UP REDUX STORE FOR TESTS
================================================================================

CREATE A TEST STORE FACTORY:
  // src/store/index.ts (app's real store setup)
  import { configureStore } from '@reduxjs/toolkit';
  import counterReducer from './counterSlice';
  import userReducer from './userSlice';

  export const store = configureStore({
    reducer: { counter: counterReducer, user: userReducer },
  });

  // src/test-utils/store.ts
  import { configureStore } from '@reduxjs/toolkit';
  import counterReducer from '../store/counterSlice';
  import userReducer from '../store/userSlice';

  // Factory function — creates a fresh store per test:
  export function createTestStore(preloadedState = {}) {
    return configureStore({
      reducer: { counter: counterReducer, user: userReducer },
      preloadedState,
    });
  }

WHY PRELOADED STATE MATTERS:
  // Component needs user.name to be set before it renders:
  const store = createTestStore({
    user: { name: 'Alice', role: 'admin', isAuthenticated: true },
    counter: { value: 10 },
  });
  // Component renders with that state immediately — no actions needed to set it up

6. CUSTOM RENDER WITH REDUX
================================================================================

FULL TEST UTILS WITH REDUX:
  // src/test-utils/index.tsx
  import React from 'react';
  import { render, RenderOptions } from '@testing-library/react';
  import { Provider } from 'react-redux';
  import { MemoryRouter } from 'react-router-dom';
  import { createTestStore } from './store';
  import type { RootState } from '../store';

  interface ExtendedRenderOptions extends Omit<RenderOptions, 'wrapper'> {
    preloadedState?: Partial<RootState>;
    store?: ReturnType<typeof createTestStore>;
    initialRoute?: string;
  }

  export function renderWithProviders(
    ui: React.ReactElement,
    {
      preloadedState = {},
      store = createTestStore(preloadedState),
      initialRoute = '/',
      ...renderOptions
    }: ExtendedRenderOptions = {}
  ) {
    function Wrapper({ children }: { children: React.ReactNode }) {
      return (
        <Provider store={store}>
          <MemoryRouter initialEntries={[initialRoute]}>
            {children}
          </MemoryRouter>
        </Provider>
      );
    }

    return { store, ...render(ui, { wrapper: Wrapper, ...renderOptions }) };
  }

  export * from '@testing-library/react';

USAGE:
  import { renderWithProviders, screen } from '../test-utils';

  it('shows admin panel for admin users', () => {
    renderWithProviders(<Dashboard />, {
      preloadedState: {
        user: { name: 'Alice', role: 'admin', isAuthenticated: true },
      },
    });

    expect(screen.getByRole('link', { name: /admin panel/i })).toBeInTheDocument();
  });

  it('hides admin panel for regular users', () => {
    renderWithProviders(<Dashboard />, {
      preloadedState: {
        user: { name: 'Bob', role: 'viewer', isAuthenticated: true },
      },
    });

    expect(screen.queryByRole('link', { name: /admin panel/i })).not.toBeInTheDocument();
  });

7. TESTING DISPATCHED ACTIONS
================================================================================

PATTERN: USE THE RETURNED STORE TO INSPECT STATE AFTER INTERACTION
  it('increments counter when button clicked', async () => {
    const user = userEvent.setup();
    const { store } = renderWithProviders(<Counter />, {
      preloadedState: { counter: { value: 5 } },
    });

    await user.click(screen.getByRole('button', { name: /\+/i }));

    // Check the Redux state changed:
    expect(store.getState().counter.value).toBe(6);

    // Also check the UI updated:
    expect(screen.getByText('6')).toBeInTheDocument();
  });

PATTERN: SPY ON DISPATCH
  it('dispatches logout action', async () => {
    const user = userEvent.setup();
    const store = createTestStore({ user: { isAuthenticated: true } });
    const dispatchSpy = jest.spyOn(store, 'dispatch');

    renderWithProviders(<NavBar />, { store });

    await user.click(screen.getByRole('button', { name: /logout/i }));

    expect(dispatchSpy).toHaveBeenCalledWith(
      expect.objectContaining({ type: 'user/logout' })
    );
  });

PATTERN: TEST ASYNC THUNKS (API calls via Redux Thunk)
  it('loads users and populates store', async () => {
    // MSW handles the /api/users endpoint
    const { store } = renderWithProviders(<UserList />, {
      preloadedState: { users: { items: [], loading: false } },
    });

    await user.click(screen.getByRole('button', { name: /load users/i }));

    // Wait for async thunk to complete:
    await waitFor(() => {
      expect(store.getState().users.items).toHaveLength(2);
    });

    expect(screen.getByText('Alice')).toBeInTheDocument();
    expect(screen.getByText('Bob')).toBeInTheDocument();
  });

8. TESTING SELECTORS AND DERIVED STATE
================================================================================

TESTING MEMOIZED SELECTORS DIRECTLY (no render needed):
  import { selectUserRole, selectIsAdmin } from '../store/userSlice';

  it('selectIsAdmin returns true for admin role', () => {
    const state = {
      user: { id: 1, name: 'Alice', role: 'admin' }
    };
    expect(selectIsAdmin(state)).toBe(true);
  });

  it('selectIsAdmin returns false for viewer role', () => {
    const state = {
      user: { id: 2, name: 'Bob', role: 'viewer' }
    };
    expect(selectIsAdmin(state)).toBe(false);
  });

TESTING RESELECT SELECTORS:
  import { createSelector } from 'reselect';

  const selectFilteredTodos = createSelector(
    [state => state.todos, state => state.filter],
    (todos, filter) => todos.filter(t => t.status === filter)
  );

  it('filters todos by status', () => {
    const state = {
      todos: [
        { id: 1, text: 'Buy groceries', status: 'active' },
        { id: 2, text: 'Read book', status: 'completed' },
      ],
      filter: 'active',
    };
    expect(selectFilteredTodos(state)).toHaveLength(1);
    expect(selectFilteredTodos(state)[0].text).toBe('Buy groceries');
  });

9. TESTING WITH REACT QUERY
================================================================================

React Query uses a QueryClient and QueryClientProvider.
Each test needs a fresh QueryClient to prevent cache contamination.

CUSTOM RENDER WITH REACT QUERY:
  import { QueryClient, QueryClientProvider } from '@tanstack/react-query';
  import { render } from '@testing-library/react';

  function createQueryWrapper() {
    const queryClient = new QueryClient({
      defaultOptions: {
        queries: {
          retry: false,         // Don't retry failed queries in tests
          gcTime: Infinity,     // Keep cache for entire test
          staleTime: Infinity,  // Don't auto-refetch in tests
        },
      },
    });
    return ({ children }) => (
      <QueryClientProvider client={queryClient}>
        {children}
      </QueryClientProvider>
    );
  }

  export function renderWithQuery(ui, options = {}) {
    return render(ui, {
      wrapper: createQueryWrapper(),
      ...options,
    });
  }

TESTING A COMPONENT THAT USES useQuery:
  // With MSW intercepting /api/users:
  it('fetches and displays users', async () => {
    renderWithQuery(<UserList />);

    expect(screen.getByText(/loading/i)).toBeInTheDocument();

    const alice = await screen.findByText('Alice');
    expect(alice).toBeInTheDocument();
  });

  it('shows error when fetch fails', async () => {
    server.use(
      http.get('/api/users', () => HttpResponse.error())
    );

    renderWithQuery(<UserList />);
    expect(await screen.findByRole('alert')).toHaveTextContent(/error/i);
  });

10. COMMON PATTERNS AND PITFALLS
================================================================================

PATTERN: MOCK ENTIRE CONTEXT MODULE
  jest.mock('../context/AuthContext', () => ({
    useAuth: () => ({
      user: { id: 1, name: 'Alice', role: 'admin' },
      isAuthenticated: true,
      logout: jest.fn(),
    }),
    AuthProvider: ({ children }) => children, // Pass-through
  }));

  // Then test without providing the actual context:
  render(<NavBar />);
  // NavBar calls useAuth() and gets the mocked value

PITFALL 1: FORGETTING TO RESET STORE BETWEEN TESTS
  // Each test should get a FRESH store via createTestStore()
  // Don't share a store instance between tests
  ❌ const store = createTestStore(); // Defined outside tests — shared!
  ✅ const { store } = renderWithProviders(<Component />, { preloadedState: ... });

PITFALL 2: USING ACTUAL REDUX STORE IN TESTS
  // Don't import the real app store in tests:
  ❌ import { store } from '../store'; render(<Component />, { store });
  // The real store has state from previous tests — tests are not isolated

  ✅ Always use createTestStore() for a fresh instance

PITFALL 3: COUPLING TESTS TO IMPLEMENTATION DETAILS
  ❌ expect(store.getState().counter.value).toBe(6); // Fine, but avoid over-testing state
  ❌ expect(dispatchSpy).toHaveBeenCalledWith({ type: 'counter/increment' }); // Action implementation detail

  ✅ Prefer testing user-visible behavior:
  expect(screen.getByText('6')).toBeInTheDocument(); // What the user sees

PITFALL 4: NOT HANDLING ASYNC CONTEXT UPDATES
  // If context updates asynchronously (e.g., after auth check):
  ❌ render(<Protected />);
  expect(screen.getByRole('heading', { name: /dashboard/i })).toBeInTheDocument();
  // Auth check is still pending!

  ✅ Wait for the async update:
  render(<Protected />);
  await screen.findByRole('heading', { name: /dashboard/i });

QUICK REFERENCE:
  // Minimal Provider wrapper:
  render(<Component />, {
    wrapper: ({ children }) => <ThemeProvider>{children}</ThemeProvider>,
  });

  // Custom render from test-utils (preferred):
  import { render } from '../test-utils'; // Has all providers
  render(<Component />);

  // Redux with preloaded state:
  renderWithProviders(<Component />, { preloadedState: { user: mockUser } });

  // React Query wrapper:
  renderWithQuery(<Component />);
