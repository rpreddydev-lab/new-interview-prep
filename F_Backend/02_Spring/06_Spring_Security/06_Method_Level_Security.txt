  1. OVERVIEW
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Method-level security allows authorization checks directly on
individual service/repository methods using annotations.

Benefits over URL-based security:
  - Works on any Spring bean (not just controllers)
  - Uses method parameters and return values in SpEL expressions
  - Cleaner — security close to the business logic
  - Testable in isolation

Enable in configuration:
  @Configuration
  @EnableMethodSecurity          // Spring Security 6.x (Boot 3.x)
  public class SecurityConfig { ... }

  // Spring Security 5.x:
  @EnableGlobalMethodSecurity(prePostEnabled = true, securedEnabled = true)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  2. @PreAuthorize
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Evaluated BEFORE the method executes. If expression evaluates to
false → AccessDeniedException (403).

// Simple role check
@PreAuthorize("hasRole('ADMIN')")
public void deleteUser(Long userId) { ... }

// Any of multiple roles
@PreAuthorize("hasAnyRole('ADMIN', 'MODERATOR')")
public void banUser(Long userId) { ... }

// Authority check (no ROLE_ prefix)
@PreAuthorize("hasAuthority('USER_WRITE')")
public User createUser(UserRequest request) { ... }

// Check method parameter (SpEL — #paramName)
@PreAuthorize("#userId == authentication.name")
public User getProfile(@PathVariable String userId) { ... }

// Combined condition
@PreAuthorize("hasRole('ADMIN') or #userId == authentication.name")
public User getProfile(String userId) { ... }

// Access principal properties
@PreAuthorize("@userService.isOwner(#documentId, authentication.principal)")
public Document getDocument(Long documentId) { ... }

// Use custom bean (@ syntax)
@PreAuthorize("@permissionEvaluator.hasPermission(authentication, #orderId, 'ORDER', 'READ')")
public Order getOrder(Long orderId) { ... }

  3. @PostAuthorize
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Evaluated AFTER the method executes, can inspect the return value
via returnObject.

Note: the method RUNS first even if PostAuthorize denies — use
only when the method itself has no side effects.

// Only allow owner to see their document
@PostAuthorize("returnObject.owner == authentication.name")
public Document getDocument(Long id) {
    return documentRepo.findById(id).orElseThrow();
}

// Ensure returned user is the current user (or admin)
@PostAuthorize("returnObject.email == authentication.name or hasRole('ADMIN')")
public User getUser(Long id) {
    return userRepo.findById(id).orElseThrow();
}

  4. @PreFilter AND @PostFilter
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

@PreFilter  — filters an input Collection before method executes
@PostFilter — filters a return Collection after method executes

// Only process documents owned by current user
@PreFilter("filterObject.owner == authentication.name")
public void processDocuments(List<Document> docs) { ... }

// Only return documents the current user owns
@PostFilter("filterObject.owner == authentication.name")
public List<Document> getAllDocuments() {
    return documentRepo.findAll();  // all fetched, then filtered
}

⚠️  @PostFilter loads ALL data first, then removes items not matching
    the predicate. For large datasets, this is VERY inefficient.
    Prefer to filter at the query level (WHERE owner = :user).

  5. @Secured
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

A simpler, legacy annotation — only supports role checking, no SpEL:

@Secured("ROLE_ADMIN")
public void adminOnly() { ... }

@Secured({"ROLE_ADMIN", "ROLE_MANAGER"})
public void adminOrManager() { ... }

Enable: @EnableMethodSecurity(securedEnabled = true)
Prefer @PreAuthorize — it is more flexible.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  7. CUSTOM PERMISSION EVALUATOR
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Implement complex domain-object security with PermissionEvaluator:

@Component("permissionEvaluator")
public class CustomPermissionEvaluator implements PermissionEvaluator {

    @Override
    public boolean hasPermission(Authentication auth,
                                  Object targetDomainObject, Object permission) {
        if (targetDomainObject instanceof Order order) {
            String perm = (String) permission;
            return switch (perm) {
                case "READ" -> order.getOwnerId().equals(auth.getName())
                                || isAdmin(auth);
                case "WRITE" -> order.getOwnerId().equals(auth.getName());
                default -> false;
            };
        }
        return false;
    }

    @Override
    public boolean hasPermission(Authentication auth, Serializable targetId,
                                  String targetType, Object permission) {
        if ("ORDER".equals(targetType)) {
            Order order = orderRepo.findById((Long) targetId).orElseThrow();
            return hasPermission(auth, order, permission);
        }
        return false;
    }
}

// Usage:
@PreAuthorize("hasPermission(#order, 'READ')")
public Order viewOrder(Order order) { ... }

@PreAuthorize("hasPermission(#orderId, 'ORDER', 'WRITE')")
public void updateOrder(Long orderId, OrderUpdateRequest req) { ... }

// Register:
@Bean
public MethodSecurityExpressionHandler expressionHandler(
        CustomPermissionEvaluator permEval) {
    DefaultMethodSecurityExpressionHandler handler =
        new DefaultMethodSecurityExpressionHandler();
    handler.setPermissionEvaluator(permEval);
    return handler;
}

class OrderServiceSecurityTest {

    @Autowired
    private OrderService orderService;

    @Test
    @WithMockUser(roles = "USER")
    void regularUserCannotDeleteOrder() {
        assertThrows(AccessDeniedException.class,
            () -> orderService.deleteOrder(1L));
    }

    @Test
    @WithMockUser(roles = "ADMIN")
    void adminCanDeleteOrder() {
        assertDoesNotThrow(() -> orderService.deleteOrder(1L));
    }

    @Test
    @WithMockUser(username = "user1")
    void userCanAccessOwnProfile() {
        assertDoesNotThrow(() -> orderService.getProfile("user1"));
    }

    // Custom user details
    @Test
    @WithUserDetails("admin@example.com")  // loads from UserDetailsService
    void adminCanDoEverything() { ... }
}

Q: What is the difference between @PreAuthorize and @PostAuthorize?
A: @PreAuthorize evaluates BEFORE the method runs — blocks execution
   if denied. @PostAuthorize evaluates AFTER the method runs — can
   access the returnObject. Use @PostAuthorize when authorization
   depends on the data returned. Warning: the method executes either
   way with @PostAuthorize, so avoid side-effectful methods.

Q: How does method security work internally?
A: @EnableMethodSecurity registers a MethodInterceptor using Spring AOP.
   Every call to a @PreAuthorize-annotated method is intercepted, the
   SpEL expression is evaluated against the current SecurityContext
   and method arguments. If false, AccessDeniedException is thrown.

Q: Why might @PreAuthorize not work on a method?
A: Common causes: (1) @EnableMethodSecurity not added to config,
   (2) the method is called from within the same class (self-invocation
   bypasses the AOP proxy), (3) the method is private (Spring AOP
   cannot proxy private methods).

Q: When should you use @PostFilter vs filtering in the query?
A: @PostFilter is convenient for small datasets but loads ALL records
   from DB first, then removes items. For large datasets, filter in
   the repository query (WHERE owner = :user) — much more efficient.

Q: What is a PermissionEvaluator and when do you need it?
A: PermissionEvaluator enables domain-object security — checking
   permission on a specific object instance (e.g., can this user
   read ORDER #42?). Used with hasPermission() in SpEL. It is needed
   when role-based checks are insufficient and you need fine-grained,
   object-level authorization logic.
