  1. MOCKITO CORE CONCEPTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Mock        → fake object with no real implementation; all methods return defaults
Stub        → mock configured to return specific values for specific inputs
Spy         → real object with some methods overridden (partial mock)
Verify      → assert that certain methods were called with expected args
Captor      → capture the argument passed to a mock for assertion
ArgumentMatcher → flexible matching (any(), eq(), argThat(), etc.)

  2. SETUP — TWO STYLES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

// Style 1: @ExtendWith (preferred for JUnit 5 — no Spring context)
@ExtendWith(MockitoExtension.class)
class OrderServiceTest {

    @Mock
    private OrderRepository orderRepository;

    @Mock
    private PaymentService paymentService;

    @InjectMocks                // injects @Mock fields via constructor/setter/field
    private OrderService orderService;

    // ...tests
}

// Style 2: Manual (useful when you need full control)
class OrderServiceTest {

    private OrderRepository orderRepository;
    private PaymentService paymentService;
    private OrderService orderService;

    @BeforeEach
    void setUp() {
        orderRepository = Mockito.mock(OrderRepository.class);
        paymentService = Mockito.mock(PaymentService.class);
        orderService = new OrderService(orderRepository, paymentService);
    }
}

  3. STUBBING — WHEN / GIVEN
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

// when().thenReturn() — classic
when(orderRepository.findById(1L)).thenReturn(Optional.of(order));
when(orderRepository.findAll()).thenReturn(List.of(order1, order2));
when(orderRepository.save(any(Order.class))).thenReturn(savedOrder);

// BDD style — given().willReturn() (same behavior, better readability)
given(orderRepository.findById(1L)).willReturn(Optional.of(order));

// Throw exception
when(paymentService.charge(any())).thenThrow(new PaymentException("declined"));
doThrow(new RuntimeException()).when(orderRepository).delete(any());

// Return different values on successive calls
when(orderRepository.findById(1L))
    .thenReturn(Optional.of(pendingOrder))  // first call
    .thenReturn(Optional.of(shippedOrder)); // second call

// Answer (compute return value dynamically)
when(orderRepository.save(any(Order.class)))
    .thenAnswer(invocation -> {
        Order o = invocation.getArgument(0);
        o.setId(123L);  // simulate DB-assigned ID
        return o;
    });

// Void methods
doNothing().when(emailService).sendOrderConfirmation(anyString());
// or just don't stub — void methods do nothing by default

  4. ARGUMENT MATCHERS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

// Common matchers
any()                → any object (including null)
any(Order.class)     → any instance of Order (not null)
anyString()          → any non-null String
anyLong()            → any Long
anyList()            → any List
eq("specific")       → exactly "specific" (use when mixing matchers)
isNull()             → null value
isNotNull()          → non-null value
argThat(o -> o.getId() > 0)  → custom predicate

// IMPORTANT: if you use ANY matcher for one arg, ALL args must use matchers
when(repo.findByIdAndStatus(eq(1L), eq("ACTIVE"))).thenReturn(order);  // ✓
when(repo.findByIdAndStatus(1L, eq("ACTIVE"))).thenReturn(order);      // ✗ mixing not allowed

  5. VERIFY — INTERACTION TESTING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

// Verify called exactly once (default)
verify(orderRepository).save(order);
verify(emailService).sendOrderConfirmation(order.getCustomerEmail());

// Verify call count
verify(orderRepository, times(2)).save(any());
verify(orderRepository, never()).delete(any());
verify(orderRepository, atLeastOnce()).findById(anyLong());
verify(orderRepository, atMost(3)).findAll();

// Verify order of calls (InOrder)
InOrder inOrder = inOrder(orderRepository, paymentService);
inOrder.verify(paymentService).charge(any());
inOrder.verify(orderRepository).save(any());

// Verify no more interactions
verifyNoMoreInteractions(orderRepository);
verifyNoInteractions(emailService);  // never called at all

// BDD style
then(orderRepository).should().save(order);
then(emailService).should(never()).sendAlert(any());

  6. ARGUMENT CAPTOR
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Used to capture the exact argument passed to a mock, then assert on it.

@Captor
private ArgumentCaptor<Order> orderCaptor;

@Test
void shouldSaveOrderWithCorrectStatus() {
    // Act
    orderService.placeOrder(request);

    // Capture the Order passed to save()
    verify(orderRepository).save(orderCaptor.capture());
    Order savedOrder = orderCaptor.getValue();

    assertThat(savedOrder.getStatus()).isEqualTo(OrderStatus.PENDING);
    assertThat(savedOrder.getTotal()).isEqualByComparingTo("99.99");
}

// Capture multiple invocations
verify(emailService, times(2)).send(emailCaptor.capture());
List<Email> emails = emailCaptor.getAllValues();
assertThat(emails).extracting(Email::getTo)
    .containsExactly("alice@example.com", "bob@example.com");

  7. SPY — PARTIAL MOCK
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

A spy wraps a real object. Real methods are called UNLESS you stub them.

@Spy
private NotificationService notificationService = new EmailNotificationService();

// Only stub what you need
doReturn("mocked-message-id").when(notificationService).sendEmail(any());

// Other methods (not stubbed) execute the REAL implementation
notificationService.formatMessage(event);  // real code runs

// IMPORTANT: Use doReturn() not when() with spies:
// when(spy.someMethod()).thenReturn(x)  → calls the REAL method first (may NPE)
// doReturn(x).when(spy).someMethod()    → safe, skips real call

  8. MOCKING STATIC METHODS (Mockito 3.4+)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

try (MockedStatic<UUIDUtils> mocked = Mockito.mockStatic(UUIDUtils.class)) {
    mocked.when(UUIDUtils::generate).thenReturn("fixed-uuid");

    String id = orderService.createOrderId();
    assertThat(id).isEqualTo("fixed-uuid");
}
// Static mock is automatically closed at end of try block

// Mock construction (new operator)
try (MockedConstruction<PaymentGateway> mocked =
        Mockito.mockConstruction(PaymentGateway.class,
            (mock, ctx) -> when(mock.charge(any())).thenReturn("SUCCESS"))) {

    paymentService.processPayment(request);
    verify(mocked.constructed().get(0)).charge(any());
}

@ExtendWith(MockitoExtension.class)
class OrderServiceTest {

    @Mock private OrderRepository orderRepository;
    @Mock private PaymentService paymentService;
    @Mock private EmailService emailService;
    @Captor private ArgumentCaptor<Order> orderCaptor;

    @InjectMocks
    private OrderService orderService;

    private CreateOrderRequest validRequest;

    @BeforeEach
    void setUp() {
        validRequest = new CreateOrderRequest(1L, 2, new BigDecimal("49.99"));
    }

    @Test
    @DisplayName("Should place order, charge payment, and send confirmation")
    void shouldPlaceOrderSuccessfully() {
        // Given
        given(paymentService.charge(any())).willReturn("TX-001");
        given(orderRepository.save(any())).willAnswer(inv -> {
            Order o = inv.getArgument(0);
            o.setId(1L);
            return o;
        });

        // When
        Order result = orderService.placeOrder(validRequest);

        // Then
        assertThat(result.getId()).isEqualTo(1L);
        assertThat(result.getStatus()).isEqualTo(OrderStatus.CONFIRMED);

        verify(orderRepository).save(orderCaptor.capture());
        assertThat(orderCaptor.getValue().getTotal())
            .isEqualByComparingTo("49.99");

        verify(emailService).sendOrderConfirmation(anyString());
    }

    @Test
    @DisplayName("Should throw exception when payment fails")
    void shouldThrowWhenPaymentFails() {
        // Given
        given(paymentService.charge(any()))
            .willThrow(new PaymentException("Card declined"));

        // When & Then
        assertThatThrownBy(() -> orderService.placeOrder(validRequest))
            .isInstanceOf(PaymentException.class)
            .hasMessageContaining("Card declined");

        verify(orderRepository, never()).save(any());
        verify(emailService, never()).sendOrderConfirmation(any());
    }
}

Q: What is the difference between @Mock and @Spy?
A: @Mock creates a pure mock — all methods return default values (null,
   0, false, empty collections) unless stubbed. @Spy wraps a real object
   — real methods are called unless you explicitly stub them. Use @Mock
   for dependencies you want to fully control; use @Spy when you need
   mostly real behavior with a few overrides.

Q: What is @InjectMocks and how does injection work?
A: @InjectMocks creates an instance of the class under test and tries
   to inject @Mock/@Spy fields into it via: (1) constructor injection
   (preferred — picks the largest constructor it can satisfy), (2)
   setter injection, (3) field injection as a last resort.

Q: When should you use ArgumentCaptor vs custom matchers?
A: Use ArgumentCaptor when you need to assert multiple properties of the
   captured argument AFTER the verify. Use argThat() or custom matchers
   when you want to restrict stubbing/verification to arguments matching
   specific criteria. Captor is cleaner for complex assertions; matchers
   are better for reusable matching logic.

Q: What is the difference between verify() and assertThat() in a test?
A: assertThat() verifies the STATE of the system (the return value, the
   object's properties). verify() verifies BEHAVIOR — that specific
   interactions with mocks happened. A good unit test uses both:
   assertThat checks the result; verify checks that the correct
   collaborators were called.

Q: What are the problems with using @Spy over @Mock?
A: With @Spy, the real constructor is called, which may require the object
   to be initialized (dependencies, connections). If using when().thenReturn()
   on a spy, the real method is invoked before being stubbed (potential side
   effects, NPEs). Use doReturn() instead. Prefer @Mock unless you specifically
   need real behavior.
