━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

WebFlux is the reactive-stack web framework in Spring 5+. It runs on:
  - Netty (default): NIO event-loop, single-threaded per core
  - Undertow: supports reactive IO
  - Tomcat / Jetty: with non-blocking NIO connectors

Dependency:
<dependency>
    <groupId>org.springframework.boot</groupId>
    <artifactId>spring-boot-starter-webflux</artifactId>
</dependency>

IMPORTANT: Do not mix spring-boot-starter-web (Servlet) with
           spring-boot-starter-webflux — they conflict.

  2. ANNOTATED CONTROLLER MODEL
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Same annotations as Spring MVC but return types are Mono/Flux.

@RestController
@RequestMapping("/api/orders")
public class OrderController {

    private final OrderService orderService;

    // GET single — returns Mono (0 or 1 result)
    @GetMapping("/{id}")
    public Mono<ResponseEntity<Order>> findById(@PathVariable Long id) {
        return orderService.findById(id)
            .map(ResponseEntity::ok)
            .defaultIfEmpty(ResponseEntity.notFound().build());
    }

    // GET list — returns Flux
    @GetMapping
    public Flux<Order> findAll(
        @RequestParam(defaultValue = "0") int page,
        @RequestParam(defaultValue = "20") int size) {
        return orderService.findAll(PageRequest.of(page, size));
    }

    // POST — accepts reactive body
    @PostMapping
    @ResponseStatus(HttpStatus.CREATED)
    public Mono<Order> create(@Valid @RequestBody Mono<CreateOrderRequest> request) {
        return request.flatMap(orderService::create);
    }

    // PUT — update
    @PutMapping("/{id}")
    public Mono<Order> update(
        @PathVariable Long id,
        @RequestBody Mono<UpdateOrderRequest> request) {
        return request.flatMap(r -> orderService.update(id, r));
    }

    // DELETE
    @DeleteMapping("/{id}")
    @ResponseStatus(HttpStatus.NO_CONTENT)
    public Mono<Void> delete(@PathVariable Long id) {
        return orderService.delete(id);
    }

    // Server-Sent Events (streaming)
    @GetMapping(value = "/stream", produces = MediaType.TEXT_EVENT_STREAM_VALUE)
    public Flux<Order> stream() {
        return orderService.findAll()
            .delayElements(Duration.ofMillis(200));
    }
}

RouterFunction + HandlerFunction replace @RequestMapping annotations.
More explicit, functional, and testable.

@Configuration
public class OrderRouter {

    @Bean
    public RouterFunction<ServerResponse> orderRoutes(OrderHandler handler) {
        return route()
            .GET("/api/orders/{id}", handler::findById)
            .GET("/api/orders",      handler::findAll)
            .POST("/api/orders",     handler::create)
            .PUT("/api/orders/{id}", handler::update)
            .DELETE("/api/orders/{id}", handler::delete)
            .build();
    }
}

@Component
public class OrderHandler {

    private final OrderService service;

    public Mono<ServerResponse> findById(ServerRequest request) {
        Long id = Long.parseLong(request.pathVariable("id"));
        return service.findById(id)
            .flatMap(order -> ServerResponse.ok()
                .contentType(MediaType.APPLICATION_JSON)
                .bodyValue(order))
            .switchIfEmpty(ServerResponse.notFound().build());
    }

    public Mono<ServerResponse> findAll(ServerRequest request) {
        return ServerResponse.ok()
            .contentType(MediaType.APPLICATION_JSON)
            .body(service.findAll(), Order.class);
    }

    public Mono<ServerResponse> create(ServerRequest request) {
        return request.bodyToMono(CreateOrderRequest.class)
            .flatMap(service::create)
            .flatMap(order -> ServerResponse
                .created(URI.create("/api/orders/" + order.getId()))
                .bodyValue(order));
    }

    public Mono<ServerResponse> update(ServerRequest request) {
        Long id = Long.parseLong(request.pathVariable("id"));
        return request.bodyToMono(UpdateOrderRequest.class)
            .flatMap(r -> service.update(id, r))
            .flatMap(order -> ServerResponse.ok().bodyValue(order));
    }

    public Mono<ServerResponse> delete(ServerRequest request) {
        Long id = Long.parseLong(request.pathVariable("id"));
        return service.delete(id)
            .then(ServerResponse.noContent().build());
    }
}

  4. WEBCLIENT — REACTIVE HTTP CLIENT
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

WebClient is the non-blocking replacement for RestTemplate.

@Configuration
public class WebClientConfig {

    @Bean
    public WebClient productClient(@Value("${product-service.url}") String url) {
        return WebClient.builder()
            .baseUrl(url)
            .defaultHeader(HttpHeaders.CONTENT_TYPE, MediaType.APPLICATION_JSON_VALUE)
            .codecs(c -> c.defaultCodecs().maxInMemorySize(1024 * 1024))
            .build();
    }
}

@Service
public class ProductServiceClient {

    private final WebClient client;

    // GET single
    public Mono<Product> findById(Long id) {
        return client.get()
            .uri("/api/products/{id}", id)
            .retrieve()
            .onStatus(HttpStatusCode::is4xxClientError, response ->
                Mono.error(new ProductNotFoundException(id)))
            .onStatus(HttpStatusCode::is5xxServerError, response ->
                Mono.error(new ServiceException("product-service error")))
            .bodyToMono(Product.class);
    }

    // GET list
    public Flux<Product> findAll() {
        return client.get()
            .uri("/api/products")
            .retrieve()
            .bodyToFlux(Product.class);
    }

    // POST
    public Mono<Product> create(CreateProductRequest request) {
        return client.post()
            .uri("/api/products")
            .bodyValue(request)
            .retrieve()
            .bodyToMono(Product.class);
    }

    // Parallel calls
    public Mono<OrderDetails> enrichOrder(Long orderId) {
        Mono<Order> order = findOrderById(orderId);
        Mono<Product> product = findProductById(1L);
        Mono<Customer> customer = findCustomerById(2L);

        return Mono.zip(order, product, customer)
            .map(tuple -> OrderDetails.of(tuple.getT1(), tuple.getT2(), tuple.getT3()));
    }
}

  5. ERROR HANDLING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

// @ControllerAdvice works with WebFlux too
@RestControllerAdvice
public class GlobalExceptionHandler {

    @ExceptionHandler(ProductNotFoundException.class)
    @ResponseStatus(HttpStatus.NOT_FOUND)
    public ProblemDetail handleNotFound(ProductNotFoundException ex) {
        return ProblemDetail.forStatusAndDetail(HttpStatus.NOT_FOUND, ex.getMessage());
    }

    @ExceptionHandler(WebExchangeBindException.class)
    @ResponseStatus(HttpStatus.BAD_REQUEST)
    public ProblemDetail handleValidation(WebExchangeBindException ex) {
        Map<String, String> errors = ex.getFieldErrors().stream()
            .collect(Collectors.toMap(FieldError::getField, FieldError::getDefaultMessage));
        ProblemDetail pd = ProblemDetail.forStatus(HttpStatus.BAD_REQUEST);
        pd.setProperty("errors", errors);
        return pd;
    }
}

// In operator chain (local error handling)
service.findById(id)
    .onErrorReturn(ProductNotFoundException.class, Product.empty())
    .onErrorMap(TimeoutException.class, e -> new ServiceException("timeout"))
    .doOnError(ex -> log.error("Error: {}", ex.getMessage()));

  6. TESTING WEBFLUX WITH WebTestClient
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

@WebFluxTest(OrderController.class)
class OrderControllerTest {

    @Autowired
    private WebTestClient client;

    @MockBean
    private OrderService orderService;

    @Test
    void shouldReturnOrder() {
        given(orderService.findById(1L))
            .willReturn(Mono.just(new Order(1L, OrderStatus.PENDING)));

        client.get().uri("/api/orders/1")
            .exchange()
            .expectStatus().isOk()
            .expectBody()
            .jsonPath("$.id").isEqualTo(1)
            .jsonPath("$.status").isEqualTo("PENDING");
    }

    @Test
    void shouldReturn404WhenOrderNotFound() {
        given(orderService.findById(99L)).willReturn(Mono.empty());

        client.get().uri("/api/orders/99")
            .exchange()
            .expectStatus().isNotFound();
    }
}

Q: What is the difference between the annotated controller model and
   the functional endpoint model in WebFlux?
A: Annotated controller uses @RestController and @GetMapping (same as Spring MVC)
   but returns Mono/Flux. Functional endpoints use RouterFunction + HandlerFunction
   — routes are declared programmatically as beans. Functional is more explicit and
   easier to compose, while annotated is familiar to MVC developers.

Q: What is WebClient and when should you use it?
A: WebClient is the non-blocking HTTP client introduced in Spring 5. It returns
   Mono/Flux and integrates naturally with reactive pipelines. Use it in WebFlux
   applications for calling external services. In Spring MVC, it can still be used
   by calling .block() to get a synchronous result (though this defeats its purpose).

Q: How do you handle blocking code in WebFlux?
A: Wrap the blocking call in Mono.fromCallable() and subscribe on Schedulers
   .boundedElastic() — a thread pool designed for I/O blocking:
   Mono.fromCallable(() -> jdbcTemplate.query(...))
       .subscribeOn(Schedulers.boundedElastic())

Q: How does error handling differ between Spring MVC and WebFlux?
A: Both support @ControllerAdvice/@ExceptionHandler. The difference is that
   errors in WebFlux are part of the reactive pipeline — you can handle them
   inline with onErrorReturn, onErrorMap, onErrorResume. In Spring MVC,
   exceptions propagate out of methods to the advice.

Q: What is @WebFluxTest and what does it load?
A: @WebFluxTest loads only the WebFlux layer (controllers, filters, @ControllerAdvice,
   WebFluxConfigurer). It does not load @Service, @Repository, or the full context.
   Use @MockBean for service dependencies. It auto-configures a WebTestClient.
