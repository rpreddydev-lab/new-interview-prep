  // Custom converter: String → Money
  @Component
  public class StringToMoneyConverter implements Converter<String, Money> {
      @Override
      public Money convert(String source) {
          String[] parts = source.split(" ");
          return new Money(new BigDecimal(parts[0]),
                           Currency.getInstance(parts[1]));
      }
  }

  // Jackson annotations on response DTO
  @JsonNaming(PropertyNamingStrategies.SnakeCaseStrategy.class)
  public class UserDto {
      private Long id;
      private String firstName;      // → "first_name" in JSON
      private String email;
      @JsonIgnore
      private String passwordHash;   // excluded from output
      @JsonInclude(Include.NON_NULL)
      private String phone;          // excluded if null
  }

  // Global Jackson configuration
  @Bean
  public Jackson2ObjectMapperBuilderCustomizer jsonCustomizer() {
      return builder -> builder
          .featuresToDisable(SerializationFeature.WRITE_DATES_AS_TIMESTAMPS)
          .featuresToEnable(DeserializationFeature.FAIL_ON_UNKNOWN_PROPERTIES)
          .modules(new JavaTimeModule())
          .propertyNamingStrategy(PropertyNamingStrategies.SNAKE_CASE);
  }

================================================================
HTTP STATUS CODE GUIDE
================================================================

  2xx Success:
    200 OK              -- successful GET, PUT, PATCH
    201 Created         -- successful POST (with Location header)
    202 Accepted        -- async processing initiated
    204 No Content      -- successful DELETE or PUT with no body

  3xx Redirection:
    301 Moved Permanently -- resource moved
    302 Found             -- temporary redirect
    304 Not Modified      -- cached response still valid (ETag match)

  4xx Client Errors:
    400 Bad Request        -- invalid input
    401 Unauthorized       -- authentication required
    403 Forbidden          -- authenticated but no permission
    404 Not Found          -- resource doesn't exist
    405 Method Not Allowed -- wrong HTTP method
    409 Conflict           -- state conflict (duplicate, version mismatch)
    410 Gone               -- resource permanently deleted
    415 Unsupported Media Type -- wrong Content-Type
    422 Unprocessable Entity   -- validation failed (semantically wrong)
    429 Too Many Requests      -- rate limited

  5xx Server Errors:
    500 Internal Server Error -- unexpected exception
    502 Bad Gateway           -- upstream service error
    503 Service Unavailable   -- overloaded / maintenance
    504 Gateway Timeout       -- upstream timeout

  // Via HttpServletResponse (imperative)
  @GetMapping("/report")
  public void downloadReport(HttpServletResponse response) throws IOException {
      response.setContentType("application/pdf");
      response.setHeader("Content-Disposition",
                         "attachment; filename=report.pdf");
      try (OutputStream out = response.getOutputStream()) {
          reportService.writePdf(out);
      }
  }

  // Programmatic cache control
  @GetMapping("/products/{id}")
  public ResponseEntity<Product> getProduct(@PathVariable Long id) {
      Product p = productService.findById(id);
      return ResponseEntity.ok()
          .cacheControl(CacheControl.maxAge(10, TimeUnit.MINUTES)
                                    .mustRevalidate())
          .eTag(p.getVersion().toString())
          .body(p);
  }

INPUT VALIDATION WITH @Valid
================================================================

  @PostMapping("/registrations")
  public ResponseEntity<RegistrationResponse> register(
          @Valid @RequestBody RegistrationRequest request,
          BindingResult result) {

      if (result.hasErrors()) {
          // Manual error handling
          List<String> errors = result.getFieldErrors().stream()
              .map(e -> e.getField() + ": " + e.getDefaultMessage())
              .toList();
          return ResponseEntity.badRequest().body(new RegistrationResponse(errors));
      }

      // Without BindingResult → Spring throws MethodArgumentNotValidException
      return ResponseEntity.ok(registrationService.register(request));
  }

READING/WRITING RAW HTTP STREAMS
================================================================

  // Streaming upload
  @PostMapping("/stream-upload")
  public ResponseEntity<?> streamUpload(
          InputStream inputStream,          // raw request body stream
          @RequestHeader("Content-Length") long length) {
      storageService.store(inputStream, length);
      return ResponseEntity.noContent().build();
  }

  // Streaming download — avoid loading large files into memory
  @GetMapping("/files/{name}")
  public ResponseEntity<StreamingResponseBody> download(
          @PathVariable String name) {
      StreamingResponseBody body = outputStream -> {
          try (InputStream in = storageService.getStream(name)) {
              in.transferTo(outputStream);
          }
      };
      return ResponseEntity.ok()
          .contentType(MediaType.APPLICATION_OCTET_STREAM)
          .header("Content-Disposition", "attachment; filename=" + name)
          .body(body);
  }

================================================================
INTERVIEW QUESTIONS — REQUEST/RESPONSE HANDLING
================================================================

  Q: How does Spring convert a String query parameter to LocalDate?
  A: Via @DateTimeFormat annotation on the parameter + ConversionService.
     Spring reads the annotation and applies the format pattern.

  Q: What is the difference between 400 and 422?
  A: 400 Bad Request means the request is syntactically invalid
     (malformed JSON, wrong type). 422 Unprocessable Entity means
     the request is syntactically valid but semantically incorrect
     (failed business validation rules).

  Q: How do you stream a large file download without OutOfMemoryError?
  A: Return a StreamingResponseBody from the controller. Spring writes
     to the response output stream in a task executor, without
     buffering the entire file in memory.

  Q: How does content negotiation work?
  A: DispatcherServlet's ContentNegotiationManager inspects the
     Accept header, URL extension, or request parameter to determine
     desired media type. It then selects the matching
     HttpMessageConverter to serialize the response.

  Q: What happens when you omit BindingResult after @Valid?
  A: Spring automatically throws MethodArgumentNotValidException
     (HTTP 400) with all constraint violations. Handle it globally
     with @RestControllerAdvice + @ExceptionHandler.
