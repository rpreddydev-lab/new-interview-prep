REDIS TESTING IN JAVA/SPRING
==============================

=== OPTIONS OVERVIEW ===

1. Testcontainers (RECOMMENDED) — real Redis in Docker
2. Embedded Redis (okumin/embedded-redis) — lightweight, no Docker needed
3. Redis Mock (testredis) — in-memory mock, limited commands
4. Integration test against real Redis — requires running Redis

=== TESTCONTAINERS (RECOMMENDED) ===

Most reliable: uses real Redis Docker container.
Requires Docker daemon running during tests.

Dependency:
  <dependency>
      <groupId>org.springframework.boot</groupId>
      <artifactId>spring-boot-testcontainers</artifactId>
      <scope>test</scope>
  </dependency>
  <dependency>
      <groupId>org.testcontainers</groupId>
      <artifactId>junit-jupiter</artifactId>
      <scope>test</scope>
  </dependency>

Test with Testcontainers:

@SpringBootTest
@Testcontainers
class RedisCacheTest {

    @Container
    static GenericContainer<?> redis = new GenericContainer<>(
        DockerImageName.parse("redis:7.2-alpine"))
        .withExposedPorts(6379)
        .waitingFor(Wait.forLogMessage(".*Ready to accept connections.*\\n", 1));

    @DynamicPropertySource
    static void configureProperties(DynamicPropertyRegistry registry) {
        registry.add("spring.data.redis.host", redis::getHost);
        registry.add("spring.data.redis.port",
                     () -> redis.getMappedPort(6379).toString());
    }

    @Autowired
    private ProductService productService; // uses @Cacheable internally

    @Autowired
    private RedisTemplate<String, Object> redisTemplate;

    @Test
    void shouldCacheProductOnFirstCall() {
        // First call — hits DB, caches result
        Product product1 = productService.findById("P001");
        // Second call — should return from cache
        Product product2 = productService.findById("P001");

        assertThat(product1).isEqualTo(product2);
        // Verify it's actually in Redis
        assertThat(redisTemplate.hasKey("products::P001")).isTrue();
    }

    @Test
    void shouldEvictCacheOnUpdate() {
        productService.findById("P001");          // populate cache
        productService.updateProduct("P001", ...); // @CacheEvict triggers

        assertThat(redisTemplate.hasKey("products::P001")).isFalse();
    }
}

=== SPRING BOOT 3 TESTCONTAINERS AUTO-CONFIGURATION ===

Simpler approach with Spring Boot 3.1+ service connections:

@SpringBootTest
@Testcontainers
class RedisServiceTest {

    @Container
    @ServiceConnection  // auto-configures Spring Data Redis
    static RedisContainer redis = new RedisContainer(
        DockerImageName.parse("redis:7.2-alpine"));

    // No @DynamicPropertySource needed!
    // Spring Boot auto-detects RedisContainer and configures host/port

    @Autowired
    private StringRedisTemplate stringRedisTemplate;

    @Test
    void testSetAndGet() {
        stringRedisTemplate.opsForValue().set("key", "value");
        assertThat(stringRedisTemplate.opsForValue().get("key")).isEqualTo("value");
    }
}

Note: import org.springframework.boot.testcontainers.service.connection.ServiceConnection

=== EMBEDDED REDIS ===

No Docker required — runs Redis in-process.
Useful for CI without Docker or lightweight unit tests.

Dependency:
  <dependency>
      <groupId>it.ozimov</groupId>
      <artifactId>embedded-redis</artifactId>
      <version>0.7.3</version>
      <scope>test</scope>
  </dependency>

Configuration:

@TestConfiguration
public class EmbeddedRedisConfig {

    private RedisServer redisServer;

    @PostConstruct
    public void startRedis() throws IOException {
        redisServer = new RedisServer(6370); // custom port to avoid conflicts
        redisServer.start();
    }

    @PreDestroy
    public void stopRedis() {
        if (redisServer != null && redisServer.isActive()) {
            redisServer.stop();
        }
    }
}

application-test.yml:
  spring:
    data:
      redis:
        host: localhost
        port: 6370  # match embedded server port

Test:
  @SpringBootTest
  @ActiveProfiles("test")
  @Import(EmbeddedRedisConfig.class)
  class CacheTest {
      @Test
      void testCaching() { ... }
  }

Limitation: embedded-redis is based on Redis 3.x — some newer commands missing.

=== MOCKING REDISTEMPLATE (UNIT TESTS) ===

For pure unit tests of services that use RedisTemplate:

@ExtendWith(MockitoExtension.class)
class SessionServiceTest {

    @Mock
    private StringRedisTemplate redisTemplate;

    @Mock
    private ValueOperations<String, String> valueOps;

    @InjectMocks
    private SessionService sessionService;

    @BeforeEach
    void setup() {
        when(redisTemplate.opsForValue()).thenReturn(valueOps);
    }

    @Test
    void shouldStoreSession() {
        sessionService.createSession("user123", "session-data");

        verify(valueOps).set(
            eq("session:user123"),
            eq("session-data"),
            eq(Duration.ofHours(1))
        );
    }

    @Test
    void shouldReturnNullWhenSessionExpired() {
        when(valueOps.get("session:user123")).thenReturn(null);

        String session = sessionService.getSession("user123");

        assertThat(session).isNull();
    }
}

=== TESTING @CACHEABLE ===

@SpringBootTest
@Testcontainers
class CacheableTest {

    @Container
    @ServiceConnection
    static RedisContainer redis = new RedisContainer(DockerImageName.parse("redis:7.2"));

    @Autowired
    private ProductRepository productRepository; // the real repo

    @Autowired
    private ProductService productService; // @Cacheable annotated

    @Autowired
    private CacheManager cacheManager;

    @Test
    void shouldCallRepositoryOnlyOnce() {
        productService.getProduct("P001");  // populates cache
        productService.getProduct("P001");  // from cache

        // Verify DB was called only once
        verify(productRepository, times(1)).findById("P001");
    }

    @Test
    void shouldReturnCachedValue() {
        // Pre-populate cache manually
        Cache cache = cacheManager.getCache("products");
        cache.put("P002", new Product("P002", "Laptop", 999.99));

        // Service should use cached value, not hit DB
        Product result = productService.getProduct("P002");
        assertThat(result.getName()).isEqualTo("Laptop");
        verify(productRepository, never()).findById("P002");
    }

    @Test
    void cacheEvictionTest() {
        productService.getProduct("P001");        // populate
        productService.invalidateProduct("P001"); // @CacheEvict

        Cache cache = cacheManager.getCache("products");
        assertThat(cache.get("P001")).isNull();
    }
}

=== TESTING REDIS PUB/SUB ===

@SpringBootTest
@Testcontainers
class RedisPubSubTest {

    @Container
    @ServiceConnection
    static RedisContainer redis = new RedisContainer(DockerImageName.parse("redis:7.2"));

    @Autowired
    private RedisTemplate<String, String> redisTemplate;

    @Test
    void shouldReceivePublishedMessage() throws InterruptedException {
        CountDownLatch latch = new CountDownLatch(1);
        List<String> received = new ArrayList<>();

        // Subscribe in background thread
        Thread subscriber = new Thread(() -> {
            redisTemplate.execute(new RedisCallback<Void>() {
                @Override
                public Void doInRedis(RedisConnection conn) {
                    conn.subscribe((message, pattern) -> {
                        received.add(new String(message.getBody()));
                        latch.countDown();
                    }, "test-channel".getBytes());
                    return null;
                }
            });
        });
        subscriber.start();
        Thread.sleep(100); // let subscriber connect

        // Publish
        redisTemplate.convertAndSend("test-channel", "hello");

        assertThat(latch.await(2, TimeUnit.SECONDS)).isTrue();
        assertThat(received).contains("hello");
    }
}
