1. WHY CRITERIA API?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

JPQL with string concatenation is fragile for dynamic queries:

  // Fragile string building:
  String jpql = "SELECT u FROM User u WHERE 1=1";
  if (firstName != null) jpql += " AND u.firstName = :firstName";
  if (status != null)    jpql += " AND u.status = :status";

Problems:
  - No compile-time checking of field names
  - Runtime errors if typo in "firstName" vs "first_name"
  - Hard to maintain as complexity grows

Criteria API solution:
  - Type-safe, compile-time validated
  - Programmatic predicate building
  - Used under the hood by Spring Specifications

2. CORE CLASSES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

CriteriaBuilder  → factory for predicates, expressions, queries
CriteriaQuery<T> → the query itself (typed by result type)
Root<T>          → the FROM clause (entity root)
Predicate        → a WHERE condition (can be ANDed/ORed)
Path<X>          → navigation into fields (root.get("field"))

Obtain CriteriaBuilder:
  CriteriaBuilder cb = em.getCriteriaBuilder();

3. BASIC QUERY
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

// Find all active users ordered by lastName
CriteriaBuilder cb = em.getCriteriaBuilder();
CriteriaQuery<User> cq = cb.createQuery(User.class);
Root<User> user = cq.from(User.class);

cq.select(user)
  .where(cb.equal(user.get("active"), true))
  .orderBy(cb.asc(user.get("lastName")));

List<User> results = em.createQuery(cq).getResultList();

4. DYNAMIC PREDICATES — THE MAIN USE CASE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

public List<User> searchUsers(String firstName, String email, Boolean active) {
    CriteriaBuilder cb = em.getCriteriaBuilder();
    CriteriaQuery<User> cq = cb.createQuery(User.class);
    Root<User> root = cq.from(User.class);

    List<Predicate> predicates = new ArrayList<>();

    if (firstName != null && !firstName.isBlank()) {
        predicates.add(cb.like(cb.lower(root.get("firstName")),
                               "%" + firstName.toLowerCase() + "%"));
    }
    if (email != null && !email.isBlank()) {
        predicates.add(cb.equal(root.get("email"), email));
    }
    if (active != null) {
        predicates.add(cb.equal(root.get("active"), active));
    }

    cq.select(root)
      .where(cb.and(predicates.toArray(new Predicate[0])))
      .orderBy(cb.asc(root.get("lastName")));

    return em.createQuery(cq).getResultList();
}

5. COMMON CRITERIABUILDER PREDICATES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

// Equality
cb.equal(root.get("status"), "ACTIVE")
cb.notEqual(root.get("status"), "DELETED")

// Comparisons
cb.greaterThan(root.get("age"), 18)
cb.lessThanOrEqualTo(root.get("total"), new BigDecimal("1000"))
cb.between(root.get("createdAt"), start, end)

// Null checks
cb.isNull(root.get("deletedAt"))
cb.isNotNull(root.get("email"))

// Like / pattern
cb.like(root.get("email"), "%@gmail.com")
cb.like(cb.lower(root.get("firstName")), "al%")

// In
cb.in(root.get("status")).value("ACTIVE").value("PENDING")
root.get("id").in(idList)

// Logical
cb.and(predicate1, predicate2)
cb.or(predicate1, predicate2)
cb.not(predicate)

// Collection operations
cb.isEmpty(root.get("orders"))
cb.isNotEmpty(root.get("orders"))
cb.isMember(product, root.get("wishlist"))

// Aggregates
cb.count(root)
cb.sum(root.get("salary"))
cb.avg(root.get("score"))
cb.max(root.get("total"))
cb.min(root.get("total"))

6. JOINS WITH CRITERIA API
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

// Find users who have orders with total > 500
CriteriaBuilder cb = em.getCriteriaBuilder();
CriteriaQuery<User> cq = cb.createQuery(User.class);
Root<User> userRoot = cq.from(User.class);

// JOIN users → orders
Join<User, Order> orderJoin = userRoot.join("orders", JoinType.INNER);

cq.select(userRoot).distinct(true)
  .where(cb.greaterThan(orderJoin.get("total"), new BigDecimal("500")));

// FETCH JOIN (eager load)
Fetch<User, Order> orderFetch = userRoot.fetch("orders", JoinType.LEFT);

7. TUPLE QUERY (SELECT SPECIFIC FIELDS)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

CriteriaBuilder cb = em.getCriteriaBuilder();
CriteriaQuery<Tuple> cq = cb.createTupleQuery();
Root<User> root = cq.from(User.class);

cq.multiselect(
    root.get("id").alias("id"),
    root.get("email").alias("email"),
    root.get("firstName").alias("firstName")
);

List<Tuple> results = em.createQuery(cq).getResultList();
for (Tuple t : results) {
    Long id = t.get("id", Long.class);
    String email = t.get("email", String.class);
}

8. TYPE-SAFE METAMODEL (JPA Static Metamodel)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Generate with: hibernate-jpamodelgen (annotation processor)

// Auto-generated User_ metamodel class:
@StaticMetamodel(User.class)
public class User_ {
    public static volatile SingularAttribute<User, Long> id;
    public static volatile SingularAttribute<User, String> email;
    public static volatile SingularAttribute<User, String> firstName;
    public static volatile ListAttribute<User, Order> orders;
}

// Type-safe query using metamodel:
cq.where(cb.equal(root.get(User_.email), "alice@example.com"))
  .orderBy(cb.asc(root.get(User_.firstName)));

9. SPRING DATA SPECIFICATION PATTERN
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Spring Specification wraps Criteria API in a reusable predicate interface:

// Specification = (Root<T>, CriteriaQuery<?>, CriteriaBuilder) → Predicate
public class UserSpecifications {

    public static Specification<User> hasEmail(String email) {
        return (root, query, cb) -> cb.equal(root.get("email"), email);
    }

    public static Specification<User> isActive() {
        return (root, query, cb) -> cb.equal(root.get("active"), true);
    }

    public static Specification<User> createdAfter(LocalDate date) {
        return (root, query, cb) ->
            cb.greaterThan(root.get("createdAt"), date.atStartOfDay());
    }
}

// Repository:
public interface UserRepository extends JpaRepository<User, Long>,
                                         JpaSpecificationExecutor<User> {}

// Usage — compose specs:
Specification<User> spec = UserSpecifications.isActive()
    .and(UserSpecifications.createdAfter(LocalDate.of(2024, 1, 1)));

List<User> users = userRepository.findAll(spec);
Page<User> page  = userRepository.findAll(spec, PageRequest.of(0, 20));

Q: What is the Criteria API and when would you use it?
A: The Criteria API is a programmatic, type-safe way to build JPA queries
   at runtime. Use it when you need DYNAMIC queries (some filters optional),
   when compile-time type safety matters, or when building a reusable
   filter/search framework. JPQL is preferred for static queries.

Q: What is a JPA Specification?
A: A Specification is a Spring abstraction over the Criteria API.
   It encapsulates a single predicate/condition as a reusable object.
   Specifications can be composed with .and(), .or(), .not().
   The repository must extend JpaSpecificationExecutor<T>.

Q: What is the JPA Static Metamodel?
A: Annotation-processor-generated classes (User_, Order_) with static
   SingularAttribute/ListAttribute fields for each entity field. They
   allow refactor-safe, compile-time-checked Criteria API field references
   instead of "magic strings" like root.get("email").

Q: Criteria API vs JPQL: when to choose each?
A: JPQL: static queries, readable, concise, prefer for most cases.
   Criteria API: dynamic queries (search filters), type safety requirement,
   reusable predicate composition (Specifications pattern).
