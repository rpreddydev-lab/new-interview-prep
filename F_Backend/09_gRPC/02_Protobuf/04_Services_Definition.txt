PROTOBUF SERVICES DEFINITION
================================================================

----------------------------------------------------------------
1. SERVICE DEFINITION SYNTAX
----------------------------------------------------------------

Services define the API contract — the methods that a gRPC server
exposes for clients to call.

Basic syntax:
  service ServiceName {
    rpc MethodName (RequestType) returns (ResponseType);
  }

The four communication patterns:
  // 1. Unary - one request, one response
  rpc GetOrder (GetOrderRequest) returns (Order);
  
  // 2. Server streaming - one request, stream of responses
  rpc ListOrders (ListOrdersRequest) returns (stream Order);
  
  // 3. Client streaming - stream of requests, one response
  rpc BatchCreateOrders (stream CreateOrderRequest) returns (BatchResult);
  
  // 4. Bidirectional streaming - stream both ways
  rpc ProcessOrderStream (stream OrderEvent) returns (stream OrderResult);

----------------------------------------------------------------
2. COMPLETE EXAMPLE: E-COMMERCE API
----------------------------------------------------------------

syntax = "proto3";
package com.example.ecommerce.v1;

option java_package = "com.example.ecommerce.v1";
option java_multiple_files = true;
option java_outer_classname = "EcommerceProto";

import "google/protobuf/empty.proto";
import "google/protobuf/field_mask.proto";
import "google/protobuf/timestamp.proto";

// ── ENUMS ────────────────────────────────────────────────────

enum OrderStatus {
  ORDER_STATUS_UNSPECIFIED = 0;
  ORDER_STATUS_PENDING     = 1;
  ORDER_STATUS_CONFIRMED   = 2;
  ORDER_STATUS_PROCESSING  = 3;
  ORDER_STATUS_SHIPPED     = 4;
  ORDER_STATUS_DELIVERED   = 5;
  ORDER_STATUS_CANCELLED   = 6;
}

// ── COMMON MESSAGES ──────────────────────────────────────────

message Money {
  int64  amount_cents   = 1;
  string currency_code  = 2;
}

message PageRequest {
  int32  page_size  = 1;
  string page_token = 2;
}

message PageResponse {
  string next_page_token = 1;
  int32  total_count     = 2;
}

// ── DOMAIN MESSAGES ──────────────────────────────────────────

message OrderItem {
  string product_id  = 1;
  int32  quantity    = 2;
  Money  unit_price  = 3;
}

message Order {
  string      order_id    = 1;
  string      customer_id = 2;
  OrderStatus status      = 3;
  repeated OrderItem items  = 4;
  Money       total        = 5;
  google.protobuf.Timestamp created_at = 6;
  google.protobuf.Timestamp updated_at = 7;
}

// ── REQUEST/RESPONSE MESSAGES ────────────────────────────────

message CreateOrderRequest {
  string         customer_id = 1;
  repeated OrderItem items   = 2;
}

message GetOrderRequest {
  string order_id = 1;
}

message UpdateOrderRequest {
  Order                    order      = 1;
  google.protobuf.FieldMask field_mask = 2;
}

message CancelOrderRequest {
  string order_id = 1;
  string reason   = 2;
}

message ListOrdersRequest {
  string      customer_id = 1;
  OrderStatus status      = 2;  // filter by status (0 = all)
  PageRequest page        = 3;
}

message ListOrdersResponse {
  repeated Order orders = 1;
  PageResponse   page   = 2;
}

message BatchCreateOrdersRequest {
  // For client-streaming; each stream message is one of these
  string customer_id = 1;
  OrderItem item     = 2;
}

message BatchCreateResult {
  int32  total_received = 1;
  int32  success_count  = 2;
  int32  failure_count  = 3;
  repeated string failed_item_ids = 4;
}

message OrderEvent {
  string order_id   = 1;
  string event_type = 2;  // "STATUS_CHANGE", "ITEM_ADDED", etc.
  string payload    = 3;
}

message OrderResult {
  string order_id = 1;
  bool   success  = 2;
  string message  = 3;
}

// ── SERVICE DEFINITIONS ──────────────────────────────────────

service OrderService {
  // Unary: Create a new order
  rpc CreateOrder (CreateOrderRequest) returns (Order);
  
  // Unary: Get order by ID
  rpc GetOrder (GetOrderRequest) returns (Order);
  
  // Unary: Partial update (using FieldMask)
  rpc UpdateOrder (UpdateOrderRequest) returns (Order);
  
  // Unary: Cancel order
  rpc CancelOrder (CancelOrderRequest) returns (Order);
  
  // Unary with pagination: List orders
  rpc ListOrders (ListOrdersRequest) returns (ListOrdersResponse);
  
  // Server streaming: Watch order status changes in real time
  rpc WatchOrder (GetOrderRequest) returns (stream Order);
  
  // Server streaming: Stream all orders for a customer
  rpc StreamCustomerOrders (GetOrderRequest) returns (stream Order);
  
  // Client streaming: Batch create orders
  rpc BatchCreateOrders (stream BatchCreateOrdersRequest) returns (BatchCreateResult);
  
  // Bidirectional: Real-time order processing
  rpc ProcessOrderEvents (stream OrderEvent) returns (stream OrderResult);
  
  // Returns empty for void operations
  rpc DeleteOrder (GetOrderRequest) returns (google.protobuf.Empty);
}

3. REQUEST AND RESPONSE DESIGN PATTERNS
----------------------------------------------------------------

PATTERN 1: Dedicated Request/Response Messages (Recommended)
  
  Always create dedicated request and response messages per RPC.
  Never reuse a domain entity as a request/response directly.
  
  ✅ GOOD:
    service UserService {
      rpc GetUser(GetUserRequest) returns (GetUserResponse);
    }
    message GetUserRequest { string user_id = 1; }
    message GetUserResponse { User user = 1; }
  
  ❌ AVOID:
    service UserService {
      rpc GetUser(UserIdMessage) returns (User);  // reusing domain type
    }
  
  Why dedicated messages?
    - Can add fields to request/response without changing domain type
    - Clear separation of transport layer from domain model
    - Request can include metadata (page_size, filters, field_mask)
    - Response can include metadata (pagination tokens, totals)

PATTERN 2: Standard Method Names (Google API Design Guide)
  
  Follow Google's AIP (API Improvement Proposals) naming:
    List   → ListXxx (returns collection with pagination)
    Get    → GetXxx  (returns single resource by ID)
    Create → CreateXxx (creates new resource)
    Update → UpdateXxx (full or partial update)
    Delete → DeleteXxx (deletes resource)
    
  Custom methods use verb + noun:
    SearchOrders, BatchCreateOrders, CancelOrder, ValidateAddress

PATTERN 3: Pagination
  
  message ListRequest {
    int32  page_size  = 1;   // max items per page (client sets)
    string page_token = 2;   // token from previous response (empty for first page)
  }
  
  message ListResponse {
    repeated Order orders         = 1;
    string         next_page_token = 2;  // empty string means last page
    int32          total_count     = 3;  // optional: total matching items
  }
  
  Client usage:
    String pageToken = "";
    do {
      ListOrdersResponse response = stub.listOrders(
          ListOrdersRequest.newBuilder()
              .setPageSize(50)
              .setPageToken(pageToken)
              .build());
      process(response.getOrdersList());
      pageToken = response.getNextPageToken();
    } while (!pageToken.isEmpty());

PATTERN 4: Partial Updates with FieldMask
  
  Instead of full replacement (PUT), use FieldMask for partial updates (PATCH):
  
  // gRPC UpdateOrder with FieldMask
  UpdateOrderRequest request = UpdateOrderRequest.newBuilder()
      .setOrder(Order.newBuilder()
          .setOrderId("ORD-001")
          .setStatus(OrderStatus.ORDER_STATUS_SHIPPED)
          .build())
      .setFieldMask(FieldMask.newBuilder()
          .addPaths("status")   // ONLY update status field
          .build())
      .build();
  
  Server-side handling:
  @Override
  public void updateOrder(UpdateOrderRequest req, StreamObserver<Order> resp) {
    Order existingOrder = repository.findById(req.getOrder().getOrderId());
    // Apply only the fields listed in field_mask
    FieldMaskUtil.merge(req.getFieldMask(), req.getOrder(), existingOrder.toBuilder());
    Order updated = repository.save(existingOrder.toBuilder().build());
    resp.onNext(updated);
    resp.onCompleted();
  }

PATTERN 5: Resource Names (Google API style)
  
  Use hierarchical resource names as string IDs:
    users/user123
    users/user123/orders/ord456
    customers/cust789/invoices/inv001
  
  This embeds hierarchy in the ID itself:
  message GetOrderRequest {
    string name = 1;  // "customers/cust789/orders/ord456"
  }

4. SERVICE VERSIONING
----------------------------------------------------------------

Option A: Package versioning (recommended)
  // order/v1/order_service.proto
  package com.example.order.v1;
  service OrderService { ... }
  
  // order/v2/order_service.proto (breaking changes)
  package com.example.order.v2;
  service OrderService { ... }  // same service name, different package

Option B: Service name versioning
  service OrderServiceV1 { ... }
  service OrderServiceV2 { ... }

Option C: Method name versioning (avoid)
  rpc CreateOrderV2(...) returns (...);  // messy, hard to manage

Best practices for versioning:
  - Additive changes (new fields, new methods) don't need new version
  - Breaking changes (remove field, change type) need new major version
  - Run multiple versions in parallel during migration
  - Deprecate old methods before removing them
  - Mark deprecated RPCs with comment and potentially a response header

Deprecation annotation (in .proto comments):
  service OrderService {
    // Deprecated: Use CreateOrderV2 instead. Will be removed 2025-01.
    rpc CreateOrder (CreateOrderRequest) returns (Order) {
      option deprecated = true;
    }
    
    rpc CreateOrderV2 (CreateOrderV2Request) returns (Order);
  }

5. SERVICE OPTIONS
----------------------------------------------------------------

Service-level options:
  service OrderService {
    option (google.api.default_host) = "orders.example.com";
    // ... methods
  }

Method-level options (common in gRPC-Gateway transcoding):
  import "google/api/annotations.proto";
  
  service OrderService {
    rpc GetOrder (GetOrderRequest) returns (Order) {
      option (google.api.http) = {
        get: "/v1/orders/{order_id}"
      };
    }
    
    rpc CreateOrder (CreateOrderRequest) returns (Order) {
      option (google.api.http) = {
        post: "/v1/orders"
        body: "*"
      };
    }
    
    rpc UpdateOrder (UpdateOrderRequest) returns (Order) {
      option (google.api.http) = {
        patch: "/v1/orders/{order.order_id}"
        body: "order"
      };
    }
  }

These HTTP annotations are used by gRPC-Gateway to generate a
REST/JSON proxy in front of your gRPC service automatically.

6. GOOGLE AIP PATTERNS FOR SERVICE DESIGN
----------------------------------------------------------------

Standard lifecycle methods:
  rpc CreateOrder  (CreateOrderRequest)  returns (Order);  // POST
  rpc GetOrder     (GetOrderRequest)     returns (Order);  // GET
  rpc ListOrders   (ListOrdersRequest)   returns (ListOrdersResponse);  // GET with filter
  rpc UpdateOrder  (UpdateOrderRequest)  returns (Order);  // PATCH
  rpc DeleteOrder  (DeleteOrderRequest)  returns (google.protobuf.Empty);  // DELETE

Long-running operations pattern:
  import "google/longrunning/operations.proto";
  
  service ExportService {
    // Returns an Operation immediately; client polls for completion
    rpc ExportOrders(ExportRequest) returns (google.longrunning.Operation) {
      option (google.longrunning.operation_info) = {
        response_type: "ExportResponse"
        metadata_type: "ExportMetadata"
      };
    }
  }
  
  // Client polls:
  Operation op = stub.exportOrders(request);
  while (!op.getDone()) {
    Thread.sleep(1000);
    op = operationsStub.getOperation(GetOperationRequest.newBuilder()
        .setName(op.getName()).build());
  }
  ExportResponse result = op.getResponse().unpack(ExportResponse.class);

7. BIDIRECTIONAL STREAMING: PROTOCOL DESIGN CONSIDERATIONS
----------------------------------------------------------------

When designing a bidi streaming API, think about:

1. Who closes first?
  - Typically client closes first (sends all data), then server closes
  - Or either side can close on error

2. Request/Response matching:
  - Correlation IDs allow matching async responses to requests
  message TradeRequest {
    string correlation_id = 1;  // client-generated UUID
    // ... order details
  }
  message TradeResponse {
    string correlation_id = 1;  // echoed back from request
    // ... fill details
  }

3. Heartbeat/Keepalive messages:
  message OrderEvent {
    oneof event {
      OrderCreated   created   = 1;
      OrderCancelled cancelled = 2;
      Heartbeat      heartbeat = 3;  // empty ping to keep stream alive
    }
  }

4. Error propagation:
  - Use onError() with appropriate status code to terminate stream
  - Or use a status field in message for non-fatal per-message errors
  message OrderResult {
    string order_id = 1;
    bool   success  = 2;
    string error    = 3;  // per-message error, doesn't terminate stream
  }

8. SERVICE DEFINITION: WHAT TO AVOID
----------------------------------------------------------------

❌ Don't use generic request/response types
  rpc DoStuff(Request) returns (Response);
  // "Request" and "Response" are meaningless names
  
  ✅ Use specific names:
  rpc CreateOrder(CreateOrderRequest) returns (Order);

❌ Don't put domain logic in proto definitions
  // DON'T add validation ranges in comments and hope people follow them
  // Use protoc-gen-validate or application-level validation

❌ Don't model exceptions as error response fields
  message OrderResponse {
    Order order      = 1;
    string error_msg = 2;  // WRONG: use gRPC status codes for errors
  }
  
  ✅ Use gRPC status codes:
  if (orderNotFound) {
    responseObserver.onError(
        Status.NOT_FOUND.withDescription("Order not found").asException());
  }

❌ Don't design for single-language usage
  // Using Java enum names in descriptions assumes Java clients
  // Use cross-language patterns (integer enum values, standard names)

----------------------------------------------------------------
9. INTERVIEW KEY POINTS
----------------------------------------------------------------

Q: What is the difference between server streaming and bidirectional streaming?
A: In server streaming, the client sends ONE request and receives a
   STREAM of responses. The client's sending is done after the first
   message. In bidirectional streaming, BOTH the client AND server can
   send multiple messages at any time, independently. Both use a single
   HTTP/2 stream.

Q: When would you use client streaming vs a single request with repeated fields?
A: Client streaming is better when:
   1. The data set is too large for a single message (> 4MB default limit)
   2. You want to start processing on the server before receiving all data
   3. The client generates data over time (e.g., real-time sensor readings)
   A single request with repeated fields is simpler and preferred for
   small, known-size batches.

Q: How do you handle pagination in gRPC?
A: Use cursor-based pagination with page_token. The client sends
   page_size and optionally a page_token from the previous response.
   The server returns the next_page_token with each page (empty string
   indicates the last page). This is safer than offset-based pagination
   for concurrent data changes.
