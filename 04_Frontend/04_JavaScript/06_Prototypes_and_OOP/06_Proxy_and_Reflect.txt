1. PROXY OVERVIEW
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Proxy wraps an object and intercepts fundamental operations:
  property get, set, delete, function call, new, in operator, etc.

  const proxy = new Proxy(target, handler);
  // target  — the original object to wrap
  // handler — object with TRAP methods that intercept operations

  // Without any traps, proxy is transparent:
  const p = new Proxy({ x: 1 }, {});
  p.x;  // 1 — passes through to target

3. PRACTICAL EXAMPLES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // 1. VALIDATION:
  function createValidatedUser(user) {
    return new Proxy(user, {
      set(target, key, value) {
        if (key === "age") {
          if (typeof value !== "number" || value < 0)
            throw new TypeError("Age must be a positive number");
        }
        if (key === "email") {
          if (!value.includes("@"))
            throw new TypeError("Invalid email");
        }
        target[key] = value;
        return true;  // MUST return true for set traps (strict mode)
      }
    });
  }
  const user = createValidatedUser({});
  user.age = 25;      // ok
  user.age = -1;      // TypeError
  user.email = "bad"; // TypeError

  // 2. DEFAULT VALUES:
  const withDefaults = new Proxy({}, {
    get(target, key) {
      return key in target ? target[key] : `DEFAULT(${key})`;
    }
  });
  withDefaults.foo;   // "DEFAULT(foo)"
  withDefaults.x = 5;
  withDefaults.x;     // 5

  // 3. LOGGING / TRACING:
  function createLogged(target) {
    return new Proxy(target, {
      get(t, key) {
        console.log(`GET ${key}`);
        return Reflect.get(t, key);
      },
      set(t, key, value) {
        console.log(`SET ${key} = ${value}`);
        return Reflect.set(t, key, value);
      }
    });
  }

  // 4. READONLY OBJECT:
  const frozen = new Proxy({ name: "Alice" }, {
    set() { throw new Error("Object is read-only"); },
    deleteProperty() { throw new Error("Object is read-only"); }
  });
  frozen.name;       // "Alice"
  frozen.name = "X"; // throws

4. REFLECT
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Reflect is a built-in object with static methods that mirror Proxy traps.
  Use it inside trap handlers to forward operations to the original target.

  // Each Proxy trap has a corresponding Reflect method:
  Reflect.get(target, key, receiver)      // get property
  Reflect.set(target, key, value, recv)   // set property (returns boolean)
  Reflect.has(target, key)                // "key" in target
  Reflect.deleteProperty(target, key)     // delete target[key]
  Reflect.apply(target, thisArg, args)    // call function
  Reflect.construct(target, args, newTarget)
  Reflect.ownKeys(target)                 // all own property keys
  Reflect.defineProperty(target, key, desc)
  Reflect.getPrototypeOf(target)
  Reflect.setPrototypeOf(target, proto)

  // WHY use Reflect in traps?
  // 1. Correct return values (set must return boolean)
  // 2. Handles edge cases correctly (prototype chain, receiver)
  // 3. Consistent with language semantics

  // Example — always prefer Reflect inside traps:
  const handler = {
    get(target, key, receiver) {
      console.log("Getting:", key);
      return Reflect.get(target, key, receiver);  // pass receiver for correct `this`
    }
  };

5. REVOCABLE PROXY
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  const { proxy, revoke } = Proxy.revocable({ x: 1 }, {});
  proxy.x;   // 1
  revoke();  // disable the proxy
  proxy.x;   // TypeError: Cannot perform 'get' on a proxy that has been revoked

  // Use case: API tokens — revoke access after a session ends.

6. PROXY LIMITATIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  - Private class fields (#field) bypass Proxy traps (internal slot access)
  - Some built-in objects (Map, Set internals) can't be fully proxied
  - Proxy adds overhead — not suitable for hot paths
  - Cannot be transpiled/polyfilled (needs native engine support)
  - identity checks: proxy !== target (can cause issues with === checks)

7. COMMON INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What is a Proxy in JavaScript?
A: An object that wraps another object and intercepts fundamental operations
   (get, set, delete, function call, etc.) via trap functions in a handler
   object. Used for validation, logging, default values, and more.

Q: What is the difference between Proxy and Object.defineProperty()?
A: defineProperty can intercept get/set for specific properties but requires
   knowing the properties in advance. Proxy intercepts ALL property accesses
   dynamically, even for properties that don't exist yet.

Q: Why use Reflect inside Proxy traps?
A: Reflect methods mirror trap operations, handle edge cases correctly (like
   passing the receiver for correct `this` binding), and return the expected
   types (e.g. boolean for set). It ensures default behavior is preserved
   while adding custom logic.

Q: What is a revocable proxy?
A: Created with Proxy.revocable(), it returns a proxy and a revoke function.
   After revoke() is called, all operations on the proxy throw TypeError.
   Useful for access control (revoke credentials after session ends).

Q: Can Proxy intercept private class fields?
A: No. Private fields use internal slots that bypass the Proxy mechanism.
   Accessing a private field on a proxy of a class instance throws a TypeError
   because the proxy is not the actual instance.
