  AbortController (Web API) lets you cancel async operations like:
  - fetch() requests
  - Event listeners
  - Any operation that accepts an AbortSignal

  const controller = new AbortController();
  const signal = controller.signal;  // AbortSignal

  controller.abort();       // cancel the operation
  signal.aborted;           // true if aborted
  signal.reason;            // the abort reason (ES2022)

  const controller = new AbortController();
  const { signal } = controller;

  fetch("https://api.example.com/large-data", { signal })
    .then(res => res.json())
    .then(data => console.log(data))
    .catch(err => {
      if (err.name === "AbortError") {
        console.log("Fetch was aborted");  // handle cancellation
      } else {
        console.error("Fetch error:", err);
      }
    });

  // Cancel after 3 seconds:
  setTimeout(() => controller.abort(), 3000);

  // Cancel on user action:
  cancelButton.addEventListener("click", () => controller.abort());

3. FETCH WITH TIMEOUT (COMMON PATTERN)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Manual timeout (before AbortSignal.timeout):
  async function fetchWithTimeout(url, ms = 5000) {
    const controller = new AbortController();
    const timeoutId = setTimeout(() => controller.abort(), ms);
    try {
      const res = await fetch(url, { signal: controller.signal });
      if (!res.ok) throw new Error(`HTTP ${res.status}`);
      return await res.json();
    } catch (err) {
      if (err.name === "AbortError") throw new Error(`Request timed out after ${ms}ms`);
      throw err;
    } finally {
      clearTimeout(timeoutId);  // always clear timer
    }
  }

  // Modern shorthand (ES2022+) — AbortSignal.timeout():
  const signal = AbortSignal.timeout(5000);  // auto-aborts after 5s
  try {
    const res = await fetch(url, { signal });
    return await res.json();
  } catch (err) {
    if (err.name === "TimeoutError") console.log("Request timed out");
  }

  // Prevents "Can't perform state update on unmounted component" warnings:
  useEffect(() => {
    const controller = new AbortController();

    async function loadData() {
      try {
        const res = await fetch("/api/data", { signal: controller.signal });
        if (!res.ok) throw new Error("Network error");
        const data = await res.json();
        setData(data);
      } catch (err) {
        if (err.name !== "AbortError") {
          setError(err.message);
        }
        // AbortError is expected — just ignore it
      }
    }

    loadData();

    return () => controller.abort();  // cleanup: abort on unmount
  }, []);

  // addEventListener accepts a signal option to auto-remove the listener:
  const controller = new AbortController();
  const { signal } = controller;

  document.addEventListener("keydown", handleKey, { signal });
  document.addEventListener("click", handleClick, { signal });
  window.addEventListener("resize", handleResize, { signal });

  // Remove ALL listeners at once:
  controller.abort();
  // No need to call removeEventListener for each — very convenient!

8. CHECKING ABORT BEFORE ASYNC WORK
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  async function doWork(signal) {
    // Check before starting expensive work:
    signal.throwIfAborted();  // throws if already aborted

    const step1 = await step1();
    signal.throwIfAborted();  // check between steps

    const step2 = await step2(step1);
    signal.throwIfAborted();

    return step2;
  }

Q: What is AbortController used for?
A: Cancelling asynchronous operations. Most commonly used to cancel fetch()
   requests, but also works for event listeners and any custom operation that
   accepts an AbortSignal.

Q: How do you abort a fetch request?
A: Create an AbortController, pass signal to fetch options, call abort():
   const ctrl = new AbortController();
   fetch(url, { signal: ctrl.signal }).catch(e => ...);
   ctrl.abort();

Q: How do you identify an AbortError vs other errors?
A: Check err.name === "AbortError". AbortError is a DOMException with that
   specific name. Never silence all catch errors — only the AbortError.

Q: Why is AbortController important in React?
A: In useEffect, if a component unmounts while a fetch is in flight, the
   response would try to call setState on an unmounted component. Aborting
   the fetch in the cleanup function (return () => controller.abort()) prevents
   this and avoids memory leaks.

Q: Can one AbortController cancel multiple requests?
A: Yes. The same signal can be passed to multiple fetch() calls. One
   controller.abort() cancels all of them simultaneously.

Q: What is AbortSignal.timeout()?
A: A static method (ES2022) that returns a signal pre-configured to abort
   automatically after a specified number of milliseconds. Eliminates the
   need to manually set up setTimeout + controller.abort().
