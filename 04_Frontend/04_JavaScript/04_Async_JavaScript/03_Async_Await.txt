1. WHAT IS ASYNC/AWAIT?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  async/await (ES2017) is syntactic sugar built ON TOP of Promises.
  It makes async code LOOK synchronous, improving readability.

  async function → always returns a Promise
  await          → pauses execution in the async function until the Promise settles

2. BASIC SYNTAX
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  async function fetchUser(id) {
    const res = await fetch(`/api/users/${id}`);   // pauses here
    const user = await res.json();                  // pauses here
    return user;                                    // wraps in Promise.resolve()
  }

  fetchUser(1).then(user => console.log(user));  // returns Promise

  // Arrow function:
  const fetchUser = async (id) => {
    const res = await fetch(`/api/users/${id}`);
    return res.json();
  };

  // async method in class:
  class UserService {
    async getUser(id) {
      return await fetch(`/api/users/${id}`).then(r => r.json());
    }
  }

  async function c() { return Promise.resolve(42); }

4. ERROR HANDLING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // try/catch (recommended):
  async function fetchData(url) {
    try {
      const res = await fetch(url);
      if (!res.ok) throw new Error(`HTTP Error: ${res.status}`);
      return await res.json();
    } catch (err) {
      console.error("Fetch failed:", err.message);
      throw err;  // re-throw to let caller handle
    }
  }

  // Handling at call site:
  try {
    const data = await fetchData("/api/data");
    render(data);
  } catch (err) {
    showError(err);
  }

  // .catch() alternative:
  const data = await fetchData("/api/data").catch(err => null);
  if (!data) return showFallback();

  // Helper: returns [error, result] tuple (Go-style):
  async function safeAwait(promise) {
    try {
      return [null, await promise];
    } catch (err) {
      return [err, null];
    }
  }
  const [err, user] = await safeAwait(fetchUser(1));
  if (err) return handleError(err);

5. SEQUENTIAL vs PARALLEL EXECUTION
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // SEQUENTIAL — each waits for the previous (slower):
  const user = await fetchUser(1);       // waits
  const posts = await fetchPosts(1);     // then waits
  // Total time: user_time + posts_time

  // PARALLEL — concurrent (faster):
  const [user, posts] = await Promise.all([
    fetchUser(1),
    fetchPosts(1)
  ]);
  // Total time: max(user_time, posts_time)

  // Parallel with individual error handling:
  const [userResult, postsResult] = await Promise.allSettled([
    fetchUser(1),
    fetchPosts(1)
  ]);

  // Start multiple, await later:
  const userPromise = fetchUser(1);    // starts immediately
  const postsPromise = fetchPosts(1);  // starts immediately
  const user = await userPromise;      // wait now
  const posts = await postsPromise;    // wait now (likely already done)

6. ASYNC IN LOOPS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  const ids = [1, 2, 3, 4, 5];

  // WRONG — forEach doesn't await:
  ids.forEach(async (id) => {
    const user = await fetchUser(id);  // forEach ignores returned promise!
    console.log(user);
  });

  // SEQUENTIAL with for...of (awaits each):
  for (const id of ids) {
    const user = await fetchUser(id);  // waits for each
    console.log(user);
  }
  // Total time: n * fetchTime

  // PARALLEL with Promise.all + map:
  const users = await Promise.all(ids.map(id => fetchUser(id)));
  // Total time: max(fetchTime) — much faster!

  // Rate-limited parallel (chunks):
  async function parallel(arr, limit, fn) {
    const results = [];
    for (let i = 0; i < arr.length; i += limit) {
      const chunk = arr.slice(i, i + limit);
      const chunkResults = await Promise.all(chunk.map(fn));
      results.push(...chunkResults);
    }
    return results;
  }

8. ASYNC/AWAIT vs PROMISES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Same operation — Promise chain vs async/await:

  // Promise chain:
  function getUser(id) {
    return fetch(`/api/users/${id}`)
      .then(res => {
        if (!res.ok) throw new Error(res.status);
        return res.json();
      })
      .then(user => user.name)
      .catch(err => { console.error(err); return "Unknown"; });
  }

  // async/await (equivalent, more readable):
  async function getUser(id) {
    try {
      const res = await fetch(`/api/users/${id}`);
      if (!res.ok) throw new Error(res.status);
      const user = await res.json();
      return user.name;
    } catch (err) {
      console.error(err);
      return "Unknown";
    }
  }

9. COMMON INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What does async/await do under the hood?
A: It's syntactic sugar for Promises. An async function returns a Promise.
   await pauses the function's execution context (as a microtask), lets
   the call stack empty, then resumes when the awaited Promise resolves.

Q: What happens if you don't await a Promise inside an async function?
A: The Promise runs but the result is ignored. Errors may be unhandled.
   The async function won't wait for it and continues immediately.

Q: Why does await in forEach not work as expected?
A: forEach is not async-aware. It fires all callbacks but doesn't await
   their promises. The async function completes before the async callbacks
   are done. Use for...of or Promise.all with .map() instead.

Q: How do you run multiple async operations in parallel?
A: const [a, b] = await Promise.all([op1(), op2()]);
   Both promises start concurrently and resolve when both are done.

Q: What is top-level await?
A: ES2022 feature allowing await at the top level of ES modules (without
   being inside an async function). The module waits for the promise before
   being available to importers.

Q: What is the difference between await and .then()?
A: Functionally equivalent. await makes async code look synchronous and
   works naturally with try/catch. .then() is more explicit and better
   for scenarios like parallel execution without async/await.

Q: What does this return?
   async function fn() { return 42; }
A: A Promise that resolves with 42. All async functions return Promises.
   The returned value is wrapped in Promise.resolve() automatically.
