  fetch() is the modern replacement for XMLHttpRequest (XHR).
  It returns a Promise that resolves to a Response object.
  Available in browsers and Node.js 18+.

  fetch(url, options?)  →  Promise<Response>

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  const response = await fetch(url);

  // Properties:
  response.ok          // true if status 200-299
  response.status      // HTTP status code (200, 404, 500 ...)
  response.statusText  // "OK", "Not Found" ...
  response.headers     // Headers object
  response.url         // final URL (after redirects)
  response.redirected  // true if redirected
  response.bodyUsed    // true if body already consumed

  // Body reading methods (each returns a Promise, can only be called ONCE):
  response.json()      // parse body as JSON
  response.text()      // body as string
  response.blob()      // body as Blob (images, files)
  response.arrayBuffer()  // body as ArrayBuffer (binary data)
  response.formData()  // body as FormData

  // Clone to read body multiple times:
  const clone = response.clone();
  const json = await response.json();
  const text = await clone.text();  // still works on clone

4. CRITICAL: HTTP ERRORS vs NETWORK ERRORS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  fetch() only REJECTS for network failures (no connection, DNS failure).
  HTTP errors (404, 500) STILL RESOLVE — the promise fulfills!

  // WRONG — doesn't catch 404 or 500:
  fetch("/api/users/999")
    .then(res => res.json())  // runs even on 404
    .catch(err => console.error(err));  // only for network errors

  // CORRECT — check res.ok:
  async function fetchWithErrorHandling(url) {
    const response = await fetch(url);

    if (!response.ok) {
      throw new Error(`HTTP Error: ${response.status} ${response.statusText}`);
    }

    return response.json();
  }

  // Or a reusable wrapper:
  async function apiFetch(url, options = {}) {
    const res = await fetch(url, options);
    if (!res.ok) {
      const error = new Error(`API Error: ${res.status}`);
      error.status = res.status;
      try {
        error.body = await res.json();
      } catch (_) {}
      throw error;
    }
    return res.json();
  }

5. REQUEST OPTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  fetch(url, {
    method: "POST",             // GET, POST, PUT, DELETE, PATCH ...
    headers: {
      "Content-Type": "application/json",
      "Authorization": "Bearer token123"
    },
    body: JSON.stringify({ name: "Alice" }),  // string, FormData, Blob, ...
    mode: "cors",               // cors, no-cors, same-origin
    credentials: "include",     // omit, same-origin, include (send cookies)
    cache: "no-cache",          // default, no-store, no-cache, reload ...
    redirect: "follow",         // follow, error, manual
    signal: abortController.signal  // for cancellation
  });

  // POST with JSON:
  async function createUser(userData) {
    const response = await fetch("/api/users", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(userData)
    });
    if (!response.ok) throw new Error("Failed to create user");
    return response.json();
  }

  // POST with FormData (multipart/form-data — for file uploads):
  async function uploadFile(file) {
    const formData = new FormData();
    formData.append("file", file);
    formData.append("name", file.name);
    // Do NOT set Content-Type — browser sets it with boundary automatically
    const response = await fetch("/api/upload", {
      method: "POST",
      body: formData
    });
    return response.json();
  }

8. RETRY PATTERN
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  async function fetchWithRetry(url, options = {}, retries = 3, delay = 500) {
    for (let i = 0; i < retries; i++) {
      try {
        const res = await fetch(url, options);
        if (!res.ok) throw new Error(`HTTP ${res.status}`);
        return await res.json();
      } catch (err) {
        if (i === retries - 1) throw err;          // last attempt: rethrow
        await new Promise(r => setTimeout(r, delay * (i + 1)));  // backoff
      }
    }
  }

9. COMMON INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: Does fetch() reject on HTTP errors like 404 or 500?
A: NO. fetch() only rejects on network failures (no connection). HTTP error
   responses (4xx, 5xx) are treated as successful fetches. You must check
   response.ok or response.status manually and throw if needed.

Q: What is the difference between fetch() and XMLHttpRequest?
A: fetch() returns a Promise (cleaner async code), has a cleaner API,
   supports streams, and integrates with Service Workers. XHR is older,
   callback-based, but has upload progress events that fetch() lacks
   (though ReadableStream can simulate that).

Q: How do you send JSON data with fetch?
A: Set method: "POST", headers: {"Content-Type": "application/json"},
   and body: JSON.stringify(data). Also check response.ok after the call.

Q: How do you handle timeouts with fetch?
A: Use AbortController with setTimeout:
   const controller = new AbortController();
   const timeout = setTimeout(() => controller.abort(), 5000);
   fetch(url, { signal: controller.signal }).finally(() => clearTimeout(timeout));

Q: What does response.clone() do?
A: Creates a copy of the Response object, allowing the body to be read
   more than once (the original body is a stream and can only be consumed once).

Q: How does credentials differ in fetch?
A: "omit" (default) never sends cookies. "same-origin" sends cookies for
   same-origin only. "include" always sends cookies including cross-origin
   (requires server to set Access-Control-Allow-Credentials: true).
