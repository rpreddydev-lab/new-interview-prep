1. OVERVIEW
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  JavaScript has two major module systems:
  - CommonJS (CJS)  — Node.js original system (require/module.exports)
  - ES Modules (ESM) — Official standard (import/export), ES2015+

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
2. CommonJS (CJS)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Exporting:
  // math.js
  function add(a, b) { return a + b; }
  const PI = 3.14;

  module.exports = { add, PI };       // export object
  module.exports.add = add;           // add single property
  exports.PI = PI;                    // shortcut (exports === module.exports initially)
  // WARNING: exports = {...} breaks the reference!

  // Single default:
  module.exports = function add(a, b) { return a + b; };

  // Importing:
  const { add, PI } = require("./math");
  const add = require("./math");           // if single export
  const math = require("./math");          // whole object
  const path = require("path");            // built-in Node module

  // Dynamic require (runtime):
  const name = "lodash";
  const _ = require(name);  // evaluated at runtime

3. ES MODULES (ESM)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Exporting (math.mjs or math.js with "type":"module" in package.json):
  export function add(a, b) { return a + b; }
  export const PI = 3.14;
  export default class Calculator { ... }

  // Importing:
  import { add, PI } from "./math.js";         // named
  import Calculator from "./math.js";          // default
  import { add as sum } from "./math.js";      // rename
  import * as math from "./math.js";           // namespace
  import Calculator, { add, PI } from "./math.js"; // mixed

  // Dynamic import (returns Promise):
  const { add } = await import("./math.js");

  // Re-exporting (barrel files):
  export { add } from "./math.js";
  export * from "./helpers.js";
  export { default } from "./calculator.js";

  ┌─────────────────────────┬───────────────────────┬───────────────────────┐
  │ Feature                 │ CommonJS (CJS)        │ ES Modules (ESM)      │
  ├─────────────────────────┼───────────────────────┼───────────────────────┤
  │ Syntax                  │ require / exports     │ import / export       │
  │ Loading                 │ SYNCHRONOUS           │ ASYNCHRONOUS          │
  │ Resolution              │ Runtime               │ Static (parse time)   │
  │ Exports type            │ Copied values         │ LIVE BINDINGS         │
  │ Tree shaking            │ Not possible          │ Supported             │
  │ Top-level await         │ Not supported         │ Supported             │
  │ this at top level       │ module.exports        │ undefined             │
  │ Named exports           │ via object props      │ native export syntax  │
  │ Dynamic import          │ require() (sync)      │ import() (async)      │
  │ Circular dependencies   │ Gets current value    │ Gets live binding     │
  │ File extension          │ .js (default)         │ .mjs or "type":"module"│
  └─────────────────────────┴───────────────────────┴───────────────────────┘

5. LIVE BINDINGS — KEY ESM DIFFERENCE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // counter.mjs:
  export let count = 0;
  export function increment() { count++; }

  // main.mjs:
  import { count, increment } from "./counter.mjs";
  console.log(count);  // 0
  increment();
  console.log(count);  // 1 — live binding reflects the change!

  // With CJS:
  // const { count } = require("./counter"); // count is a copy
  // Modifying in counter won't update the local copy.

6. NODE.js SETUP
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Enable ESM in Node.js:
  // Option 1: Use .mjs extension
  // Option 2: Add to package.json:
  { "type": "module" }
  // Then all .js files are treated as ESM; use .cjs for CommonJS.

  // __dirname / __filename don't exist in ESM:
  import { fileURLToPath } from "url";
  import { dirname } from "path";
  const __filename = fileURLToPath(import.meta.url);
  const __dirname = dirname(__filename);

  // import.meta:
  import.meta.url;      // file URL: "file:///path/to/module.mjs"
  import.meta.resolve;  // resolve path relative to module

  // ESM can import CJS (gets module.exports as default):
  import lodash from "lodash";  // CJS package — lodash._... is the default

  // CJS cannot use static import — must use dynamic import:
  // In CJS file:
  const { add } = await import("./math.mjs");  // async
  // Or inside an async IIFE.

8. COMMON INTERVIEW QUESTIONS

Q: What is the main difference between CommonJS and ES Modules?
A: CJS uses require/module.exports and loads synchronously at runtime.
   ESM uses import/export with static analysis at parse time and loads
   asynchronously. ESM supports live bindings and tree-shaking; CJS copies values.

Q: What are live bindings in ESM?
A: Named imports in ESM are live references to the exported variables. If the
   exporting module changes the value, the importing module sees the new value.
   CommonJS exports copies — changes in the source don't propagate.

Q: Why can't you conditionally use import statements?
A: Static import must be at the top level — it's evaluated before the code
   runs. For conditional/dynamic loading use import() which returns a Promise.

Q: How do you use __dirname in an ES module?
A: __dirname doesn't exist in ESM. Use fileURLToPath(import.meta.url) to get
   the file path, then dirname() from the path module.

Q: What is tree shaking and why does ESM enable it?
A: Tree shaking is dead code elimination during bundling. ESM's static import/
   export structure lets bundlers (Webpack, Rollup, Vite) statically analyze
   which exports are used and exclude unused ones. CJS's dynamic require()
   makes this impossible.
