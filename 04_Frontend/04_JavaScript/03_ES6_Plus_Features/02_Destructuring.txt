1. ARRAY DESTRUCTURING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  const [a, b, c] = [1, 2, 3];
  a; // 1, b; // 2, c; // 3

  // Skip elements:
  const [first, , third] = [1, 2, 3];  // skip index 1
  first; // 1, third; // 3

  // Default values:
  const [x = 10, y = 20] = [5];
  x; // 5 (provided), y; // 20 (default)

  // Rest in array destructuring:
  const [head, ...tail] = [1, 2, 3, 4];
  head; // 1, tail; // [2, 3, 4]

  // Swap variables:
  let p = 1, q = 2;
  [p, q] = [q, p];
  p; // 2, q; // 1

  // Nested array destructuring:
  const [[x1, x2], [y1, y2]] = [[1, 2], [3, 4]];
  x1; // 1, y2; // 4

  // Ignoring return values:
  function getCoords() { return [10, 20, 30]; }
  const [lat, lon] = getCoords();  // ignore z

  // From iterables:
  const [c1, c2, c3] = "abc";  // c1="a", c2="b", c3="c"
  const [v1, v2] = new Set([1, 2, 3]);  // v1=1, v2=2

2. OBJECT DESTRUCTURING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  const user = { name: "Alice", age: 30, role: "admin" };

  // Basic:
  const { name, age } = user;
  name; // "Alice", age; // 30

  // Rename (alias):
  const { name: userName, age: userAge } = user;
  userName; // "Alice", userAge; // 30

  // Default values:
  const { name: n = "Guest", active = false } = user;
  n;      // "Alice" (provided)
  active; // false (default, not in user)

  // Default + rename:
  const { role: userRole = "viewer" } = user;
  userRole; // "admin" (provided)

  // Rest in object destructuring:
  const { name: nm, ...rest } = user;
  nm;   // "Alice"
  rest; // { age: 30, role: "admin" }

  // Nested object destructuring:
  const profile = {
    user: {
      name: "Bob",
      address: { city: "NYC", country: "USA" }
    }
  };
  const { user: { name: pName, address: { city } } } = profile;
  pName; // "Bob", city; // "NYC"

  // Computed property names in destructuring:
  const key = "name";
  const { [key]: value } = user;  // value = "Alice"

3. FUNCTION PARAMETER DESTRUCTURING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Object parameter destructuring:
  function greet({ name, greeting = "Hello" }) {
    return `${greeting}, ${name}!`;
  }
  greet({ name: "Alice" });            // "Hello, Alice!"
  greet({ name: "Bob", greeting: "Hi" }); // "Hi, Bob!"

  // Array parameter destructuring:
  function first([head]) { return head; }
  first([1, 2, 3]);  // 1

  // Destructure with rest in params:
  function sum([first, ...rest]) {
    return rest.reduce((acc, n) => acc + n, first);
  }
  sum([1, 2, 3, 4]);  // 10

  // Nested param destructuring:
  function displayUser({ name, address: { city } }) {
    return `${name} lives in ${city}`;
  }
  displayUser({ name: "Alice", address: { city: "LA" } });

  second;  // "Bob"

  const [{ x: x1 }, { x: x2 }] = [{ x: 1 }, { x: 2 }];

5. DESTRUCTURING IN LOOPS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  const users = [
    { name: "Alice", age: 30 },
    { name: "Bob", age: 25 }
  ];

  for (const { name, age } of users) {
    console.log(`${name}: ${age}`);
  }

  // Object.entries with destructuring:
  const scores = { math: 90, english: 85 };
  for (const [subject, score] of Object.entries(scores)) {
    console.log(`${subject}: ${score}`);
  }

6. DESTRUCTURING GOTCHAS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  A) Destructuring null/undefined throws:
  const { x } = null;       // TypeError: Cannot destructure null
  const { x } = undefined;  // TypeError: Cannot destructure undefined

  // Guard:
  const { x } = obj || {};  // safe default

  B) Default values only trigger for undefined (not null):
  const { a = 10 } = { a: null };
  a;  // null (default NOT used — null is not undefined!)

  const { b = 10 } = { b: undefined };
  b;  // 10 (default IS used)

  C) Destructuring in assignment (without declaration) needs parens:
  let name, age;
  ({ name, age } = { name: "Alice", age: 30 });  // parens required!
  // Without parens: { name, age } = ... is a block statement → SyntaxError

7. COMMON INTERVIEW QUESTIONS

Q: What is destructuring?
A: Destructuring is an ES6 syntax to unpack values from arrays or properties
   from objects into separate variables.

Q: What's the difference between array and object destructuring?
A: Array destructuring uses position (order matters, uses []).
   Object destructuring uses property names (order doesn't matter, uses {}).

Q: How do you provide default values in destructuring?
A: const { x = 10 } = {}  → x = 10 (triggered only when undefined).
   const [a = 5] = []      → a = 5 (triggered only when undefined).
   null does NOT trigger defaults.

Q: How do you rename a destructured property?
A: const { originalName: newName } = obj;
   "originalName" is the key to look up, "newName" is the local variable.

Q: What happens when you destructure undefined?
A: TypeError is thrown. Always guard: const { x } = val || {};

Q: How do you swap two variables using destructuring?
A: [a, b] = [b, a]; — elegant one-liner, no temp variable needed.
