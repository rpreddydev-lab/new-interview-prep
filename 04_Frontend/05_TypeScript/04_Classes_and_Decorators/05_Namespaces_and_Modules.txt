1. MODULES vs NAMESPACES — OVERVIEW
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  TypeScript has TWO code organization mechanisms:

  MODULES (ES Modules / CommonJS):
  - File-based: each file is its own module
  - Code is isolated unless explicitly exported/imported
  - Recommended for ALL modern TypeScript projects
  - Compiled to ESM, CommonJS, AMD, UMD, etc.

  NAMESPACES (internal modules, formerly "modules" pre-TS 1.5):
  - TypeScript-specific concept for grouping related types/values
  - Used mainly for global scripts and .d.ts declaration files
  - Compiled to nested JavaScript object literals
  - Not recommended for new projects using module bundlers

2. ES MODULES IN TYPESCRIPT
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // mathUtils.ts — export:
  export function add(a: number, b: number): number { return a + b; }
  export const PI = 3.14159;
  export type Point = { x: number; y: number };
  export interface Shape { area(): number; }

  export default class Calculator {
    add(a: number, b: number) { return a + b; }
  }

  // main.ts — import:
  import Calculator from "./mathUtils";           // default import
  import { add, PI, Point } from "./mathUtils";   // named imports
  import * as MathUtils from "./mathUtils";       // namespace import
  import type { Shape } from "./mathUtils";       // type-only import (TS 3.8+)

  // Re-export:
  export { add, PI } from "./mathUtils";
  export type { Shape } from "./mathUtils";
  export * from "./mathUtils";                     // re-export everything
  export * as Math from "./mathUtils";             // re-export with alias

  // Conditional/async import:
  const utils = await import("./mathUtils");
  utils.add(1, 2);

3. TYPE-ONLY IMPORTS (TypeScript 3.8+)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  import type { User } from "./models";
  import { type User, type Post, createUser } from "./models"; // inline type

  Benefits:
  - Guaranteed to be erased at runtime — never included in JS output
  - Prevents accidentally importing runtime code when you only need types
  - Required when verbatimModuleSyntax: true (TS 5.0+) is enabled
  - Helps bundlers tree-shake type-only dependencies

4. MODULE RESOLUTION STRATEGIES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  moduleResolution options in tsconfig:

  "node" — Classic Node.js resolution (for CommonJS, older setup):
    1. Exact file match
    2. Add .ts, .tsx, .d.ts
    3. package.json "types" / "typings"
    4. index.ts / index.d.ts

  "bundler" — Modern bundler resolution (Vite, esbuild, webpack 5):
    Like "node16" but without requiring .js extensions on imports.
    Recommended for React/Vite projects. (TS 4.7+)

  "node16" / "nodenext" — Strict ESM resolution:
    Requires .js extensions on imports even for .ts files.
    Enforces ESM interop correctly.

  "classic" — Deprecated TypeScript-specific resolution.

  // Path aliases (baseUrl + paths in tsconfig):
  {
    "baseUrl": "src",
    "paths": {
      "@utils/*": ["utils/*"],
      "@components/*": ["components/*"]
    }
  }
  // Now you can: import { Button } from "@components/Button";

5. NAMESPACES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  namespace Validation {
    export interface StringValidator {
      isAcceptable(s: string): boolean;
    }

    const lettersRegExp = /^[A-Za-z]+$/;
    const numberRegExp  = /^[0-9]+$/;

    export class LettersOnlyValidator implements StringValidator {
      isAcceptable(s: string) { return lettersRegExp.test(s); }
    }

    export class ZipCodeValidator implements StringValidator {
      isAcceptable(s: string) {
        return s.length === 5 && numberRegExp.test(s);
      }
    }
  }

  const v1 = new Validation.LettersOnlyValidator();
  const v2 = new Validation.ZipCodeValidator();
  console.log(v1.isAcceptable("Hello")); // true
  console.log(v2.isAcceptable("12345")); // true

  // Compiled output — namespaces become nested JS objects:
  var Validation;
  (function (Validation) {
    Validation.LettersOnlyValidator = class { ... };
  })(Validation || (Validation = {}));

6. NAMESPACE SPLITTING AND MERGING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // File: shapes.ts
  namespace Shapes {
    export class Circle { constructor(public radius: number) {} }
  }

  // File: more-shapes.ts
  namespace Shapes {
    export class Rectangle { constructor(public w: number, public h: number) {} }
  }

  // Both files contribute to the same Shapes namespace.
  // Use triple-slash reference to tell TS about the split:
  /// <reference path="shapes.ts" />
  /// <reference path="more-shapes.ts" />
  const c = new Shapes.Circle(5);
  const r = new Shapes.Rectangle(3, 4);

  // Namespace aliasing:
  import Shapes = Validation; // alias
  import Circle = Shapes.Circle;  // shorter reference

7. AMBIENT MODULES AND GLOBAL AUGMENTATION
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Declare types for modules TypeScript doesn't know about:
  declare module "*.svg" {
    const content: string;
    export default content;
  }

  declare module "some-js-library" {
    export function doThing(x: number): string;
    export const version: string;
  }

  // Augment an existing module's types:
  declare module "express" {
    interface Request {
      user?: { id: string; email: string };
    }
  }

  // Global augmentation (add to global scope):
  declare global {
    interface Window {
      analytics: { track(event: string): void };
    }
    function myGlobalFn(): void;
  }
  export {}; // Make this file a module (required for declare global)

8. INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q1: What is the difference between modules and namespaces in TypeScript?
    Modules are file-based (ESM/CommonJS), each file is isolated, and they're
    the modern standard. Namespaces are TypeScript-specific groupings compiled
    to nested JS objects — useful for global scripts and .d.ts files but
    superseded by modules in all modern projects.

Q2: What is a type-only import and why is it useful?
    `import type { User } from "./models"` ensures the import is completely
    erased at runtime (no JS import statement emitted). It prevents bundling
    runtime code when only a type is needed and is required by some strict
    module configs like verbatimModuleSyntax.

Q3: What does `export *` do?
    It re-exports every named export from another module:
    export * from "./utils"; — makes all of utils' exports available from this module.
    export * as Utils from "./utils" creates a namespace re-export.

Q4: What is the `moduleResolution` setting in tsconfig?
    It controls how TypeScript resolves import paths to files. Common values:
    "node" (classic Node), "bundler" (modern bundlers like Vite), "node16"
    (strict ESM with .js extensions required), "nodenext".

Q5: How do you create path aliases in TypeScript?
    Set `baseUrl` and `paths` in tsconfig.json:
    "paths": { "@utils/*": ["src/utils/*"] }
    Then import { helper } from "@utils/helper". Note: paths don't work at
    runtime without a bundler plugin or `tsconfig-paths` Node.js package.

Q6: When would you still use TypeScript namespaces today?
    For global scripts (no module bundler), in .d.ts declaration files to
    namespace types for a library, or when augmenting existing global APIs.
    They're also still used in older Angular codebases and for UMD builds.

Q7: What is a wildcard module declaration?
    `declare module "*.png" { const content: string; export default content; }`
    This tells TypeScript how to type imports of non-JS files (images, CSS, SVG)
    that bundlers process. Without it, importing ".png" is a type error.

Q8: What is module augmentation?
    Adding to an existing module's type declarations:
    declare module "express" { interface Request { user?: User } }
    This extends Express's Request type without modifying the original package.

Q9: What is `declare global` used for?
    It augments the global TypeScript scope inside a module file. The file must
    have at least one import/export to be treated as a module before `declare global`
    can be used.

Q10: What is the difference between `import` and `import type`?
     `import { User }` may keep the import at runtime if the value is used as a
     value. `import type { User }` is always erased — it signals "this is only
     needed for types, never as a runtime value." With `verbatimModuleSyntax:
     true`, TypeScript enforces type-only imports for type-only usages.
