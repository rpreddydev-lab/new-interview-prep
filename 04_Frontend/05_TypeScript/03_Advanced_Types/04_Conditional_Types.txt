1. WHAT ARE CONDITIONAL TYPES?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Conditional types choose between two types based on a condition at the TYPE
  level. They use the ternary-like syntax:

    T extends U ? X : Y

  "If T is assignable to U, the type is X; otherwise, the type is Y."

  type IsString<T> = T extends string ? "yes" : "no";
  type A = IsString<string>;   // "yes"
  type B = IsString<number>;   // "no"
  type C = IsString<"hello">;  // "yes" — "hello" extends string

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
2. BASIC CONDITIONAL TYPE PATTERNS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  a) Check if a type extends another:
     type IsArray<T> = T extends any[] ? true : false;
     type IsArray<number[]>   // true
     type IsArray<string>     // false

  b) Extract element type from an array:
     type ElementType<T> = T extends (infer E)[] ? E : never;
     type E = ElementType<string[]>; // string
     type F = ElementType<number>;   // never

  c) Check for callable types:
     type IsFunction<T> = T extends (...args: any[]) => any ? true : false;

  d) Nested conditional (chained ternary):
     type TypeName<T> =
       T extends string  ? "string"  :
       T extends number  ? "number"  :
       T extends boolean ? "boolean" :
       T extends null    ? "null"    :
       T extends undefined ? "undefined" :
       "object";

3. DISTRIBUTIVE CONDITIONAL TYPES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  When a BARE type parameter is used on the left side of `extends`, conditional
  types DISTRIBUTE over union types automatically:

    (A | B) extends U ? X : Y
    → (A extends U ? X : Y) | (B extends U ? X : Y)

  type IsString<T> = T extends string ? "yes" : "no";
  type R = IsString<"hello" | 42>;
  // = IsString<"hello"> | IsString<42>
  // = "yes" | "no"

  IMPORTANT: This only happens when T is a BARE (naked) type parameter — not
  when wrapped in another type like Array<T> or [T].

  // Distributive (bare T):
  type Distribute<T> = T extends string ? "string" : "other";
  type R1 = Distribute<string | number>; // "string" | "other"

  // NOT distributive (T is wrapped):
  type NoDistribute<T> = [T] extends [string] ? "string" : "other";
  type R2 = NoDistribute<string | number>; // "other" (whole union checked)

  PRACTICAL USE — Exclude<T, U>:
  type Exclude<T, U> = T extends U ? never : T;

  type E = Exclude<"a" | "b" | "c", "a">;
  // = ("a" extends "a" ? never : "a") | ("b" extends "a" ? never : "b") | ("c" extends "a" ? never : "c")
  // = never | "b" | "c"
  // = "b" | "c"

4. THE `infer` KEYWORD IN CONDITIONAL TYPES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  `infer` introduces a new type variable within the true branch of a conditional
  type. TypeScript "infers" the value of that variable from the type being matched.

  Syntax: T extends Pattern<infer X> ? X : fallback

  a) Extract function return type:
     type ReturnType<T> = T extends (...args: any[]) => infer R ? R : never;
     type R = ReturnType<() => string>;   // string
     type S = ReturnType<(n: number) => boolean[]>; // boolean[]

  b) Extract function parameter types:
     type Parameters<T> = T extends (...args: infer P) => any ? P : never;
     type P = Parameters<(a: string, b: number) => void>; // [string, number]

  c) Extract the element type of an array:
     type ElementType<T> = T extends (infer E)[] ? E : never;
     type E = ElementType<string[]>; // string

  d) Extract the value type from a Promise:
     type Awaited<T> = T extends Promise<infer U> ? U : T;
     type A = Awaited<Promise<string>>;           // string
     type B = Awaited<Promise<Promise<number>>>;  // Promise<number> (one level)

     // Recursive version:
     type DeepAwaited<T> = T extends Promise<infer U> ? DeepAwaited<U> : T;
     type C = DeepAwaited<Promise<Promise<number>>>; // number

  e) Infer from a specific property:
     type GetId<T> = T extends { id: infer I } ? I : never;
     type Id = GetId<{ id: number; name: string }>; // number

  f) Extract constructor parameter types:
     type ConstructorParams<T> =
       T extends new (...args: infer P) => any ? P : never;
     class MyClass { constructor(a: string, b: number) {} }
     type CP = ConstructorParams<typeof MyClass>; // [string, number]

  g) Extract first/last element of a tuple:
     type First<T extends any[]> = T extends [infer F, ...any[]] ? F : never;
     type Last<T extends any[]> = T extends [...any[], infer L] ? L : never;
     type Head = First<[1, 2, 3]>; // 1
     type Tail = Last<[1, 2, 3]>;  // 3

5. PREVENTING DISTRIBUTION (NON-DISTRIBUTIVE)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Wrap T in a tuple [T] to prevent distribution:

  type IsNever<T> = [T] extends [never] ? true : false;
  type A = IsNever<never>;           // true  ✅
  type B = IsNever<string | never>;  // false ✅

  // Without wrapping:
  type IsNeverBad<T> = T extends never ? true : false;
  type C = IsNeverBad<never>; // never ❌ (distributes over empty union)

  // Check if a union has exactly one member:
  type IsUnion<T, U = T> =
    T extends U
      ? [U] extends [T] ? false : true
      : never;
  type D = IsUnion<string | number>; // true
  type E = IsUnion<string>;          // false

6. BUILT-IN CONDITIONAL UTILITY TYPES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  ┌────────────────────┬──────────────────────────────────────────┐
  │ Utility Type       │ Implementation (conditional type)        │
  ├────────────────────┼──────────────────────────────────────────┤
  │ Exclude<T, U>      │ T extends U ? never : T                  │
  │ Extract<T, U>      │ T extends U ? T : never                  │
  │ NonNullable<T>     │ T extends null | undefined ? never : T   │
  │ ReturnType<T>      │ T extends (...) => infer R ? R : never   │
  │ Parameters<T>      │ T extends (...args: infer P) => ? P :    │
  │ Awaited<T>         │ T extends Promise<infer U> ? ... : T     │
  │ InstanceType<T>    │ T extends new (...) => infer I ? I :     │
  └────────────────────┴──────────────────────────────────────────┘

  Examples:
  type E = Exclude<"a" | "b" | "c", "a">;   // "b" | "c"
  type X = Extract<"a" | "b" | 1 | 2, string>; // "a" | "b"
  type N = NonNullable<string | null | undefined>; // string

7. ADVANCED CONDITIONAL TYPE PATTERNS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  a) UnwrapPromise (fully recursive):
     type UnwrapPromise<T> = T extends Promise<infer U> ? UnwrapPromise<U> : T;
     type R = UnwrapPromise<Promise<Promise<string>>>; // string

  b) UnwrapArray:
     type UnwrapArray<T> = T extends (infer U)[] ? UnwrapArray<U> : T;
     type R = UnwrapArray<string[][][]>; // string

  c) Flatten nested type:
     type Flatten<T> = T extends Array<infer Item> ? Item : T;
     type S = Flatten<string[]>; // string
     type N = Flatten<number>;   // number (passthrough)

  d) Overloaded function: get LAST overload's return type:
     type OverloadedReturn<T> =
       T extends { (...args: any[]): infer R; (...args: any[]): any } ? R :
       T extends { (...args: any[]): infer R } ? R : never;

  e) Convert union to intersection (advanced):
     type UnionToIntersection<U> =
       (U extends any ? (x: U) => void : never) extends (x: infer I) => void
         ? I : never;
     type I = UnionToIntersection<{ a: string } | { b: number }>;
     // { a: string } & { b: number }

  f) Conditional based on function arity:
     type IsVariadic<T> = T extends (...args: infer P) => any
       ? P extends [any, ...any[]] ? false : true
       : false;

8. DEFERRED EVALUATION
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  When a conditional type involves a free type parameter (not yet resolved),
  TypeScript defers evaluation until the type is applied:

  function strip<T>(val: T): T extends string[] ? string : T {
    return val as any;
  }

  const a = strip(["hello"]);  // string — evaluated when T = string[]
  const b = strip(42);         // number — T = number, no array

  Deferred evaluation ensures correct behavior in generic contexts even when
  TypeScript cannot immediately resolve the type.

9. INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q1: What are conditional types in TypeScript?
    Conditional types express "if-else" logic at the type level using the syntax
    T extends U ? X : Y. If T is assignable to U, the resulting type is X;
    otherwise it's Y. They're used for advanced type transformations.

Q2: What are distributive conditional types?
    When a bare type parameter T appears as the left-hand side of `extends`,
    conditional types distribute over union types:
    type D<T> = T extends string ? 1 : 2;
    D<string | number> → D<string> | D<number> → 1 | 2

Q3: How do you prevent distribution in a conditional type?
    Wrap T in a tuple: [T] extends [U] ? X : Y.
    type IsNever<T> = [T] extends [never] ? true : false;
    This prevents the conditional from distributing over each union member.

Q4: What does the `infer` keyword do?
    `infer` introduces a type variable that TypeScript infers from the matched
    type pattern. It's only valid inside the `extends` clause of a conditional:
    type ReturnType<T> = T extends (...args: any[]) => infer R ? R : never;
    Here R is inferred as the return type of T.

Q5: How do you extract the element type from an array?
    type ElementType<T> = T extends (infer E)[] ? E : never;
    type E = ElementType<string[]>; // string

Q6: How does Exclude<T, U> work internally?
    type Exclude<T, U> = T extends U ? never : T;
    Because T is a bare type param, it distributes: each member of T is tested
    against U; if it extends U, it becomes never (removed); otherwise it stays.

Q7: What is the difference between Extract and Exclude?
    Exclude<T, U> keeps what is NOT in U: T extends U ? never : T
    Extract<T, U> keeps what IS in U:    T extends U ? T : never

Q8: Can conditional types be recursive?
    Yes. A common example:
    type DeepAwaited<T> = T extends Promise<infer U> ? DeepAwaited<U> : T;
    TypeScript supports circular conditional types in limited ways.

Q9: How do conditional types handle `never`?
    Distributing over `never` (empty union) produces `never`:
    type D<T> = T extends string ? "s" : "o";
    type R = D<never>; // never (empty distribution)
    To test for never: [T] extends [never] ? true : false.

Q10: What is a deferred conditional type?
     When a conditional type involves a free (unresolved) type parameter, TS
     cannot evaluate it immediately, so it's stored as-is and resolved when the
     type is instantiated. This means conditional return types in generic
     functions only resolve to the correct branch once called with a concrete type.
