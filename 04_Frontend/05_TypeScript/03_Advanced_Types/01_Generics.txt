1. WHAT ARE GENERICS?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Generics allow you to write REUSABLE, TYPE-SAFE code that works with multiple
types without sacrificing type information.

The problem generics solve:

  // Without generics — not type-safe (any):
  function identity(value: any): any { return value; }
  const result = identity(42);   // result is `any` — type info LOST

  // Without generics — not reusable (separate for each type):
  function identityStr(value: string): string { return value; }
  function identityNum(value: number): number { return value; }

  // WITH generics — type-safe AND reusable:
  function identity<T>(value: T): T { return value; }
  const str = identity("hello"); // T = string → result is string
  const num = identity(42);      // T = number → result is number

The angle brackets <T> declare a TYPE PARAMETER — a placeholder for a type
that will be filled in when the function is called.

2. GENERIC FUNCTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Single type parameter:
  function wrap<T>(value: T): { value: T } {
    return { value };
  }
  wrap(42);       // { value: number }
  wrap("hello");  // { value: string }

  // Multiple type parameters:
  function pair<A, B>(first: A, second: B): [A, B] {
    return [first, second];
  }
  pair(1, "hello");   // [number, string]
  pair(true, [1,2]);  // [boolean, number[]]

  // Generic array functions:
  function first<T>(arr: T[]): T | undefined {
    return arr[0];
  }
  first([1, 2, 3]);       // number | undefined
  first(["a", "b"]);      // string | undefined

  function map<T, U>(arr: T[], fn: (item: T) => U): U[] {
    return arr.map(fn);
  }
  map([1, 2, 3], n => n.toString()); // string[]
  map(["a", "b"], s => s.length);    // number[]

  // Explicit type argument (when inference needs help):
  function createArray<T>(length: number, fill: T): T[] {
    return Array(length).fill(fill);
  }
  createArray<string>(3, "x");  // explicitly pass T = string
  createArray(3, "x");          // T inferred as string

3. GENERIC INTERFACES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  interface Box<T> {
    value: T;
    transform: (val: T) => T;
    isEmpty: boolean;
  }

  const numBox: Box<number> = {
    value: 42,
    transform: n => n * 2,
    isEmpty: false
  };

  const strBox: Box<string> = {
    value: "hello",
    transform: s => s.toUpperCase(),
    isEmpty: false
  };

  // Repository pattern:
  interface Repository<T, ID = number> {
    findById(id: ID): Promise<T | null>;
    findAll(): Promise<T[]>;
    save(entity: T): Promise<T>;
    delete(id: ID): Promise<void>;
    count(): Promise<number>;
  }

  interface User { id: number; name: string }

  class UserRepo implements Repository<User> {
    async findById(id: number): Promise<User | null> { ... }
    async findAll(): Promise<User[]> { ... }
    async save(user: User): Promise<User> { ... }
    async delete(id: number): Promise<void> { ... }
    async count(): Promise<number> { ... }
  }

4. GENERIC CLASSES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  class Stack<T> {
    private items: T[] = [];

    push(item: T): void {
      this.items.push(item);
    }

    pop(): T | undefined {
      return this.items.pop();
    }

    peek(): T | undefined {
      return this.items[this.items.length - 1];
    }

    get size(): number {
      return this.items.length;
    }

    isEmpty(): boolean {
      return this.items.length === 0;
    }
  }

  const numStack = new Stack<number>();
  numStack.push(1);
  numStack.push(2);
  numStack.pop();     // number | undefined

  const strStack = new Stack<string>();
  strStack.push("a");

  // Generic class with multiple type params:
  class KeyValueStore<K, V> {
    private map = new Map<K, V>();
    set(key: K, value: V) { this.map.set(key, value); }
    get(key: K): V | undefined { return this.map.get(key); }
    has(key: K): boolean { return this.map.has(key); }
  }

5. GENERIC CONSTRAINTS (extends)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Constraints restrict what types can be used as type parameters.
  Use `extends` to say "T must have at least this shape."

  // T must be an object with a length property:
  function longest<T extends { length: number }>(a: T, b: T): T {
    return a.length >= b.length ? a : b;
  }

  longest("hello", "hi");     // ✅ strings have .length
  longest([1, 2, 3], [1, 2]); // ✅ arrays have .length
  longest(10, 20);             // ❌ numbers don't have .length

  // Key constraint (very common):
  function getProperty<T, K extends keyof T>(obj: T, key: K): T[K] {
    return obj[key];
  }
  getProperty({ name: "Alice", age: 30 }, "name"); // string
  getProperty({ name: "Alice", age: 30 }, "age");  // number
  getProperty({ name: "Alice" }, "salary");         // ❌ Error: "salary" not in type

  // Constrain to specific types:
  function formatNum<T extends number | bigint>(value: T): string {
    return value.toString();
  }

  // Must extend a class:
  class Animal { constructor(public name: string) {} }
  function createAndName<T extends Animal>(ctor: new(name: string) => T, name: string): T {
    return new ctor(name);
  }

  // Default generic type parameters (TS 2.3+):
  interface Container<T = string> {
    value: T;
  }
  type DefaultContainer = Container;           // Container<string>
  type NumContainer = Container<number>;       // Container<number>

  type Maybe<T> = T | null | undefined;
  type Optional<T> = T | undefined;

  type Functor<F extends <T>(x: T) => any, T, U> = {

  };

7. ADVANCED GENERIC PATTERNS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

a) Conditional Generics:
   type IsArray<T> = T extends any[] ? true : false;
   type FlattenArray<T> = T extends (infer U)[] ? U : T;

   type A = FlattenArray<number[]>;   // number
   type B = FlattenArray<string[][]>; // string[]
   type C = FlattenArray<string>;     // string (passthrough)

b) Variance — Covariance and Contravariance (TS 4.7+):
   // Covariant (out — producer):
   type Getter<out T> = () => T;

   // Contravariant (in — consumer):
   type Setter<in T> = (value: T) => void;

   // Invariant:
   type ReadWrite<in out T> = { get(): T; set(v: T): void };

c) Generic Currying:
   function curry<A, B, C>(fn: (a: A, b: B) => C): (a: A) => (b: B) => C {
     return (a: A) => (b: B) => fn(a, b);
   }
   const add = curry((a: number, b: number) => a + b);
   const add5 = add(5);  // (b: number) => number
   add5(3);              // 8

d) Generic Function Composition:
   function compose<A, B, C>(f: (b: B) => C, g: (a: A) => B): (a: A) => C {
     return (a: A) => f(g(a));
   }
   const trim = (s: string) => s.trim();
   const upper = (s: string) => s.toUpperCase();
   const trimAndUpper = compose(upper, trim);
   trimAndUpper("  hello  "); // "HELLO"

e) Generic Builder Pattern:
   class QueryBuilder<T> {
     private conditions: ((item: T) => boolean)[] = [];
     private _limit: number = Infinity;

     where(predicate: (item: T) => boolean): this {
       this.conditions.push(predicate);
       return this;
     }
     limit(n: number): this {
       this._limit = n;
       return this;
     }
     execute(items: T[]): T[] {
       return items
         .filter(item => this.conditions.every(c => c(item)))
         .slice(0, this._limit);
     }
   }

   const results = new QueryBuilder<User>()
     .where(u => u.age > 18)
     .where(u => u.active)
     .limit(10)
     .execute(users);

8. `const` TYPE PARAMETERS (TypeScript 5.0+)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  function makeArray<const T>(value: T): T[] {
    return [value];
  }

  const arr = makeArray(["a", "b"] as const);
  // Without `const`: T = string[] → arr is string[][]
  // With `const`:    T = readonly ["a", "b"] → arr is (readonly ["a","b"])[]

  // Practical: preserving literal types in generics
  function createState<const T extends Record<string, unknown>>(initial: T) {
    return { state: initial };
  }
  const s = createState({ theme: "dark", lang: "en" });
  // s.state.theme: "dark" (literal!) not string

9. GENERICS IN STANDARD LIBRARY
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  The entire TypeScript standard library uses generics:

  Array<T>        → T[]
  Promise<T>      → resolves to T
  Map<K, V>       → key-value store
  Set<T>          → unique values
  Record<K, V>    → object with keys K and values V
  Partial<T>      → all properties optional
  Required<T>     → all properties required
  Readonly<T>     → all properties readonly
  Pick<T, K>      → subset of T's properties
  Omit<T, K>      → T without K properties
  Exclude<T, U>   → T excluding types in U
  Extract<T, U>   → intersection of T and U
  NonNullable<T>  → T without null/undefined
  ReturnType<T>   → return type of function T
  Parameters<T>   → parameter types of function T
  InstanceType<T> → instance type of constructor T

10. COMMON INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q1: What are generics and why are they useful?
    Generics allow writing reusable functions, classes, and interfaces that
    work with multiple types while preserving type information. Without
    generics, you'd either use `any` (losing type safety) or write duplicate
    code for each type.

Q2: What is a type parameter?
    A type parameter (written as <T>, <K>, <V> etc.) is a placeholder for
    a type that will be determined when the generic is used. TypeScript
    infers it from arguments or you can specify it explicitly.

Q3: What is a generic constraint and why would you use it?
    A generic constraint (<T extends SomeType>) restricts what types can
    be used as T. Used when you need to access a property or call a method
    on T: function len<T extends { length: number }>(x: T) allows .length.

Q4: What is the difference between `T extends string` and `T: string`?
    There is no `T: string` syntax in TypeScript.
    `T extends string` is a constraint meaning T must be assignable to string.
    T could be "hello" (literal) or string itself.
    In conditional types, `T extends string ? A : B` distributes over unions.

Q5: Can a generic type have a default?
    Yes: interface Container<T = string> { value: T }
    Container without argument is Container<string>.
    Defaults can reference previous type parameters: <A, B = A[]>

Q6: How do you constrain a generic to be a constructor?
    Use: T extends new (...args: any[]) => SomeType
    Or: T extends abstract new (...args: any[]) => SomeType
    This ensures T is a class (constructor function).

Q7: What is the difference between Array<T> and T[]?
    They're equivalent. T[] is shorthand for Array<T>.
    Array<T> makes the generic nature explicit.
    Prefer T[] for readability, use Array<T> when needed (e.g., generics).

Q8: What does `infer` do in generics?
    `infer` is used in conditional types to extract (infer) a type from
    another type at the type level:
    type ReturnType<T> = T extends (...args: any[]) => infer R ? R : never;
    Here R is inferred from the return type of T.

Q9: What is generic variance in TypeScript 4.7+?
    Variance describes how subtyping relates when types have generic parameters.
    `out T` (covariant): can only be produced/returned — widening is safe.
    `in T` (contravariant): can only be consumed/accepted — narrowing is safe.
    TypeScript 4.7 added explicit variance annotations for better precision
    and performance.

Q10: What is the `const` type parameter in TypeScript 5.0?
     Adding `const` to a type parameter prevents type widening, similar to
     `as const` at the call site: `function fn<const T>(x: T)`
     TypeScript infers literal/narrower types instead of widening to string,
     number, etc. Useful when you want to preserve exact literal types
     passed to a generic function.
