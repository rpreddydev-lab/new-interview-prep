1. WHAT ARE UTILITY TYPES?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Utility types are built-in generic type transformers provided by TypeScript.
They let you derive new types from existing ones without rewriting the whole type.

All utility types are in the global scope — no imports needed.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
2. OBJECT TRANSFORMATION UTILITIES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Base type used in examples:
  interface User {
    id: number;
    name: string;
    email: string;
    password: string;
    role: "admin" | "user";
    age?: number;
  }

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Partial<T>
  Makes ALL properties optional.

  type UpdateUser = Partial<User>;
  // { id?: number; name?: string; email?: string; ... }

  Use case: Update payloads, form state, mock data
  function updateUser(id: number, changes: Partial<User>): User { ... }
  updateUser(1, { name: "Bob" }); // only pass what you want to update

  Implementation:
  type Partial<T> = { [K in keyof T]?: T[K] };

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Required<T>
  Makes ALL properties required (removes ? from optional properties).

  type RequiredUser = Required<User>;
  // { id: number; name: string; email: string; age: number; ... }
  // age is now required (was optional)

  Use case: Ensure all fields are present after validation
  function processUser(user: Required<User>): void { ... }

  Implementation:
  type Required<T> = { [K in keyof T]-?: T[K] };
  // The minus sign removes optionality

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Readonly<T>
  Makes ALL properties readonly (prevents mutation).

  type ReadonlyUser = Readonly<User>;
  // { readonly id: number; readonly name: string; ... }

  const user: ReadonlyUser = { id: 1, name: "Alice", ... };
  user.name = "Bob"; // ❌ Error: Cannot assign to 'name' because it is read-only

  Use case: Immutable state, frozen config, Redux state
  const state: Readonly<AppState> = getInitialState();

  Implementation:
  type Readonly<T> = { readonly [K in keyof T]: T[K] };

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Pick<T, K>
  Creates a type with only the specified keys from T.
  K must be a union of keyof T.

  type UserPreview = Pick<User, "id" | "name">;
  // { id: number; name: string }

  type PublicUser = Pick<User, "id" | "name" | "email">;
  // Remove sensitive fields (password)

  Use case: API response types, displaying subsets of data
  function getPublicProfile(id: number): Pick<User, "id" | "name"> { ... }

  Implementation:
  type Pick<T, K extends keyof T> = { [P in K]: T[P] };

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Omit<T, K>
  Creates a type WITHOUT the specified keys. The inverse of Pick.

  type CreateUserDTO = Omit<User, "id">;
  // { name: string; email: string; password: string; role: ...; age?: number }

  type SafeUser = Omit<User, "password">;
  // Removes the password field

  type UpdateUserDTO = Omit<Partial<User>, "id">;
  // All fields optional except id is completely removed

  Use case: DTOs, removing sensitive fields, creating from existing
  function createUser(data: Omit<User, "id">): User { ... }

  Implementation:
  type Omit<T, K extends keyof any> = Pick<T, Exclude<keyof T, K>>;

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Record<K, V>
  Creates an object type with keys K and values V.

  type RoleMap = Record<string, string[]>;
  // { [key: string]: string[] }

  type UsersByRole = Record<"admin" | "user" | "guest", User[]>;
  // { admin: User[]; user: User[]; guest: User[] }

  const permissions: Record<"admin" | "user", string[]> = {
    admin: ["read", "write", "delete"],
    user:  ["read"],
  };

  type PageConfig = Record<string, { title: string; path: string }>;

  Use case: Dictionaries, lookup tables, maps
  const cache: Record<string, unknown> = {};

  Implementation:
  type Record<K extends keyof any, T> = { [P in K]: T };

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
3. UNION TRANSFORMATION UTILITIES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Exclude<T, U>
  Removes types from T that are assignable to U (set subtraction).

  type T = Exclude<string | number | boolean, string>;
  // number | boolean  (string was excluded)

  type NonNullableStr = Exclude<string | null | undefined, null | undefined>;
  // string

  type AnimalWithoutDog = Exclude<"cat" | "dog" | "bird", "dog">;
  // "cat" | "bird"

  Implementation:
  type Exclude<T, U> = T extends U ? never : T;

Extract<T, U>
  Keeps ONLY types from T that are assignable to U. (inverse of Exclude)

  type T = Extract<string | number | boolean, string | number>;
  // string | number

  type StringOrNumber = Extract<string | number | object | boolean, string | number>;
  // string | number

  Implementation:
  type Extract<T, U> = T extends U ? T : never;

NonNullable<T>
  Removes null and undefined from T.

  type T = NonNullable<string | null | undefined>;
  // string

  type SafeID = NonNullable<string | number | null | undefined>;
  // string | number

  Implementation:
  type NonNullable<T> = T extends null | undefined ? never : T;

4. FUNCTION UTILITIES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

ReturnType<T>
  Extracts the return type of a function type T.

  function fetchUser(id: number): Promise<User> { ... }
  type FetchResult = ReturnType<typeof fetchUser>; // Promise<User>

  const getUser = () => ({ id: 1, name: "Alice" });
  type UserType = ReturnType<typeof getUser>; // { id: number; name: string }

  // Useful when you don't control the function signature:
  type UseStateResult = ReturnType<typeof useState>; // from React

  Implementation:
  type ReturnType<T extends (...args: any) => any> = T extends (...args: any) => infer R ? R : any;

Parameters<T>
  Extracts the parameter types of a function as a tuple.

  function greet(name: string, age: number): string { ... }
  type Params = Parameters<typeof greet>; // [string, number]
  type FirstParam = Parameters<typeof greet>[0]; // string

  // Use case: wrapping functions
  function loggedFn<T extends (...args: any[]) => any>(
    fn: T,
    ...args: Parameters<T>
  ): ReturnType<T> {
    console.log("Calling with:", args);
    return fn(...args);
  }

  Implementation:
  type Parameters<T extends (...args: any) => any> = T extends (...args: infer P) => any ? P : never;

ConstructorParameters<T>
  Extracts constructor parameter types.

  class Person {
    constructor(public name: string, public age: number) {}
  }
  type PersonArgs = ConstructorParameters<typeof Person>; // [string, number]

  Implementation:
  type ConstructorParameters<T extends abstract new (...args: any) => any> =
    T extends abstract new (...args: infer P) => any ? P : never;

InstanceType<T>
  Extracts the instance type of a constructor.

  class MyClass { value = 42 }
  type MyInstance = InstanceType<typeof MyClass>; // MyClass

  // Useful for factory functions:
  function create<T extends new (...args: any[]) => any>(
    ctor: T,
    ...args: ConstructorParameters<T>
  ): InstanceType<T> {
    return new ctor(...args);
  }

5. AWAITED<T> (TypeScript 4.5+)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Unwraps nested Promise types recursively.

  type T1 = Awaited<Promise<string>>;               // string
  type T2 = Awaited<Promise<Promise<number>>>;       // number
  type T3 = Awaited<string | Promise<boolean>>;      // string | boolean

  async function fetchUser(): Promise<User> { ... }
  type ResolvedUser = Awaited<ReturnType<typeof fetchUser>>; // User

  Implementation:
  type Awaited<T> = T extends null | undefined
    ? T
    : T extends object & { then(onfulfilled: infer F, ...args: any): any }
      ? F extends (value: infer V, ...args: any) => any
        ? Awaited<V>
        : never
      : T;

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
6. NoInfer<T> (TypeScript 5.4+)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Prevents TypeScript from using this position to infer T.

  // Problem: TypeScript infers T from both the array AND the default
  function createState<T>(initial: T[], defaultVal: T): T[] { ... }
  createState([1, 2, 3], "fallback"); // T inferred as number | string — too wide!

  // Solution: NoInfer on the default forces T to be inferred from initial only
  function createState<T>(initial: T[], defaultVal: NoInfer<T>): T[] { ... }
  createState([1, 2, 3], "fallback"); // ❌ Error: string not assignable to number

7. COMBINING UTILITY TYPES (REAL-WORLD PATTERNS)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  type User = { id: number; name: string; email: string; password: string; age?: number };

  // Create: all fields except id
  type CreateUserInput = Omit<User, "id">;

  // Update: all fields optional, no id
  type UpdateUserInput = Partial<Omit<User, "id">>;

  // Public view: no password
  type PublicUser = Omit<Readonly<User>, "password">;

  // Stored: always has id, all fields required
  type StoredUser = Required<User>;

  // Custom: make only some keys required
  type PartialBy<T, K extends keyof T> = Omit<T, K> & Partial<Pick<T, K>>;
  type UserWithOptionalEmail = PartialBy<User, "email" | "age">;

  // Custom: make only some keys optional
  type RequiredBy<T, K extends keyof T> = Omit<T, K> & Required<Pick<T, K>>;

  // Deep partial (recursive):
  type DeepPartial<T> = {
    [K in keyof T]?: T[K] extends object ? DeepPartial<T[K]> : T[K];
  };

  // Deep readonly (recursive):
  type DeepReadonly<T> = {
    readonly [K in keyof T]: T[K] extends object ? DeepReadonly<T[K]> : T[K];
  };

8. STRING MANIPULATION UTILITIES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  type U = Uppercase<"hello">;     // "HELLO"
  type L = Lowercase<"WORLD">;     // "world"
  type C = Capitalize<"hello">;    // "Hello"
  type N = Uncapitalize<"Hello">; // "hello"

  // Used with mapped types to generate property name variants:
  type Getters<T> = {
    [K in keyof T as `get${Capitalize<string & K>}`]: () => T[K];
  };
  type Setters<T> = {
    [K in keyof T as `set${Capitalize<string & K>}`]: (val: T[K]) => void;
  };

  type UserGetters = Getters<{ name: string; age: number }>;
  // { getName: () => string; getAge: () => number }

10. COMMON INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q1: What are utility types in TypeScript?
    Built-in generic types that transform existing types: Partial<T>, Pick<T,K>,
    Omit<T,K>, Record<K,V>, etc. They derive new types without rewriting them.
    All are available globally without imports.

Q2: What is the difference between Pick and Omit?
    Pick<T, K> keeps only the listed keys: Pick<User, "id" | "name"> → { id; name }
    Omit<T, K> removes the listed keys: Omit<User, "password"> → everything except password
    Use Pick when you want a few properties; Omit when you want all but a few.

Q3: What is the difference between Partial and Required?
    Partial<T> makes ALL properties optional (adds ?).
    Required<T> makes ALL properties required (removes ?).
    Use Partial for update/patch operations; Required for ensuring complete data.

Q4: How do Exclude and Extract differ?
    Exclude<T, U> removes types from T that match U.
    Extract<T, U> keeps only types from T that match U.
    They're inverses. Both operate on union types.
    Exclude<A | B | C, B> = A | C
    Extract<A | B | C, B | C> = B | C

Q5: What does NonNullable<T> do?
    Removes null and undefined from T:
    NonNullable<string | null | undefined> = string
    Useful when you've already null-checked and want to reflect that in types.

Q6: When would you use Record<K, V>?
    When you need an object type with dynamic keys:
    Record<string, number> → { [key: string]: number }
    Record<"a" | "b", string> → { a: string; b: string }
    Common for lookup tables, caches, dictionaries, and config objects.

Q7: What does ReturnType<T> extract?
    The return type of a function type T.
    type R = ReturnType<() => string>; // string
    Useful when you don't control a function but need to know its return type.

Q8: How do you create a "deep partial" type?
    The built-in Partial<T> is shallow. For deep:
    type DeepPartial<T> = { [K in keyof T]?: T[K] extends object ? DeepPartial<T[K]> : T[K] }
    This recursively makes all nested properties optional too.

Q9: What is the Awaited<T> utility type?
    Awaited<T> recursively unwraps Promise types to get the resolved value type.
    Awaited<Promise<Promise<string>>> = string
    Added in TypeScript 4.5. Used to get the return type of async functions.

Q10: How would you create a type that makes only specific keys optional?
     type PartialBy<T, K extends keyof T> = Omit<T, K> & Partial<Pick<T, K>>;
     type UserWithOptionalEmail = PartialBy<User, "email">;
     // email is optional, other fields required
