1. useState
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Simple inferred types:
  const [count, setCount] = useState(0);         // number
  const [name, setName]   = useState("");         // string
  const [flag, setFlag]   = useState(false);      // boolean

  // Explicit type parameter (needed when initial value doesn't match all states):
  const [user, setUser] = useState<User | null>(null);
  const [items, setItems] = useState<string[]>([]);
  const [error, setError] = useState<Error | null>(null);

  // Complex state with interface:
  interface FormState {
    name: string;
    email: string;
    age: number;
  }
  const [form, setForm] = useState<FormState>({
    name: "",
    email: "",
    age: 0,
  });

  // Partial update:
  setForm(prev => ({ ...prev, name: "Alice" }));

  // Discriminated union state (instead of multiple booleans):
  type Status = "idle" | "loading" | "success" | "error";
  const [status, setStatus] = useState<Status>("idle");

2. useEffect
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // useEffect is typed to return void or a cleanup function () => void:
  useEffect(() => {
    const subscription = subscribe(callback);
    return () => subscription.unsubscribe(); // cleanup: () => void
  }, [dependency]);

  // The callback can be async IF you wrap it:
  useEffect(() => {
    // ❌ Don't make the effect callback async directly:
    // async () => { await fetch(...) } returns a Promise, not () => void

    // ✅ Declare async function inside and call it:
    async function fetchData() {
      const data = await fetch("/api/data").then(r => r.json());
      setItems(data);
    }
    fetchData();
  }, []);

  // Typed cleanup with AbortController:
  useEffect(() => {
    const controller = new AbortController();
    fetch("/api/data", { signal: controller.signal })
      .then(r => r.json())
      .then(setData);
    return () => controller.abort();
  }, []);

3. useRef
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // DOM element ref — initial value null, becomes populated by React:
  const inputRef = useRef<HTMLInputElement>(null);
  // inputRef.current is HTMLInputElement | null
  inputRef.current?.focus();

  // Mutable ref (for storing any value — NOT rendered):
  const countRef = useRef<number>(0);
  // countRef.current is always number (no null, initial value provided)

  // When to use which:
  // useRef<T>(null)  → DOM refs (React fills it in)
  // useRef<T>(value) → mutable container (you control the value)

  // Common use cases:
  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null);
  const prevValueRef = useRef<number>(0);

  useEffect(() => {
    timerRef.current = setInterval(() => setCount(c => c + 1), 1000);
    return () => {
      if (timerRef.current) clearInterval(timerRef.current);
    };
  }, []);

4. useReducer
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Define State and Action types (discriminated union for actions):
  interface State {
    count: number;
    loading: boolean;
    error: string | null;
  }

  type Action =
    | { type: "INCREMENT"; amount: number }
    | { type: "DECREMENT"; amount: number }
    | { type: "RESET" }
    | { type: "SET_LOADING"; loading: boolean }
    | { type: "SET_ERROR"; error: string };

  function reducer(state: State, action: Action): State {
    switch (action.type) {
      case "INCREMENT":    return { ...state, count: state.count + action.amount };
      case "DECREMENT":    return { ...state, count: state.count - action.amount };
      case "RESET":        return { count: 0, loading: false, error: null };
      case "SET_LOADING":  return { ...state, loading: action.loading };
      case "SET_ERROR":    return { ...state, error: action.error };
      default:
        const _: never = action; // exhaustiveness check
        return state;
    }
  }

  // In component:
  const [state, dispatch] = useReducer(reducer, { count: 0, loading: false, error: null });
  dispatch({ type: "INCREMENT", amount: 1 }); // ✅ typed
  dispatch({ type: "RESET" });               // ✅
  dispatch({ type: "BAD" });                 // ❌ Error: not a valid action type

  const handleClick = useCallback(
    (event: React.MouseEvent<HTMLButtonElement>) => {
      console.log(event.target);
    },
    []

  // handleClick: (event: React.MouseEvent<HTMLButtonElement>) => void

  // With parameters:
  const handleChange = useCallback(
    (id: number, value: string) => {
      setItems(prev => prev.map(item => item.id === id ? { ...item, value } : item));
    },
    []
  );

7. CUSTOM HOOKS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  a) Basic data fetching hook:
     interface UseFetchResult<T> {
       data: T | null;
       loading: boolean;
       error: Error | null;
     }

     function useFetch<T>(url: string): UseFetchResult<T> {
       const [data, setData] = useState<T | null>(null);
       const [loading, setLoading] = useState(true);
       const [error, setError] = useState<Error | null>(null);

       useEffect(() => {
         let cancelled = false;
         setLoading(true);
         fetch(url)
           .then(r => r.json() as Promise<T>)
           .then(d => { if (!cancelled) { setData(d); setLoading(false); } })
           .catch(e => { if (!cancelled) { setError(e); setLoading(false); } });
         return () => { cancelled = true; };
       }, [url]);

       return { data, loading, error };
     }

     // Usage:
     const { data, loading } = useFetch<User[]>("/api/users");
     // data: User[] | null, loading: boolean

  b) useLocalStorage hook:
     function useLocalStorage<T>(key: string, initialValue: T): [T, (val: T) => void] {
       const [stored, setStored] = useState<T>(() => {
         try {
           const item = window.localStorage.getItem(key);
           return item ? JSON.parse(item) as T : initialValue;
         } catch {
           return initialValue;
         }
       });

       const setValue = (value: T) => {
         try {
           setStored(value);
           window.localStorage.setItem(key, JSON.stringify(value));
         } catch (e) {
           console.error(e);
         }
       };

       return [stored, setValue];
     }

     const [theme, setTheme] = useLocalStorage<"dark" | "light">("theme", "light");

  c) Returning a tuple with const assertion:
     // Without const assertion:
     function useToggle(init = false) {
       const [on, setOn] = useState(init);
       const toggle = () => setOn(v => !v);
       return [on, toggle]; // type: (boolean | (() => void))[]  ❌ too wide
     }

     // With `as const`:
     function useToggle(init = false) {
       const [on, setOn] = useState(init);
       const toggle = () => setOn(v => !v);
       return [on, toggle] as const; // type: [boolean, () => void] ✅
     }

8. INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q1: When do you need to explicitly type useState?
    When the initial value is null/undefined (so TypeScript can't infer the full
    type) or when the state can hold multiple distinct types:
    useState<User | null>(null), useState<string[]>([])

Q2: How do you type useRef for a DOM element?
    useRef<HTMLInputElement>(null) — provide the element type and null as the
    initial value. React will assign the actual element on mount, so
    .current is HTMLInputElement | null.

Q3: How do you type useRef for a mutable container (not a DOM element)?
    Provide a non-null initial value: useRef<number>(0)
    The .current is then always number, not null.

Q4: Why shouldn't useEffect's callback be async?
    useEffect expects its callback to return either void or a cleanup function
    (() => void). An async function returns a Promise, not a cleanup function.
    The fix: declare an async function inside the callback and call it.

Q5: How do you add exhaustiveness checking to useReducer?
    In the default branch of the reducer's switch, assign action to a never variable:
    const _: never = action;
    If a new action type is added but not handled, TypeScript reports an error here.

Q6: How do you type a custom hook that returns a tuple?
    Return `as const` to preserve the tuple type instead of a wide array type:
    return [value, setter] as const; // [T, (v: T) => void]
    Without `as const`, TypeScript infers (T | ((v: T) => void))[] — too loose.

Q7: How do you type useMemo?
    Type is usually inferred from the factory function. For explicit typing:
    useMemo<ReturnType>(() => computeValue(), [deps])

Q8: What is the ReturnType<typeof setInterval>?
    In a Node.js environment: NodeJS.Timeout
    In a browser environment: number
    For portability: ReturnType<typeof setInterval> captures whichever environment.

Q9: How do you type a generic data-fetching custom hook?
    Make it generic: function useFetch<T>(url: string): { data: T | null; loading: boolean; error: Error | null }
    The caller specifies T: useFetch<User[]>("/api/users")

Q10: How do you handle the cleanup function type in useEffect?
     The callback should return (() => void) | void. Cleanup functions must
     return void (not a Promise). If no cleanup is needed, return nothing
     or return undefined. TypeScript enforces this automatically.
