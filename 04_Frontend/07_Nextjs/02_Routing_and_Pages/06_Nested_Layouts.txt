1. WHAT ARE LAYOUTS?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

A layout is UI that is SHARED across multiple pages. It wraps its children and
persists across navigations (does not re-mount = state preserved).

In the App Router, layout.tsx is a first-class concept — each route segment
can have its own layout, and they nest hierarchically.

2. ROOT LAYOUT (REQUIRED)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // app/layout.tsx — required, wraps EVERY page
  import type { Metadata } from 'next';
  import { Inter } from 'next/font/google';
  import './globals.css';

  const inter = Inter({ subsets: ['latin'] });

  export const metadata: Metadata = {
    title: { template: '%s | My App', default: 'My App' },
    description: 'My awesome application',
  };

  export default function RootLayout({ children }: { children: React.ReactNode }) {
    return (
      <html lang="en">
        <body className={inter.className}>
          <Navbar />
          <main>{children}</main>
          <Footer />
        </body>
      </html>
    );
  }

  Key rules:
  - MUST include <html> and <body> tags (replaces _document.tsx)
  - Cannot be a Client Component (cannot use useState in root layout)
  - Providers (context, TanStack Query, Zustand) need a 'use client' wrapper

3. NESTED LAYOUTS — HOW NESTING WORKS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  app/
  ├── layout.tsx            ← Root layout (wraps everything)
  ├── page.tsx              → /
  ├── dashboard/
  │   ├── layout.tsx        ← Dashboard layout (wraps /dashboard/*)
  │   ├── page.tsx          → /dashboard
  │   └── settings/
  │       ├── layout.tsx    ← Settings layout (wraps /dashboard/settings/*)
  │       └── page.tsx      → /dashboard/settings

When visiting /dashboard/settings:
  RootLayout
    └── DashboardLayout
          └── SettingsLayout
                └── SettingsPage

  All three layouts render — root layout is ALWAYS outermost.

  // app/dashboard/layout.tsx
  export default function DashboardLayout({ children }: { children: React.ReactNode }) {
    return (
      <div className="flex">
        <Sidebar links={dashboardLinks} />
        <div className="flex-1 p-6">
          {children}   {/* renders nested layouts or pages */}
        </div>
      </div>
    );
  }

4. LAYOUT vs TEMPLATE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

┌──────────────┬────────────────────────────────────────────────────────────┐
│ layout.tsx   │ Persists across navigation. Does NOT re-mount.             │
│              │ State, scroll position, and focus preserved.               │
│              │ Use for: nav bars, sidebars, persistent UI.                │
├──────────────┼────────────────────────────────────────────────────────────┤
│ template.tsx │ Re-mounts on EVERY navigation. State resets each time.     │
│              │ Same API as layout (wraps children).                       │
│              │ Use for: page enter animations, resetting state per route. │
└──────────────┴────────────────────────────────────────────────────────────┘

  // When does layout re-render?
  // - Layout itself does NOT re-render on same-segment navigation
  // - The {children} slot is replaced (the page/nested layout changes)
  // - Layout's own state/effects only run once (on mount)

  // Example: this counter persists when navigating /dashboard → /dashboard/settings
  'use client';
  export default function DashboardLayout({ children }) {
    const [count, setCount] = useState(0);  // persists across child navigation
    return (
      <div>
        <button onClick={() => setCount(c => c + 1)}>{count}</button>
        {children}
      </div>
    );
  }

5. DATA FETCHING IN LAYOUTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Layouts can be async Server Components and fetch data directly:

  // app/dashboard/layout.tsx
  import { auth } from '@/auth';
  import { redirect } from 'next/navigation';

  export default async function DashboardLayout({ children }) {
    const session = await auth();
    if (!session) redirect('/login');

    const user = await prisma.user.findUnique({ where: { id: session.user.id } });

    return (
      <div className="flex">
        <Sidebar user={user} />
        <main>{children}</main>
      </div>
    );
  }

  // Important: layouts CANNOT receive searchParams (only page.tsx gets them)
  // Layouts DO receive params from dynamic segments

  // app/[team]/layout.tsx
  export default async function TeamLayout({ children, params }) {
    const team = await getTeam(params.team);
    return (
      <div>
        <h1>{team.name}</h1>
        {children}
      </div>
    );
  }

6. PROVIDING CONTEXT IN LAYOUTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Server Components cannot use context providers directly. Pattern:

  // components/providers.tsx  — Client Component wrapper
  'use client';
  import { QueryClient, QueryClientProvider } from '@tanstack/react-query';
  import { ThemeProvider } from 'next-themes';
  import { useState } from 'react';

  export function Providers({ children }: { children: React.ReactNode }) {
    const [queryClient] = useState(() => new QueryClient());
    return (
      <QueryClientProvider client={queryClient}>
        <ThemeProvider>{children}</ThemeProvider>
      </QueryClientProvider>
    );
  }

  // app/layout.tsx  — Server Component
  import { Providers } from '@/components/providers';
  export default function RootLayout({ children }) {
    return (
      <html lang="en">
        <body>
          <Providers>    {/* Client boundary — providers here */}
            {children}   {/* children can still be Server Components */}
          </Providers>
        </body>
      </html>
    );
  }

7. LOADING UI & STREAMING WITH LAYOUTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  app/dashboard/
  ├── layout.tsx      ← Rendered immediately (no Suspense)
  ├── loading.tsx     ← Automatically wrapped in <Suspense> for the page
  └── page.tsx        ← Async Server Component (streamed when ready)

  // app/dashboard/loading.tsx
  export default function DashboardLoading() {
    return <DashboardSkeleton />;  // shown while page.tsx fetches data
  }

  // app/dashboard/page.tsx
  export default async function DashboardPage() {
    const data = await fetchHeavyData();  // triggers loading.tsx while waiting
    return <Dashboard data={data} />;
  }

  // How it works:
  // 1. Browser requests /dashboard
  // 2. layout.tsx renders immediately (nav/sidebar shown)
  // 3. loading.tsx skeleton shown in the {children} slot
  // 4. page.tsx finishes fetching → streams in, replaces skeleton

8. ERROR HANDLING IN LAYOUTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // app/dashboard/error.tsx — MUST be a Client Component
  'use client';
  import { useEffect } from 'react';

  export default function DashboardError({
    error,
    reset,
  }: {
    error: Error & { digest?: string };
    reset: () => void;
  }) {
    useEffect(() => {
      console.error(error);
    }, [error]);

    return (
      <div>
        <h2>Something went wrong!</h2>
        <button onClick={() => reset()}>Try again</button>
      </div>
    );
  }

  // error.tsx handles errors in the SAME segment's page and nested layouts
  // It does NOT catch errors in the parent layout (use parent's error.tsx)
  // To catch root layout errors, use app/global-error.tsx

  // app/global-error.tsx — catches root layout errors
  'use client';
  export default function GlobalError({ error, reset }) {
    return (
      <html><body>
        <h1>Global Error</h1>
        <button onClick={reset}>Retry</button>
      </body></html>
    );
  }

9. PAGES ROUTER: LAYOUT PATTERNS (LEGACY)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

In the Pages Router, there's no built-in layout system. Common patterns:

  // Pattern 1: Per-page layout (getLayout)
  // _app.tsx
  export default function MyApp({ Component, pageProps }) {
    const getLayout = Component.getLayout ?? (page => page);
    return getLayout(<Component {...pageProps} />);
  }

  // pages/dashboard/index.tsx
  DashboardPage.getLayout = function(page) {
    return <DashboardLayout>{page}</DashboardLayout>;
  };

  // Pattern 2: Persistent layout via _app.tsx
  export default function MyApp({ Component, pageProps }) {
    return (
      <Layout>
        <Component {...pageProps} />
      </Layout>
    );
  }
  // Problem: this wraps ALL pages in the same layout

10. COMMON INTERVIEW QUESTIONS & ANSWERS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q1: How do nested layouts work in Next.js App Router?
A:  Each folder in the app/ directory can have a layout.tsx. Layouts wrap their
    children and all nested segments. When navigating within the same layout's
    subtree, the layout does NOT re-mount — only the page (children) changes.
    Multiple layouts compose hierarchically: RootLayout → DashboardLayout →
    SettingsLayout → SettingsPage.

Q2: What is the difference between layout.tsx and template.tsx?
A:  layout.tsx persists across navigation (does not re-mount, state preserved).
    template.tsx creates a new instance on every navigation (state resets,
    useEffect re-runs). Use template for per-page animations or when you need
    fresh state on each route visit.

Q3: Can layouts receive searchParams?
A:  No. In App Router, only page.tsx receives searchParams as a prop. Layouts
    only receive params (from dynamic segments) and children. If you need
    searchParams in a layout, use useSearchParams() hook (Client Component only).

Q4: How do you add global providers (like React Query) in App Router?
A:  Create a Client Component wrapper (e.g., providers.tsx with 'use client')
    that wraps children in context providers. Import it in the root layout.tsx
    (which remains a Server Component). This way, providers are client-side but
    the layout itself (with its HTML structure) stays a Server Component.

Q5: What happens to scroll position when navigating with persistent layouts?
A:  By default, Next.js scrolls to the top when navigating to a new route.
    This behavior can be controlled with <Link scroll={false}> to preserve
    scroll position, or router.push(url, { scroll: false }).

Q6: How do you handle authentication in a layout?
A:  In the layout Server Component, check auth (await auth() or check cookies).
    If not authenticated, call redirect('/login') — this is more efficient than
    checking per-page because the layout wraps all protected routes.
    Example: dashboard/layout.tsx checks auth once for all /dashboard/* routes.

================================================================================
  KEY INSIGHT: Layouts in App Router are a superpower. One layout.tsx protects
  all nested routes, provides shared UI, and doesn't re-mount on navigation.
  This was complex to implement in the Pages Router; it's trivial in App Router.
