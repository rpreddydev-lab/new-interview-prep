1. WHAT IS useRef?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

useRef returns a mutable object { current: initialValue } that persists across
renders. Unlike state, CHANGING .current does NOT trigger a re-render.

  const ref = useRef(initialValue);
  // ref = { current: initialValue }

Two primary use cases:
  1. Accessing DOM nodes directly (the most common use)
  2. Storing mutable values that persist across renders without causing re-renders

2. USE CASE 1: DOM REFERENCES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Auto-focus an input on mount
  function SearchInput() {
    const inputRef = useRef(null);

    useEffect(() => {
      inputRef.current.focus();  // access DOM node
    }, []);

    return <input ref={inputRef} type="search" />;
  }

  // Scroll to bottom (e.g., chat window)
  function Chat({ messages }) {
    const bottomRef = useRef(null);

    useEffect(() => {
      bottomRef.current?.scrollIntoView({ behavior: 'smooth' });
    }, [messages]);

    return (
      <div className="chat">
        {messages.map(m => <Message key={m.id} message={m} />)}
        <div ref={bottomRef} />   {/* invisible anchor at bottom */}
      </div>
    );
  }

  // Video player controls
  function VideoPlayer({ src }) {
    const videoRef = useRef(null);

    const play = () => videoRef.current.play();
    const pause = () => videoRef.current.pause();

    return (
      <div>
        <video ref={videoRef} src={src} />
        <button onClick={play}>Play</button>
        <button onClick={pause}>Pause</button>
      </div>
    );
  }

  // Measuring DOM element size
  function MeasuredComponent() {
    const divRef = useRef(null);
    const [size, setSize] = useState({ width: 0, height: 0 });

    useLayoutEffect(() => {
      const rect = divRef.current.getBoundingClientRect();
      setSize({ width: rect.width, height: rect.height });
    }, []);

    return <div ref={divRef}>Width: {size.width}, Height: {size.height}</div>;
  }

3. USE CASE 2: PERSISTING VALUES WITHOUT RE-RENDER
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Store interval ID for cleanup
  function Timer() {
    const [seconds, setSeconds] = useState(0);
    const intervalRef = useRef(null);

    function start() {
      intervalRef.current = setInterval(() => setSeconds(s => s + 1), 1000);
    }

    function stop() {
      clearInterval(intervalRef.current);
    }

    useEffect(() => () => clearInterval(intervalRef.current), []);  // cleanup on unmount

    return (
      <div>
        <p>Time: {seconds}s</p>
        <button onClick={start}>Start</button>
        <button onClick={stop}>Stop</button>
      </div>
    );
  }

  // Tracking previous value
  function usePrevious(value) {
    const prevRef = useRef();
    useEffect(() => {
      prevRef.current = value;
    }); // runs after render — captures the "just-past" value
    return prevRef.current;
  }

  function Counter() {
    const [count, setCount] = useState(0);
    const prevCount = usePrevious(count);
    return <p>Current: {count}, Previous: {prevCount}</p>;
  }

  // Track if component is still mounted (prevent stale updates)
  function AsyncComponent() {
    const [data, setData] = useState(null);
    const isMountedRef = useRef(true);

    useEffect(() => {
      isMountedRef.current = true;
      fetchData().then(data => {
        if (isMountedRef.current) setData(data);
      });
      return () => { isMountedRef.current = false; };
    }, []);

  }

  // Track if first render (skip initial effect)
  function useUpdateEffect(fn, deps) {
    const isFirstRenderRef = useRef(true);
    useEffect(() => {
      if (isFirstRenderRef.current) {
        isFirstRenderRef.current = false;
        return;
      }
      return fn();
    }, deps);
  }

4. FORWARDING REFS (forwardRef)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

By default, you cannot attach a ref to a custom component. Use React.forwardRef:

  // Without forwardRef: ref points to the wrapper component, not the input
  // <CustomInput ref={inputRef} />  // ✗ ref is null or points to nothing useful

  // WITH forwardRef: ref is forwarded to the inner <input>
  const CustomInput = React.forwardRef(function CustomInput(props, ref) {
    return (
      <div className="input-wrapper">
        <input ref={ref} {...props} />
      </div>
    );
  });

  // Usage:
  function Form() {
    const inputRef = useRef(null);
    useEffect(() => { inputRef.current.focus(); }, []);
    return <CustomInput ref={inputRef} placeholder="Enter text" />;
  }

  // TypeScript:
  const CustomInput = React.forwardRef<HTMLInputElement, InputProps>(
    (props, ref) => <input ref={ref} {...props} />
  );

  // React 19 Note: forwardRef is no longer needed — ref is a regular prop

5. useImperativeHandle
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

useImperativeHandle customizes what is exposed via a ref to the parent.
Instead of exposing the raw DOM node, you expose specific methods.

  const CustomInput = React.forwardRef((props, ref) => {
    const inputRef = useRef(null);

    useImperativeHandle(ref, () => ({
      focus() { inputRef.current.focus(); },
      clear() { inputRef.current.value = ''; },
      getValue() { return inputRef.current.value; },
      // Does NOT expose the raw DOM node — only these three methods
    }));

    return <input ref={inputRef} {...props} />;
  });

  // Parent:
  function Form() {
    const inputRef = useRef(null);
    inputRef.current.focus();       // works (exposed)
    inputRef.current.clear();       // works (exposed)
    inputRef.current.style.color;   // undefined (not exposed)
  }

6. useRef vs useState
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  | Aspect              | useRef                     | useState                    |
  |---------------------|----------------------------|-----------------------------|
  | Triggers re-render  | NO                         | YES                         |
  | Mutable             | Yes (.current = value)     | Via setter function only    |
  | Value persists      | Yes (across renders)       | Yes (across renders)        |
  | Read in render      | Yes (but value is "stale") | Yes (always current)        |
  | Primary use         | DOM refs, instance vars    | UI state                    |

  When to use ref vs state:
  - Use STATE if changing the value should update the UI
  - Use REF if changing the value should NOT update the UI
    (timers, interval IDs, previous values, flags, DOM nodes)

7. COMMON INTERVIEW QUESTIONS ON useRef
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What is useRef and what is it used for?
A: A hook that returns a mutable { current } object persisting across renders.
   Primary uses: (1) Access DOM nodes directly, (2) Store mutable values that
   don't trigger re-renders (timer IDs, previous values, instance variables).

Q: What is the difference between useRef and useState?
A: Updating a ref's .current does NOT trigger re-render. Updating state does.
   Use refs for DOM access and values that affect logic but not UI rendering.

Q: How does forwardRef work?
A: React.forwardRef() wraps a functional component and passes the ref as a
   second argument alongside props, allowing parents to attach refs to the
   component's internal DOM elements.

Q: When would you use useImperativeHandle?
A: When you want to expose a limited, controlled API via a ref instead of
   exposing the raw DOM node. Useful for components like DatePicker or MediaPlayer
   where you want to expose focus(), play(), reset() but not the internal DOM.

Q: What is the value of a ref before the component mounts?
A: The initial value passed to useRef(). For DOM refs, it's typically null
   (since the DOM element doesn't exist yet). It becomes the DOM node after mount.

Q: Can you use a ref to read state inside an async callback without stale closure?
A: Yes. Keep a ref synchronized with state:
   const countRef = useRef(count);
   useEffect(() => { countRef.current = count; }, [count]);
   // Now async callbacks can read countRef.current for the latest value
