1. WHAT IS CONDITIONAL RENDERING?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Conditional rendering in React means showing different UI elements based on
certain conditions — just like if/else logic in regular JavaScript.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
2. METHOD 1: if/else (outside JSX)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

The most readable approach for complex conditions:

  function Greeting({ isLoggedIn, user }) {
    if (!user) return <p>Loading...</p>;

    if (isLoggedIn) {
      return <h1>Welcome back, {user.name}!</h1>;
    } else {
      return <h1>Please log in.</h1>;
    }
  }

Early return pattern (guard clauses):
  function UserProfile({ user, loading, error }) {
    if (loading) return <Spinner />;
    if (error) return <ErrorMessage error={error} />;
    if (!user) return <p>No user found</p>;

    // At this point we know we have valid data
    return (
      <div>
        <h1>{user.name}</h1>
        <p>{user.email}</p>
      </div>
    );
  }

3. METHOD 2: TERNARY OPERATOR (inside JSX)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Best for simple two-branch conditionals inline in JSX:

  function LoginButton({ isLoggedIn, onLogin, onLogout }) {
    return (
      <button onClick={isLoggedIn ? onLogout : onLogin}>
        {isLoggedIn ? 'Log Out' : 'Log In'}
      </button>
    );
  }

  // Conditional rendering of entire component
  function Page({ isLoggedIn }) {
    return (
      <div>
        <Header />
        {isLoggedIn ? <Dashboard /> : <LoginForm />}
        <Footer />
      </div>
    );
  }

  // Nested ternary (readable only if well-formatted)
  function Status({ status }) {
    return (
      <span>
        {status === 'loading'
          ? 'Loading...'
          : status === 'error'
          ? 'Error!'
          : 'Loaded'}
      </span>
    );
  }
  // For 3+ conditions, prefer if/else or a lookup object

4. METHOD 3: LOGICAL AND (&&)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Best for "show this OR nothing":

  function Notification({ message }) {
    return (
      <div>
        {message && <Alert>{message}</Alert>}
      </div>
    );
  }

  // Show badge only if count > 0
  function CartIcon({ count }) {
    return (
      <div>
        <svg>...</svg>
        {count > 0 && <Badge>{count}</Badge>}
      </div>
    );
  }

COMMON GOTCHA WITH &&:
  // If count is 0 (falsy), React renders "0" (the number), not nothing!
  {count && <Badge>{count}</Badge>}    // WRONG: renders "0" if count is 0
  {count > 0 && <Badge>{count}</Badge>} // CORRECT: renders nothing if count is 0
  {!!count && <Badge>{count}</Badge>}  // CORRECT: double negation converts to boolean

  // Other falsy values that DON'T render (safe to use):
  {false && <Component />}     // renders nothing ✓
  {null && <Component />}      // renders nothing ✓
  {undefined && <Component />} // renders nothing ✓

  // Falsy values that DO render (be careful):
  {0 && <Component />}         // renders "0" ✗
  {'' && <Component />}        // renders "" (empty string, nothing visible but still a node) ✗

5. METHOD 4: LOOKUP OBJECTS (for multiple conditions)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Cleaner than chained ternaries for multiple states:

  const STATUS_COMPONENTS = {
    loading:  <Spinner />,
    error:    <ErrorView />,
    empty:    <EmptyState />,
    success:  <DataList />,
  };

  function Page({ status }) {
    return (
      <div>
        {STATUS_COMPONENTS[status] ?? <UnknownStatus />}
      </div>
    );
  }

  // With component mapping (dynamic components):
  const views = {
    home: HomeView,
    about: AboutView,
    contact: ContactView,
  };

  function Router({ currentPage, ...props }) {
    const View = views[currentPage] ?? NotFound;
    return <View {...props} />;
  }

Allows complex logic with if/switch inside JSX:

  function ComplexComponent({ user }) {
    return (
      <div>
        {(() => {
          switch(user.role) {
            case 'admin':    return <AdminPanel user={user} />;
            case 'editor':   return <EditorPanel user={user} />;
            case 'viewer':   return <ViewerPanel user={user} />;
            default:         return <GuestPanel />;
          }
        })()}
      </div>
    );
  }

  // NOTE: Prefer extracting to a separate component or variable for readability

7. PREVENTING COMPONENTS FROM RENDERING (return null)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Returning null from a component renders nothing but the component still
goes through mount/update/unmount lifecycle:

  function Alert({ message, visible }) {
    if (!visible) return null; // renders nothing
    return <div className="alert">{message}</div>;
  }

  // Note: returning null doesn't skip lifecycle effects
  // useEffect still runs even if component returns null

8. CONDITIONAL RENDERING vs MOUNTING/UNMOUNTING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Important distinction: showing/hiding vs mounting/unmounting

  // Conditionally MOUNT/UNMOUNT (component state is reset on hide)
  {isVisible && <Counter />}
  // When isVisible → false: Counter unmounts, state is lost
  // When isVisible → true: Counter mounts fresh with initial state

  // Show/HIDE via CSS (component stays mounted, state persists)
  <div style={{ display: isVisible ? 'block' : 'none' }}>
    <Counter />
  </div>
  // Counter always mounted, state preserved even when hidden

  // Which to use?
  // - Use unmounting when: component has heavy resources (subscriptions,
  //   large DOM trees) that should be freed when hidden
  // - Use CSS hiding when: you need to preserve state, or component is
  //   expensive to remount, or for animation (can't animate unmounted elements)

9. COMMON INTERVIEW QUESTIONS ON CONDITIONAL RENDERING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What are the different ways to do conditional rendering in React?
A: if/else before return, ternary operator (?:), logical AND (&&), lookup objects,
   IIFE inside JSX, and early return (guard clauses).

Q: What is the gotcha with && for conditional rendering?
A: If the left side is a falsy number (0) or empty string (''), React renders
   that value instead of nothing. Use boolean checks: count > 0 && ... instead
   of count && ...

Q: What is the difference between conditional rendering and CSS display:none?
A: Conditional rendering (using &&, ternary, or null) mounts/unmounts the component
   (state resets, lifecycle runs). CSS hiding keeps the component mounted (state
   preserved, no unmount/remount cost). Use CSS when preserving state matters.

Q: Can you use if statements inside JSX?
A: No. If statements are statements, not expressions. JSX only allows expressions
   inside {}. Use ternary, logical &&, or move the logic outside JSX.
