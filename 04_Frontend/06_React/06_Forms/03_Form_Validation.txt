  1. HTML5 native validation (required, minlength, pattern, type="email")
  2. Manual React validation (validate on submit or onChange)
  3. Library-based (React Hook Form built-in rules)
  4. Schema validation (Zod, Yup — define the shape and constraints)

  WHEN TO VALIDATE:
  - On submit (least intrusive, shows all errors at once)
  - On blur (validate when user leaves a field — good balance)
  - On change (most responsive but can feel aggressive)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  function validate(values) {
    const errors = {};

    if (!values.name.trim()) {
      errors.name = 'Name is required';
    } else if (values.name.trim().length < 2) {
      errors.name = 'Name must be at least 2 characters';
    }

    if (!values.email) {
      errors.email = 'Email is required';
    } else if (!/\S+@\S+\.\S+/.test(values.email)) {
      errors.email = 'Email is not valid';
    }

    if (!values.age) {
      errors.age = 'Age is required';
    } else if (isNaN(values.age) || values.age < 18) {
      errors.age = 'Must be 18 or older';
    }

    return errors;   // empty object = valid
  }

  function RegistrationForm() {
    const [values, setValues] = useState({ name: '', email: '', age: '' });
    const [errors, setErrors] = useState({});
    const [touched, setTouched] = useState({});

    function handleChange(e) {
      const { name, value } = e.target;
      setValues(v => ({ ...v, [name]: value }));
      // Validate on change but only for touched fields
      if (touched[name]) {
        const newErrors = validate({ ...values, [name]: value });
        setErrors(e => ({ ...e, [name]: newErrors[name] }));
      }
    }

    function handleBlur(e) {
      const { name, value } = e.target;
      setTouched(t => ({ ...t, [name]: true }));
      // Validate on blur
      const newErrors = validate({ ...values, [name]: value });
      setErrors(e => ({ ...e, [name]: newErrors[name] }));
    }

    function handleSubmit(e) {
      e.preventDefault();
      const allErrors = validate(values);
      setErrors(allErrors);
      setTouched({ name: true, email: true, age: true });
      if (Object.keys(allErrors).length === 0) {
        // Submit!
      }
    }

    return (
      <form onSubmit={handleSubmit}>
        <div>
          <input name="name" value={values.name}
            onChange={handleChange} onBlur={handleBlur} />
          {touched.name && errors.name && <span>{errors.name}</span>}
        </div>
        <div>
          <input name="email" type="email" value={values.email}
            onChange={handleChange} onBlur={handleBlur} />
          {touched.email && errors.email && <span>{errors.email}</span>}
        </div>
        <button type="submit">Register</button>
      </form>
    );
  }

  Checking if username is available (server call):

  // With React Hook Form:
  register('username', {
    required: 'Required',
    validate: async (value) => {
      const res = await fetch(`/api/check-username?username=${value}`);
      const { available } = await res.json();
      return available || 'Username is taken';
    },
  })

  // With debounce to avoid too many requests:
  const debouncedCheckUsername = useMemo(
    () => debounce(async (username, setError) => {
      const res = await fetch(`/api/check-username?username=${username}`);
      const { available } = await res.json();
      if (!available) setError('username', { message: 'Username is taken' });
    }, 500),
    [setError]
  );

  Best practices for accessibility:
  - Use aria-invalid to signal invalid state to screen readers
  - Use aria-describedby to link the input to its error message
  - Use role="alert" on error messages to announce them
  - Move focus to the first error on submit

  <div>
    <label htmlFor="email">Email</label>
    <input
      id="email"
      name="email"
      type="email"
      aria-invalid={!!errors.email}
      aria-describedby={errors.email ? 'email-error' : undefined}
      value={values.email}
      onChange={handleChange}
    />
    {errors.email && (
      <span id="email-error" role="alert" style={{ color: 'red' }}>
        {errors.email}
      </span>
    )}
  </div>

Q: What is the "touched" pattern in form validation?
A: Touched tracks which fields the user has interacted with. Without it,
   validation errors appear on all fields immediately on page load (bad UX).
   With touched, errors only appear for fields the user has visited and left.

Q: How do you handle async validation (e.g., checking if a username exists)?
A: Use React Hook Form's validate with an async function, or use setError to
   manually set a field error after an API response. Add debouncing to avoid
   firing requests on every keystroke.

Q: What is the difference between client-side and server-side validation?
A: Client-side validation gives immediate feedback before form submission.
   Server-side validation is the authoritative check (cannot be bypassed).
   Always implement BOTH — client-side for UX, server-side for security.

Q: Why should you always validate on the server too, even with client-side validation?
A: Client-side code can be bypassed by disabling JavaScript or sending
   requests directly to the API. Server-side validation is the only reliable
   security boundary.
