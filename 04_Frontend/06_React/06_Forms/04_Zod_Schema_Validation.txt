1. WHAT IS ZOD
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Zod is a TypeScript-first schema declaration and validation library.
It lets you define the shape and constraints of data once, then:
  - Validate data against the schema at runtime
  - Infer TypeScript types from the schema (no duplication)
  - Parse and transform data with type safety

  Install:
    npm install zod

  Key features:
  - No dependencies, tiny (~8KB gzipped)
  - Composable — schemas can be combined
  - Transformations — parse and transform in one step
  - Refinements — custom validation logic
  - Inference — TypeScript types from schema

2. BASIC SCHEMAS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  import { z } from 'zod';

  // Primitive schemas
  const nameSchema = z.string().min(2, 'Too short').max(50, 'Too long');
  const ageSchema  = z.number().int().min(18, 'Must be 18+').max(120);
  const emailSchema = z.string().email('Invalid email');
  const urlSchema  = z.string().url('Invalid URL');
  const boolSchema = z.boolean();
  const dateSchema = z.date();

  // Parse (throws on invalid) vs safeParse (returns { success, data/error })
  nameSchema.parse('Alice');           // 'Alice' — valid
  nameSchema.parse('A');               // ZodError: Too short

  const result = nameSchema.safeParse('A');
  if (!result.success) {
    console.log(result.error.issues);
    // [{ code: 'too_small', message: 'Too short', path: [] }]
  }

4. FORM SCHEMA WITH REFINEMENTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  const registerSchema = z
    .object({
      username: z
        .string()
        .min(3, 'Min 3 characters')
        .max(20, 'Max 20 characters')
        .regex(/^[a-zA-Z0-9_]+$/, 'Only letters, numbers, underscores'),

      email: z.string().email('Invalid email address'),

      password: z
        .string()
        .min(8, 'Min 8 characters')
        .regex(/[A-Z]/, 'Must contain uppercase letter')
        .regex(/[0-9]/, 'Must contain a number'),

      confirmPassword: z.string(),

      age: z.coerce.number().int().min(18, 'Must be 18 or older'),
      //    ↑ coerce: converts string "25" → number 25 (needed for HTML inputs)

      terms: z.literal(true, {
        errorMap: () => ({ message: 'You must accept the terms' }),
      }),
    })
    .refine(
      (data) => data.password === data.confirmPassword,
      {
        message: 'Passwords do not match',
        path: ['confirmPassword'],  // which field to show the error on
      }
    );

  type RegisterFormData = z.infer<typeof registerSchema>;

  import { zodResolver } from '@hookform/resolvers/zod';

  function RegisterForm() {
    const {
      register,
      handleSubmit,
      formState: { errors, isSubmitting },
    } = useForm<RegisterFormData>({
      resolver: zodResolver(registerSchema),
      defaultValues: {
        username: '',
        email: '',
        password: '',
        confirmPassword: '',
        age: undefined,
        terms: false,
      },
    });

    const onSubmit = async (data: RegisterFormData) => {
      // data is fully typed and validated
      await createUser(data);
    };

    return (
      <form onSubmit={handleSubmit(onSubmit)}>
        <input {...register('username')} placeholder="Username" />
        {errors.username && <p>{errors.username.message}</p>}

        <input {...register('email')} type="email" placeholder="Email" />
        {errors.email && <p>{errors.email.message}</p>}

        <input {...register('password')} type="password" placeholder="Password" />
        {errors.password && <p>{errors.password.message}</p>}

        <input {...register('confirmPassword')} type="password" placeholder="Confirm" />
        {errors.confirmPassword && <p>{errors.confirmPassword.message}</p>}

        <input {...register('age')} type="number" placeholder="Age" />
        {errors.age && <p>{errors.age.message}</p>}

        <label>
          <input {...register('terms')} type="checkbox" />
          I accept the terms
        </label>
        {errors.terms && <p>{errors.terms.message}</p>}

        <button type="submit" disabled={isSubmitting}>Register</button>
      </form>
    );

6. USEFUL ZOD METHODS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  z.string().optional()           // string | undefined
  z.string().nullable()           // string | null
  z.string().nullish()            // string | null | undefined
  z.string().default('fallback')  // provides default if undefined
  z.coerce.number()               // converts "42" → 42 (for HTML inputs)
  z.union([z.string(), z.number()]) // string | number
  z.discriminatedUnion('type', [  // efficient union with discriminator
    z.object({ type: z.literal('cat'), meows: z.boolean() }),
    z.object({ type: z.literal('dog'), barks: z.boolean() }),
  ])
  z.array(z.string()).min(1)      // non-empty array of strings
  z.record(z.string(), z.number()) // { [key: string]: number }
  z.enum(['a', 'b', 'c'])         // union of string literals
  z.nativeEnum(MyEnum)            // TypeScript enum

7. COMMON INTERVIEW QUESTIONS

Q: What is the advantage of Zod over writing validation manually?
A: Zod provides type inference (z.infer<typeof schema> gives you a TypeScript
   type), composable schemas, clear error messages, and automatic coercion.
   You define validation once and get both runtime validation and TypeScript types.

Q: What is the difference between parse and safeParse?
A: parse throws a ZodError on invalid input. safeParse returns
   { success: true, data } or { success: false, error } and never throws.
   Use safeParse when you want to handle errors gracefully.

Q: Why is z.coerce.number() needed for HTML form inputs?
A: HTML inputs always return strings. z.number() would fail on "25" (a string).
   z.coerce.number() converts the string to a number before validation.

Q: How do you validate cross-field constraints (e.g., passwords match)?
A: Use .refine() on the object schema. It receives the whole object, so you
   can compare multiple fields. Specify the path option to associate the error
   with a specific field.

Q: What is z.infer and why is it useful?
A: z.infer<typeof schema> extracts the TypeScript type corresponding to a Zod
   schema. This means you define your validation schema once and get free
   TypeScript types — no more keeping a type and a validation schema in sync.
