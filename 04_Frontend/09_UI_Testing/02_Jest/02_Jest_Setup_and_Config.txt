1. INSTALLATION
================================================================================

BASIC INSTALLATION:
  npm install --save-dev jest
  # or
  yarn add --dev jest

REACT + TYPESCRIPT + RTL (recommended stack):
  npm install --save-dev \
    jest \
    @types/jest \
    ts-jest \
    @testing-library/react \
    @testing-library/user-event \
    @testing-library/jest-dom \
    jest-environment-jsdom \
    identity-obj-proxy

FOR VITEST INSTEAD (Vite projects):
  npm install --save-dev \
    vitest \
    @testing-library/react \
    @testing-library/user-event \
    @testing-library/jest-dom \
    jsdom

2. TYPESCRIPT SETUP
================================================================================

OPTION A: ts-jest (Most Common)
  npm install --save-dev ts-jest @types/jest

  // jest.config.ts
  export default {
    preset: 'ts-jest',
    testEnvironment: 'jsdom',
  };

OPTION B: @swc/jest (Faster — uses Rust compiler)
  npm install --save-dev @swc/jest @swc/core

  // jest.config.ts
  export default {
    testEnvironment: 'jsdom',
    transform: {
      '^.+\\.(t|j)sx?$': ['@swc/jest'],
    },
  };

OPTION C: babel-jest with @babel/preset-typescript
  npm install --save-dev babel-jest @babel/preset-typescript @babel/preset-env

  // babel.config.js
  module.exports = {
    presets: [
      ['@babel/preset-env', { targets: { node: 'current' } }],
      '@babel/preset-typescript',
      ['@babel/preset-react', { runtime: 'automatic' }],
    ],
  };

TYPESCRIPT STRICT MODE IN TESTS:
  Configure ts-jest to use a specific tsconfig for tests:

  // jest.config.ts
  export default {
    preset: 'ts-jest',
    globals: {
      'ts-jest': {
        tsconfig: {
          jsx: 'react-jsx',
          strict: true,
        },
      },
    },
  };


================================================================================
3. BABEL SETUP (for CRA / non-TypeScript projects)
================================================================================

WHEN TO USE:
  - Create React App (uses babel-jest under the hood)
  - Pure JavaScript projects
  - Projects that already use Babel

INSTALLATION:
  npm install --save-dev \
    babel-jest \
    @babel/core \
    @babel/preset-env \
    @babel/preset-react

BABEL.CONFIG.JS:
  module.exports = {
    presets: [
      ['@babel/preset-env', {
        targets: { node: 'current' },
        modules: 'commonjs',  // Important: convert ESM to CJS for Jest
      }],
      ['@babel/preset-react', {
        runtime: 'automatic',  // React 17+ JSX transform
      }],
    ],
  };

CRA NOTE:
  If using Create React App, Jest is already configured. You don't need a
  jest.config.js — CRA manages it internally. You can extend it via:
    "jest" field in package.json (limited customization)
    react-app-rewired (for full control without ejecting)

4. JEST.CONFIG.TS FULL REFERENCE
================================================================================

import type { Config } from 'jest';

const config: Config = {
  // ─── Test Environment ───────────────────────────────────────────────────────
  // 'jsdom'  → Browser-like env (for React/DOM testing)
  // 'node'   → Node.js env (for server-side code, API routes)
  testEnvironment: 'jsdom',

  // ─── TypeScript Transform ───────────────────────────────────────────────────
  preset: 'ts-jest',
  // OR manual transform:
  // transform: { '^.+\\.tsx?$': 'ts-jest' }

  // ─── Setup Files ────────────────────────────────────────────────────────────
  // Run before each test file (before test framework is installed):
  setupFiles: ['./src/__mocks__/setupEnvVars.ts'],

  // Run after test framework is installed (can use expect(), etc.):
  setupFilesAfterFramework: ['./src/setupTests.ts'],

  // ─── Module Resolution ───────────────────────────────────────────────────────
  moduleNameMapper: {
    // CSS Modules
    '\\.(css|scss|sass|less)$': 'identity-obj-proxy',
    // Static files
    '\\.(jpg|jpeg|png|gif|svg|ico|webp|woff|woff2|eot|ttf|otf)$':
      '<rootDir>/src/__mocks__/fileMock.ts',
    // Path aliases (match your tsconfig.json "paths")
    '^@/(.*)$': '<rootDir>/src/$1',
    '^@components/(.*)$': '<rootDir>/src/components/$1',
    '^@hooks/(.*)$': '<rootDir>/src/hooks/$1',
    '^@utils/(.*)$': '<rootDir>/src/utils/$1',
  },

  // ─── File Extensions ─────────────────────────────────────────────────────────
  moduleFileExtensions: ['ts', 'tsx', 'js', 'jsx', 'json'],

  // ─── Test Discovery ──────────────────────────────────────────────────────────
  testMatch: [
    '**/__tests__/**/*.{ts,tsx}',
    '**/*.{test,spec}.{ts,tsx}',
  ],

  // Ignore these paths entirely
  testPathIgnorePatterns: [
    '/node_modules/',
    '/dist/',
    '/build/',
    '/coverage/',
    '/storybook-static/',
  ],

  // ─── Coverage ────────────────────────────────────────────────────────────────
  collectCoverage: false,  // Only collect when --coverage flag used
  collectCoverageFrom: [
    'src/**/*.{ts,tsx}',
    '!src/**/*.d.ts',
    '!src/**/*.test.{ts,tsx}',
    '!src/**/*.spec.{ts,tsx}',
    '!src/**/*.stories.{ts,tsx}',
    '!src/**/index.ts',             // Barrel exports
    '!src/main.tsx',                // App entry point
    '!src/vite-env.d.ts',
  ],
  coverageDirectory: 'coverage',
  coverageReporters: ['text', 'text-summary', 'lcov', 'html'],
  coverageThreshold: {
    global: {
      branches: 80,
      functions: 80,
      lines: 80,
      statements: 80,
    },
    // Per-file thresholds for critical modules:
    './src/utils/': {
      branches: 95,
      functions: 95,
      lines: 95,
    },
  },

  // ─── Performance ─────────────────────────────────────────────────────────────
  maxWorkers: '50%',       // Use 50% of CPU cores
  // maxWorkers: 4,         // Or use a fixed number

  // ─── Timeouts ────────────────────────────────────────────────────────────────
  testTimeout: 10000,      // 10 seconds per test (default: 5000)

  // ─── Globals ─────────────────────────────────────────────────────────────────
  globals: {
    'ts-jest': {
      tsconfig: './tsconfig.json',
      diagnostics: true,  // Show TypeScript errors in tests
    },
  },

  // ─── Verbose ─────────────────────────────────────────────────────────────────
  verbose: false,           // true = show each test name in output

  // ─── Custom Reporter ─────────────────────────────────────────────────────────
  // reporters: ['default', 'jest-junit'],  // For CI XML reports

  // ─── Module Directories ──────────────────────────────────────────────────────
  modulePaths: ['<rootDir>/src'],  // Allow imports relative to src/

  // ─── Cache ───────────────────────────────────────────────────────────────────
  cacheDirectory: '/tmp/jest_cache',
};

export default config;

5. SETUPTESTS.TS CONFIGURATION
================================================================================

This file runs after the test framework is loaded. It's where you:
  - Import jest-dom matchers
  - Set up MSW server
  - Set up global mocks
  - Configure custom matchers

// src/setupTests.ts

import '@testing-library/jest-dom';
import { server } from './mocks/server';

// ─── MSW Server Setup ───────────────────────────────────────────────────────
beforeAll(() => server.listen({ onUnhandledRequest: 'warn' }));
afterEach(() => server.resetHandlers());
afterAll(() => server.close());

// ─── Clean up after each test ───────────────────────────────────────────────
afterEach(() => {
  jest.clearAllMocks();
  localStorage.clear();
  sessionStorage.clear();
});

// ─── Mock window.matchMedia (not supported in jsdom) ────────────────────────
Object.defineProperty(window, 'matchMedia', {
  writable: true,
  value: jest.fn().mockImplementation(query => ({
    matches: false,
    media: query,
    onchange: null,
    addListener: jest.fn(),
    removeListener: jest.fn(),
    addEventListener: jest.fn(),
    removeEventListener: jest.fn(),
    dispatchEvent: jest.fn(),
  })),
});

// ─── Mock IntersectionObserver ───────────────────────────────────────────────
global.IntersectionObserver = class IntersectionObserver {
  observe = jest.fn();
  unobserve = jest.fn();
  disconnect = jest.fn();
} as any;

// ─── Mock ResizeObserver ────────────────────────────────────────────────────
global.ResizeObserver = class ResizeObserver {
  observe = jest.fn();
  unobserve = jest.fn();
  disconnect = jest.fn();
} as any;

// ─── Suppress console.error for known test warnings ─────────────────────────
const originalConsoleError = console.error;
beforeEach(() => {
  console.error = jest.fn();
});
afterEach(() => {
  console.error = originalConsoleError;
});

6. MODULE NAME MAPPING (CSS, SVG, ALIASES)
================================================================================

CSS MODULES:
  npm install --save-dev identity-obj-proxy

  // Behavior: className.module.css returns { container: 'container', ... }
  moduleNameMapper: {
    '\\.(css|scss|sass)$': 'identity-obj-proxy',
  }

STATIC FILE MOCK:
  // src/__mocks__/fileMock.ts
  export default 'test-file-stub';
  // or for SVG with React component:
  export const ReactComponent = () => null;
  export default 'test-file-stub';

  // jest.config.ts
  moduleNameMapper: {
    '\\.(jpg|png|gif|webp|ico)$': '<rootDir>/src/__mocks__/fileMock.ts',
    '\\.svg$': '<rootDir>/src/__mocks__/svgMock.ts',
  }

PATH ALIASES:
  // tsconfig.json has:
  "paths": {
    "@/*": ["./src/*"],
    "@components/*": ["./src/components/*"]
  }

  // jest.config.ts must mirror this:
  moduleNameMapper: {
    '^@/(.*)$': '<rootDir>/src/$1',
    '^@components/(.*)$': '<rootDir>/src/components/$1',
  }

7. ENVIRONMENT SETUP (jsdom vs node)
================================================================================

JSDOM ENVIRONMENT (default for React):
  - Provides: window, document, HTMLElement, fetch (in newer jsdom)
  - Missing by default: matchMedia, IntersectionObserver, ResizeObserver
  - Use for: Any component tests, browser-side logic

  // Per-file override using docblock:
  /**
   * @jest-environment jsdom
   */

NODE ENVIRONMENT:
  - Pure Node.js runtime
  - No window or document
  - Use for: API utilities, Node.js scripts, server-side code

  // Per-file override:
  /**
   * @jest-environment node
   */

MULTIPLE ENVIRONMENTS IN ONE PROJECT:
  // jest.config.ts
  projects: [
    {
      displayName: 'dom',
      testEnvironment: 'jsdom',
      testMatch: ['<rootDir>/src/components/**/*.test.tsx'],
    },
    {
      displayName: 'node',
      testEnvironment: 'node',
      testMatch: ['<rootDir>/src/api/**/*.test.ts'],
    },
  ]

8. COVERAGE CONFIGURATION
================================================================================

RUNNING COVERAGE:
  npx jest --coverage
  npx jest --coverage --coverageReporters=text

COVERAGE OUTPUT:
  ------------------|---------|----------|---------|---------|
  File              | % Stmts | % Branch | % Funcs | % Lines |
  ------------------|---------|----------|---------|---------|
  components/       |         |          |         |         |
    Button.tsx      |   100   |    100   |   100   |   100   |
    Form.tsx        |    82   |     75   |    80   |    82   |
  utils/            |         |          |         |         |
    formatDate.ts   |   100   |   85.71  |   100   |   100   |
  ------------------|---------|----------|---------|---------|

HTML REPORT:
  Open coverage/lcov-report/index.html in browser for visual report.

EXCLUDING FROM COVERAGE:
  Method 1: collectCoverageFrom excludes in jest.config.ts (recommended)

  Method 2: Inline ignore comments:
    /* istanbul ignore next */
    /* istanbul ignore if */
    /* istanbul ignore else */
    // Example:
    if (process.env.NODE_ENV === 'development') { /* istanbul ignore next */
      console.log('debug info');
    }

V8 COVERAGE (faster alternative):
  coverageProvider: 'v8',  // Instead of 'babel' (default)
  // v8 is faster but has less accurate branch coverage for some patterns

      },

NEXT.JS + JEST

  import nextJest from 'next/jest';
  const createJestConfig = nextJest({ dir: './' });

  const customJestConfig = {
    testEnvironment: 'jest-environment-jsdom',
    setupFilesAfterFramework: ['./jest.setup.ts'],
    moduleNameMapper: {
      '^@/(.*)$': '<rootDir>/src/$1',
    },
  };

  export default createJestConfig(customJestConfig);
