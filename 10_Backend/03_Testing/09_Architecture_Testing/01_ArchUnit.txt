ARCHUNIT — ARCHITECTURE TESTING
================================

=== CONCEPT ===

ArchUnit is a Java testing library to enforce architectural rules as unit tests.
It analyzes bytecode and verifies that code structure follows defined rules.

Why:
  - Architecture documentation gets outdated; tests don't
  - Enforce package boundaries (no circular dependencies)
  - Prevent layering violations (controllers calling repositories directly)
  - Ensure naming conventions are followed
  - Prevent forbidden imports (e.g., don't use Guava, use only standard Java)

=== DEPENDENCY ===

<dependency>
    <groupId>com.tngtech.archunit</groupId>
    <artifactId>archunit-junit5</artifactId>
    <version>1.2.1</version>
    <scope>test</scope>
</dependency>

=== BASIC SETUP ===

@AnalyzeClasses(packages = "com.example.myapp")  // your root package
public class ArchitectureTest {

    @ArchTest
    static final ArchRule noFieldInjection =
        noFields().should().beAnnotatedWith(Autowired.class)
            .because("Use constructor injection instead");

    @ArchTest
    static final ArchRule servicesShouldNotDependOnControllers =
        noClasses().that().resideInAPackage("..service..")
            .should().dependOnClassesThat()
            .resideInAPackage("..controller..");
}

=== LAYERED ARCHITECTURE ENFORCEMENT ===

@AnalyzeClasses(packages = "com.example.myapp")
public class LayeredArchitectureTest {

    @ArchTest
    static final ArchRule layeredArchitecture =
        layeredArchitecture().consideringAllDependencies()
            .layer("Web").definedBy("..web..")
            .layer("Service").definedBy("..service..")
            .layer("Persistence").definedBy("..repository..")
            .whereLayer("Web").mayNotBeAccessedByAnyLayer()
            .whereLayer("Service").mayOnlyBeAccessedByLayers("Web")
            .whereLayer("Persistence").mayOnlyBeAccessedByLayers("Service");
}

This enforces:
  Controller → Service → Repository
  Repository CANNOT import Service or Controller classes
  Service CANNOT import Controller classes

=== NAMING CONVENTION RULES ===

@AnalyzeClasses(packages = "com.example.myapp")
public class NamingConventionTest {

    @ArchTest
    static final ArchRule serviceNameRule =
        classes().that().resideInAPackage("..service..")
            .should().haveSimpleNameEndingWith("Service");

    @ArchTest
    static final ArchRule repositoryNameRule =
        classes().that().resideInAPackage("..repository..")
            .should().haveSimpleNameEndingWith("Repository");

    @ArchTest
    static final ArchRule controllerNameRule =
        classes().that().areAnnotatedWith(RestController.class)
            .should().haveSimpleNameEndingWith("Controller");

    @ArchTest
    static final ArchRule dtoNameRule =
        classes().that().resideInAPackage("..dto..")
            .should().haveSimpleNameEndingWith("Dto")
                .orShould().haveSimpleNameEndingWith("Request")
                .orShould().haveSimpleNameEndingWith("Response");
}

=== DEPENDENCY RULES ===

@AnalyzeClasses(packages = "com.example.myapp")
public class DependencyTest {

    // Controllers should only be in web package
    @ArchTest
    static final ArchRule controllersInWebPackage =
        classes().that().areAnnotatedWith(RestController.class)
            .should().resideInAPackage("..web..")
                .orShould().resideInAPackage("..controller..");

    // No class should use java.util.Date (use java.time instead)
    @ArchTest
    static final ArchRule noJavaUtilDate =
        noClasses().should().dependOnClassesThat()
            .haveFullyQualifiedName("java.util.Date")
            .because("Use java.time.LocalDate/LocalDateTime instead");

    // Repositories must be interfaces
    @ArchTest
    static final ArchRule repositoriesMustBeInterfaces =
        classes().that().resideInAPackage("..repository..")
            .and().haveSimpleNameEndingWith("Repository")
            .should().beInterfaces();

    // No circular dependencies between packages
    @ArchTest
    static final ArchRule noCircularDependencies =
        slices().matching("com.example.myapp.(*)..").should().beFreeOfCycles();
}

=== ANNOTATION RULES ===

@AnalyzeClasses(packages = "com.example.myapp")
public class AnnotationTest {

    // All controllers should be REST controllers (not @Controller)
    @ArchTest
    static final ArchRule controllerAnnotation =
        classes().that().haveSimpleNameEndingWith("Controller")
            .should().beAnnotatedWith(RestController.class);

    // All service classes should be @Transactional or methods should be
    @ArchTest
    static final ArchRule servicesAnnotated =
        classes().that().haveSimpleNameEndingWith("Service")
            .and().resideInAPackage("..service..")
            .should().beAnnotatedWith(Service.class);

    // No @Autowired on fields (enforce constructor injection)
    @ArchTest
    static final ArchRule noFieldInjection =
        noFields().should().beAnnotatedWith(Autowired.class);
}

=== FORBIDDEN IMPORTS ===

@AnalyzeClasses(packages = "com.example.myapp")
public class ForbiddenImportsTest {

    // Domain layer must not depend on infrastructure (hexagonal arch)
    @ArchTest
    static final ArchRule domainMustNotUseSpringSingletons =
        noClasses().that().resideInAPackage("..domain..")
            .should().dependOnClassesThat()
            .resideInAPackage("org.springframework..")
            .because("Domain layer must be framework-agnostic");

    // No Lombok in service/domain (use explicit code)
    @ArchTest
    static final ArchRule noLombokInDomain =
        noClasses().that().resideInAPackage("..domain..")
            .should().dependOnClassesThat()
            .resideInAPackage("lombok..");
}

=== HEXAGONAL ARCHITECTURE ===

@AnalyzeClasses(packages = "com.example.myapp")
public class HexagonalArchitectureTest {

    @ArchTest
    static final ArchRule hexagonalArchitecture =
        onionArchitecture()
            .domainModels("..domain.model..")
            .domainServices("..domain.service..")
            .applicationServices("..application..")
            .adapter("web", "..adapter.web..")
            .adapter("persistence", "..adapter.persistence..");
}

=== RUNNING ARCHUNIT ===

@ArchTest → runs as part of normal JUnit 5 test suite (./mvnw test)
Fast: analyzes bytecode, no Spring context needed
Can be run in CI/CD as standard unit tests

IDE: Results show as failed tests with clear violation descriptions.
