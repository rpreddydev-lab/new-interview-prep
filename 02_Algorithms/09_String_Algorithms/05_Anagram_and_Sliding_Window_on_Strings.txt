----------------------------------------------------------------
ANAGRAM CHECK (LC 242)
----------------------------------------------------------------
Two strings are anagrams if they have same characters with same frequencies.
  Sort both: O(n log n) -- simple
  Frequency array: O(n) -- optimal

  boolean isAnagram(String s, String t) {
    if (s.length() != t.length()) return false;
    int[] count = new int[26];
    for (char c : s.toCharArray()) count[c-'a']++;
    for (char c : t.toCharArray()) count[c-'a']--;
    for (int x : count) if (x != 0) return false;
    return true;
  }

----------------------------------------------------------------
FIND ALL ANAGRAMS IN STRING (LC 438)
----------------------------------------------------------------
Fixed-size sliding window with frequency tracking.
  Track `formed` = number of chars with correct frequency.

  List<Integer> findAnagrams(String s, String p) {
    int[] pCount = new int[26], wCount = new int[26];
    for (char c : p.toCharArray()) pCount[c-'a']++;
    List<Integer> res = new ArrayList<>();
    int need = 0;
    for (int i = 0; i < 26; i++) if (pCount[i] > 0) need++;
    int formed = 0, l = 0;
    for (int r = 0; r < s.length(); r++) {
      int rc = s.charAt(r) - 'a';
      wCount[rc]++;
      if (pCount[rc] > 0 && wCount[rc] == pCount[rc]) formed++;
      if (r - l + 1 == p.length()) {
        if (formed == need) res.add(l);
        int lc = s.charAt(l++) - 'a';
        if (pCount[lc] > 0 && wCount[lc] == pCount[lc]) formed--;
        wCount[lc]--;
      }
    }
    return res;
  }

----------------------------------------------------------------
MINIMUM WINDOW SUBSTRING (LC 76) -- Variable window
----------------------------------------------------------------
Find smallest window in s containing all chars of t.

  String minWindow(String s, String t) {
    int[] tCount = new int[128];
    for (char c : t.toCharArray()) tCount[c]++;
    int l=0, r=0, need=t.length(), minL=0, minLen=Integer.MAX_VALUE;
    while (r < s.length()) {
      if (tCount[s.charAt(r++)]-- > 0) need--;  // valid char from t
      while (need == 0) {                        // window is valid: try to shrink
        if (r - l < minLen) { minL = l; minLen = r - l; }
        if (tCount[s.charAt(l++)]++ == 0) need++;  // shrink: if needed char removed
      }
    }
    return minLen == Integer.MAX_VALUE ? "" : s.substring(minL, minL + minLen);
  }

  Key: tCount[c]-- > 0 means c was needed (decrement before compare)
       tCount[c]++ == 0 means c is exactly needed (we lost a needed char)

----------------------------------------------------------------
LONGEST SUBSTRING WITHOUT REPEATING (LC 3)
----------------------------------------------------------------
  int lengthOfLongestSubstring(String s) {
    int[] last = new int[128]; Arrays.fill(last, -1);
    int l = 0, res = 0;
    for (int r = 0; r < s.length(); r++) {
      char c = s.charAt(r);
      if (last[c] >= l) l = last[c] + 1;  // shrink left past last occurrence
      last[c] = r;
      res = Math.max(res, r - l + 1);
    }
    return res;
  }

----------------------------------------------------------------
INTERVIEW TIPS
----------------------------------------------------------------
  1. Fixed window: slide by removing left and adding right simultaneously
  2. Variable window: expand right; shrink left when valid; update answer
  3. Minimum Window: need counter tracks how many chars still needed
  4. tCount[c]-- > 0 pattern: elegant because check happens BEFORE decrement
  5. `last[c] >= l` handles character last seen BEFORE window start
