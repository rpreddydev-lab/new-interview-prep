================================================================
  GRAPH TRAVERSAL: DFS (DEPTH-FIRST SEARCH)
================================================================

----------------------------------------------------------------
CORE IDEA
----------------------------------------------------------------
Explore as deep as possible before backtracking.
Uses call stack (recursive) or explicit Deque (iterative).

  Data structure: Stack (recursion or ArrayDeque)
  Use cases: connected components, cycle detection,
             topological sort, strongly connected components

----------------------------------------------------------------
RECURSIVE DFS
----------------------------------------------------------------
  void dfs(List<List<Integer>> adj, int node, boolean[] visited) {
    visited[node] = true;
    System.out.println("visiting " + node);
    for (int neighbor : adj.get(node))
      if (!visited[neighbor])
        dfs(adj, neighbor, visited);
  }

  void dfsAll(List<List<Integer>> adj, int n) {  // handles disconnected graphs
    boolean[] visited = new boolean[n];
    for (int i = 0; i < n; i++)
      if (!visited[i]) dfs(adj, i, visited);
  }

  Time: O(V + E)  Space: O(V) for recursion stack

----------------------------------------------------------------
ITERATIVE DFS
----------------------------------------------------------------
  void dfsIterative(List<List<Integer>> adj, int start, int n) {
    boolean[] visited = new boolean[n];
    Deque<Integer> stack = new ArrayDeque<>();
    stack.push(start);
    while (!stack.isEmpty()) {
      int node = stack.pop();
      if (visited[node]) continue;  // iterative DFS: check after pop
      visited[node] = true;
      for (int neighbor : adj.get(node))
        if (!visited[neighbor]) stack.push(neighbor);
    }
  }

  Note: iterative DFS visits neighbors in REVERSE order vs recursive.
  For same order: push in reverse order (reverse neighbor list before pushing).

----------------------------------------------------------------
DFS ON A GRID
----------------------------------------------------------------
  int[][] grid;
  boolean[][] visited;
  int[] dr = {0,0,1,-1};  // direction rows
  int[] dc = {1,-1,0,0};  // direction cols

  void dfsGrid(int r, int c) {
    if (r < 0 || r >= grid.length || c < 0 || c >= grid[0].length) return;
    if (visited[r][c] || grid[r][c] == 0) return;
    visited[r][c] = true;
    for (int d = 0; d < 4; d++) dfsGrid(r + dr[d], c + dc[d]);
  }

----------------------------------------------------------------
INTERVIEW TIPS
----------------------------------------------------------------
  1. Recursive DFS: mark visited BEFORE recursing (on entry)
  2. Iterative DFS: check visited AFTER popping (can have duplicates in stack)
  3. DFS for cycle detection, topological sort, SCCs
  4. BFS for shortest path; DFS for everything else
  5. For very deep graphs/trees (> 10K nodes): use iterative to avoid
     StackOverflowError
