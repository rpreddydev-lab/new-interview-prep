================================================================
  KRUSKAL'S MINIMUM SPANNING TREE
================================================================

----------------------------------------------------------------
CORE IDEA
----------------------------------------------------------------
Builds MST by greedily adding the CHEAPEST EDGE that doesn't
create a cycle. Uses Union-Find to detect cycles.

  Steps:
  1. Sort ALL edges by weight.
  2. For each edge (u, v, w) in sorted order:
     - If u and v are in different components: add edge to MST, union them.
     - If same component: skip (would create cycle).
  3. Stop when V-1 edges added.

  Time: O(E log E) for sorting + O(E * alpha(V)) for Union-Find
       ≈ O(E log E)  Space: O(V)

----------------------------------------------------------------
UNION-FIND DATA STRUCTURE
----------------------------------------------------------------
  int[] parent, rank;
  void init(int n) {
    parent = new int[n]; rank = new int[n];
    for (int i = 0; i < n; i++) parent[i] = i;
  }
  int find(int x) {
    return parent[x] == x ? x : (parent[x] = find(parent[x]));  // path compression
  }
  boolean union(int x, int y) {
    int px = find(x), py = find(y);
    if (px == py) return false;  // same component
    if (rank[px] < rank[py]) parent[px] = py;
    else if (rank[px] > rank[py]) parent[py] = px;
    else { parent[py] = px; rank[px]++; }
    return true;
  }

----------------------------------------------------------------
IMPLEMENTATION
----------------------------------------------------------------
  int kruskalsMST(int n, int[][] edges) {
    Arrays.sort(edges, (a, b) -> a[2] - b[2]);  // sort by weight
    init(n);
    int totalCost = 0, edgesAdded = 0;
    for (int[] edge : edges) {
      if (union(edge[0], edge[1])) {
        totalCost += edge[2];
        edgesAdded++;
        if (edgesAdded == n - 1) break;  // MST complete
      }
    }
    return edgesAdded == n - 1 ? totalCost : -1;  // -1 if disconnected
  }

----------------------------------------------------------------
CONNECTING CITIES WITH MINIMUM COST (LC 1135)
----------------------------------------------------------------
  int minimumCost(int n, int[][] connections) {
    Arrays.sort(connections, (a,b) -> a[2] - b[2]);
    init(n + 1);  // 1-indexed
    int cost = 0, edges = 0;
    for (int[] c : connections) {
      if (union(c[0], c[1])) { cost += c[2]; edges++; }
      if (edges == n - 1) return cost;
    }
    return -1;
  }

----------------------------------------------------------------
INTERVIEW TIPS
----------------------------------------------------------------
  1. Kruskal's = sort edges + Union-Find (no PQ needed)
  2. Path compression in find() is essential for O(alpha(n)) per op
  3. Union by rank prevents tall trees
  4. Kruskal's preferred when edges are few (sparse graphs)
  5. Can verify MST weight = sum; verify edges added = n-1
