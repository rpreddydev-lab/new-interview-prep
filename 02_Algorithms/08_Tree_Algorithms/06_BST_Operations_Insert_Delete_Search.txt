================================================================
  BST OPERATIONS: INSERT, DELETE, SEARCH
================================================================

----------------------------------------------------------------
BST PROPERTY
----------------------------------------------------------------
  For every node: left subtree < node < right subtree.
  Inorder traversal of BST = sorted sequence.
  Search, Insert, Delete: O(h) where h = height.
  Balanced BST (AVL/Red-Black): h = O(log n).
  Degenerate BST (sorted insertion): h = O(n).

----------------------------------------------------------------
SEARCH
----------------------------------------------------------------
  TreeNode search(TreeNode root, int val) {
    if (root == null || root.val == val) return root;
    return val < root.val ? search(root.left, val) : search(root.right, val);
  }
  // Iterative:
  while (root != null && root.val != val)
    root = val < root.val ? root.left : root.right;

----------------------------------------------------------------
INSERT
----------------------------------------------------------------
  TreeNode insert(TreeNode root, int val) {
    if (root == null) return new TreeNode(val);
    if (val < root.val) root.left = insert(root.left, val);
    else if (val > root.val) root.right = insert(root.right, val);
    return root;  // return root unchanged (val == root.val: duplicate ignored)
  }

----------------------------------------------------------------
DELETE (most complex BST operation)
----------------------------------------------------------------
  Three cases:
  1. No children (leaf): simply remove.
  2. One child: replace node with its child.
  3. Two children: replace with INORDER SUCCESSOR (smallest in right subtree),
     then delete the successor.

  TreeNode delete(TreeNode root, int val) {
    if (root == null) return null;
    if (val < root.val) { root.left = delete(root.left, val); }
    else if (val > root.val) { root.right = delete(root.right, val); }
    else {  // found node to delete
      if (root.left == null) return root.right;   // cases 1 and 2 (left missing)
      if (root.right == null) return root.left;   // case 2 (right missing)
      // case 3: find inorder successor (min of right subtree)
      TreeNode successor = findMin(root.right);
      root.val = successor.val;                   // copy successor value
      root.right = delete(root.right, successor.val);  // delete successor
    }
    return root;
  }
  TreeNode findMin(TreeNode node) {
    while (node.left != null) node = node.left;
    return node;
  }

----------------------------------------------------------------
BST VALIDATION (LC 98)
----------------------------------------------------------------
Check if tree is a valid BST.
  Use min/max bounds (not just left < node < right -- that's wrong for ancestors).

  boolean isValidBST(TreeNode root) { return validate(root, Long.MIN_VALUE, Long.MAX_VALUE); }
  boolean validate(TreeNode node, long min, long max) {
    if (node == null) return true;
    if (node.val <= min || node.val >= max) return false;
    return validate(node.left, min, node.val) && validate(node.right, node.val, max);
  }
  // Use Long.MIN/MAX to handle Integer.MIN/MAX edge cases

----------------------------------------------------------------
INTERVIEW TIPS
----------------------------------------------------------------
  1. BST delete: three cases; case 3 uses inorder successor (min of right)
  2. BST validation: use long bounds to avoid Integer overflow edge case
  3. Common mistake: checking only direct parent, not all ancestors
  4. Inorder = sorted: use for kth-smallest, validate, range queries
  5. Average O(log n) but worst O(n) for degenerate BST -- mention balanced variants
