GENERIC INTERFACES AND FUNCTIONAL INTERFACES
==============================================

GENERIC INTERFACES
-------------------
  An interface can have one or more type parameters, just like a class.
  Any class implementing the interface must bind those type parameters.

  // Simple generic interface
  public interface Transformer<T, R> {
      R transform(T input);
  }

  // Implementation: bind both type parameters
  class StringToInt implements Transformer<String, Integer> {
      @Override
      public Integer transform(String input) { return input.length(); }
  }

  // Implementation: keep one generic, bind another
  class StringTransformer<R> implements Transformer<String, R> {
      private final Function<String, R> fn;
      public StringTransformer(Function<String, R> fn) { this.fn = fn; }
      @Override
      public R transform(String input) { return fn.apply(input); }
  }

FUNCTIONAL INTERFACES WITH GENERICS
--------------------------------------
  Java 8 introduced java.util.function package — a collection of ready-made
  functional interfaces, all parameterized with generics.

  Key interfaces:
  ┌─────────────────────────┬────────────────────────────┬───────────────────────────┐
  │ Interface               │ Signature                  │ Use case                  │
  ├─────────────────────────┼────────────────────────────┼───────────────────────────┤
  │ Function<T, R>          │ R apply(T t)               │ Transform T → R           │
  │ BiFunction<T, U, R>     │ R apply(T t, U u)          │ Combine two inputs → R    │
  │ Consumer<T>             │ void accept(T t)           │ Do something with T       │
  │ BiConsumer<T, U>        │ void accept(T t, U u)      │ Do something with T and U │
  │ Supplier<T>             │ T get()                    │ Produce a T               │
  │ Predicate<T>            │ boolean test(T t)          │ Test a T                  │
  │ BiPredicate<T, U>       │ boolean test(T t, U u)     │ Test T and U              │
  │ UnaryOperator<T>        │ T apply(T t)               │ Transform T → T           │
  │ BinaryOperator<T>       │ T apply(T t1, T t2)        │ Combine two T → T         │
  └─────────────────────────┴────────────────────────────┴───────────────────────────┘

  // UnaryOperator extends Function<T,T>
  // BinaryOperator extends BiFunction<T,T,T>

FUNCTION<T, R>
---------------
  @FunctionalInterface
  public interface Function<T, R> {
      R apply(T t);
      default <V> Function<V, R> compose(Function<? super V, ? extends T> before) { ... }
      default <V> Function<T, V> andThen(Function<? super R, ? extends V> after) { ... }
      static <T> Function<T, T> identity() { return t -> t; }
  }

  // Basic usage
  Function<String, Integer> length = String::length;
  System.out.println(length.apply("hello")); // 5

  // Composing functions
  Function<String, String> trim = String::trim;
  Function<String, String> upper = String::toUpperCase;
  Function<String, String> trimThenUpper = trim.andThen(upper);
  System.out.println(trimThenUpper.apply("  hello  ")); // "HELLO"

  Function<Integer, String> intToStr = i -> "val:" + i;
  Function<String, Integer> strLen = String::length;
  Function<Integer, Integer> composed = intToStr.andThen(strLen);
  System.out.println(composed.apply(42)); // 6 ("val:42".length())

PREDICATE<T>
-------------
  @FunctionalInterface
  public interface Predicate<T> {
      boolean test(T t);
      default Predicate<T> and(Predicate<? super T> other) { ... }
      default Predicate<T> or(Predicate<? super T> other) { ... }
      default Predicate<T> negate() { ... }
      static <T> Predicate<T> not(Predicate<? super T> target) { ... }  // Java 11+
  }

  Predicate<String> notEmpty = s -> !s.isEmpty();
  Predicate<String> longerThan5 = s -> s.length() > 5;
  Predicate<String> longAndNotEmpty = notEmpty.and(longerThan5);

  List<String> words = Arrays.asList("", "hi", "hello", "goodbye");
  words.stream().filter(longAndNotEmpty).forEach(System.out::println);
  // hello, goodbye

  // Predicate.not (Java 11)
  words.stream().filter(Predicate.not(String::isEmpty)).forEach(System.out::println);

CONSUMER<T> AND SUPPLIER<T>
-----------------------------
  // Consumer — does something with input, returns void
  Consumer<String> print = System.out::println;
  Consumer<String> printUpperCase = s -> System.out.println(s.toUpperCase());
  Consumer<String> both = print.andThen(printUpperCase); // chain consumers

  List<String> names = List.of("Alice", "Bob");
  names.forEach(print);           // Alice, Bob
  names.forEach(both);            // Alice\nALICE\nBob\nBOB

  // Supplier — produces value without input
  Supplier<List<String>> listFactory = ArrayList::new;
  List<String> list1 = listFactory.get(); // new ArrayList
  List<String> list2 = listFactory.get(); // another new ArrayList

  // Lazy initialization pattern
  private String expensiveField;
  private final Supplier<String> supplier = () -> computeExpensiveValue();
  public String getField() {
      if (expensiveField == null) expensiveField = supplier.get();
      return expensiveField;
  }

DEFINING CUSTOM FUNCTIONAL INTERFACES
---------------------------------------
  // When built-in interfaces don't fit (different signatures)
  @FunctionalInterface
  public interface TriFunction<A, B, C, R> {
      R apply(A a, B b, C c);
  }

  TriFunction<String, Integer, Boolean, String> fn = (s, n, b) -> s + n + b;
  System.out.println(fn.apply("x=", 5, true)); // x=5true

  @FunctionalInterface
  public interface ThrowingSupplier<T> {
      T get() throws Exception;  // allows checked exceptions
  }

  ThrowingSupplier<Connection> connSupplier = () -> DriverManager.getConnection(url);

GENERIC INTERFACE WITH SELF-REFERENTIAL BOUND
----------------------------------------------
  // Builder pattern — enables fluent API with correct return type in subclasses
  public interface Builder<T, B extends Builder<T, B>> {
      B withName(String name);
      T build();
  }

  // Self-referential enum type:
  public abstract class Enum<E extends Enum<E>> implements Comparable<E>, Serializable {
      // E is always the concrete enum type itself
  }
  // This guarantees: MyEnum.compareTo() compares MyEnum with MyEnum (not with other enums)

COMBINING FUNCTIONAL INTERFACES
---------------------------------
  // Pipeline using Function composition
  Function<String, String[]> split = s -> s.split(",");
  Function<String[], List<String>> toList = Arrays::asList;
  Function<String, List<String>> csvToList = split.andThen(toList);

  List<String> parts = csvToList.apply("a,b,c"); // [a, b, c]

  // Composing predicates for validation
  public static <T> Predicate<T> allOf(List<Predicate<T>> predicates) {
      return predicates.stream().reduce(t -> true, Predicate::and);
  }

  List<Predicate<String>> validators = Arrays.asList(
      s -> !s.isEmpty(),
      s -> s.length() <= 20,
      s -> s.matches("[a-zA-Z0-9]+")
  );
  Predicate<String> combined = allOf(validators);
  System.out.println(combined.test("Alice123")); // true
  System.out.println(combined.test(""));         // false

COMMON INTERVIEW QUESTIONS
---------------------------
Q: What is a functional interface and what makes it functional?
A: A functional interface has EXACTLY ONE abstract method. It may have default and static methods.
   The @FunctionalInterface annotation is optional but recommended — compiler enforces the constraint.
   Functional interfaces are the target types for lambda expressions and method references.
   Examples: Runnable, Callable, Comparator, Function, Predicate, Consumer, Supplier.

Q: What is the difference between Function<T,R> and UnaryOperator<T>?
A: UnaryOperator<T> extends Function<T,T> — it's a special case where input and output
   types are the SAME. Function<T,R> allows input and output to be different types.
   Use UnaryOperator when transforming a type into itself (e.g., String → String, int → int).

Q: How do you chain multiple Predicates?
A: Using default methods: and(), or(), negate().
   pred1.and(pred2): both must be true
   pred1.or(pred2):  either can be true
   pred.negate():    logical NOT
   Java 11: Predicate.not(pred) — static method for negation (useful with method references).

Q: What is the difference between Consumer and Function?
A: Function<T,R>: takes T, returns R. Used for transformation.
   Consumer<T>: takes T, returns void. Used for side effects (print, save, update).
   Consumers can be chained with andThen(); the chain runs both for side effects.
