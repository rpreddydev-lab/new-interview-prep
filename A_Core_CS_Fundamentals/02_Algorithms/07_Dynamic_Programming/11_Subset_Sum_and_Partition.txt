================================================================
  SUBSET SUM & PARTITION

----------------------------------------------------------------
SUBSET SUM
----------------------------------------------------------------
Given an array of non-negative integers and target S,
determine if there's a subset summing to S.

  Boolean DP (0-1 knapsack variant):
  dp[i] = can sum i be achieved?
  dp[0] = true; Reverse iteration for 0-1 (no reuse).

  boolean subsetSum(int[] nums, int target) {
    boolean[] dp = new boolean[target + 1];
    dp[0] = true;
    for (int num : nums)
      for (int i = target; i >= num; i--)  // REVERSE for 0-1
        dp[i] |= dp[i - num];
    return dp[target];
  }

  Time: O(n * target)  Space: O(target)

----------------------------------------------------------------
PARTITION EQUAL SUBSET SUM (LC 416)
----------------------------------------------------------------
Divide array into two subsets with equal sum.
  1. If total is odd: impossible (return false)
  2. If any element > total/2: impossible
  3. Find subset summing to total/2 (subset sum)

  boolean canPartition(int[] nums) {
    int total = Arrays.stream(nums).sum();
    if (total % 2 != 0) return false;
    int target = total / 2;
    boolean[] dp = new boolean[target + 1];
    dp[0] = true;
    for (int num : nums) {
      for (int i = target; i >= num; i--)  // reverse for 0-1
        dp[i] |= dp[i - num];
      if (dp[target]) return true;  // early exit
    }
    return dp[target];
  }

  Time: O(n * total/2)  Space: O(total)

----------------------------------------------------------------
LAST STONE WEIGHT II (LC 1049)
----------------------------------------------------------------
Divide stones into two groups, minimize |sum1 - sum2|.
  Minimize |sum1 - sum2| = minimize |total - 2 * sum1|
  Maximize sum1 <= total/2 = subset sum optimization.
  Answer = total - 2 * (max sum1 <= total/2)

  int lastStoneWeightII(int[] stones) {
    int total = Arrays.stream(stones).sum();
    int target = total / 2;
    boolean[] dp = new boolean[target + 1];
    dp[0] = true;
    for (int s : stones)
      for (int i = target; i >= s; i--)
        dp[i] |= dp[i - s];
    for (int i = target; i >= 0; i--)  // find max achievable sum <= target
      if (dp[i]) return total - 2 * i;
    return 0;
  }

----------------------------------------------------------------
TARGET SUM (LC 494)
----------------------------------------------------------------
Assign + or - to each number, count ways to reach target.
  sum(P) - sum(N) = target; sum(P) + sum(N) = total
  sum(P) = (target + total) / 2
  Count subsets with sum (target + total) / 2

  int findTargetSumWays(int[] nums, int target) {
    int total = Arrays.stream(nums).sum();
    if ((total + target) % 2 != 0 || Math.abs(target) > total) return 0;
    int S = (total + target) / 2;
    int[] dp = new int[S + 1];
    dp[0] = 1;
    for (int num : nums)
      for (int i = S; i >= num; i--)  // count, reverse for 0-1
        dp[i] += dp[i - num];
    return dp[S];
  }

----------------------------------------------------------------
INTERVIEW TIPS
----------------------------------------------------------------
  1. Subset sum: boolean dp[], 0-1 -> reverse iteration
  2. Count variants: int dp[] += (instead of |=)
  3. Partition Equal Subset Sum: target = total/2, must check total is even
  4. Last Stone Weight II: elegant transform to subset sum
  5. Target Sum: algebraic transform sum(P) = (target + total)/2
     -- recognize this pattern for any problem with +/- assignment
