================================================================
  SERIALIZE AND DESERIALIZE BINARY TREE (LC 297)
================================================================

----------------------------------------------------------------
CORE CONCEPT
----------------------------------------------------------------
Serialization: convert tree to string/array.
Deserialization: reconstruct tree from string/array.

Preorder DFS with null markers is the simplest approach.
  Preorder guarantees root always appears FIRST (needed for rebuild).
  Inorder alone cannot reconstruct; need preorder + inorder OR null markers.

----------------------------------------------------------------
DFS PREORDER (PREFERRED)
----------------------------------------------------------------
  class Codec {
    public String serialize(TreeNode root) {
      StringBuilder sb = new StringBuilder();
      serDFS(root, sb);
      return sb.toString();
    }
    private void serDFS(TreeNode node, StringBuilder sb) {
      if (node == null) { sb.append("#,"); return; }
      sb.append(node.val).append(',');
      serDFS(node.left, sb);
      serDFS(node.right, sb);
    }

    public TreeNode deserialize(String data) {
      Queue<String> queue = new ArrayDeque<>(Arrays.asList(data.split(",")));
      return desDFS(queue);
    }
    private TreeNode desDFS(Queue<String> queue) {
      String val = queue.poll();
      if (val.equals("#")) return null;
      TreeNode node = new TreeNode(Integer.parseInt(val));
      node.left = desDFS(queue);
      node.right = desDFS(queue);
      return node;
    }
  }

  Time: O(n) serialize, O(n) deserialize  Space: O(n)

----------------------------------------------------------------
BFS LEVEL ORDER APPROACH (LeetCode's actual format)
----------------------------------------------------------------
  Serialize: BFS, write "null" for null nodes.
  Deserialize: BFS, pair each non-null node with next two tokens as children.

  public String serialize(TreeNode root) {
    if (root == null) return "";
    StringBuilder sb = new StringBuilder();
    Queue<TreeNode> q = new ArrayDeque<>();
    q.offer(root);
    while (!q.isEmpty()) {
      TreeNode node = q.poll();
      if (node == null) { sb.append("null,"); continue; }
      sb.append(node.val).append(',');
      q.offer(node.left); q.offer(node.right);
    }
    return sb.toString();
  }

----------------------------------------------------------------
RECONSTRUCT FROM PREORDER + INORDER (LC 105)
----------------------------------------------------------------
  TreeNode build(int[] preorder, int preStart, int[] inorder, int inStart, int inEnd,
                 Map<Integer, Integer> inMap) {
    if (preStart >= preorder.length || inStart > inEnd) return null;
    int rootVal = preorder[preStart];
    TreeNode root = new TreeNode(rootVal);
    int mid = inMap.get(rootVal);  // position in inorder
    int leftSize = mid - inStart;
    root.left = build(preorder, preStart+1, inorder, inStart, mid-1, inMap);
    root.right = build(preorder, preStart+leftSize+1, inorder, mid+1, inEnd, inMap);
    return root;
  }
  // HashMap maps inorder value -> index for O(1) lookup

----------------------------------------------------------------
INTERVIEW TIPS
----------------------------------------------------------------
  1. Preorder DFS with '#' markers: cleanest serialization
  2. Key insight: null markers make preorder uniquely decodable
  3. Inorder alone = ambiguous (can't reconstruct); need two traversals or markers
  4. Preorder + Inorder reconstruction: inorder split = left/right size
  5. HashMap for inorder index lookups: reduces O(n^2) to O(n)
