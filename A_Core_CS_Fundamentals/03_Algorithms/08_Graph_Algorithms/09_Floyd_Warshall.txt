================================================================
  FLOYD-WARSHALL ALL-PAIRS SHORTEST PATH
================================================================

----------------------------------------------------------------
CORE IDEA
----------------------------------------------------------------
All-pairs shortest path: find shortest path between EVERY pair of nodes.
Handles negative weights. Detects negative cycles.

  dp[i][j] = shortest path from i to j
  Key: try every intermediate node k:
    dp[i][j] = min(dp[i][j], dp[i][k] + dp[k][j])

  Three nested loops: for k, for i, for j (k must be OUTERMOST)
  Time: O(V^3)  Space: O(V^2)

  Best for DENSE graphs or when all-pairs result is needed.
  For sparse graphs: run Dijkstra from every vertex = O(V * E log V).

----------------------------------------------------------------
IMPLEMENTATION
----------------------------------------------------------------
  int[][] floydWarshall(int n, int[][] dist) {
    // dist[i][j] = weight of edge i->j, INF if no edge, 0 if i==j
    int INF = Integer.MAX_VALUE / 2;  // avoid overflow when adding
    for (int k = 0; k < n; k++)      // k MUST be outermost loop
      for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
          if (dist[i][k] + dist[k][j] < dist[i][j])
            dist[i][j] = dist[i][k] + dist[k][j];
    // Negative cycle check: dist[i][i] < 0 for any i
    return dist;
  }

  void initDistMatrix(int n, int[][] edges, int[][] dist) {
    int INF = Integer.MAX_VALUE / 2;
    for (int[] row : dist) Arrays.fill(row, INF);
    for (int i = 0; i < n; i++) dist[i][i] = 0;
    for (int[] e : edges) dist[e[0]][e[1]] = e[2];  // directed edge
  }

----------------------------------------------------------------
NEGATIVE CYCLE DETECTION
----------------------------------------------------------------
After running Floyd-Warshall:
  if dist[i][i] < 0 for any i --> i is on a negative cycle.

----------------------------------------------------------------
FIND THE CITY (LC 1334)
----------------------------------------------------------------
Find city with fewest reachable cities within distance threshold.
  Run Floyd-Warshall. For each city, count how many cities are
  reachable (dist <= threshold). Return city with minimum count
  (ties: largest city number).

  int findTheCity(int n, int[][] edges, int distanceThreshold) {
    int INF = Integer.MAX_VALUE / 2;
    int[][] dist = new int[n][n];
    for (int[] row : dist) Arrays.fill(row, INF);
    for (int i = 0; i < n; i++) dist[i][i] = 0;
    for (int[] e : edges) {
      dist[e[0]][e[1]] = e[2]; dist[e[1]][e[0]] = e[2]; // undirected
    }
    for (int k = 0; k < n; k++)
      for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
          dist[i][j] = Math.min(dist[i][j], dist[i][k] + dist[k][j]);
    int res = 0, minCount = n;
    for (int i = 0; i < n; i++) {
      int count = 0;
      for (int j = 0; j < n; j++) if (i != j && dist[i][j] <= distanceThreshold) count++;
      if (count <= minCount) { minCount = count; res = i; }
    }
    return res;
  }

----------------------------------------------------------------
INTERVIEW TIPS
----------------------------------------------------------------
  1. Floyd-Warshall: k MUST be the outermost loop (common bug: wrong loop order)
  2. INF = MAX_VALUE/2 (not MAX_VALUE!) to avoid addition overflow
  3. Negative cycles: check dist[i][i] < 0 after running
  4. For sparse graph + all-pairs: prefer V * Dijkstra over Floyd-Warshall
  5. Use case: small dense graphs (V <= 500), when all-pairs needed
