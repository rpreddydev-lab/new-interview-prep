GENERICS AND COLLECTIONS
==========================

HOW GENERICS POWER THE COLLECTIONS FRAMEWORK
----------------------------------------------
  The Java Collections Framework (JCF) is almost entirely built on generics.
  Understanding this connection demonstrates mastery of both topics.

  // Core interfaces — all parameterized
  Collection<E>
  List<E> extends Collection<E>
  Set<E>  extends Collection<E>
  Map<K,V>
  Comparable<T>
  Comparator<T>
  Iterator<E>

  // Key utility methods in Collections class — generic methods
  static <T extends Comparable<? super T>> void sort(List<T> list)
  static <T> void sort(List<T> list, Comparator<? super T> c)
  static <T> int binarySearch(List<? extends Comparable<? super T>> list, T key)
  static <T extends Object & Comparable<? super T>> T min(Collection<? extends T> coll)
  static <T> T min(Collection<? extends T> coll, Comparator<? super T> comp)

WRITING GENERIC COLLECTION UTILITIES
--------------------------------------
  // Find element by predicate
  public static <T> Optional<T> findFirst(Collection<T> collection, Predicate<T> predicate) {
      for (T element : collection) {
          if (predicate.test(element)) return Optional.of(element);
      }
      return Optional.empty();
  }

  List<String> names = List.of("Alice", "Bob", "Charlie");
  Optional<String> found = findFirst(names, s -> s.startsWith("B"));
  found.ifPresent(System.out::println); // Bob

  // Filter to new list
  public static <T> List<T> filter(List<T> list, Predicate<T> predicate) {
      List<T> result = new ArrayList<>();
      for (T item : list) {
          if (predicate.test(item)) result.add(item);
      }
      return result;
  }

  // Transform list (map operation)
  public static <T, R> List<R> map(List<T> list, Function<T, R> mapper) {
      List<R> result = new ArrayList<>(list.size());
      for (T item : list) result.add(mapper.apply(item));
      return result;
  }

  List<Integer> lengths = map(names, String::length); // [5, 3, 7]

  // Group by key
  public static <T, K> Map<K, List<T>> groupBy(Collection<T> collection, Function<T, K> classifier) {
      Map<K, List<T>> result = new HashMap<>();
      for (T item : collection) {
          K key = classifier.apply(item);
          result.computeIfAbsent(key, k -> new ArrayList<>()).add(item);
      }
      return result;
  }

  Map<Integer, List<String>> byLength = groupBy(names, String::length);
  // {5=[Alice], 3=[Bob], 7=[Charlie]}

GENERIC STACK IMPLEMENTATION
------------------------------
  public class GenericStack<T> {
      private final Deque<T> deque = new ArrayDeque<>();

      public void push(T item) { deque.addFirst(item); }
      public T pop()           { return deque.removeFirst(); }
      public T peek()          { return deque.peekFirst(); }
      public boolean isEmpty() { return deque.isEmpty(); }
      public int size()        { return deque.size(); }

      @Override
      public String toString() { return deque.toString(); }
  }

  GenericStack<String> stack = new GenericStack<>();
  stack.push("A"); stack.push("B"); stack.push("C");
  System.out.println(stack.pop()); // C
  System.out.println(stack.peek()); // B

GENERIC GRAPH (ADJACENCY LIST)
--------------------------------
  public class Graph<T> {
      private final Map<T, List<T>> adjacency = new HashMap<>();

      public void addVertex(T vertex) {
          adjacency.putIfAbsent(vertex, new ArrayList<>());
      }

      public void addEdge(T from, T to) {
          adjacency.computeIfAbsent(from, k -> new ArrayList<>()).add(to);
      }

      public List<T> neighbors(T vertex) {
          return adjacency.getOrDefault(vertex, Collections.emptyList());
      }

      public Set<T> bfs(T start) {
          Set<T> visited = new LinkedHashSet<>();
          Queue<T> queue = new ArrayDeque<>();
          queue.offer(start);
          visited.add(start);
          while (!queue.isEmpty()) {
              T current = queue.poll();
              for (T neighbor : neighbors(current)) {
                  if (visited.add(neighbor)) queue.offer(neighbor);
              }
          }
          return visited;
      }
  }

  Graph<String> g = new Graph<>();
  g.addEdge("A", "B"); g.addEdge("A", "C"); g.addEdge("B", "D");
  System.out.println(g.bfs("A")); // [A, B, C, D]

WILDCARDS IN COLLECTION METHODS
---------------------------------
  // Correct: accepts List<Integer>, List<Double>, etc.
  public static double sum(List<? extends Number> list) {
      return list.stream().mapToDouble(Number::doubleValue).sum();
  }

  // Correct: accepts List<Number>, List<Object>, etc. — can add Integer
  public static void fill(List<? super Integer> list, int value, int count) {
      for (int i = 0; i < count; i++) list.add(value);
  }

  // Collections.copy uses both:
  // dest must accept elements of same type as src
  public static <T> void copy(List<? super T> dest, List<? extends T> src) {
      for (T item : src) dest.add(item);
  }

GENERIC TYPE TOKENS (CLASS<T>)
--------------------------------
  Class<T> objects carry type information at runtime — useful for reflection
  and frameworks that need to work around type erasure.

  public class TypeSafeMap {
      private final Map<Class<?>, Object> map = new HashMap<>();

      public <T> void put(Class<T> type, T value) {
          map.put(type, value);
      }

      public <T> T get(Class<T> type) {
          return type.cast(map.get(type)); // safe cast using Class.cast()
      }
  }

  TypeSafeMap tsm = new TypeSafeMap();
  tsm.put(String.class, "hello");
  tsm.put(Integer.class, 42);

  String s = tsm.get(String.class);   // "hello" — no cast warning
  Integer n = tsm.get(Integer.class); // 42

  // Class<T> as factory:
  public static <T> T createInstance(Class<T> type) throws Exception {
      return type.getDeclaredConstructor().newInstance();
  }
  Person p = createInstance(Person.class);

COMMON INTERVIEW QUESTIONS
---------------------------
Q: How do Collections.sort() use generics and wildcards?
A: The signature is:
     public static <T extends Comparable<? super T>> void sort(List<T> list)
   T must implement Comparable. The <? super T> allows T's compareTo to be defined
   on a supertype — so Animal can define compareTo, and Dog (extends Animal) can be sorted
   using Animal's natural ordering without explicitly implementing Comparable<Dog>.

Q: What is an unbounded wildcard useful for in collection methods?
A: Unbounded wildcard List<?> is useful when:
   - The method only needs to read elements as Object (printing, size check)
   - You want to accept a Collection of ANY type (most flexible signature)
   - You don't add to the collection (read-only operations)
   Example: static int count(Collection<?> c) { return c.size(); }

Q: How do you implement a type-safe heterogeneous container?
A: Use Map<Class<?>, Object> as the backing store (Joshua Bloch's "typesafe heterogeneous
   container" pattern). Store with Class<T> as key and T as value. Retrieve using
   Class.cast(map.get(type)) — safe unchecked cast. The type token (Class<T>) carries
   the type information at runtime, working around erasure.

Q: Why does Java use invariant generics instead of covariant (like arrays)?
A: Arrays are covariant (String[] is-a Object[]) but this is unsound — you can put the
   wrong type at runtime (ArrayStoreException). Generics are invariant by design — you
   cannot assign List<String> to List<Object>. This prevents type errors at compile time
   rather than runtime. Wildcards provide flexibility where needed with proper bounds.
