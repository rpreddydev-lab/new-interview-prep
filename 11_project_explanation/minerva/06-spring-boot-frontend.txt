REST Controller Pattern

@RestController
@RequestMapping("/api/dashboards")
@RequiredArgsConstructor
public class DashboardController {

    private final DashboardService dashboardService;

    @GetMapping
    public ResponseEntity> listDashboards(
            @AuthenticationPrincipal Jwt jwt) {

        String tenantId = jwt.getClaimAsString("tenant_id");
        String department = jwt.getClaimAsString("department");
        List roles = jwt.getClaimAsStringList("roles");
        int roleLevel = RoleUtils.getHighestLevel(roles);

        List dashboards = dashboardService
            .listAccessible(tenantId, department, roleLevel);

        return ResponseEntity.ok(dashboards);
    }

    @GetMapping("/{id}")
    public ResponseEntity getDashboard(
            @PathVariable String id,
            @AuthenticationPrincipal Jwt jwt) {

        String tenantId = jwt.getClaimAsString("tenant_id");
        DashboardConfigDto config = dashboardService.getFullConfig(id, tenantId);

        return ResponseEntity.ok(config);
    }

    @PostMapping
    @PreAuthorize("hasAnyRole('CONFIGURATOR', 'TENANTADMIN', 'PLATFORMADMIN')")
    public ResponseEntity createDashboard(
            @Valid @RequestBody CreateDashboardRequest request,
            @AuthenticationPrincipal Jwt jwt) {

        String tenantId = jwt.getClaimAsString("tenant_id");
        String createdBy = jwt.getSubject();

        DashboardConfigDto created = dashboardService.create(request, tenantId, createdBy);

        return ResponseEntity.status(HttpStatus.CREATED).body(created);
    }
}

Service Layer Pattern (with Caching + Events)

@Service
@RequiredArgsConstructor
@Transactional(readOnly = true)
public class DashboardService {

    private final DashboardRepository dashboardRepo;
    private final DashboardFilterRepository filterRepo;
    private final DashboardChartRepository chartRepo;
    private final DashboardGridRepository gridRepo;
    private final DashboardMapLayerRepository mapLayerRepo;
    private final ConfigCacheService cacheService;
    private final DashboardEventProducer eventProducer;
    private final SqlValidationService sqlValidator;

    public DashboardConfigDto getFullConfig(String dashboardId, String tenantId) {
        // Try cache first
        Optional cached = cacheService.getConfig(dashboardId);
        if (cached.isPresent()) {
            return cached.get();
        }

        // Cache miss — load from DB
        Dashboard dashboard = dashboardRepo.findByIdAndTenantId(dashboardId, tenantId)
            .orElseThrow(() -> new NotFoundException("Dashboard not found"));

        List filters = filterRepo.findByDashboardIdOrderByDisplayOrder(dashboardId);
        List charts = chartRepo.findByDashboardIdOrderByDisplayOrder(dashboardId);
        List grids = gridRepo.findByDashboardIdOrderByDisplayOrder(dashboardId);
        List mapLayers = mapLayerRepo.findByDashboardIdOrderByDisplayOrder(dashboardId);

        DashboardConfigDto config = DashboardConfigDto.builder()
            .id(dashboard.getId())
            .name(dashboard.getName())
            .department(dashboard.getDepartment())
            .layout(dashboard.getLayoutJson())
            .filters(filters.stream().map(FilterDto::from).toList())
            .charts(charts.stream().map(ChartDto::from).toList())
            .grids(grids.stream().map(GridDto::from).toList())
            .mapLayers(mapLayers.stream().map(MapLayerDto::from).toList())
            .build();

        // Cache for next request
        cacheService.cacheConfig(dashboardId, config);

        return config;
    }

    @Transactional
    public DashboardConfigDto create(CreateDashboardRequest request, String tenantId, String createdBy) {
        // Validate all SQL in the request
        for (ChartRequest chart : request.getCharts()) {
            SqlValidationResult result = sqlValidator.validateSql(chart.getDataSql(), tenantId);
            if (result.isRejected()) {
                throw new ValidationException("Chart SQL invalid: " + result.getMessage());
            }
        }

        // Save all components in one transaction
        Dashboard dashboard = Dashboard.builder()
            .id(generateId())
            .tenantId(tenantId)
            .name(request.getName())
            .department(request.getDepartment())
            .layoutJson(request.getLayout())
            .createdBy(createdBy)
            .build();

        dashboardRepo.save(dashboard);
        // Save filters, charts, grids, map layers...

        // Publish events
        eventProducer.publishDashboardCreated(dashboard.getId(), tenantId, createdBy);

        return getFullConfig(dashboard.getId(), tenantId);
    }
}

Global Exception Handler

@RestControllerAdvice
@Slf4j
public class GlobalExceptionHandler {

    @ExceptionHandler(NotFoundException.class)
    public ResponseEntity handleNotFound(NotFoundException ex) {
        return ResponseEntity.status(HttpStatus.NOT_FOUND)
            .body(new ErrorResponse("NOT_FOUND", ex.getMessage()));
    }

    @ExceptionHandler(AccessDeniedException.class)
    public ResponseEntity handleAccessDenied(AccessDeniedException ex) {
        return ResponseEntity.status(HttpStatus.FORBIDDEN)
            .body(new ErrorResponse("FORBIDDEN", "You don't have permission for this action"));
    }

    @ExceptionHandler(ValidationException.class)
    public ResponseEntity handleValidation(ValidationException ex) {
        return ResponseEntity.status(HttpStatus.BAD_REQUEST)
            .body(new ErrorResponse("VALIDATION_ERROR", ex.getMessage()));
    }

    @ExceptionHandler(CallNotPermittedException.class) // Resilience4j circuit breaker
    public ResponseEntity handleCircuitBreaker(CallNotPermittedException ex) {
        return ResponseEntity.status(HttpStatus.SERVICE_UNAVAILABLE)
            .body(new ErrorResponse("SERVICE_UNAVAILABLE",
                "Analytics service temporarily unavailable. Please try again in 30 seconds."));
    }

    @ExceptionHandler(Exception.class)
    public ResponseEntity handleGeneral(Exception ex) {
        log.error("Unhandled exception", ex);
        return ResponseEntity.status(HttpStatus.INTERNALSERVERERROR)
            .body(new ErrorResponse("INTERNAL_ERROR", "An unexpected error occurred"));
    }

    record ErrorResponse(String code, String message) {}
}

WebSocket Integration

// useWebSocket hook — connects to Notification Service
export const useWebSocket = (dashboardId: string) => {
  const { token } = useAuth();
  const [notifications, setNotifications] = useState([]);

  useEffect(() => {
    const client = new Client({
      brokerURL: wss://minerva.vz/ws,
      connectHeaders: { Authorization: Bearer ${token} },
      onConnect: () => {
        // Subscribe to dashboard-specific channel
        client.subscribe(
          /topic/tenant/${getTenantId(token)}/dashboard/${dashboardId},
          (message) => {
            const notification = JSON.parse(message.body);
            setNotifications(prev => [notification, ...prev]);

            // Auto-refresh dashboard data on data-refresh event
            if (notification.type === 'DATA_REFRESHED') {
              queryClient.invalidateQueries(['dashboard', dashboardId]);
            }
          },
          { Authorization: Bearer ${token} } // STOMP header for ChannelInterceptor
        );
      },
      onStompError: (frame) => {
        console.error('WebSocket error:', frame.body);
      }
    });

    client.activate();
    return () => client.deactivate();
  }, [dashboardId, token]);

  return { notifications };
};

Axios Interceptor (Token Management)

// Auto-attach Bearer token, handle 401 with silent refresh
const api = axios.create({ baseURL: '/api' });

let isRefreshing = false;
let failedQueue: Array = [];

api.interceptors.request.use((config) => {
  const token = getAccessToken(); // from memory, not localStorage
  if (token) {
    config.headers.Authorization = Bearer ${token};
  }
  return config;
});

api.interceptors.response.use(
  (response) => response,
  async (error) => {
    if (error.response?.status !== 401) throw error;

    if (isRefreshing) {
      // Queue this request — will retry after refresh completes
      return new Promise((resolve, reject) => {
        failedQueue.push({ resolve, reject });
      }).then((token) => {
        error.config.headers.Authorization = Bearer ${token};
        return api(error.config);
      });
    }

    isRefreshing = true;

    try {
      const newToken = await refreshToken(); // POST to ForgeRock /token
      setAccessToken(newToken);

      // Retry all queued requests with new token
      failedQueue.forEach(({ resolve }) => resolve(newToken));
      failedQueue = [];

      error.config.headers.Authorization = Bearer ${newToken};
      return api(error.config);
    } catch (refreshError) {
      failedQueue.forEach(({ reject }) => reject(refreshError));
      failedQueue = [];
      logout(); // redirect to ForgeRock login
      throw refreshError;
    } finally {
      isRefreshing = false;
    }
  }
);

Q&A Bank (Spring Boot + Frontend)
---------------------------------

Q: "How do you handle database migrations?"
"Flyway. Every schema change is a versioned migration file (V1createdashboards.sql, V2addalert_configs.sql). Runs automatically on startup. Hibernate is set to ddl-auto: validate — it verifies the schema matches entities but never modifies the DB. This gives us repeatable, versioned schema evolution."

Q: "How do you handle configuration across environments?"
"Spring profiles (application-dev.yml, application-prod.yml) for per-environment config. Secrets (DB passwords, API keys) are Kubernetes Secrets mounted as environment variables. The application.yml uses ${ENV_VAR:default} syntax — defaults for local dev, overrides in K8s."

Q: "Why React and not Angular or Vue?"
"Team familiarity and the React ecosystem. Highcharts has official React bindings. React-Leaflet gives us declarative map components. React Query (TanStack Query) handles our server state beautifully — auto-refetch, cache invalidation, optimistic updates. Plus, the hooks pattern maps naturally to our custom hooks for config loading and query execution."

Q: "How do you prevent XSS?"
"React automatically escapes rendered content (JSX). We store tokens in memory (not localStorage). CSP headers restrict script sources. The only user-generated content is dashboard names/descriptions, which are always escaped on render."

Q: "What about accessibility?"
"Highcharts has built-in accessibility features (screen reader descriptions for charts). Our DataGrid uses semantic HTML tables with ARIA attributes. Filter dropdowns use headless UI components from Radix for keyboard navigation."
