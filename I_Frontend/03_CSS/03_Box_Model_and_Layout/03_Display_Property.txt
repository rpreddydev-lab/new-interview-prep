1. THE DISPLAY PROPERTY
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

The display property controls two things:
  1. The OUTER display — how the element participates in its parent's flow
     (block | inline)
  2. The INNER display — how the element's CHILDREN are laid out
     (flow | flex | grid | table | etc.)

Modern CSS two-value syntax:
  display: block         ≡ display: block flow
  display: inline        ≡ display: inline flow
  display: inline-block  ≡ display: inline flow-root
  display: flex          ≡ display: block flex
  display: inline-flex   ≡ display: inline flex
  display: grid          ≡ display: block grid
  display: inline-grid   ≡ display: inline grid

2. display: block
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  • Takes up the FULL width of its parent (by default)
  • Starts on a NEW LINE; creates a line break before and after
  • Respects width, height, margin, padding
  • width: auto → stretches to fill container

  Default block elements: div, p, h1-h6, ul, ol, li, section, article,
                          header, footer, nav, main, form, blockquote, hr, pre,
                          table, figure, figcaption, address, fieldset

  .box {
    display: block;
    width: 300px;       /* works */
    height: 100px;      /* works */
    margin: 20px auto;  /* centers horizontally */
    padding: 16px;      /* works */
  }

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
3. display: inline
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  • Sits inline with text; no line break before/after
  • Width and height have NO effect
  • Horizontal margin and padding work; vertical ONLY visual (don't push other elements)
  • Cannot be centered with margin: auto

  Default inline elements: span, a, strong, em, code, img (technically replaced),
                           label, input, select, textarea, button, abbr, cite,
                           small, sub, sup, time, b, i, s, u

  .link {
    display: inline;
    color: blue;
    padding: 2px 4px;    /* visual padding, but top/bottom don't affect flow */
    width: 200px;        /* IGNORED for inline */
    height: 50px;        /* IGNORED for inline */
    margin-top: 20px;    /* IGNORED for inline (horizontal margin works) */
  }

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
4. display: inline-block
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  • Sits inline (no line break, can have siblings on same line)
  • Respects width, height, all padding and margin (like block)
  • The best of both worlds

  Use cases: horizontal nav items, buttons in text, image + text layouts

  .nav-item {
    display: inline-block;
    width: 120px;
    height: 40px;
    padding: 8px 16px;
    margin-right: 8px;
    text-align: center;
  }

  Gotcha — whitespace gap:
  /* inline-block elements have whitespace between them if there's whitespace in HTML */
  /* Solutions: */
  /* 1. font-size: 0 on parent, restore on children */
  /* 2. Remove whitespace in HTML (ugly) */
  /* 3. Use flexbox instead (modern solution) */

  .nav { font-size: 0; }
  .nav-item { font-size: 1rem; }

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
5. display: none
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  • Element is completely removed from the rendering tree
  • Takes up NO space in the layout
  • Not accessible to screen readers
  • Cannot be transitioned/animated (use visibility + opacity instead)

  .hidden { display: none; }

  vs visibility: hidden:
  • visibility: hidden → hides visually but PRESERVES space
  • display: none → hides and removes from flow entirely

  vs opacity: 0:
  • opacity: 0 → invisible but still TAKES SPACE and is accessible to events

  Transitioning visibility (common pattern):
  .menu {
    visibility: hidden;
    opacity: 0;
    transition: opacity 0.3s, visibility 0.3s;
  }
  .menu.open {
    visibility: visible;
    opacity: 1;
  }
  /* visibility delays with opacity for smooth show/hide */

  Accessible hiding (visually hidden but available to screen readers):
  .sr-only {
    position: absolute;
    width: 1px;
    height: 1px;
    padding: 0;
    margin: -1px;
    overflow: hidden;
    clip: rect(0, 0, 0, 0);
    white-space: nowrap;
    border: 0;
  }

6. display: flex and display: grid
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Covered in detail in 03_Flexbox_and_Grid.
  Brief overview:

  display: flex
    • Creates a flex container; children become flex items
    • One-dimensional layout (row OR column)
    • Powerful alignment along main and cross axes
    • Direct children only are flex items

  display: inline-flex
    • Like flex but the container itself is inline (sits in text flow)

  display: grid
    • Creates a grid container; children become grid items
    • Two-dimensional layout (rows AND columns)
    • Explicit or implicit grid tracks
    • fr unit for flexible track sizing

  display: inline-grid
    • Like grid but container is inline

7. display: table and table-related values
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  display: table           → like <table>
  display: table-row       → like <tr>
  display: table-cell      → like <td>
  display: table-header-group → like <thead>
  display: table-row-group    → like <tbody>
  display: table-column       → like <col>
  display: table-caption      → like <caption>

  Rarely used directly in modern CSS (flexbox/grid replace most use cases).
  One classic use: equal-height columns before flexbox:
  .wrapper { display: table; width: 100%; }
  .col     { display: table-cell; vertical-align: top; }

  display: table-cell enables vertical-align on non-table elements:
  .center-vertical {
    display: table-cell;
    vertical-align: middle;
    text-align: center;
  }

8. display: flow-root
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  • Creates a Block Formatting Context (BFC) without side effects
  • Equivalent to overflow: hidden but WITHOUT clipping overflow content
  • Ideal for containing floats (modern replacement for clearfix hack)

  .container { display: flow-root; }
  /* Now contains any floated children */

9. display: contents
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  • The element itself is not rendered; its CHILDREN replace it in the layout
  • Useful for wrapping elements in flex/grid without affecting layout

  <div class="wrapper">     /* display: contents */
    <div class="flex-item">Item 1</div>
    <div class="flex-item">Item 2</div>
  </div>

  .flex-container { display: flex; }
  .wrapper { display: contents; }
  /* .flex-item elements now participate directly in the flex container */

  ⚠ Accessibility issue: display: contents removes the element from the
  accessibility tree in some browsers. Avoid on interactive/semantic elements.

11. KEY INTERVIEW QUESTIONS — DISPLAY
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What is the difference between block, inline, and inline-block?
A: block: full-width, new line before/after, respects all margin/padding/width/height.
   inline: only content width, no new line, ignores width/height and vertical margin.
   inline-block: sits inline like inline, but respects width/height and all margin/padding.

Q: What is the difference between display:none and visibility:hidden?
A: display:none removes the element entirely — no space, not accessible.
   visibility:hidden hides the element visually but preserves its space in layout.
   Also: opacity:0 makes element invisible but it still occupies space and catches events.

Q: What does display:contents do?
A: Makes the element itself "invisible" to layout — its children take its place
   in the layout flow. Useful for flex/grid layouts where you need a wrapper in
   HTML but don't want it to break the grid/flex structure.

Q: What is display:flow-root?
A: Creates a Block Formatting Context (BFC) without the side effects of overflow:hidden
   (which clips content) or other BFC-creating properties. It's the clean, modern
   way to contain floats or prevent margin collapsing from a container.

Q: Why does margin:0 auto center a block element?
A: margin:0 auto sets top/bottom margin to 0 and sets left/right margin to auto.
   For a block element with an explicit width, the browser divides the remaining
   horizontal space equally between left and right auto margins, centering the element.
   This only works for horizontal centering on block elements with a set width.
