1. THE PROBLEM WITH FIXED IMAGES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  <img src="hero.jpg" width="1200">   <!-- BAD on mobile -->

  Problems:
  • Oversized images waste bandwidth on mobile (2MB image on a 375px screen)
  • Layout breaks — image overflows or is squashed
  • Retina screens (2x/3x) need higher resolution images to look sharp

  The solution: responsive images using CSS and HTML attributes.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
2. CSS BASICS FOR RESPONSIVE IMAGES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  /* Never overflow container */
  img {
    max-width: 100%;
    height: auto;          /* maintain aspect ratio */
    display: block;        /* removes baseline gap */
  }

  /* Full-bleed hero image */
  .hero-img {
    width: 100%;
    height: 400px;
    object-fit: cover;     /* fill box, crop if needed */
    object-position: center top;
  }

  object-fit values:
    fill     — stretch to fill (default, may distort)
    contain  — fit inside box, may letterbox
    cover    — fill box, crop to maintain aspect ratio
    none     — original size, may overflow
    scale-down — smaller of none or contain

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
3. THE srcset ATTRIBUTE — RESOLUTION SWITCHING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  srcset lets the browser choose the best image based on:
    a) Screen pixel density (x descriptor)
    b) Viewport width + sizes attribute (w descriptor)

  ── Pixel Density (x descriptor) ──────────────────────────
  <img src="logo.png"
       srcset="logo.png 1x, logo@2x.png 2x, logo@3x.png 3x"
       alt="Logo">

  • 1x = standard screens
  • 2x = Retina/HiDPI displays
  • 3x = High-end phone screens (iPhone Plus models)

  ── Width Descriptor (w descriptor) ───────────────────────
  <img src="small.jpg"
       srcset="small.jpg 400w,
               medium.jpg 800w,
               large.jpg 1200w,
               xlarge.jpg 1600w"
       sizes="(max-width: 600px) 100vw,
              (max-width: 1200px) 50vw,
              800px"
       alt="Article image">

  srcset lists available images and their actual widths (in pixels).
  sizes tells the browser how wide the image will be displayed:
    • "at max 600px viewport → image is 100vw wide"
    • "at max 1200px viewport → image is 50vw wide"
    • "otherwise → 800px wide"
  The browser calculates: needed pixels = display width × device pixel ratio,
  then picks the closest image from srcset.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
4. THE <picture> ELEMENT — ART DIRECTION
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Use <picture> when you want to show a DIFFERENT image (not just a different
  size) based on conditions — different crop, different aspect ratio, or
  a different format.

  <picture>
    <!-- WebP for modern browsers, JPEG fallback -->
    <source type="image/webp" srcset="hero.webp">
    <source type="image/jpeg" srcset="hero.jpg">
    <img src="hero.jpg" alt="Hero image">   <!-- required fallback -->
  </picture>

  <!-- Art direction: different crops per viewport -->
  <picture>
    <source media="(max-width: 600px)"
            srcset="hero-mobile-portrait.jpg">
    <source media="(max-width: 1200px)"
            srcset="hero-tablet.jpg">
    <img src="hero-desktop.jpg" alt="Team photo">
  </picture>

  <!-- Modern formats with responsive sizes -->
  <picture>
    <source
      type="image/avif"
      srcset="photo-400.avif 400w, photo-800.avif 800w, photo-1200.avif 1200w"
      sizes="(max-width: 600px) 100vw, 50vw">
    <source
      type="image/webp"
      srcset="photo-400.webp 400w, photo-800.webp 800w, photo-1200.webp 1200w"
      sizes="(max-width: 600px) 100vw, 50vw">
    <img src="photo-800.jpg"
         srcset="photo-400.jpg 400w, photo-800.jpg 800w, photo-1200.jpg 1200w"
         sizes="(max-width: 600px) 100vw, 50vw"
         alt="Product photo">
  </picture>

5. MODERN IMAGE FORMATS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Format   | Quality   | File Size  | Browser Support    | Best For
  ───────────────────────────────────────────────────────────────────────
  JPEG     | Good      | Medium     | Universal          | Photos
  PNG      | Lossless  | Large      | Universal          | Icons, logos
  GIF      | Poor      | Medium     | Universal          | Simple animations
  SVG      | Vector    | Tiny       | Universal          | Icons, logos, UI
  WebP     | Excellent | 25-30% ↓  | Modern (94%+)      | Photos, UI
  AVIF     | Best      | 50%+ ↓    | Modern (90%+)      | Photos
  HEIC     | Excellent | Very small | Limited (Safari)   | Apple devices

  Use WebP/AVIF with JPEG fallback via <picture> for maximum compatibility.

6. LAZY LOADING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  <img src="photo.jpg" loading="lazy" alt="...">

  loading="lazy"  — defer loading until image is near the viewport
  loading="eager" — load immediately (default behavior)

  The browser handles the intersection detection natively — no JS needed.
  ⚠️ Do NOT lazy-load images above the fold (hero images) — it delays LCP.
  ✅ Use lazy loading for all below-the-fold images.

  /* Prevent layout shift while loading — always set dimensions */
  img {
    width: 100%;
    height: auto;
    aspect-ratio: 16 / 9;  /* reserve space before image loads */
  }

7. BACKGROUND IMAGES IN CSS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  /* Responsive background image */
  .hero {
    background-image: url('hero.jpg');
    background-size: cover;
    background-position: center;
    background-repeat: no-repeat;
    aspect-ratio: 16 / 9;
  }

  /* Serve different background images per breakpoint */
  .hero {
    background-image: url('hero-mobile.jpg');  /* mobile first */
  }

  @media (min-width: 768px) {
    .hero { background-image: url('hero-tablet.jpg'); }
  }

  @media (min-width: 1200px) {
    .hero { background-image: url('hero-desktop.jpg'); }
  }

  /* Retina background image */
  @media (-webkit-min-device-pixel-ratio: 2), (min-resolution: 192dpi) {
    .logo { background-image: url('logo@2x.png'); background-size: 100px 50px; }
  }

8. KEY INTERVIEW QUESTIONS — RESPONSIVE IMAGES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What is the difference between srcset and the <picture> element?
A: srcset provides multiple versions of the SAME image at different resolutions
   or sizes — the browser picks the most appropriate one. <picture> allows
   ART DIRECTION — showing completely different images (different crop, subject
   framing, or format) based on media queries. <picture> also supports multiple
   image formats for progressive enhancement.

Q: What does the sizes attribute do?
A: It tells the browser how wide the image will be displayed (in CSS) before
   the browser has rendered the layout. Without sizes, the browser assumes
   the image is 100vw wide. With sizes, the browser can calculate the actual
   needed image width and pick the most efficient option from srcset.

Q: What is object-fit and when would you use it?
A: object-fit controls how an <img> or <video> fills its content box.
   cover fills the box and crops if needed (maintains aspect ratio), like
   background-size: cover. contain fits the whole image inside (may letterbox).
   Use it when you need consistent image dimensions (e.g., a card grid where
   all images must be the same height) without distortion.

Q: What is lazy loading and when should you avoid it?
A: loading="lazy" tells the browser to defer loading an image until it's near
   the viewport. It reduces initial page load time and bandwidth. However,
   never use it for images above the fold (hero images, logos) — it will
   delay the Largest Contentful Paint (LCP), hurting Core Web Vitals.

Q: Why should you specify width and height on images in HTML?
A: To prevent Cumulative Layout Shift (CLS). When the browser doesn't know
   image dimensions, it renders with 0 height and then reflows when the image
   loads. Setting width/height attributes (or aspect-ratio in CSS) reserves
   the right amount of space before the image loads.
