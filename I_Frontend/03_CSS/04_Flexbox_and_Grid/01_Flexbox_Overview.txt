1. WHAT IS FLEXBOX?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Flexbox (Flexible Box Layout Module) is a ONE-DIMENSIONAL layout method.
It handles layout in either a ROW or a COLUMN direction, but not both simultaneously.

Key concepts:
  • Flex container — parent element with display: flex
  • Flex items — DIRECT children of the flex container
  • Main axis — the primary axis (row = horizontal, column = vertical)
  • Cross axis — perpendicular to main axis
  • Main start/end — where main axis begins/ends
  • Cross start/end — where cross axis begins/ends

  display: flex;         /* block-level flex container */
  display: inline-flex;  /* inline-level flex container */

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
2. FLEX-DIRECTION — THE MAIN AXIS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  flex-direction: row;             /* default: left → right */
  flex-direction: row-reverse;     /* right → left */
  flex-direction: column;          /* top → bottom */
  flex-direction: column-reverse;  /* bottom → top */

  row    : main axis = horizontal, cross axis = vertical
  column : main axis = vertical,   cross axis = horizontal

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
3. JUSTIFY-CONTENT — ALIGNMENT ON MAIN AXIS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  justify-content: flex-start;    /* default: items at start */
  justify-content: flex-end;      /* items at end */
  justify-content: center;        /* items centered */
  justify-content: space-between; /* equal space BETWEEN items (no edge space) */
  justify-content: space-around;  /* equal space around items (half at edges) */
  justify-content: space-evenly;  /* equal space between and at edges */
  justify-content: normal;        /* default: same as flex-start */

  Visual:
  flex-start:    [A][B][C]         ]
  flex-end:      [         ][A][B][C]
  center:        [    ][A][B][C][  ]
  space-between: [A]   [B]   [C]
  space-around:  [ A ][ B ][ C ]
  space-evenly:  [  A  ][  B  ][  C  ]

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
4. ALIGN-ITEMS — ALIGNMENT ON CROSS AXIS (per row/column)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  align-items: stretch;     /* default: items stretch to fill cross axis */
  align-items: flex-start;  /* items aligned to cross start */
  align-items: flex-end;    /* items aligned to cross end */
  align-items: center;      /* items centered on cross axis */
  align-items: baseline;    /* items aligned by text baseline */

  /* Center everything: */
  .container {
    display: flex;
    justify-content: center;  /* main axis */
    align-items: center;      /* cross axis */
  }

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
5. ALIGN-CONTENT — ALIGNMENT ON CROSS AXIS (multi-line)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Only applies when flex-wrap: wrap is set and there are multiple lines:

  align-content: flex-start;    /* lines packed to start */
  align-content: flex-end;      /* lines packed to end */
  align-content: center;        /* lines centered */
  align-content: space-between; /* equal space between lines */
  align-content: space-around;  /* equal space around lines */
  align-content: space-evenly;
  align-content: stretch;       /* default: lines stretch to fill */

6. FLEX-WRAP
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  flex-wrap: nowrap;        /* default: all items in one line, may overflow */
  flex-wrap: wrap;          /* items wrap to next line if needed */
  flex-wrap: wrap-reverse;  /* wrap, but in reverse */

  flex-flow shorthand (direction + wrap):
  flex-flow: row wrap;
  flex-flow: column nowrap;

7. GAP (gap / row-gap / column-gap)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  gap: 16px;               /* gap between ALL items (both axes) */
  gap: 16px 32px;          /* row-gap column-gap */
  row-gap: 16px;           /* gap between flex rows */
  column-gap: 32px;        /* gap between flex columns (items in a row) */

  /* gap replaced the need for margin hacks on flex items */
  /* No margin on first/last items needed */

8. PLACE-CONTENT / PLACE-ITEMS SHORTHANDS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  place-items: center;          /* align-items: center + justify-items: center */
  /* (note: justify-items has no effect in flexbox) */

  place-content: center;        /* align-content + justify-content */
  place-content: space-between center;

  /* Perfect centering with grid (works with place-items): */
  .center-box {
    display: grid;
    place-items: center;
  }

9. COMPLETE CENTERING REFERENCE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Method 1 — Flexbox:
  .parent {
    display: flex;
    justify-content: center;
    align-items: center;
  }

Method 2 — Grid:
  .parent {
    display: grid;
    place-items: center;
  }

Method 3 — Grid shorthand:
  .parent { display: grid; }
  .child  { margin: auto; }

Method 4 — Absolute + transform:
  .child {
    position: absolute;
    top: 50%;
    left: 50%;
    transform: translate(-50%, -50%);
  }

Method 5 — Absolute + inset + margin:
  .child {
    position: absolute;
    inset: 0;
    margin: auto;
    width: 200px;   /* must have explicit dimensions */
    height: 100px;
  }

10. KEY INTERVIEW QUESTIONS — FLEXBOX OVERVIEW
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What is flexbox and when would you use it?
A: Flexbox is a one-dimensional layout method for arranging items in a row or
   column with powerful alignment and distribution controls. Use it for:
   navbars, card rows, centering elements, equal-height columns, spacing items
   evenly. For two-dimensional layouts (rows AND columns), prefer CSS Grid.

Q: What is the difference between justify-content and align-items?
A: justify-content aligns items along the MAIN axis (horizontal in row, vertical in column).
   align-items aligns items along the CROSS axis (vertical in row, horizontal in column).
   Mnemonic: "justify = main axis" — the axis you declared with flex-direction.

Q: What is the difference between align-items and align-content?
A: align-items: aligns individual items within a flex line (cross axis).
   align-content: aligns multiple flex lines when flex-wrap is enabled.
   align-content has no effect when all items are on a single line.

Q: How do you center an element both horizontally and vertically using flexbox?
A: Set on the parent: display:flex; justify-content:center; align-items:center;
   Or use grid: display:grid; place-items:center;

Q: What is the default value of align-items?
A: stretch — items stretch to fill the cross axis of the flex line. This makes
   all flex items equal height (in row direction) by default, which is the
   "equal-height columns" effect that used to require JavaScript hacks.

Q: What does flex-wrap do and when do you need it?
A: By default (nowrap) all items are forced onto one line, possibly overflowing
   the container. flex-wrap: wrap allows items to move to the next line when they
   don't fit. Combined with align-content to control how the multiple lines are
   distributed. Essential for responsive layouts without media queries.
