1. NAMED GRID LINES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

You can give grid lines meaningful names using square brackets:

  grid-template-columns: [start] 1fr [middle] 2fr [end];
  grid-template-rows:    [header-start] 60px [header-end content-start] 1fr [content-end];

  /* Use names when placing items */
  .item {
    grid-column: start / end;
    grid-row: header-start / content-end;
  }

  /* Lines can have multiple names */
  grid-template-rows: [header-start] 60px [header-end main-start] 1fr [main-end];

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
2. grid-template-areas — VISUAL LAYOUT TEMPLATES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

grid-template-areas lets you define the layout visually with named areas.
This is one of the most readable and powerful CSS Grid features.

  .page {
    display: grid;
    grid-template-columns: 200px 1fr;
    grid-template-rows: 60px 1fr 50px;
    grid-template-areas:
      "header  header"
      "sidebar main  "
      "footer  footer";
  }

  /* Place items using area names */
  .header  { grid-area: header; }
  .sidebar { grid-area: sidebar; }
  .main    { grid-area: main; }
  .footer  { grid-area: footer; }

Rules for grid-template-areas:
  • Each area name must form a contiguous rectangle
  • Use "." for an empty cell
  • Each row is a quoted string
  • Each name in the string = one cell
  • Rows must have the same number of columns

  Example with empty cell:
  grid-template-areas:
    "header  header"
    "sidebar main  "
    ".       footer";
  /* Bottom-left is empty */

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
3. RESPONSIVE LAYOUT WITH grid-template-areas
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  /* Mobile: single column, stacked */
  .page {
    display: grid;
    grid-template-areas:
      "header"
      "main"
      "sidebar"
      "footer";
  }

  /* Tablet and desktop: sidebar layout */
  @media (min-width: 768px) {
    .page {
      grid-template-columns: 240px 1fr;
      grid-template-areas:
        "header  header"
        "sidebar main  "
        "footer  footer";
    }
  }

  /* Items don't need to change — they keep their grid-area names */
  .header  { grid-area: header; }
  .sidebar { grid-area: sidebar; }
  .main    { grid-area: main; }
  .footer  { grid-area: footer; }

4. grid-template SHORTHAND
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Combines grid-template-rows, grid-template-columns, and grid-template-areas:

  grid-template:
    "header  header"  60px
    "sidebar main  "  1fr
    "footer  footer"  50px
    / 200px 1fr;   /* row sizes defined after each line; / = column sizes */

  Equivalent to:
  grid-template-columns: 200px 1fr;
  grid-template-rows: 60px 1fr 50px;
  grid-template-areas:
    "header  header"
    "sidebar main  "
    "footer  footer";

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
5. grid SHORTHAND (Everything)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

The grid shorthand sets ALL grid properties:

  grid: <grid-template> | <grid-template-rows> / <grid-auto-flow> <grid-auto-columns>

  /* Simple 3-column grid */
  grid: auto / repeat(3, 1fr);

  /* With auto rows */
  grid: auto-flow minmax(100px, auto) / repeat(3, 1fr);

  /* Complex shorthand with areas (prefer verbose for readability) */
  grid:
    "header" 60px
    "main"   1fr
    "footer" 50px
    / 1fr;

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
6. MASONRY-LIKE LAYOUTS WITH dense
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  .masonry {
    display: grid;
    grid-template-columns: repeat(3, 1fr);
    grid-auto-rows: 100px;
    grid-auto-flow: dense;  /* fills in holes left by spanning items */
    gap: 8px;
  }

  .item-tall   { grid-row: span 2; }
  .item-wide   { grid-column: span 2; }
  .item-large  { grid-column: span 2; grid-row: span 2; }

7. NAMED grid-area CREATES NAMED LINES AUTOMATICALLY
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

When you define grid-template-areas, CSS automatically creates named lines
for each area:
  area name "main" → lines: main-start (row-start, col-start), main-end (row-end, col-end)

  grid-template-areas: "header" "main" "footer";
  /* Creates: header-start, header-end, main-start, main-end, footer-start, footer-end */

  .item {
    grid-row: header-start / footer-end;  /* use auto-generated names */
  }

8. COMPLETE PAGE LAYOUT EXAMPLE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  <div class="page">
    <header class="header">Header</header>
    <nav class="nav">Nav</nav>
    <main class="main">Main Content</main>
    <aside class="aside">Sidebar</aside>
    <footer class="footer">Footer</footer>
  </div>

  .page {
    display: grid;
    min-height: 100vh;
    grid-template-columns: 200px 1fr 200px;
    grid-template-rows: auto 1fr auto;
    grid-template-areas:
      "header header header"
      "nav    main   aside "
      "footer footer footer";
  }

  .header { grid-area: header; }
  .nav    { grid-area: nav; }
  .main   { grid-area: main; }
  .aside  { grid-area: aside; }
  .footer { grid-area: footer; }

9. KEY INTERVIEW QUESTIONS — GRID AREAS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What is grid-template-areas?
A: It lets you define a visual text map of your layout using named areas. Each
   quoted string represents a row; each word in the string represents a column.
   Items are placed into areas using grid-area: <name>. Areas must be rectangular.
   Use "." for empty cells.

Q: What is the benefit of named grid areas?
A: (1) Extremely readable — the layout looks like a visual ASCII art map.
   (2) Responsive changes are done by redeclaring template-areas in media queries —
   items don't need to change their placement rules.
   (3) Grid lines are automatically named (header-start, header-end, etc.)

Q: What is the difference between grid-template and the grid shorthand?
A: grid-template sets template-rows, template-columns, and template-areas (the explicit grid).
   grid sets all of those PLUS grid-auto-rows, grid-auto-columns, and grid-auto-flow
   (both explicit and implicit grid settings).

Q: How do you create a classic header/sidebar/main/footer layout with CSS Grid?
A: Using grid-template-areas:
   grid-template-columns: sidebar-width 1fr;
   grid-template-rows: header-height 1fr footer-height;
   grid-template-areas:
     "header header"
     "sidebar main"
     "footer footer";
   Then: .header { grid-area: header; } etc.

Q: How would you create an overlap in CSS Grid?
A: Place two items in the same grid cell/area using the same line numbers or area name.
   Control visual order with z-index. This doesn't require position: absolute.
   .background { grid-area: 1/1/3/3; }
   .overlay    { grid-area: 1/1/3/3; z-index: 1; }
