SECTION 1: CLASSIC GOTCHAS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

GOTCHA 1: typeof null
─────────────────────
typeof null === "object" // at runtime (historical JS bug)
TypeScript knows null has type `null`, not `object`. But runtime typeof checks
won't help:

  function isObject(val: unknown): val is object {
    return typeof val === "object"; // ❌ also true for null!
  }
  isObject(null); // returns true at runtime

  // Correct:
  function isObject(val: unknown): val is object {
    return typeof val === "object" && val !== null;
  }

─────────────────────────────────────────────────────────────────────────────
GOTCHA 2: The `void` return type doesn't prevent returning a value
──────────────────────────────────────────────────────────────────
type VoidFn = () => void;
const fn: VoidFn = () => "hello"; // ✅ TypeScript allows this!

// Why? So that you can pass non-void callbacks to higher-order functions:
const nums: number[] = [];
[1, 2, 3].forEach((n) => nums.push(n)); // Array.push returns number, but forEach expects () => void

// EXCEPTION: a function with an explicit void return type annotation
function explicitVoid(): void {
  return "hello"; // ❌ Error — explicit annotation IS enforced
}

─────────────────────────────────────────────────────────────────────────────
GOTCHA 3: Excess property check only on fresh object literals
──────────────────────────────────────────────────────────────
interface Point { x: number; y: number }

// Direct object literal → excess property check
const p1: Point = { x: 1, y: 2, z: 3 }; // ❌ 'z' does not exist in type 'Point'

// Through a variable (not "fresh") → no excess property check
const obj = { x: 1, y: 2, z: 3 };
const p2: Point = obj; // ✅ structural check passes — obj has at least x and y

// Same issue with function calls:
function move(p: Point) {}
move({ x: 1, y: 2, z: 3 });      // ❌ excess property check fires
const pos = { x: 1, y: 2, z: 3 };
move(pos);                         // ✅ no check on variable

─────────────────────────────────────────────────────────────────────────────
GOTCHA 4: Enum numeric values accept any number
─────────────────────────────────────────────────
enum Direction { Up = 0, Down = 1 }

function go(d: Direction) {}
go(Direction.Up); // ✅
go(99);           // ✅ TypeScript allows any number for numeric enums — this is by design
go("Up");         // ❌ string enums don't have this problem

// This is why many TypeScript experts prefer string enums or as-const objects.

─────────────────────────────────────────────────────────────────────────────
GOTCHA 5: Intersection with incompatible types creates never
──────────────────────────────────────────────────────────────
type A = { x: string } & { x: number };
// A.x is string & number = never
// You can create a value of type A but can't ever set x to anything!
const a = {} as A;
a.x; // type: never — you can never meaningfully set this

─────────────────────────────────────────────────────────────────────────────
GOTCHA 6: `readonly` doesn't deep-freeze
─────────────────────────────────────────
interface Config {
  readonly server: { host: string; port: number };
}

const config: Config = { server: { host: "localhost", port: 3000 } };
config.server = { host: "x", port: 80 }; // ❌ Cannot assign to 'server' (readonly)
config.server.port = 9000;                // ✅ nested property is NOT readonly!

// To freeze deeply, use DeepReadonly<T> or Object.freeze (runtime).

SECTION 2: TYPE EVALUATION PUZZLES — "WHAT IS THE TYPE?"
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

PUZZLE 1: What is the type of x?
──────────────────────────────────
const x = [1, "hello", true];
Answer: (string | number | boolean)[]
TypeScript widens the array literal and unifies all element types into a union.
To get the tuple: const x = [1, "hello", true] as const; → readonly [1, "hello", true]

PUZZLE 2: What does this evaluate to?
───────────────────────────────────────
type T = "a" | "b" extends "a" ? "yes" : "no";
Answer: "no"
"a" | "b" does NOT extend "a" (a union is wider than a single member).
Note: This is NOT a naked type parameter so there's no distribution.
Compare: type Check<T> = T extends "a" ? "yes" : "no"; type R = Check<"a" | "b">; → "yes" | "no"

PUZZLE 3: What is the type of result?
───────────────────────────────────────
function fn() {
  if (Math.random() > 0.5) return "hello";
  return 42;
}
type Result = ReturnType<typeof fn>;
Answer: string | number
TypeScript unions all possible return types.

PUZZLE 4: What's wrong here?
──────────────────────────────
const arr: string[] = [];
const first: string = arr[0];
// This compiles fine but first is `undefined` at runtime!
// With `noUncheckedIndexedAccess: true`, arr[0] would be string | undefined.
// This tsconfig option catches this class of bugs.

PUZZLE 5: What is the type of `key`?
──────────────────────────────────────
const obj = { a: 1, b: 2, c: 3 };
for (const key in obj) {
  // key is: string — NOT "a" | "b" | "c"
}
// The for...in loop gives string keys. To get the literal union:
(Object.keys(obj) as Array<keyof typeof obj>).forEach(key => {
  // key is: "a" | "b" | "c"
});

PUZZLE 6: What type does satisfies produce?
────────────────────────────────────────────
type Palette = Record<string, [number, number, number]>;
const palette = {
  red: [255, 0, 0],
  green: "#00ff00",  // ❌ satisfies catches this — string is not [number,number,number]
} satisfies Palette;

// If valid, palette.red is [number, number, number], not a wider Palette value type.

PUZZLE 7: What happens with double assertion?
──────────────────────────────────────────────
const x = "hello" as unknown as number;
// TypeScript: x is `number` — no compile error
// Runtime: x is still "hello" — no conversion
// double assertion is always a code smell

PUZZLE 8: What does this conditional type return?
──────────────────────────────────────────────────
type IsNever<T> = T extends never ? true : false;
type A = IsNever<never>;
Answer: never (NOT true!)

When T = never, the conditional type distributes over never. Distributing over
an empty union produces never. This is a famous TypeScript quirk.

Fix:
type IsNever<T> = [T] extends [never] ? true : false; // Wrapping prevents distribution
type A = IsNever<never>; // true ✅

PUZZLE 9: What is the type of head?
──────────────────────────────────────
function head<T extends readonly unknown[]>(arr: readonly [...T]): T[0] {
  return arr[0] as T[0];
}
const h = head([1, "hello", true]); // h: number
// T is inferred as [number, string, boolean], T[0] = number ✅

PUZZLE 10: What is the type of Result?
────────────────────────────────────────
type Flatten<T> = T extends (infer E)[] ? E : T;
type Result = Flatten<string[][]>;
Answer: string[]
Only one level is flattened — (string[])[] → string[]. Flatten is not recursive.
To fully flatten: type DeepFlatten<T> = T extends (infer E)[] ? DeepFlatten<E> : T;

SECTION 3: FUNCTION AND CALLABLE TYPE EDGE CASES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

GOTCHA 7: Function parameter types are bivariant in method shorthand syntax
────────────────────────────────────────────────────────────────────────────
// strictFunctionTypes only affects function-type properties, NOT method shorthand

interface A {
  method(x: string): void;    // method shorthand — bivariant (both ✅)
  fn: (x: string) => void;    // function property — strictly contravariant
}

// With strictFunctionTypes:
const a: A = {
  method(x: number) {},       // ✅ allowed (bivariant)
  fn: (x: number) => {},      // ❌ Error (strict — number is not contravariant to string)
};

GOTCHA 8: Optional parameters and overloads
────────────────────────────────────────────
// Optional params are assignable to required params:
type FnA = (x: string, y?: number) => void;
type FnB = (x: string) => void;

const a: FnA = (x, y) => {};
const b: FnB = a; // ✅ FnA is compatible with FnB

// This is why Array.forEach works with callbacks of different arities

GOTCHA 9: `this` typing in callbacks
──────────────────────────────────────
interface EventEmitter {
  on(event: string, cb: (this: EventEmitter, ...args: any[]) => void): void;
}

// The `this` parameter is ERASED in emitted JS but enforced at compile time:
emitter.on("data", function() {
  this; // ✅ typed as EventEmitter (not the outer class)
});
emitter.on("data", () => {
  this; // ❌ Arrow function captures outer `this`, not the emitter
});

GOTCHA 10: Inference from two positions conflicts
──────────────────────────────────────────────────
function combine<T>(a: T, b: T): T[] {
  return [a, b];
}
combine(1, "hello"); // ❌ Error: string not assignable to number (T can't be both)

// Fix: use two type params, or widen the constraint
function combine<T, U>(a: T, b: U): (T | U)[] {
  return [a, b];
}
combine(1, "hello"); // ✅ [number, string]

GOTCHA 11: Generic default doesn't help with inference
────────────────────────────────────────────────────────
function wrap<T = string>(val: T): { value: T } {
  return { value: val };
}
wrap(42); // T is inferred as number — default is IGNORED when argument is present
// Defaults only matter when no argument is provided or T isn't inferable

GOTCHA 12: Mapped type on union distributes — but differently than conditional
────────────────────────────────────────────────────────────────────────────────
type M<T> = { [K in keyof T]: T[K] };
type R = M<string | number>;
// R = { [K in keyof string | keyof number]: ... } — NOT distributed per member
// Result is the mapped intersection, not separate mapped types unioned

Conditional types on naked T distribute; mapped types do NOT.

SECTION 5: COMMON INTERVIEW "TRAP" QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

TRAP 1: Is TypeScript completely safe with `strictNullChecks`?
Answer: No. You can still get runtime null/undefined errors through:
- Type assertions bypassing checks (x as string)
- DOM APIs returning T | null when you expected T
- JSON.parse returning any
- Array out-of-bounds access (unless noUncheckedIndexedAccess)
- External data with wrong shape (need runtime validation like Zod/io-ts)

TRAP 2: Does TypeScript always compile to the same JS regardless of types?
Answer: Mostly yes — type information is erased. But NOT always:
- const enum: replaced with literal values
- Legacy decorators (experimentalDecorators): emit additional descriptor code
- Parameter decorators: emit __metadata
- namespace: compiled to an IIFE
- class fields with useDefineForClassFields: changes initialization semantics

TRAP 3: Is a class also a type in TypeScript?
Answer: Yes — a class declaration creates BOTH a value (constructor function)
and a type (the instance type). The type of the class itself (constructor)
is accessed with `typeof ClassName`:
  type UserInstance = User;         // type of instances
  type UserConstructor = typeof User; // type of the class itself

TRAP 4: Can you use `await` outside async in TypeScript?
Answer: With `"module": "esnext"` (or system/node16) and TypeScript 3.8+,
top-level await is supported at the module level. However, files must be
treated as modules (have at least one import/export). In scripts (no import/export),
top-level await is not available.

TRAP 5: What's the difference between Array<T> and T[]?
Answer: They're functionally identical — purely stylistic. However:
- T[] is shorthand syntax
- Array<T> works better with readonly: ReadonlyArray<T> vs readonly T[]
- In some complex generic scenarios Array<T> is clearer
- TypeScript might resolve them slightly differently in error messages

TRAP 6: Does `noImplicitAny` affect all `any` usage?
Answer: No. `noImplicitAny` only catches IMPLICIT `any` — where TypeScript
would infer `any` without you asking for it (e.g., untyped parameters).
Explicit `any` annotations are always allowed regardless of noImplicitAny.
// ❌ with noImplicitAny: function fn(x) {} — x implicitly has type any
// ✅ with noImplicitAny: function fn(x: any) {} — explicit is fine

TRAP 7: What does `never` extend?
Answer: `never` extends EVERYTHING. `never extends T` is always true for any T.
This is because never is the bottom type — a subtype of every type (empty set ⊆ any set).
But nothing except `never` extends `never` (only the empty set is a subset of the empty set).

type Test1<T> = T extends string ? "yes" : "no";
type T1 = Test1<never>; // never (distributes over empty union)

// And:
type T2 = never extends string ? "yes" : "no"; // "yes" — never extends everything

TRAP 8: Why does `Object.keys()` return string[] not (keyof T)[]?
Answer: TypeScript intentionally widens Object.keys to string[] because an object
may have MORE keys at runtime than its declared type (structural typing + excess
properties through assignment). TypeScript keeps this safe.
Workaround when you know the shape:
  (Object.keys(obj) as (keyof typeof obj)[]).forEach(...)

SCENARIO 1: You have a JavaScript library with no types. What do you do?
Steps:
1. Check if @types/library-name exists on DefinitelyTyped (npm i -D @types/lib)
2. If not, write an ambient declaration in src/types/library-name.d.ts:
   declare module "library-name" {
     export function doThing(x: string): number;
   }
3. For rapid iteration, use `declare module "library-name"` with no body
   (allows `import x from "library-name"` typed as any)

SCENARIO 2: How do you model a Redux-like action system in TypeScript?
// Use discriminated unions for actions, generics for reducers:
type Action =
  | { type: "INCREMENT"; payload: number }
  | { type: "RESET" }
  | { type: "SET_USER"; payload: User };

function reducer(state: State, action: Action): State {
  switch (action.type) {
    case "INCREMENT": return { ...state, count: state.count + action.payload };
    case "RESET":     return initialState;
    case "SET_USER":  return { ...state, user: action.payload };
    default:
      const _: never = action; // exhaustiveness check
      return state;
  }
}

SCENARIO 3: How would you make a function's return type depend on a boolean parameter?
// Use function overloads:
function parse(raw: string, strict: true): ParsedData;
function parse(raw: string, strict: false): ParsedData | null;
function parse(raw: string, strict: boolean): ParsedData | null {
  try { return JSON.parse(raw); }
  catch { return strict ? throw new Error("parse error") : null; }
}

const a = parse(input, true);  // ParsedData — never null
const b = parse(input, false); // ParsedData | null

SCENARIO 4: You need to accept either a callback or a promise-based API — what type?
type AsyncOrSync<T> = T | Promise<T>;
type Callback<T> = (err: Error | null, result: T) => void;

// Overloads for dual API:
function fetchData(id: string): Promise<User>;
function fetchData(id: string, cb: Callback<User>): void;
function fetchData(id: string, cb?: Callback<User>): Promise<User> | void { ... }
