SECTION 1: GENERICS FUNDAMENTALS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q1: What are generics and why do we need them?
A:  Generics allow you to write reusable code that works with multiple types
    while preserving type safety. Without generics, you'd use `any` and lose all
    type information, or write separate functions for each type.

    Without generics (loses type information):
    function identity(arg: any): any { return arg; }

    With generics (preserves type information):
    function identity<T>(arg: T): T { return arg; }

    const result = identity("hello"); // TypeScript infers: string
    const result2 = identity(42);     // TypeScript infers: number

    Use generics when the type relationship between inputs and outputs matters,
    or when you want to constrain multiple types to be compatible.

Q2: How does TypeScript infer generic type parameters?
A:  TypeScript infers generic type parameters from the arguments passed to
    the function. You only need to specify them explicitly when inference fails
    or when calling without arguments.

    // Inferred — TypeScript deduces T = string from argument
    const r1 = identity("hello");           // T inferred as string

    // Explicit — necessary when no argument to infer from
    const r2 = createArray<number>(5, 0);   // T = number

    // Inference with multiple params — T must satisfy both
    function merge<T, U>(a: T, b: U): T & U { ... }
    const m = merge({ name: "Alice" }, { age: 30 }); // T = {name:string}, U = {age:number}

Q3: What are generic constraints and how do you express them?
A:  `extends` constrains a type parameter to only accept types that are
    assignable to the constraint. This lets you safely access properties on T.

    // Without constraint: can't access .length
    function log<T>(arg: T): T {
      console.log(arg.length); // ❌ Error: Property 'length' does not exist on type T
      return arg;
    }

    // With constraint: T must have a length property
    function log<T extends { length: number }>(arg: T): T {
      console.log(arg.length); // ✅ safe
      return arg;
    }

    log("hello");      // ✅ string has .length
    log([1, 2, 3]);    // ✅ array has .length
    log(42);           // ❌ Error: number doesn't satisfy { length: number }

Q4: What is the `keyof` constraint and how is it used with generics?
A:  `keyof T` produces a union of all property keys of T. Combining it with
    a generic parameter enforces that a value is a valid key of another type:

    function getProperty<T, K extends keyof T>(obj: T, key: K): T[K] {
      return obj[key];
    }

    const user = { name: "Alice", age: 30 };
    const name = getProperty(user, "name"); // ✅ TypeScript knows return is string
    const age  = getProperty(user, "age");  // ✅ TypeScript knows return is number
    const bad  = getProperty(user, "email"); // ❌ not a key of user

    This pattern is fundamental to typed object lookup utilities.

Q5: What is the difference between generic constraints and generic defaults?
A:  Constraint (extends): restricts which types can be passed — a minimum requirement.
    Default: provides a fallback when no type argument is given — optional.

    // Constraint: T must extend object
    function clone<T extends object>(obj: T): T { ... }

    // Default: T defaults to string if not specified
    function createList<T = string>(): T[] { return []; }

    const nums = createList<number>(); // T = number
    const strs = createList();         // T = string (default)

    You can combine them: <T extends object = User>

Q6: Can you write a generic interface? Show an example.
A:  Yes. Generic interfaces parameterize the shape description:

    interface Repository<T, ID = number> {
      findById(id: ID): Promise<T | null>;
      findAll(): Promise<T[]>;
      save(entity: T): Promise<T>;
      delete(id: ID): Promise<void>;
    }

    class UserRepository implements Repository<User, string> {
      async findById(id: string): Promise<User | null> { ... }
      async findAll(): Promise<User[]> { ... }
      async save(user: User): Promise<User> { ... }
      async delete(id: string): Promise<void> { ... }
    }

Q7: How do you write a generic class and what are the restrictions?
A:  Generic classes parameterize one or more type parameters at the class level.
    The type parameter is available on all instance members but NOT static members.

    class Stack<T> {
      private items: T[] = [];

      push(item: T): void { this.items.push(item); }
      pop(): T | undefined { return this.items.pop(); }
      peek(): T | undefined { return this.items[this.items.length - 1]; }
      isEmpty(): boolean { return this.items.length === 0; }
    }

    const numStack = new Stack<number>();
    numStack.push(1);
    numStack.push("oops"); // ❌ Error: Argument of type 'string' not assignable to 'number'

    // Static members CANNOT use the class's type parameter
    class Container<T> {
      static defaultValue: T; // ❌ Error: static member cannot use type parameter
    }

Q8: What is a generic constraint using `extends` on a class/constructor type?
A:  When you need to create instances of a generic type, you must express it
    as a constructor signature. This is the "mixin pattern":

    // T must be a constructor returning an instance
    type Constructor<T = {}> = new (...args: any[]) => T;

    function withTimestamp<TBase extends Constructor>(Base: TBase) {
      return class extends Base {
        createdAt = new Date();
      };
    }

    class User { name = "" }
    const TimestampedUser = withTimestamp(User);
    const u = new TimestampedUser();
    u.createdAt; // ✅ Date
    u.name;      // ✅ string

Q9: What is conditional type inference? How does it relate to generics?
A:  The `infer` keyword inside a conditional type creates a type variable that
    TypeScript fills in during type resolution. It's how ReturnType works:

    type ReturnType<T extends (...args: any) => any> =
      T extends (...args: any) => infer R ? R : never;

    // Extracting the array element type:
    type ElementOf<T> = T extends (infer E)[] ? E : never;
    type E = ElementOf<string[]>; // string

    // Extracting promise resolution type:
    type Awaited<T> = T extends Promise<infer V> ? V : T;
    type V = Awaited<Promise<number>>; // number

Q10: What is "distributive" behavior in generic conditional types?
A:   When T is a naked type parameter in a conditional type, TypeScript distributes
     the conditional over each member of a union:

     type IsString<T> = T extends string ? "yes" : "no";
     type R = IsString<string | number>; // "yes" | "no"  ← distributed

     To prevent distribution, wrap T in a tuple or array:
     type IsString<T> = [T] extends [string] ? "yes" : "no";
     type R2 = IsString<string | number>; // "no"  ← not distributed

     Exclude is built on distribution:
     type Exclude<T, U> = T extends U ? never : T;
     type E = Exclude<"a" | "b" | "c", "a">;
     // Distributes: ("a" extends "a" ? never : "a") | ("b"...) | ("c"...)
     // = never | "b" | "c" = "b" | "c"

Q11: What are mapped types over generic parameters?
A:   You can map over a generic type parameter using `keyof T`:

     type Readonly<T> = { readonly [K in keyof T]: T[K] };
     type Partial<T> = { [K in keyof T]?: T[K] };
     type Nullable<T> = { [K in keyof T]: T[K] | null };

     // More complex: make specific keys required, rest optional
     type RequireOnly<T, K extends keyof T> =
       Required<Pick<T, K>> & Partial<Omit<T, K>>;

     type UserForm = RequireOnly<User, "name" | "email">;
     // { name: string; email: string; age?: number; ... }

Q12: What is `infer` variance — why does function parameter position matter?
A:   When inferring from function parameter position (contravariant), TypeScript
     produces an intersection; from return position (covariant) an union:

     // Infer from return type → union
     type UnionFromReturn<T> = T extends () => infer R ? R : never;
     type U = UnionFromReturn<(() => string) | (() => number)>; // string | number

     // Infer from parameter position → intersection
     type InterFromParam<T> = T extends (arg: infer P) => void ? P : never;
     type I = InterFromParam<((x: string) => void) | ((x: number) => void)>; // never (string & number)

     This is why UnionToIntersection works by converting a union to function
     parameters and then extracting via infer.

Q13: How do you type a generic React component?
A:   In TSX files, the parser might confuse `<T>` with a JSX tag. Fix with a
     trailing comma or extend constraint:

     // Option 1: trailing comma
     function Select<T,>({ options, value, onChange }: SelectProps<T>) { ... }

     // Option 2: constraint (common in practice)
     function Select<T extends object>({ options }: SelectProps<T>) { ... }

     // Option 3: explicitly in .tsx files with constraint
     interface SelectProps<T> {
       options: T[];
       value: T;
       onChange: (value: T) => void;
       getLabel: (option: T) => string;
     }

Q14: How do you write a generic custom hook?
A:   Generic hooks infer the data type from arguments or return typed state:

     function useLocalStorage<T>(key: string, initial: T): [T, (val: T) => void] {
       const [value, setValue] = useState<T>(() => {
         const stored = localStorage.getItem(key);
         return stored ? JSON.parse(stored) : initial;
       });

       const setStored = (val: T) => {
         setValue(val);
         localStorage.setItem(key, JSON.stringify(val));
       };

       return [value, setStored];
     }

     const [count, setCount] = useLocalStorage<number>("count", 0);
     // count: number, setCount: (val: number) => void

Q15: What is the generic `as const` pattern and why is it important?
A:   `as const` prevents type widening and makes all values readonly with
     their literal types. Combined with typeof, it's the modern alternative
     to enums:

     const STATUS = {
       PENDING: "pending",
       ACTIVE: "active",
       INACTIVE: "inactive",
     } as const;

     type Status = typeof STATUS[keyof typeof STATUS]; // "pending" | "active" | "inactive"

     function setStatus(s: Status) { ... }
     setStatus("pending");  // ✅
     setStatus("unknown"); // ❌ not in the type
