1. WHAT IS AN INTERFACE?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

An interface defines the SHAPE (contract) of an object — what properties and
methods it must have. Interfaces only exist at compile time and are erased
from the JavaScript output.

  interface User {
    id: number;
    name: string;
    email: string;
  }

  // Any object with at least these properties satisfies the interface:
  const alice: User = { id: 1, name: "Alice", email: "alice@example.com" };

  // Missing property → compile error:
  const bob: User = { id: 2, name: "Bob" }; // ❌ Property 'email' is missing

2. INTERFACE PROPERTY MODIFIERS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

a) Required Properties (default):
   ─────────────────────────────────
   interface Config {
     host: string;
     port: number;
   }
   // Both host and port MUST be present

b) Optional Properties (? modifier):
   ─────────────────────────────────────
   interface Config {
     host: string;
     port?: number;   // Optional — may or may not be present
     debug?: boolean; // type is boolean | undefined
   }
   const cfg: Config = { host: "localhost" }; // ✅ port omitted

c) Readonly Properties:
   ──────────────────────
   interface Point {
     readonly x: number;
     readonly y: number;
   }
   const p: Point = { x: 1, y: 2 };
   p.x = 5;  // ❌ Error: Cannot assign to 'x' because it is a read-only property

d) Index Signatures (dynamic keys):
   ──────────────────────────────────
   interface StringMap {
     [key: string]: string;
   }
   const headers: StringMap = {
     "Content-Type": "application/json",
     Authorization: "Bearer token123",
   };

   interface NumberDictionary {
     [key: string]: number;
     length: number; // specific properties must be compatible with index signature
   }

3. FUNCTION TYPES IN INTERFACES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  interface Calculator {
    // Method signature (preferred style):
    add(a: number, b: number): number;
    subtract(a: number, b: number): number;

    // Call signature (for callable objects):
    (value: number): string;  // the object itself is callable

    // Overloads via multiple call signatures:
    format(value: number): string;
    format(value: string): number;
  }

  // Interface for a standalone function:
  interface Validator {
    (value: string): boolean;
  }

  const isEmail: Validator = (value) => value.includes("@");

  // Interface with optional method:
  interface EventEmitter {
    on(event: string, handler: () => void): void;
    off?(event: string, handler: () => void): void; // optional method
  }

4. INTERFACE EXTENSION (INHERITANCE)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Interfaces can extend other interfaces using the `extends` keyword.

  interface Animal {
    name: string;
    sound(): string;
  }

  interface Pet extends Animal {
    owner: string;
  }

  interface Dog extends Pet {
    breed: string;
  }

  const rex: Dog = {
    name: "Rex",
    sound: () => "Woof",
    owner: "Alice",
    breed: "German Shepherd"
  };

  // Multiple inheritance — interfaces can extend multiple interfaces:
  interface Flyable {
    fly(): void;
  }
  interface Swimmable {
    swim(): void;
  }
  interface Duck extends Flyable, Swimmable {
    quack(): void;
  }

  // Interface extends a class (inherits class members' types):
  class Control {
    private state: string = "";
  }
  interface SelectableControl extends Control {
    select(): void;
  }
  // Only classes extending Control can implement SelectableControl

5. DECLARATION MERGING (UNIQUE TO INTERFACES)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Interfaces with the same name in the same scope are MERGED into one.
  This is called "declaration merging" or "interface augmentation".

  interface Window {
    title: string;
  }

  interface Window {            // same name — merged!
    height: number;
  }

  // TypeScript sees both declarations as one:
  // interface Window { title: string; height: number; }

  const win: Window = { title: "TS Guide", height: 768 }; // ✅

  PRACTICAL USE: Extending built-in interfaces or third-party library types.

  // Extending Express Request:
  declare module "express" {
    interface Request {
      user?: { id: string; role: string };
    }
  }

  // Adding to Window:
  interface Window {
    gtag?: (event: string, ...args: unknown[]) => void;
  }

  NOTE: Type aliases do NOT support declaration merging. This is a key
  difference between `interface` and `type`.

6. IMPLEMENTING INTERFACES IN CLASSES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  interface Printable {
    print(): void;
  }

  interface Serializable {
    serialize(): string;
  }

  class Document implements Printable, Serializable {
    constructor(private content: string) {}

    print(): void {
      console.log(this.content);
    }

    serialize(): string {
      return JSON.stringify({ content: this.content });
    }
  }

  // A class can implement multiple interfaces (unlike Java's single extends)
  // The class must implement ALL methods and properties required by the interface

  // Interface enforces the PUBLIC API of a class:
  interface Shape {
    area(): number;
    perimeter(): number;
  }

  class Circle implements Shape {
    constructor(private radius: number) {}
    area() { return Math.PI * this.radius ** 2; }
    perimeter() { return 2 * Math.PI * this.radius; }
  }

  class Square implements Shape {
    constructor(private side: number) {}
    area() { return this.side ** 2; }
    perimeter() { return 4 * this.side; }
  }

  // Polymorphism via interface:
  function printShapeInfo(shape: Shape): void {
    console.log(`Area: ${shape.area()}, Perimeter: ${shape.perimeter()}`);
  }
  printShapeInfo(new Circle(5));  // ✅
  printShapeInfo(new Square(4));  // ✅

7. INTERFACE vs OBJECT LITERAL TYPE CHECK (EXCESS PROPERTY CHECK)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  TypeScript performs EXCESS PROPERTY CHECKS when assigning object literals:

  interface Point { x: number; y: number; }

  // Direct assignment of object literal → excess property check:
  const p: Point = { x: 1, y: 2, z: 3 }; // ❌ Error: Object literal may only specify known properties

  // Via variable assignment → no excess property check:
  const obj = { x: 1, y: 2, z: 3 };
  const p2: Point = obj;  // ✅ No error (structural compatibility check)

  // Via type assertion → no excess property check:
  const p3 = { x: 1, y: 2, z: 3 } as Point; // ✅

  This is intentional: direct object literals are likely typos; variables
  may be purposely wider (Liskov substitution principle).

8. CONSTRUCTOR SIGNATURES IN INTERFACES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  An interface can describe a constructor function (factory):

  interface ClockConstructor {
    new (hour: number, minute: number): ClockInterface;
  }

  interface ClockInterface {
    tick(): void;
  }

  function createClock(ctor: ClockConstructor, hour: number, min: number) {
    return new ctor(hour, min);
  }

  class DigitalClock implements ClockInterface {
    constructor(h: number, m: number) {}
    tick() { console.log("beep beep"); }
  }

  const clock = createClock(DigitalClock, 12, 30);

9. GENERIC INTERFACES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  interface Repository<T> {
    findById(id: number): T | null;
    findAll(): T[];
    save(item: T): void;
    delete(id: number): boolean;
  }

  interface User {
    id: number;
    name: string;
  }

  class UserRepository implements Repository<User> {
    private users: User[] = [];
    findById(id: number) { return this.users.find(u => u.id === id) ?? null; }
    findAll() { return this.users; }
    save(user: User) { this.users.push(user); }
    delete(id: number) {
      const idx = this.users.findIndex(u => u.id === id);
      if (idx === -1) return false;
      this.users.splice(idx, 1);
      return true;
    }
  }

  // Generic interface with constraints:
  interface Comparable<T> {
    compareTo(other: T): number; // negative, 0, or positive
  }

  class Temperature implements Comparable<Temperature> {
    constructor(public celsius: number) {}
    compareTo(other: Temperature) { return this.celsius - other.celsius; }
  }

10. HYBRID TYPES (CALLABLE + PROPERTIES)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Some JavaScript patterns create objects that are both callable functions
  AND have properties. TypeScript interfaces can model this:

  interface Counter {
    (start: number): string;  // call signature
    interval: number;         // property
    reset(): void;            // method
  }

  function getCounter(): Counter {
    let counter = function(start: number) { return `${start}`; } as Counter;
    counter.interval = 1000;
    counter.reset = function() { console.log("reset"); };
    return counter;
  }

  const c = getCounter();
  c(10);           // ✅ callable
  c.interval = 500; // ✅ has properties
  c.reset();       // ✅ has methods

11. COMMON INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q1: What is an interface in TypeScript?
    An interface is a compile-time contract that defines the shape (structure)
    of an object. It specifies what properties and methods an object must have.
    Interfaces are erased at runtime — they produce no JavaScript output.

Q2: What is declaration merging in interfaces?
    When two interfaces with the same name exist in the same scope, TypeScript
    merges them into a single interface with all properties combined. This is
    unique to interfaces (not type aliases). It's used for augmenting existing
    types (like extending Window or Express Request).

Q3: Can an interface extend a class?
    Yes! An interface can extend a class. It inherits the class's type (including
    private and protected members). Only subclasses of the extended class can
    implement the interface — because they'd need access to the private members.

Q4: What is an index signature?
    An index signature lets you define a type for all properties with a certain
    key type: [key: string]: number allows any string key mapping to a number.
    Useful for dynamic property names (dictionaries, maps).

Q5: What is the difference between optional property (?) and the property type including undefined?
    Optional property: property can be omitted from the object entirely.
    type T = { x?: number } → x can be absent or number
    
    Union with undefined: property MUST be present but can be undefined.
    type T = { x: number | undefined } → x must be there, can be undefined
    
    With exactOptionalPropertyTypes: true, these are treated differently.

Q6: What is excess property checking?
    When you assign an OBJECT LITERAL directly to a typed variable, TypeScript
    checks for extra properties that don't exist in the type. This catches typos.
    This check does NOT happen when assigning through an intermediate variable.

Q7: Can a class implement multiple interfaces?
    Yes! Unlike class extension (single inheritance), a class can implement
    as many interfaces as needed: class A implements B, C, D {}

Q8: What is a call signature in an interface?
    A call signature makes an interface describe a callable function:
    interface Fn { (x: number): string; }
    An object implementing Fn must be callable as a function.

Q9: How do you describe a function type using an interface?
    Two ways:
    // Method signature:
    interface Adder { add(a: number, b: number): number; }
    // Call signature:
    interface Adder { (a: number, b: number): number; }

Q10: What's the difference between interface inheritance and implementation?
     interface A extends B → A IS a type of B (structural inheritance)
     class A implements B → A fulfills the contract of B (but A has impl)
     One gives you types for free (extends), the other requires you to write
     the implementation (implements).
