1. UNION TYPES (A | B)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

A union type means a value can be ONE OF several types.
Think of it as "OR" — type A OR type B.

  type StringOrNumber = string | number;

  let id: StringOrNumber = "abc-123"; // ✅
  id = 42;                            // ✅
  id = true;                          // ❌ boolean is not in the union

  // With null/undefined:
  type MaybeString = string | null | undefined;

  // String literal unions (most common usage):
  type Direction = "north" | "south" | "east" | "west";
  type Status = "pending" | "active" | "closed";
  type HttpMethod = "GET" | "POST" | "PUT" | "DELETE" | "PATCH";

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
2. WORKING WITH UNIONS — NARROWING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  You can only access members common to ALL types in the union without narrowing.

  function printId(id: string | number) {
    // id.toUpperCase(); // ❌ Error: number doesn't have toUpperCase
    // id.toFixed();     // ❌ Error: string doesn't have toFixed

    console.log(id);     // ✅ both string and number can be logged

    // Must narrow before using type-specific methods:
    if (typeof id === "string") {
      console.log(id.toUpperCase()); // ✅ narrowed to string
    } else {
      console.log(id.toFixed(2));    // ✅ narrowed to number
    }
  }

  Narrowing Techniques:
  a) typeof:
     if (typeof value === "string") { ... }
     if (typeof value === "number") { ... }
     if (typeof value === "boolean") { ... }

  b) instanceof:
     if (value instanceof Date) { ... }
     if (value instanceof Error) { ... }

  c) in operator:
     if ("name" in value) { ... }  // checks property existence

  d) Discriminant property:
     if (shape.kind === "circle") { ... }

  e) Truthiness:
     if (value) { ... }  // narrows out null, undefined, 0, ""

  f) Equality:
     if (value === null) { ... }

3. UNION WITH OBJECT TYPES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  When union members are object types, only shared properties are accessible:

  interface Cat { type: "cat"; meow(): void; purr(): void; }
  interface Dog { type: "dog"; bark(): void; fetch(): void; }

  type Pet = Cat | Dog;

  function interact(pet: Pet) {
    pet.type;    // ✅ shared property — available without narrowing

    // pet.meow();  // ❌ not in Dog
    // pet.bark();  // ❌ not in Cat

    // Must narrow:
    if (pet.type === "cat") {
      pet.meow();  // ✅
      pet.purr();  // ✅
    } else {
      pet.bark();  // ✅
      pet.fetch(); // ✅
    }
  }

4. DISCRIMINATED UNIONS (TAGGED UNIONS)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  A discriminated union uses a common literal-typed "tag" property to
  distinguish between members. This is one of TypeScript's most powerful
  patterns.

  type Circle = { kind: "circle"; radius: number };
  type Square = { kind: "square"; side: number };
  type Triangle = { kind: "triangle"; base: number; height: number };

  type Shape = Circle | Square | Triangle;

  function area(shape: Shape): number {
    switch (shape.kind) {
      case "circle":
        return Math.PI * shape.radius ** 2;  // shape is Circle here
      case "square":
        return shape.side ** 2;               // shape is Square here
      case "triangle":
        return 0.5 * shape.base * shape.height; // shape is Triangle here
      default:
        const exhaustive: never = shape;        // catches unhandled cases!
        throw new Error(`Unknown shape: ${exhaustive}`);
    }
  }

  // Adding a new shape type:
  type Pentagon = { kind: "pentagon"; side: number };
  type Shape = Circle | Square | Triangle | Pentagon; // added

  // Now the default case in area() will error:
  // Argument of type 'Pentagon' is not assignable to parameter of type 'never'
  // → TypeScript tells you to handle the new case!

5. INTERSECTION TYPES (A & B)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

An intersection type combines multiple types into one.
A value must satisfy ALL types simultaneously.
Think of it as "AND" — type A AND type B.

  type Named = { name: string };
  type Aged = { age: number };
  type Person = Named & Aged;
  // Must have BOTH name (string) AND age (number)

  const alice: Person = { name: "Alice", age: 30 }; // ✅
  const bob: Person = { name: "Bob" };              // ❌ missing age

6. PRACTICAL PATTERNS WITH INTERSECTION
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

a) Combining base type with extras:
   ─────────────────────────────────
   interface Entity { id: number; createdAt: Date; updatedAt: Date }
   interface User { name: string; email: string }
   type UserEntity = User & Entity;
   // Has: id, createdAt, updatedAt, name, email

b) Mixin pattern:
   ────────────────
   type Timestamped = { createdAt: Date; updatedAt: Date };
   type SoftDeletable = { deletedAt: Date | null };
   type Auditable = Timestamped & SoftDeletable;

   type Post = {
     title: string;
     content: string;
   } & Auditable;

c) Extending function types:
   ──────────────────────────
   type Fn = (x: number) => number;
   type FnWithMeta = Fn & { description: string };

   const double: FnWithMeta = Object.assign(
     (x: number) => x * 2,
     { description: "Doubles a number" }
   );

d) Class-based Mixin:
   ─────────────────────
   interface Serializable { serialize(): string }
   interface Loggable { log(): void }
   type SerializableLogger = Serializable & Loggable;

7. INTERSECTION WITH CONFLICTING TYPES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  When two types in an intersection have the same property with incompatible
  types, the result is `never`:

  type A = { x: string };
  type B = { x: number };
  type AB = A & B;  // { x: string & number } = { x: never }

  const obj: AB = { x: "hello" }; // ❌ Error: x must be string AND number = never
  const obj2: AB = { x: 42 };     // ❌ Error: same reason

  This is a type-system impossibility. If you see `never` in a type, check
  for conflicting property types in intersections.

  EXCEPTION — function overloads:
  type Stringify = (x: string) => string;
  type Numberify = (x: number) => number;
  type Both = Stringify & Numberify;

  const fn: Both = (x: any) => x; // callable with string or number!
  fn("hello");  // ✅ → string
  fn(42);       // ✅ → number
  // Intersection of function types = overloads!

8. UNION vs INTERSECTION: THE SET ANALOGY
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  UNION  (A | B):
    Set of values that belong to A OR B (set union).
    More possible values — "wider" type.
    Fewer guaranteed properties (only what A and B share).

  INTERSECTION (A & B):
    Set of values that belong to A AND B (set intersection).
    Fewer possible values — "narrower" type.
    More guaranteed properties (all properties from both A and B).

  Counterintuitive insight:
    When combining object types:
    - Union: FEWER guaranteed properties (only shared ones)
    - Intersection: MORE guaranteed properties (all from both)

    interface A { x: number; y: number }
    interface B { y: number; z: number }

    A | B → only y is guaranteed (shared)
    A & B → x, y, and z are all guaranteed

9. ADVANCED PATTERNS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

a) Result type (functional programming pattern):
   ─────────────────────────────────────────────
   type Ok<T> = { success: true; value: T };
   type Err<E = Error> = { success: false; error: E };
   type Result<T, E = Error> = Ok<T> | Err<E>;

   function divide(a: number, b: number): Result<number> {
     if (b === 0) return { success: false, error: new Error("Division by zero") };
     return { success: true, value: a / b };
   }

   const result = divide(10, 2);
   if (result.success) {
     console.log(result.value);  // ✅ narrowed to Ok<number>
   } else {
     console.log(result.error.message); // ✅ narrowed to Err<Error>
   }

b) State machine pattern:
   ────────────────────────
   type Idle      = { state: "idle" };
   type Loading   = { state: "loading"; requestId: string };
   type Success   = { state: "success"; data: string[] };
   type Error     = { state: "error"; message: string; retryCount: number };

   type AppState = Idle | Loading | Success | Error;

c) Option/Maybe type:
   ────────────────────
   type None = { readonly _tag: "None" };
   type Some<A> = { readonly _tag: "Some"; readonly value: A };
   type Option<A> = None | Some<A>;

   const none: None = { _tag: "None" };
   const some = <A>(value: A): Some<A> => ({ _tag: "Some", value });

10. COMMON INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q1: What is a union type?
    A union type (A | B) means a value can be either type A or type B.
    You can only access properties/methods common to ALL members without
    narrowing. Used for flexible APIs, nullable types, and discriminated unions.

Q2: What is an intersection type?
    An intersection type (A & B) means a value must satisfy BOTH type A AND B.
    The resulting type has ALL properties from both types. Used for mixing in
    additional properties, combining base types with extras, mixin patterns.

Q3: What is a discriminated union?
    A union type where each member has a common "discriminant" property with
    a unique literal type (like kind: "circle" | "square"). TypeScript uses
    this to narrow the type in switch/if-else statements. Enables exhaustive
    type checking and is TypeScript's recommended pattern for sum types.

Q4: What happens when two types in an intersection have conflicting property types?
    The property type becomes `never` — an impossible type.
    type A = { x: string } & { x: number } → { x: never }
    This means no value can satisfy the intersection. Avoid this pattern.

Q5: How does TypeScript handle property access on union types?
    Without narrowing, only properties that exist in ALL union members are
    accessible. TypeScript prevents accessing properties that might not exist.
    Use typeof, instanceof, discriminant property checks, or in operator to narrow.

Q6: What is type narrowing?
    Type narrowing is when TypeScript reduces a union type to a more specific
    type within a conditional block. After if (typeof x === "string"), x is
    narrowed to string inside the block. TypeScript tracks control flow.

Q7: What is the difference between `A | B` and `A & B` in terms of object properties?
    A | B: FEWER accessible properties (only shared ones guaranteed)
    A & B: MORE accessible properties (all from both types)
    This is counterintuitive — union has more value possibilities but fewer
    guaranteed properties; intersection has fewer value possibilities but more properties.

Q8: Can you use intersection to combine function signatures?
    Yes! Intersecting function types creates an overloaded function type:
    type Fn = ((x: string) => string) & ((x: number) => number)
    This allows the function to be called with either signature.

Q9: What is exhaustiveness checking in discriminated unions?
    By assigning the "remaining" case to `never` in the default branch:
    default: const _: never = myUnionValue;
    TypeScript will error if any union member isn't handled.
    This guarantees all cases are handled, preventing bugs when new cases are added.

Q10: When would you use a union type vs. an optional property?
     Union type: The shape of the whole object changes based on the value.
     type Action = { type: "click"; x: number; y: number } | { type: "key"; key: string }
     Optional property: The same object can have or lack a property.
     interface Config { timeout: number; debug?: boolean }
     Use union types for fundamentally different shapes; optional properties
     for the same shape with variable presence.
