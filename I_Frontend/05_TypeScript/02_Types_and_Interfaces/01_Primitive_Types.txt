1. WHAT ARE PRIMITIVE TYPES?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Primitives are the most basic data types in TypeScript. They correspond
directly to JavaScript's primitive values and are NOT objects.

TypeScript Primitive Types:
  ┌────────────────┬────────────────────────────────────────────────────────┐
  │ Type           │ Description                                            │
  ├────────────────┼────────────────────────────────────────────────────────┤
  │ string         │ Text values                                            │
  │ number         │ All numeric values (int and float)                     │
  │ boolean        │ true or false                                          │
  │ bigint         │ Arbitrary precision integers (ES2020)                  │
  │ symbol         │ Unique, immutable identifiers                          │
  │ null           │ Intentional absence of value                           │
  │ undefined      │ Uninitialized / no value                               │
  │ never          │ Values that never occur (bottom type)                  │
  │ unknown        │ Safe top type (opposite of never)                      │
  │ any            │ Opt-out of type checking                               │
  │ void           │ Function returns nothing meaningful                    │
  └────────────────┴────────────────────────────────────────────────────────┘

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
2. STRING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  let firstName: string = "Alice";
  let greeting: string = `Hello, ${firstName}!`;  // template literals
  let empty: string = "";

  // ALL of these are valid string literals:
  let a = "double quotes";
  let b = 'single quotes';
  let c = `backtick`;

  // TypeScript also supports LITERAL string types:
  let status: "active" | "inactive" | "pending" = "active";
  status = "unknown"; // ❌ Error: not one of the allowed literals

  // Common string operations TypeScript knows about:
  firstName.toUpperCase();   // ✅ TypeScript knows this method exists
  firstName.includes("Al");  // ✅

  // Important: Use lowercase `string`, not `String` (wrapper object)
  let s1: string = "hello";  // ✅ primitive (correct)
  let s2: String = "hello";  // ⚠️  wrapper object (avoid)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
3. NUMBER
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  TypeScript (like JavaScript) uses a single number type for ALL numbers:
  integers, floats, hex, octal, binary, NaN, Infinity.

  let age: number = 30;
  let price: number = 9.99;
  let hex: number = 0xff;         // 255
  let binary: number = 0b1010;    // 10
  let octal: number = 0o744;      // 484
  let inf: number = Infinity;
  let notANumber: number = NaN;   // NaN is type number!

  // Numeric separators (readability, TS 2.7+):
  let million: number = 1_000_000;
  let pi: number = 3.141_592_653;

  // Literal number types:
  let one: 1 = 1;
  let port: 8080 | 3000 | 443 = 8080;

  // Use lowercase `number`, not `Number` (wrapper)
  let n1: number = 42;  // ✅
  let n2: Number = 42;  // ⚠️  avoid

  // Pitfall: NaN === NaN is false in JavaScript
  // TypeScript cannot guard against NaN comparisons:
  const x: number = NaN;
  if (x === NaN) { ... } // always false!
  if (isNaN(x)) { ... }  // correct way

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
4. BOOLEAN
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  let isActive: boolean = true;
  let isDeleted: boolean = false;

  // Literal boolean types:
  type AlwaysTrue = true;
  type AlwaysFalse = false;

  // Functions returning boolean:
  function isEven(n: number): boolean {
    return n % 2 === 0;
  }

  // Use lowercase `boolean`, not `Boolean`
  let b1: boolean = true;   // ✅
  let b2: Boolean = true;   // ⚠️  avoid

  // Truthy/falsy: TypeScript does NOT narrow based on truthiness alone
  // unless you use if (value) — then it narrows out null/undefined/0/""

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
5. BIGINT
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  BigInt is for integers larger than Number.MAX_SAFE_INTEGER (2^53 - 1).
  Requires "target": "ES2020" or higher in tsconfig.json.

  let bigNum: bigint = 9007199254740991n;
  let anotherBig: bigint = BigInt("12345678901234567890");

  // Cannot mix bigint and number:
  let a = 10n;
  let b = 5;
  let c = a + b; // ❌ Error: Cannot mix BigInt and other types

  let d = a + BigInt(b); // ✅ convert first

  // Use cases:
  // • Cryptography
  // • Large integer IDs (e.g., database IDs > 2^53)
  // • Financial calculations requiring exact integer arithmetic

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
6. SYMBOL
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Symbols are unique and immutable values. Each Symbol() call creates a new,
  unique symbol even with the same description.

  const id = Symbol("id");
  const id2 = Symbol("id");
  console.log(id === id2); // false — always unique

  // Used as unique object property keys:
  const KEY = Symbol("key");
  const obj = { [KEY]: "value" };

  // unique symbol type (TS specific):
  const sym1: unique symbol = Symbol();
  const sym2: unique symbol = Symbol();
  type S1 = typeof sym1;
  type S2 = typeof sym2;
  // S1 and S2 are different types — guarantees uniqueness at type level

  // Use case: Well-known symbols (Symbol.iterator, Symbol.toPrimitive, etc.)
  class Range {
    [Symbol.iterator]() {
      // makes Range iterable
    }
  }

7. NULL AND UNDEFINED
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  null and undefined are distinct types in TypeScript (with strictNullChecks).

  let n: null = null;           // only value: null
  let u: undefined = undefined; // only value: undefined

  null meaning:   Intentional absence of a value (programmer set this to null)
  undefined meaning: Not yet assigned / doesn't exist

  // With strictNullChecks: false (NOT recommended):
  let name: string = null;       // ✅ allowed (dangerous!)
  let age: number = undefined;   // ✅ allowed (dangerous!)

  // With strictNullChecks: true (RECOMMENDED — default in strict mode):
  let name: string = null;       // ❌ Error
  let name: string | null = null; // ✅ Must explicitly declare

  // Handling null/undefined:
  function getUser(id: number): User | null {
    return db.find(id) ?? null;
  }

  const user = getUser(1);
  user.name;          // ❌ Error: user might be null
  user?.name;         // ✅ Optional chaining
  user!.name;         // ✅ Non-null assertion (you guarantee it's not null — use carefully!)
  if (user) user.name; // ✅ Type narrowing

  // null vs undefined in optional parameters:
  function log(msg?: string) { ... }
  // msg is: string | undefined (NOT null — unless you explicitly add null)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
8. VOID
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  void is used as the return type for functions that don't return a value.

  function logMessage(msg: string): void {
    console.log(msg);
    // no return statement, or just `return;`
  }

  // void vs undefined:
  // void: function might return undefined, but the caller shouldn't use it
  // undefined: explicitly has the value undefined

  // Callbacks with void return type:
  type Callback = () => void;
  const cb: Callback = () => {
    return 42;  // ✅ TypeScript allows this! void return type is special
    // The return value is discarded by the caller
  };

  // This is intentional: setTimeout callback can return anything, we ignore it
  setTimeout(() => 42, 1000); // ✅ setTimeout's callback is () => void

9. NEVER
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  never is the BOTTOM TYPE — represents values that never occur.
  No value is of type never. It's a subtype of every type.

  Use Cases:

  a) Functions that never return (always throw or infinite loop):
     function fail(message: string): never {
       throw new Error(message);  // never returns normally
     }
     function infiniteLoop(): never {
       while (true) {}  // never returns
     }

  b) Exhaustiveness checking in discriminated unions:
     type Shape = "circle" | "square" | "triangle";
     function area(shape: Shape): number {
       switch (shape) {
         case "circle": return Math.PI;
         case "square": return 1;
         case "triangle": return 0.5;
         default:
           const _exhaustive: never = shape; // ❌ Error if new case not handled!
           return _exhaustive;
       }
     }

  c) Impossible type intersections:
     type Impossible = string & number; // type is never (nothing is both)

  d) Conditional type branches that can't be reached:
     type NonNullable<T> = T extends null | undefined ? never : T;
     // For types that are null/undefined, the result branch is never

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
10. UNKNOWN
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  unknown is the TYPE-SAFE TOP TYPE — the opposite of never.
  Every type is assignable TO unknown. But unknown is not assignable to others
  without type narrowing.

  Introduced in TypeScript 3.0 as a safe alternative to `any`.

  let value: unknown;
  value = 42;            // ✅ any type assignable to unknown
  value = "hello";       // ✅
  value = { x: 1 };     // ✅

  // Cannot use unknown without narrowing:
  value.toUpperCase();            // ❌ Error: Object is of type 'unknown'
  const len = value.length;      // ❌ Error

  // Must narrow first:
  if (typeof value === "string") {
    value.toUpperCase();  // ✅ narrowed to string
  }
  if (value instanceof Date) {
    value.getFullYear();  // ✅ narrowed to Date
  }

  // Common use case: error handling (TS 4.0+)
  try {
    riskyOperation();
  } catch (e: unknown) {    // useUnknownInCatchVariables in strict mode
    if (e instanceof Error) {
      console.log(e.message);  // ✅
    }
  }

  // API response parsing:
  async function fetchData(): Promise<unknown> {
    const res = await fetch("/api/data");
    return res.json();  // unknown — must validate before use
  }

11. ANY
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  any opts out of TypeScript's type checking entirely. A value of type `any`
  can be assigned to any other type and any operation can be performed on it.

  let x: any = "hello";
  x = 42;               // ✅ fine
  x = true;             // ✅ fine
  x.toUpperCase();      // ✅ no error (but may fail at runtime!)
  x.nonExistentMethod(); // ✅ no error (TypeScript trusts you)

  // any is contagious — it spreads through your code:
  function doSomething(val: any) {
    return val.result; // return type is also any!
  }
  const result = doSomething(user); // result is any — type safety lost!

  When to use `any`:
  • Migrating legacy JS code to TS (temporary)
  • Third-party library with no types and you don't want to write .d.ts
  • Very dynamic code where type modeling is impractical (use sparingly)

  AVOID `any` in new TypeScript code. Use `unknown` instead when the type
  is truly unknown, and narrow it before use.

  // Implicit any (caught by noImplicitAny: true):
  function greet(name) { ... } // ❌ Parameter 'name' implicitly has 'any' type

12. ARRAYS AND TUPLES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Array Types:
  let nums: number[] = [1, 2, 3];          // preferred syntax
  let strs: Array<string> = ["a", "b"];    // generic syntax
  let matrix: number[][] = [[1, 2], [3, 4]]; // 2D array

  // Readonly arrays:
  const arr: readonly number[] = [1, 2, 3];
  const arr2: ReadonlyArray<number> = [1, 2, 3];
  arr.push(4);  // ❌ Error: Property 'push' does not exist on type 'readonly number[]'

Tuple Types:
  Tuples are fixed-length arrays where each position has a specific type.

  let pair: [string, number] = ["age", 30];
  let rgb: [number, number, number] = [255, 128, 0];

  pair[0].toUpperCase(); // ✅ TypeScript knows index 0 is string
  pair[1].toFixed();     // ✅ TypeScript knows index 1 is number
  pair[2];               // ❌ Error: Tuple type '[string, number]' has no element at index '2'

  // Optional tuple elements:
  type Maybe = [string, number?];
  let m1: Maybe = ["hello"];       // ✅
  let m2: Maybe = ["hello", 42];   // ✅

  // Rest elements in tuples:
  type StringsAndNumber = [string, ...number[]];
  let sn: StringsAndNumber = ["header", 1, 2, 3, 4]; // ✅

  // Labeled tuple elements (TS 4.0+):
  type Range = [start: number, end: number];

  // Named tuples in function return (avoids object overhead):
  function useState<T>(initial: T): [T, (val: T) => void] {
    let state = initial;
    return [state, (val) => { state = val; }];
  }

13. TYPE HIERARCHY (THE TYPE SYSTEM LATTICE)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Top types (everything assignable TO these):
    unknown — safe top type (requires narrowing before use)
    any     — unsafe top type (disables type checking)

  Primitive types (in the middle):
    string, number, boolean, bigint, symbol, null, undefined

  Literal types (subsets of primitives):
    "hello" ⊂ string
    42      ⊂ number
    true    ⊂ boolean

  Bottom type (assignable to everything, no actual values):
    never — empty set, no value is of type never

  Assignability:
    never  →  all types  →  unknown  (everything flows upward)
    any    →  all types  AND  all types → any  (bidirectional — UNSAFE)

14. COMMON INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q1: What is the difference between `any` and `unknown`?
    Both accept any value. But:
    - `any` turns OFF type checking. You can do anything with it.
    - `unknown` requires type narrowing before use. It's type-safe.
    Always prefer `unknown` over `any` when the type is genuinely unknown.

Q2: What is the difference between `void` and `undefined`?
    - `void` says "this function's return value should not be used"
    - `undefined` says "this has the specific value undefined"
    A function can return `void` but still technically return `undefined`.
    Callbacks typed as `() => void` can return any value — it's discarded.

Q3: What is `never` and what is it used for?
    `never` represents values that never occur (empty type/bottom type). Used for:
    1. Functions that always throw or loop forever
    2. Exhaustiveness checking in switch/if-else chains
    3. Type-level impossibilities (string & number = never)

Q4: Why should you avoid `any`?
    `any` completely disables TypeScript's type checking. It's "contagious" —
    operations on `any` return `any`, spreading unsafety through your code.
    Use `unknown` instead and narrow before use.

Q5: What is the difference between `null` and `undefined`?
    - `null`: intentional absence of value (developer set it to null)
    - `undefined`: value not assigned (default for uninitialized variables)
    With `strictNullChecks: true`, both are distinct types.
    Prefer `undefined` for optional values; reserve `null` for explicit absence.

Q6: What is a tuple type in TypeScript?
    A tuple is a fixed-length, fixed-type array where each position has a
    specific type. Example: [string, number] is a pair of exactly string first,
    number second. Tuples are more strict than arrays.

Q7: What is `readonly` on an array and what does it prevent?
    readonly number[] (or ReadonlyArray<number>) prevents mutation.
    You cannot use push(), pop(), splice(), sort() etc. But you can still
    read elements by index. Use for arrays that shouldn't be modified.

Q8: What are primitive wrapper types (String vs string)?
    Lowercase: `string`, `number`, `boolean` — actual primitive types (use these!)
    Uppercase: `String`, `Number`, `Boolean` — wrapper objects (avoid!)
    They're mostly interchangeable but can cause subtle issues:
    `string` extends `String` but not vice versa. Always use lowercase.

Q9: When would you use `bigint`?
    When working with integers larger than 2^53 - 1 (Number.MAX_SAFE_INTEGER):
    • Database IDs larger than JS can handle as numbers
    • Cryptographic operations
    • Financial calculations needing exact large integer arithmetic
    Requires target: "ES2020" or higher.

Q10: What does `strictNullChecks` do?
     With strictNullChecks: true, `null` and `undefined` are NOT assignable
     to other types. You must explicitly type them as `string | null` etc.
     This prevents "Cannot read property of null" runtime errors.
     It's included in `strict: true` and should always be enabled.
