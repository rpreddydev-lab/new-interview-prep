1. WHAT IS SSR IN NEXT.JS?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Server-Side Rendering (SSR) means the HTML for a page is generated ON THE
SERVER on EVERY individual request. The server fetches data, runs React,
produces HTML, and sends it to the browser.

  Request arrives → Server fetches data → React renders → HTML response

Pages Router: achieved via getServerSideProps()
App Router:  achieved by making a Server Component async and NOT caching
             (or using cache: 'no-store')

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
2. getServerSideProps — PAGES ROUTER
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Must be exported from a page file in pages/ directory.
Runs on the server for EVERY request. Never included in client bundle.

  // pages/dashboard.tsx
  import type { GetServerSideProps, NextPage } from 'next';

  interface Props {
    user: { id: string; name: string; email: string };
    stats: { orders: number; revenue: number };
  }

  const DashboardPage: NextPage<Props> = ({ user, stats }) => {
    return (
      <div>
        <h1>Welcome, {user.name}</h1>
        <p>Orders: {stats.orders}</p>
        <p>Revenue: ${stats.revenue}</p>
      </div>
    );
  };

  export const getServerSideProps: GetServerSideProps = async (context) => {
    const { req, res, params, query, resolvedUrl, locale, locales } = context;

    // Access cookies
    const token = req.cookies['auth-token'];
    if (!token) {
      return {
        redirect: {
          destination: '/login',
          permanent: false,  // 307 redirect (use true for 301/308)
        },
      };
    }

    // Fetch data
    const [user, stats] = await Promise.all([
      fetchUser(token),
      fetchStats(token),
    ]);

    // Return props
    return {
      props: {
        user,
        stats,
      },
    };
  };

  export default DashboardPage;

3. getServerSideProps RETURN VALUES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // 1. Return props (normal render)
  return { props: { data } };

  // 2. Redirect
  return {
    redirect: {
      destination: '/login',
      permanent: false,         // false = 307, true = 301
    },
  };

  // 3. Not found (renders 404 page)
  return { notFound: true };

  // 4. Set cache headers via res
  res.setHeader('Cache-Control', 'public, s-maxage=10, stale-while-revalidate=59');
  return { props: { data } };

4. CONTEXT OBJECT IN getServerSideProps
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  export const getServerSideProps: GetServerSideProps = async (context) => {
    const {
      req,          // IncomingMessage — access headers, cookies, url
      res,          // ServerResponse — set headers, cookies
      params,       // dynamic route params: { id: '123' }
      query,        // URL query string: { page: '2', filter: 'active' }
      resolvedUrl,  // resolved URL including query: '/products?page=2'
      locale,       // current locale: 'en'
      locales,      // all locales: ['en', 'fr', 'de']
      defaultLocale,// default locale: 'en'
      preview,      // boolean: is preview mode active?
      previewData,  // preview mode data
    } = context;

    // Accessing cookies (req.cookies)
    const userId = req.cookies['user-id'];

    // Accessing specific header
    const userAgent = req.headers['user-agent'];

    // Accessing query params
    const page = Number(query.page ?? 1);

    return { props: {} };
  };

5. SSR IN APP ROUTER (MODERN APPROACH)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

In App Router, SSR is the default for async Server Components that use
dynamic data (cookies, headers) or disable caching.

  // app/dashboard/page.tsx — SSR by default when using cookies()
  import { cookies } from 'next/headers';
  import { redirect } from 'next/navigation';

  export default async function DashboardPage() {
    const cookieStore = await cookies();
    const token = cookieStore.get('auth-token')?.value;

    if (!token) redirect('/login');

    // fetch with no-store = SSR (never cached)
    const data = await fetch('https://api.example.com/dashboard', {
      headers: { Authorization: `Bearer ${token}` },
      cache: 'no-store',  // force SSR behavior
    }).then(r => r.json());

    return <Dashboard data={data} />;
  }

  // Force dynamic (SSR) explicitly:
  export const dynamic = 'force-dynamic';
  // This is equivalent to getServerSideProps in behavior

  // Dynamic functions that auto-opt into SSR (force-dynamic):
  // - cookies()
  // - headers()
  // - searchParams prop in page.tsx
  // - reading request URL at runtime

6. WHEN TO USE SSR
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Use SSR (getServerSideProps / force-dynamic) when:
  ✓ Data changes on every request (live prices, stock, user-specific)
  ✓ You need request-time info: cookies, headers, user-agent, IP
  ✓ Personalized content that must be SEO-indexed (user profile pages)
  ✓ Authentication checks before render (auth-gated pages)
  ✓ Real-time data: search results, news feeds, social timelines
  ✓ A/B testing where the variant depends on the request

Do NOT use SSR when:
  ✗ Content is the same for all users (use SSG/ISR)
  ✗ Performance is critical and data rarely changes (use ISR)
  ✗ Content is user-specific and doesn't need SEO (use CSR)

7. SSR PERFORMANCE CONSIDERATIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

SSR performance depends on:
  1. Database/API speed (most critical bottleneck)
  2. Server location relative to user
  3. Number of sequential vs parallel data fetches

Optimization techniques:

  A) Parallel fetching (avoid waterfall):
     // BAD: sequential (total = t1 + t2)
     const user = await fetchUser(id);
     const posts = await fetchPosts(user.id);

     // GOOD: parallel (total = max(t1, t2))
     const [user, posts] = await Promise.all([fetchUser(id), fetchPosts(id)]);

  B) Edge SSR (Vercel Edge Runtime):
     export const runtime = 'edge';  // pages/api or route handlers
     // Runs at CDN edge — near user, low latency

  C) Cache headers for CDN caching of SSR responses:
     res.setHeader('Cache-Control', 'public, s-maxage=10, stale-while-revalidate=59');
     // CDN caches for 10s; serves stale for 59s while revalidating

  D) Streaming with App Router:
     Use Suspense to stream parts of the page progressively
     (covered in depth in 03_Streaming_and_Suspense.txt)

8. COMMON GOTCHAS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. getServerSideProps only works in pages/ — NOT in components or _app.tsx
2. Props must be JSON-serializable (no Date objects, functions, or class instances)
   → Serialize dates: date.toISOString()
3. Errors in getServerSideProps crash the page — wrap in try/catch
4. Cannot call getServerSideProps from client side — it's server-only
5. App Router: don't confuse server components with SSR — server components
   CAN be cached (SSG-like); only force-dynamic = true SSR behavior

9. COMMON INTERVIEW QUESTIONS & ANSWERS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q1: What is getServerSideProps and when does it run?
A:  getServerSideProps is a function exported from a pages/ component that runs
    on the server on EVERY request. It fetches data, has access to the request
    context (cookies, headers, query params), and returns props to the page.
    It never runs on the client.

Q2: What is the difference between SSR and SSG?
A:  SSR generates HTML per-request (fresh data every time, slower).
    SSG generates HTML at build time (served from CDN, fastest).
    SSR is for dynamic, user-specific, or frequently-changing data.
    SSG is for static or infrequently-changing public content.

Q3: Can getServerSideProps access cookies?
A:  Yes. The context.req.cookies object contains all cookies. This is how
    server-side auth works — read the session cookie, verify it, then
    either return props or redirect to /login.

Q4: How do you implement SSR in the App Router?
A:  Three ways:
    (1) Use dynamic functions: cookies(), headers(), or searchParams — these
        automatically make the page dynamic (SSR-like)
    (2) Use fetch with { cache: 'no-store' }
    (3) Export const dynamic = 'force-dynamic' from the page

Q5: What is stale-while-revalidate and how do you use it with SSR?
A:  SWR caching strategy: serve a stale (cached) response immediately while
    fetching a fresh version in the background. In SSR, set Cache-Control header:
    s-maxage=10, stale-while-revalidate=59 → CDN serves cached for 10s, then
    serves stale for 59s while revalidating. After 69s, cache is stale and a
    full re-render is required. This turns SSR into a semi-cached experience.

Q6: What happens if getServerSideProps throws an error?
A:  Next.js shows the error page (pages/_error.tsx or the default 500 page).
    In production, it shows a 500. Always wrap async operations in try/catch
    in getServerSideProps and return { notFound: true } or handle gracefully.

  SSG (CDN) < ISR (CDN + background regen) < SSR (server per request) < CSR
  SSR is slower but provides freshness and personalization that static cannot.
