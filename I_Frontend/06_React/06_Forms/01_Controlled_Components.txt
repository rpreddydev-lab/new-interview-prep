1. CONTROLLED vs UNCONTROLLED COMPONENTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  CONTROLLED component:
  - React state is the SINGLE SOURCE OF TRUTH for the input's value
  - value prop + onChange handler required
  - Every keystroke triggers a state update
  - You always know the current value

  UNCONTROLLED component:
  - DOM is the source of truth
  - Use a ref to read the value when needed (usually on submit)
  - No re-render on every keystroke (faster for large forms)
  - Harder to do real-time validation

2. CONTROLLED COMPONENT EXAMPLES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Text input
  function NameInput() {
    const [name, setName] = useState('');
    return (
      <input
        type="text"
        value={name}                        // React controls the value
        onChange={e => setName(e.target.value)}
      />
    );
  }

  // Full controlled form
  function LoginForm() {
    const [form, setForm] = useState({ email: '', password: '' });
    const [errors, setErrors] = useState({});

    function handleChange(e) {
      const { name, value } = e.target;
      setForm(prev => ({ ...prev, [name]: value }));
      // Clear error when user starts typing
      if (errors[name]) setErrors(prev => ({ ...prev, [name]: '' }));
    }

    function handleSubmit(e) {
      e.preventDefault();
      const newErrors = {};
      if (!form.email) newErrors.email = 'Email is required';
      if (!form.password) newErrors.password = 'Password is required';
      if (Object.keys(newErrors).length) {
        setErrors(newErrors);
        return;
      }
      // submit form.email, form.password
    }

    return (
      <form onSubmit={handleSubmit}>
        <div>
          <input name="email" type="email" value={form.email} onChange={handleChange} />
          {errors.email && <span className="error">{errors.email}</span>}
        </div>
        <div>
          <input name="password" type="password" value={form.password} onChange={handleChange} />
          {errors.password && <span className="error">{errors.password}</span>}
        </div>
        <button type="submit">Login</button>
      </form>
    );
  }

  // Checkbox (controlled)
  const [checked, setChecked] = useState(false);
  <input type="checkbox" checked={checked} onChange={e => setChecked(e.target.checked)} />

  // Select (controlled)
  const [color, setColor] = useState('red');
  <select value={color} onChange={e => setColor(e.target.value)}>
    <option value="red">Red</option>
    <option value="blue">Blue</option>
  </select>

  // Textarea (controlled)
  const [text, setText] = useState('');
  <textarea value={text} onChange={e => setText(e.target.value)} />

3. UNCONTROLLED COMPONENT
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  function UncontrolledForm() {
    const emailRef = useRef(null);
    const passwordRef = useRef(null);

    function handleSubmit(e) {
      e.preventDefault();
      const email = emailRef.current.value;
      const password = passwordRef.current.value;
      console.log({ email, password }); // read on submit
    }

    return (
      <form onSubmit={handleSubmit}>
        <input type="email" ref={emailRef} defaultValue="" />
        <input type="password" ref={passwordRef} />
        <button type="submit">Login</button>
      </form>
    );
  }

  // defaultValue (not value) sets initial value without controlling it
  // defaultChecked for checkboxes

  When to use uncontrolled:
  - Integrating with non-React code / DOM libraries
  - File inputs (always uncontrolled — value cannot be set programmatically)
  - When form data is only needed on submit (no real-time validation needed)

  | Feature                  | Controlled          | Uncontrolled         |
  |--------------------------|---------------------|----------------------|
  | Source of truth          | React state         | DOM                  |
  | How to get value         | State variable      | ref.current.value    |
  | Real-time validation     | Easy                | Harder               |
  | Conditional disable      | Easy                | Harder               |
  | Instant feedback         | Yes                 | No (need event)      |
  | Re-renders per keystroke | Yes                 | No                   |
  | File inputs              | ❌ Not supported    | ✅ Use ref           |
  | React Hook Form style    | Mostly controlled   | Uncontrolled by default |

5. COMMON INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What is the difference between a controlled and uncontrolled component?
A: In a controlled component, React state drives the input's value via the
   value prop + onChange handler. In uncontrolled, the DOM manages the value
   and you read it via a ref. Controlled gives you full control for validation
   and conditional logic; uncontrolled is simpler but harder to validate.

Q: What is defaultValue vs value in React forms?
A: value makes the input controlled — React always overrides the DOM value.
   defaultValue sets an initial value but the DOM controls it after that
   (uncontrolled pattern). Using both on the same input is an error.

Q: Why can't you have a controlled file input?
A: The browser's file input security model doesn't allow JavaScript to set
   the file path value. File inputs are always uncontrolled — you can only
   read the selected file via a ref or event handler, not set it.

Q: What is the "controlled to uncontrolled" warning in React?
A: It appears when you change an input's value prop from a defined value to
   undefined (or vice versa). Switching from controlled to uncontrolled is
   not allowed. Fix: always provide a string value (use '' instead of null/undefined).
