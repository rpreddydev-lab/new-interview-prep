━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. FUNCTIONAL COMPONENTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

A functional component is a plain JavaScript function that:
  - Accepts a props object as its argument
  - Returns JSX (or null)

  // Simplest functional component
  function Greeting({ name }) {
    return <h1>Hello, {name}!</h1>;
  }

  // Arrow function variant (common in modern codebases)
  const Greeting = ({ name }) => <h1>Hello, {name}!</h1>;

  // With TypeScript
  interface Props {
    name: string;
    age?: number;
  }
  const Greeting: React.FC<Props> = ({ name, age }) => (
    <div>
      <h1>Hello, {name}!</h1>
      {age && <p>Age: {age}</p>}
    </div>
  );

Key characteristics:
  - Just a function — easy to understand, test, and compose
  - Before React 16.8: stateless, no lifecycle methods (called "Stateless Functional Components")
  - After React 16.8: with Hooks, can have state, effects, context, refs — EVERYTHING
  - No 'this' keyword needed

2. CLASS COMPONENTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

A class component is an ES6 class that:
  - Extends React.Component
  - Has a render() method that returns JSX
  - Has a state property (this.state)
  - Has lifecycle methods (componentDidMount, componentDidUpdate, etc.)
  - Uses 'this' to access props and state

  import React, { Component } from 'react';

  class Counter extends Component {
    constructor(props) {
      super(props);
      this.state = { count: 0 };
      this.handleClick = this.handleClick.bind(this);
    }

    handleClick() {
      this.setState({ count: this.state.count + 1 });
    }

    componentDidMount() {
      document.title = `Count: ${this.state.count}`;
    }

    componentDidUpdate(prevProps, prevState) {
      if (prevState.count !== this.state.count) {
        document.title = `Count: ${this.state.count}`;
      }
    }

    componentWillUnmount() {
      document.title = 'App'; // cleanup
    }

    render() {
      return (
        <div>
          <p>Count: {this.state.count}</p>
          <button onClick={this.handleClick}>Increment</button>
        </div>
      );
    }
  }

  // SAME COMPONENT as a functional component with Hooks:
  import { useState, useEffect } from 'react';

  function Counter() {
    const [count, setCount] = useState(0);

    useEffect(() => {
      document.title = `Count: ${count}`;
      return () => { document.title = 'App'; };
    }, [count]);

    return (
      <div>
        <p>Count: {count}</p>
        <button onClick={() => setCount(c => c + 1)}>Increment</button>
      </div>
    );
  }

3. SIDE-BY-SIDE COMPARISON
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  | Feature               | Functional (+ Hooks)      | Class                    |
  |-----------------------|---------------------------|--------------------------|
  | Syntax                | JS function               | ES6 class                |
  | State                 | useState, useReducer      | this.state               |
  | Lifecycle             | useEffect                 | lifecycle methods        |
  | Context               | useContext                | contextType / Consumer   |
  | Refs                  | useRef                    | createRef, callback refs |
  | 'this' keyword        | Not needed                | Required everywhere      |
  | Code reuse            | Custom hooks (composable) | HOC / Render Props       |
  | Performance           | React.memo                | PureComponent, shouldUpdate|
  | Error Boundaries      | Not possible              | Yes (still needed)       |
  | getSnapshotBeforeUpdate| Not possible             | Yes                      |
  | Boilerplate           | Minimal                   | More verbose             |
  | Testing               | Easier                    | More complex (this)      |
  | React team preference | Strongly preferred        | Legacy, not deprecated   |

4. LIFECYCLE METHODS vs HOOKS EQUIVALENTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Class Lifecycle          →   Hook Equivalent
  ─────────────────────────────────────────────────────────────────────────────
  constructor()            →   useState(initialValue) / useReducer
  componentDidMount()      →   useEffect(() => { ... }, [])
  componentDidUpdate()     →   useEffect(() => { ... }, [deps])
  componentWillUnmount()   →   useEffect(() => { return () => cleanup(); }, [])
  shouldComponentUpdate()  →   React.memo / useMemo
  getDerivedStateFromProps →   Compute during render or useEffect
  getSnapshotBeforeUpdate  →   No equivalent (class only)
  componentDidCatch        →   No equivalent (class only — Error Boundaries)

5. THE 'this' BINDING PROBLEM IN CLASS COMPONENTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

A common class component bug: 'this' is undefined inside event handlers.

  class Button extends Component {
    handleClick() {
      console.log(this); // undefined! (if not bound)
    }
    render() {
      return <button onClick={this.handleClick}>Click</button>;
    }
  }

Solutions:
  1. Bind in constructor (most common legacy approach):
     this.handleClick = this.handleClick.bind(this);

  2. Arrow function class property (uses class fields proposal):
     handleClick = () => { console.log(this); } // 'this' always correct

  3. Arrow function in JSX (creates new function each render — minor perf issue):
     <button onClick={() => this.handleClick()}>Click</button>

Hooks avoid this entirely — there's no 'this' in function components.

6. WHEN TO STILL USE CLASS COMPONENTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Class components are NOT deprecated. Use them for:

  1. ERROR BOUNDARIES — The ONLY way to catch rendering errors is via class components:
     class ErrorBoundary extends Component {
       state = { hasError: false };
       static getDerivedStateFromError(error) { return { hasError: true }; }
       componentDidCatch(error, info) { logError(error, info); }
       render() {
         if (this.state.hasError) return <FallbackUI />;
         return this.props.children;
       }
     }

  2. getSnapshotBeforeUpdate — To capture scroll position before DOM update:
     getSnapshotBeforeUpdate(prevProps, prevState) {
       if (prevProps.list.length < this.props.list.length) {
         return this.listRef.scrollHeight - this.listRef.scrollTop;
       }
       return null;
     }

  3. Maintaining existing class component codebases (no need to rewrite if working)

7. COMMON INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What is the difference between functional and class components?
A: Functional components are JavaScript functions that accept props and return JSX.
   Class components extend React.Component and use render(), lifecycle methods,
   and this.state. With Hooks (React 16.8+), functional components have equivalent
   capabilities. Functional components are simpler, have no 'this', are easier
   to test, and are the current recommended approach.

Q: Can functional components replace class components completely?
A: Mostly yes, with two exceptions: Error Boundaries (require componentDidCatch
   and getDerivedStateFromError — no hook equivalent) and getSnapshotBeforeUpdate.
   All other use cases are fully covered by hooks.

Q: Why did React introduce Hooks?
A: To solve problems with class components: (1) stateful logic was hard to share
   between components (required HOC or render props), (2) lifecycle methods grouped
   unrelated logic (e.g., adding and removing event listeners in the same lifecycle
   method), (3) classes are confusing with 'this' binding and are harder to optimize.

Q: What is the 'this' binding problem in class components?
A: Event handler methods lose their 'this' context when called as callbacks.
   You must explicitly bind them in the constructor, use arrow function class
   properties, or wrap calls in arrow functions in JSX.

Q: What is PureComponent?
A: React.PureComponent is like React.Component but implements shouldComponentUpdate
   with a shallow comparison of props and state. If neither changed, it skips
   re-rendering. The functional equivalent is React.memo().

Q: What is React.memo() and how does it differ from PureComponent?
A: React.memo() is a HOC for functional components. It does a shallow props
   comparison and skips re-render if props haven't changed (like PureComponent).
   It doesn't compare state or context (those changes always trigger re-render).
