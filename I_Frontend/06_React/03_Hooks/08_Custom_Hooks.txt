1. WHAT ARE CUSTOM HOOKS?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Custom hooks are JavaScript functions that:
  - Start with "use" (naming convention enforced by ESLint react-hooks plugin)
  - Can call other hooks (built-in or custom)
  - Extract and share stateful logic between components
  - Each component using the hook gets its OWN independent state

  // Naming matters! A function named "use..." gets hook rules applied.
  // A function NOT starting with "use" cannot call hooks.

2. EXAMPLE 1: useFetch — Data Fetching
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  function useFetch(url) {
    const [data, setData] = useState(null);
    const [loading, setLoading] = useState(true);
    const [error, setError] = useState(null);

    useEffect(() => {
      if (!url) return;
      const controller = new AbortController();
      setLoading(true);
      setError(null);

      fetch(url, { signal: controller.signal })
        .then(res => {
          if (!res.ok) throw new Error(`HTTP error ${res.status}`);
          return res.json();
        })
        .then(data => { setData(data); setLoading(false); })
        .catch(err => {
          if (err.name !== 'AbortError') {
            setError(err.message);
            setLoading(false);
          }
        });

      return () => controller.abort();
    }, [url]);

    return { data, loading, error };
  }

  // Usage — reuse across many components:
  function UserProfile({ userId }) {
    const { data: user, loading, error } = useFetch(`/api/users/${userId}`);
    if (loading) return <Spinner />;
    if (error) return <p>Error: {error}</p>;
    return <div>{user.name}</div>;
  }

  function PostList({ authorId }) {
    const { data: posts, loading } = useFetch(`/api/posts?author=${authorId}`);
    return loading ? <Spinner /> : posts?.map(p => <Post key={p.id} post={p} />);
  }

3. EXAMPLE 2: useLocalStorage
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  function useLocalStorage(key, initialValue) {
    const [value, setValue] = useState(() => {
      try {
        const stored = localStorage.getItem(key);
        return stored !== null ? JSON.parse(stored) : initialValue;
      } catch {
        return initialValue;
      }
    });

    const setStoredValue = useCallback((newValue) => {
      try {
        const valueToStore = newValue instanceof Function ? newValue(value) : newValue;
        setValue(valueToStore);
        localStorage.setItem(key, JSON.stringify(valueToStore));
      } catch (err) {
        console.warn(`useLocalStorage: Error setting key "${key}":`, err);
      }
    }, [key, value]);

    return [value, setStoredValue];
  }

  // Usage:
  function Settings() {
    const [theme, setTheme] = useLocalStorage('theme', 'light');
    const [fontSize, setFontSize] = useLocalStorage('fontSize', 14);
    return (
      <div>
        <button onClick={() => setTheme(t => t === 'light' ? 'dark' : 'light')}>
          Toggle: {theme}
        </button>
      </div>
    );
  }

4. EXAMPLE 3: useDebounce
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  function useDebounce(value, delay) {
    const [debouncedValue, setDebouncedValue] = useState(value);

    useEffect(() => {
      const timer = setTimeout(() => setDebouncedValue(value), delay);
      return () => clearTimeout(timer);  // cancel on new value or unmount
    }, [value, delay]);

    return debouncedValue;
  }

  // Usage: search input that debounces API calls
  function SearchPage() {
    const [query, setQuery] = useState('');
    const debouncedQuery = useDebounce(query, 300); // wait 300ms after typing stops

    const { data: results, loading } = useFetch(
      debouncedQuery ? `/api/search?q=${debouncedQuery}` : null
    );

    return (
      <div>
        <input value={query} onChange={e => setQuery(e.target.value)} />
        {loading && <Spinner />}
        {results?.map(r => <Result key={r.id} result={r} />)}
      </div>
    );
  }

5. EXAMPLE 4: useToggle
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  function useToggle(initialValue = false) {
    const [value, setValue] = useState(initialValue);
    const toggle = useCallback(() => setValue(v => !v), []);
    const setTrue = useCallback(() => setValue(true), []);
    const setFalse = useCallback(() => setValue(false), []);
    return [value, toggle, setTrue, setFalse];
  }

  // Usage:
  function Modal() {
    const [isOpen, toggle, open, close] = useToggle(false);
    return (
      <>
        <button onClick={open}>Open</button>
        {isOpen && <Dialog onClose={close}><Content /></Dialog>}
      </>
    );
  }

6. EXAMPLE 5: useOnClickOutside
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  function useOnClickOutside(ref, handler) {
    useEffect(() => {
      function listener(event) {
        if (!ref.current || ref.current.contains(event.target)) return;
        handler(event);
      }
      document.addEventListener('mousedown', listener);
      document.addEventListener('touchstart', listener);
      return () => {
        document.removeEventListener('mousedown', listener);
        document.removeEventListener('touchstart', listener);
      };
    }, [ref, handler]);
  }

  // Usage:
  function Dropdown({ items }) {
    const [isOpen, setIsOpen] = useState(false);
    const dropdownRef = useRef(null);

    useOnClickOutside(dropdownRef, () => setIsOpen(false));

    return (
      <div ref={dropdownRef}>
        <button onClick={() => setIsOpen(o => !o)}>Toggle</button>
        {isOpen && <ul>{items.map(i => <li key={i.id}>{i.label}</li>)}</ul>}
      </div>
    );
  }

7. EXAMPLE 6: useWindowSize
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  function useWindowSize() {
    const [size, setSize] = useState({
      width: window.innerWidth,
      height: window.innerHeight,
    });

    useEffect(() => {
      const handleResize = () =>
        setSize({ width: window.innerWidth, height: window.innerHeight });
      window.addEventListener('resize', handleResize);
      return () => window.removeEventListener('resize', handleResize);
    }, []);

    return size;
  }

8. EXAMPLE 7: usePrevious
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  function usePrevious(value) {
    const ref = useRef(undefined);
    useEffect(() => {
      ref.current = value;
    }); // no deps — captures value AFTER every render
    return ref.current; // returns value from BEFORE this render
  }

  function Counter() {
    const [count, setCount] = useState(0);
    const prev = usePrevious(count);
    return <p>Now: {count}, Before: {prev}</p>;
  }

9. COMMON INTERVIEW QUESTIONS ON CUSTOM HOOKS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What is a custom hook?
A: A JavaScript function whose name starts with 'use' and that calls other hooks.
   It extracts and shares stateful logic between components. Each usage of the
   hook creates its own independent state — logic is shared, state is not.

Q: Do custom hooks share state between components?
A: No. Each component gets its own isolated state. Custom hooks share LOGIC,
   not state. To share state, use Context or external state management.

Q: Why must custom hooks start with "use"?
A: The "use" prefix is a convention that tells the React ESLint plugin to apply
   the Rules of Hooks (only call at top level, only in React functions). A function
   not named "use..." won't get these rules checked.

Q: When should you extract logic into a custom hook?
A: When the same stateful logic is needed in multiple components, or when a
   component has complex state logic that would benefit from being separated.
   Don't over-extract — simple logic can stay inline.

Q: Can a custom hook call other custom hooks?
A: Yes. Custom hooks can compose other custom hooks freely.

Q: How do custom hooks compare to HOCs and Render Props?
A: All three share logic between components. Custom hooks are cleaner (no wrapper
   nesting), more explicit (you see exactly what data you're getting), better for
   TypeScript, and can be used conditionally. They're the preferred modern approach.
