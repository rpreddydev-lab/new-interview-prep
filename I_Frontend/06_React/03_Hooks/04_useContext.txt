1. WHAT IS CONTEXT?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Context provides a way to pass data through the component tree WITHOUT manually
passing props at every level (solves prop drilling).

When to use Context:
  - Theme (dark/light mode)
  - Current authenticated user
  - Locale / language settings
  - Feature flags
  - Any data that many components at different nesting levels need

When NOT to use Context (over-used!):
  - Data that only 1-2 nearby components need (use props instead)
  - Frequently changing data (causes broad re-renders — use state management)

2. CREATING AND USING CONTEXT
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

STEP 1: Create a Context
  import { createContext } from 'react';

  const ThemeContext = createContext('light'); // default value

  // The default value is used when a component has NO matching Provider above it
  // It's rarely used in practice — mostly for testing or isolated component usage

STEP 2: Provide the value (wrap components that need access)
  function App() {
    const [theme, setTheme] = useState('light');
    return (
      <ThemeContext.Provider value={{ theme, setTheme }}>
        <Header />
        <Main />
        <Footer />
      </ThemeContext.Provider>
    );
  }

STEP 3: Consume with useContext
  import { useContext } from 'react';

  function Button({ children }) {
    const { theme } = useContext(ThemeContext);
    return (
      <button className={`btn btn-${theme}`}>{children}</button>
    );
  }

  // ANY component inside the Provider can access the context
  // No need to pass theme as a prop through Header, Main, etc.

3. PATTERN: CONTEXT + CUSTOM HOOK
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Best practice: wrap useContext in a custom hook for better DX and validation.

  // ThemeContext.js
  import { createContext, useContext, useState } from 'react';

  const ThemeContext = createContext(null);

  export function ThemeProvider({ children }) {
    const [theme, setTheme] = useState('light');
    const toggleTheme = () => setTheme(t => t === 'light' ? 'dark' : 'light');

    return (
      <ThemeContext.Provider value={{ theme, toggleTheme }}>
        {children}
      </ThemeContext.Provider>
    );
  }

  export function useTheme() {
    const context = useContext(ThemeContext);
    if (!context) {
      throw new Error('useTheme must be used within a ThemeProvider');
    }
    return context;
  }

  // Usage:
  function App() {
    return (
      <ThemeProvider>
        <Header />
      </ThemeProvider>
    );
  }

  function Header() {
    const { theme, toggleTheme } = useTheme();
    return (
      <header className={theme}>
        <button onClick={toggleTheme}>Toggle</button>
      </header>
    );
  }

4. AUTH CONTEXT EXAMPLE (Common Interview Pattern)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // AuthContext.js
  const AuthContext = createContext(null);

  export function AuthProvider({ children }) {
    const [user, setUser] = useState(null);
    const [loading, setLoading] = useState(true);

    useEffect(() => {
      // Check if user is already logged in (e.g., from token in localStorage)
      const token = localStorage.getItem('token');
      if (token) {
        validateToken(token).then(user => {
          setUser(user);
          setLoading(false);
        });
      } else {
        setLoading(false);
      }
    }, []);

    const login = async (credentials) => {
      const user = await loginAPI(credentials);
      setUser(user);
      localStorage.setItem('token', user.token);
    };

    const logout = () => {
      setUser(null);
      localStorage.removeItem('token');
    };

    return (
      <AuthContext.Provider value={{ user, loading, login, logout }}>
        {children}
      </AuthContext.Provider>
    );
  }

  export function useAuth() {
    return useContext(AuthContext);
  }

5. CONTEXT RE-RENDERING BEHAVIOR
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

When context value changes, ALL components that useContext() that context
will RE-RENDER, even if they don't use the changed part of the value.

  // PROBLEM: Both ThemeButton and LanguageButton re-render when either changes
  const AppContext = createContext();
  function AppProvider({ children }) {
    const [theme, setTheme] = useState('light');
    const [language, setLanguage] = useState('en');
    return (
      <AppContext.Provider value={{ theme, setTheme, language, setLanguage }}>
        {children}
      </AppContext.Provider>
    );
  }
  // Changing language re-renders ThemeButton (even though it only uses theme)

  SOLUTIONS:
  1. SPLIT CONTEXTS — one context per concern:
     <ThemeProvider>
       <LanguageProvider>
         {children}
       </LanguageProvider>
     </ThemeProvider>

  2. MEMOIZE CONTEXT VALUE:
     const value = useMemo(() => ({ theme, setTheme }), [theme]);
     <ThemeContext.Provider value={value}>

  3. SPLIT STABLE + CHANGING VALUES:
     <UserContext.Provider value={user}>        {/* rarely changes */}
       <UserDispatch.Provider value={dispatch}>  {/* stable reference */}
         {children}
       </UserDispatch.Provider>
     </UserContext.Provider>

7. CONTEXT vs REDUX/ZUSTAND
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  | Aspect              | Context API             | Redux/Zustand           |
  |---------------------|-------------------------|-------------------------|
  | Purpose             | Avoid prop drilling     | Global state management |
  | Performance         | Re-renders all consumers| Granular subscriptions  |
  | DevTools            | Basic                   | Rich Redux DevTools      |
  | Middleware          | No                      | Yes (thunk, saga, etc.) |
  | Time-travel debug   | No                      | Redux DevTools           |
  | Boilerplate         | Low                     | Higher (Redux), Low(Zustand)|
  | Best for            | Low-frequency global    | Complex, high-frequency |
  |                     | data (theme, auth, lang)| app-wide state          |

  Context is NOT a full state management solution. For complex, frequently
  updating global state, use Redux Toolkit or Zustand.

8. COMMON INTERVIEW QUESTIONS ON useContext
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What is Context API and why use it?
A: Context provides a way to share data across the component tree without prop
   drilling. Best for low-frequency global data: theme, auth user, locale, feature flags.

Q: What is the default value of createContext?
A: The value passed to createContext() is used when a component has no matching
   Provider above it in the tree. Mainly useful for testing components in isolation.

Q: What happens when a context value changes?
A: ALL components that call useContext() with that context re-render, regardless
   of whether they use the changed part. This is why you should split contexts
   and memoize values for performance.

Q: How do you prevent unnecessary re-renders with context?
A: 1. Split contexts by concern (separate ThemeContext from UserContext)
   2. Memoize the context value with useMemo
   3. Separate stable data (user object) from dispatch functions

Q: Is Context a replacement for Redux?
A: No. Context solves prop drilling for low-frequency data. Redux provides
   sophisticated state management: DevTools, time-travel debugging, middleware,
   efficient selective subscriptions. For simple global data, Context is sufficient.

Q: What error should a custom context hook throw?
A: If the hook is called outside its Provider, throw a descriptive error:
   throw new Error('useTheme must be used within ThemeProvider');
   This prevents silent failures from using the default context value.
