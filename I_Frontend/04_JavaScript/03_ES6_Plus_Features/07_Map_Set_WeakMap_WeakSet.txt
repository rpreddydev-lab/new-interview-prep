1. MAP
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Map is a key-value collection where KEYS CAN BE ANY TYPE (including
  objects, functions, primitives). Insertion order is preserved.

  // Creation:
  const map = new Map();
  const map2 = new Map([["key1", "val1"], ["key2", "val2"]]);  // from entries

  // Methods:
  map.set("name", "Alice");     // add/update
  map.set(42, "number key");    // key can be a number
  map.set({}, "object key");    // key can be an object
  map.get("name");              // "Alice"
  map.has("name");              // true
  map.delete("name");           // removes key
  map.size;                     // number of entries
  map.clear();                  // remove all

  // Iteration (in insertion order):
  for (const [key, value] of map) { }
  for (const key of map.keys()) { }
  for (const value of map.values()) { }
  for (const [k, v] of map.entries()) { }
  map.forEach((value, key) => { });

  // Convert to/from object:
  const obj = Object.fromEntries(map);
  const map3 = new Map(Object.entries(obj));

2. MAP vs OBJECT
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  ┌────────────────────────┬────────────────────┬────────────────────────────┐
  │ Feature                │ Object {}          │ Map                        │
  ├────────────────────────┼────────────────────┼────────────────────────────┤
  │ Key types              │ String, Symbol      │ ANY (including objects)    │
  ├────────────────────────┼────────────────────┼────────────────────────────┤
  │ Insertion order        │ Yes (mostly*)       │ Always guaranteed          │
  ├────────────────────────┼────────────────────┼────────────────────────────┤
  │ Size                   │ Manual count        │ map.size                   │
  ├────────────────────────┼────────────────────┼────────────────────────────┤
  │ Prototype keys         │ Yes (inherited)     │ No (clean)                 │
  ├────────────────────────┼────────────────────┼────────────────────────────┤
  │ Iteration              │ for...in (quirks)   │ for...of (clean)           │
  ├────────────────────────┼────────────────────┼────────────────────────────┤
  │ JSON serializable      │ Yes                 │ No (need toEntries)        │
  ├────────────────────────┼────────────────────┼────────────────────────────┤
  │ Performance (freq add) │ Slower              │ Faster for dynamic keys    │
  └────────────────────────┴────────────────────┴────────────────────────────┘

  // Use Map when:
  - Keys are not strings
  - Frequent additions/removals
  - Need to know size
  - Need guaranteed insertion order

3. SET
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Set is a collection of UNIQUE VALUES (no duplicates). Insertion order
  preserved. Values compared with SameValueZero algorithm.

  // Creation:
  const set = new Set();
  const set2 = new Set([1, 2, 3, 2, 1]);  // { 1, 2, 3 } — duplicates removed

  // Methods:
  set.add(1);          // add value
  set.has(1);          // true — check existence (O(1))
  set.delete(1);       // remove value
  set.size;            // number of unique values
  set.clear();         // remove all

  // Iteration:
  for (const val of set) { }
  for (const val of set.values()) { }
  for (const [val, val2] of set.entries()) { }  // val === val2 for Sets
  set.forEach(val => { });

  // Convert to array:
  [...set]             // [1, 2, 3]
  Array.from(set)      // [1, 2, 3]

  // Practical uses:
  // Remove duplicates from array:
  const unique = [...new Set([1, 2, 2, 3, 3, 3])];  // [1, 2, 3]

  // Set operations:
  const a = new Set([1, 2, 3, 4]);
  const b = new Set([3, 4, 5, 6]);

  // Union:
  const union = new Set([...a, ...b]);  // {1,2,3,4,5,6}

  // Intersection:
  const intersection = new Set([...a].filter(x => b.has(x)));  // {3,4}

  // Difference (a - b):
  const difference = new Set([...a].filter(x => !b.has(x)));   // {1,2}

4. WEAKMAP
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  WeakMap is like Map BUT:
  - Keys must be OBJECTS (not primitives)
  - Keys are held WEAKLY (won't prevent GC)
  - NOT iterable (no .keys(), .values(), .entries(), no .size)
  - Cannot be enumerated (privacy-friendly)

  const wm = new WeakMap();
  const key = {};          // must be object
  wm.set(key, "data");
  wm.get(key);             // "data"
  wm.has(key);             // true
  wm.delete(key);          // removes entry

  // When key object is garbage collected, entry is automatically removed!

  // Use cases:
  // 1. Private data per object:
  const _private = new WeakMap();
  class Person {
    constructor(name) {
      _private.set(this, { secret: "ssn-123" });
      this.name = name;
    }
    getSecret() {
      return _private.get(this).secret;
    }
  }

  // 2. Memoization with object keys (auto-cleanup when object GC'd):
  const cache = new WeakMap();
  function memoize(obj) {
    if (cache.has(obj)) return cache.get(obj);
    const result = expensiveCompute(obj);
    cache.set(obj, result);
    return result;
  }

  // 3. Tracking DOM elements without memory leaks:
  const domData = new WeakMap();
  function attachData(element, data) {
    domData.set(element, data);
  }
  // When element is removed from DOM and no other refs, entry GC'd

5. WEAKSET
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  WeakSet is like Set BUT:
  - Values must be OBJECTS
  - Values are weakly held (GC-able)
  - NOT iterable, no .size

  const ws = new WeakSet();
  const obj = {};
  ws.add(obj);
  ws.has(obj);     // true
  ws.delete(obj);

  // Use case: tracking objects without preventing GC:
  // Mark visited objects in a graph traversal:
  const visited = new WeakSet();
  function traverse(node) {
    if (visited.has(node)) return;  // avoid cycles
    visited.add(node);
    node.children?.forEach(traverse);
  }

  // Track processing state:
  const inProgress = new WeakSet();
  async function process(request) {
    if (inProgress.has(request)) return;
    inProgress.add(request);
    try {
      await doWork(request);
    } finally {
      inProgress.delete(request);
    }
  }

7. COMMON INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What is the difference between Map and a plain object?
A: Map allows any key type, guarantees insertion order, has .size property,
   and is more performant for frequent additions/removals. Objects only
   have string/Symbol keys and may have inherited prototype properties.

Q: When would you use a WeakMap over a Map?
A: Use WeakMap when associating data with objects where you don't control
   the object lifetime. WeakMap won't prevent GC — when the key object
   is no longer referenced elsewhere, the entry is automatically freed.
   Good for: private class data, DOM element metadata, memoization.

Q: Can you iterate over a WeakMap?
A: No. WeakMap and WeakSet are not iterable and have no .size property.
   This is intentional for memory safety — you can't observe all the keys
   because they may have been garbage collected at any time.

Q: How do you remove duplicates from an array using ES6?
A: const unique = [...new Set(array)];  — Set automatically discards
   duplicate values. For objects (comparison by reference), this only
   de-dupes identical references, not structurally equal objects.

Q: What does SameValueZero comparison mean in Set?
A: Like ===, except: NaN === NaN is true in Set (NaN is deduplicated),
   and +0 === -0. So a Set won't store two NaN values.

Q: What's the time complexity of Map/Set operations?
A: .has(), .get(), .set(), .add() are O(1) average (hash-based).
   Much faster than Array.includes() which is O(n).
