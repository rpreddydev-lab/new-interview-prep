1. WHAT IS HOISTING?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Hoisting is JavaScript's behavior of processing declarations before code
execution. During the CREATION PHASE of an execution context, the JS engine
scans the code and:
  - Registers var declarations (set to undefined)
  - Fully registers function declarations (hoisted with their body)
  - Registers let/const declarations (but does NOT initialize them → TDZ)
  - Registers class declarations (but does NOT initialize → TDZ)

Key insight: only DECLARATIONS are hoisted, not INITIALIZATIONS/ASSIGNMENTS.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
2. var HOISTING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  console.log(x);  // undefined (not ReferenceError!)
  var x = 5;
  console.log(x);  // 5

  // What JS actually does (conceptually):
  var x;           // declaration hoisted, initialized to undefined
  console.log(x);  // undefined
  x = 5;           // assignment stays in place
  console.log(x);  // 5

  // Inside functions:
  function test() {
    console.log(a);  // undefined
    var a = 10;
    console.log(a);  // 10
  }

  // var hoisted to function top, NOT to outer scope:
  function outer() {
    if (false) {
      var insideIf = "never reached";
    }
    console.log(insideIf);  // undefined (not ReferenceError — hoisted!)
  }

3. FUNCTION DECLARATION HOISTING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Function DECLARATIONS are hoisted ENTIRELY (name + body):

  sayHello();             // "Hello!" — works before declaration!

  function sayHello() {
    console.log("Hello!");
  }

  // This is why forward-calling functions works in JS

  // PITFALL: conditional function declarations (non-strict mode)
  if (true) {
    function f() { return 1; }
  } else {
    function f() { return 2; }
  }
  f(); // undefined behavior in non-strict, or 1 in most engines

4. FUNCTION EXPRESSION HOISTING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Function expressions are NOT hoisted (only the variable is):

  // var function expression:
  sayHi();                  // TypeError: sayHi is not a function
  var sayHi = function() {
    console.log("Hi!");
  };
  // var sayHi is hoisted as undefined, then called as a function → error

  // let/const function expression:
  greet();                  // ReferenceError: Cannot access 'greet' before init
  const greet = function() { ... };

  // Named function expressions:
  var fn = function namedFn() { ... };
  namedFn();  // ReferenceError — name only accessible inside the function body
  fn();       // works fine

  // Arrow functions (also expressions, not hoisted):
  arrowFn();  // TypeError or ReferenceError
  const arrowFn = () => {};

5. let / const HOISTING (TDZ)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  let and const ARE technically hoisted (the JS engine knows about them
  from the start of the block), but they are NOT initialized until the
  declaration is reached → accessing before the declaration throws
  ReferenceError (Temporal Dead Zone).

  {
    console.log(x); // ReferenceError: Cannot access 'x' before initialization
    let x = 5;
    console.log(x); // 5
  }

  // Proof they ARE hoisted (even if not accessible):
  let x = "outer";
  {
    // if let weren't hoisted, accessing x would give "outer"
    // but because it IS hoisted (in TDZ), it throws ReferenceError:
    console.log(x);  // ReferenceError
    let x = "inner";
  }

6. CLASS HOISTING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Classes are like let/const — hoisted but in TDZ:

  const obj = new MyClass();  // ReferenceError!
  class MyClass {}

  // vs function constructor (works because function decl is fully hoisted):
  const obj2 = new OldClass();  // works!
  function OldClass() {}

7. HOISTING ORDER (when both var and function exist)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Function declarations take PRIORITY over var declarations:

  console.log(typeof foo);  // "function" (not "undefined")
  var foo = 5;
  function foo() {}
  console.log(foo);  // 5 (assignment overwrites function)

  // Processing order:
  // 1. function foo() {} → hoisted first, foo = [Function: foo]
  // 2. var foo → declaration already registered, ignored
  // 3. Execution: console.log → "function"
  // 4. foo = 5 → assignment runs, foo is now 5
  // 5. console.log → 5

9. COMMON INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What is hoisting?
A: Hoisting is JS's behavior of processing declarations before execution.
   Variable declarations (var) are moved to the top of their scope and
   initialized to undefined. Function declarations are fully hoisted.
   let/const/class are hoisted but inaccessible until their line (TDZ).

Q: What is the output?
   console.log(a); var a = 1;
A: undefined — var is hoisted and initialized to undefined before execution.

Q: What is the output?
   foo(); function foo() { console.log("hi"); }
A: "hi" — function declarations are fully hoisted (callable before they appear).

Q: What is the output?
   bar(); var bar = function() { console.log("hello"); };
A: TypeError: bar is not a function — var bar is hoisted as undefined, and
   calling undefined as a function throws TypeError.

Q: Are let and const hoisted?
A: Yes, technically. They are registered in the scope from the start of
   the block, but they are NOT initialized. Accessing them before the
   declaration throws ReferenceError (Temporal Dead Zone).

Q: What happens if you declare a var and a function with the same name?
A: The function declaration takes priority. After execution, if the var
   assignment is reached, it overwrites the function.

Q: What is the output?
   console.log(typeof undeclaredVariable);
A: "undefined" — typeof doesn't throw for undeclared variables, it returns
   "undefined". This is the one place where typeof is "safe" for undeclared vars.
