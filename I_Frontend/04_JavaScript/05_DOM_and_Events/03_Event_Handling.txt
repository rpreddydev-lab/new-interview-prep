1. ATTACHING EVENT LISTENERS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Method 1 — Inline HTML (avoid):
  <button onclick="handleClick()">Click</button>

  // Method 2 — DOM property (only one handler per event):
  el.onclick = function() { console.log("clicked"); };
  el.onclick = null;  // remove

  // Method 3 — addEventListener (preferred):
  el.addEventListener("click", handler);
  el.removeEventListener("click", handler);  // requires same function reference

  // addEventListener options:
  el.addEventListener("click", handler, {
    once: true,      // auto-removes after first call
    capture: true,   // fire in capture phase (top-down)
    passive: true    // hints browser: no preventDefault() call (scroll perf)
  });

  // Named function for removal:
  function onClick(e) { console.log(e); }
  el.addEventListener("click", onClick);
  el.removeEventListener("click", onClick);  // works — same reference

  // Arrow function cannot be removed (new reference each time):
  el.addEventListener("click", () => handle());
  el.removeEventListener("click", () => handle());  // does NOT work!

2. THE EVENT OBJECT
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  el.addEventListener("click", function(event) {
    event.type          // "click"
    event.target        // element that was clicked
    event.currentTarget // element the listener is on
    event.bubbles       // true for most events
    event.cancelable    // true if preventDefault is meaningful

    event.preventDefault()    // stop default behavior (form submit, link nav)
    event.stopPropagation()   // stop bubbling/capturing up/down the tree
    event.stopImmediatePropagation()  // stop all listeners, including same el

    // Mouse events:
    event.clientX / event.clientY   // coords relative to viewport
    event.pageX / event.pageY       // coords relative to document
    event.offsetX / event.offsetY   // coords relative to target element
    event.button   // 0=left, 1=middle, 2=right

    // Keyboard events:
    event.key     // "Enter", "a", "ArrowLeft" ...
    event.code    // "KeyA", "ArrowLeft" (layout-independent)
    event.shiftKey / event.ctrlKey / event.altKey / event.metaKey

    // Touch events:
    event.touches         // all active touches
    event.changedTouches  // touches changed in this event
  });

              mouseenter, mouseleave, mouseover, mouseout, contextmenu

  Keyboard:   keydown, keyup, keypress (deprecated)

  Form:       submit, change, input, focus, blur, focusin, focusout, reset
  Document:   DOMContentLoaded, load, unload, beforeunload
  Window:     resize, scroll, hashchange, popstate, online, offline

  // DOMContentLoaded vs load:
  // DOMContentLoaded — HTML parsed, DOM ready (stylesheets may still load)
  // load — everything (images, stylesheets) fully loaded

  document.addEventListener("DOMContentLoaded", () => {
    // DOM is ready, safe to query elements
  });

  window.addEventListener("load", () => {
    // All resources fully loaded
  });

4. PREVENTING DEFAULT BEHAVIOR
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Prevent form submission:
  form.addEventListener("submit", (e) => {
    e.preventDefault();
    handleFormData(new FormData(form));
  });

  // Prevent link navigation:
  link.addEventListener("click", (e) => {
    e.preventDefault();
    router.navigate(link.href);
  });

  // Prevent context menu:
  document.addEventListener("contextmenu", (e) => e.preventDefault());

  // Prevent drag-over default (required for drop target):
  dropzone.addEventListener("dragover", (e) => e.preventDefault());
  dropzone.addEventListener("drop", (e) => {
    e.preventDefault();
    handleDrop(e.dataTransfer.files);
  });

5. CUSTOM EVENTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Create and dispatch a custom event:
  const event = new CustomEvent("user:login", {
    detail: { userId: 42, name: "Alice" },  // custom data
    bubbles: true,       // propagate up the tree
    cancelable: true     // allow preventDefault
  });
  document.dispatchEvent(event);

  // Listen for it:
  document.addEventListener("user:login", (e) => {
    console.log(e.detail.userId);  // 42
  });

  // EventTarget interface — custom event emitter without DOM:
  const emitter = new EventTarget();
  emitter.addEventListener("data", (e) => console.log(e.detail));
  emitter.dispatchEvent(new CustomEvent("data", { detail: { x: 1 } }));

6. COMMON INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What is the difference between event.target and event.currentTarget?
A: event.target is the element that actually triggered the event (the origin).
   event.currentTarget is the element the listener is attached to. When
   bubbling, currentTarget changes as the event moves up; target stays fixed.

Q: Why can't you removeEventListener with an anonymous/arrow function?
A: removeEventListener requires an identical function reference. Arrow functions
   create a new reference each call, so the reference passed to addEventListener
   and removeEventListener won't match.

Q: What does event.preventDefault() do?
A: Cancels the browser's default behavior for that event (e.g. stops form
   submission, prevents link navigation, prevents context menu). It does NOT
   stop bubbling — use stopPropagation() for that.

Q: What is passive: true in addEventListener?
A: A performance hint telling the browser that the handler will never call
   preventDefault(). This lets the browser optimize scroll performance by not
   waiting for the JS handler before scrolling (used for touch/wheel events).

Q: What is the difference between DOMContentLoaded and load?
A: DOMContentLoaded fires when the HTML is fully parsed and the DOM is ready,
   but before stylesheets, images, and subframes load. load fires after ALL
   resources (images, stylesheets) have loaded.
