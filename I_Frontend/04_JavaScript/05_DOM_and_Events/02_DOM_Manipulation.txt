2. INSERTING / REMOVING ELEMENTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  const parent = document.getElementById("container");
  const child = document.createElement("p");

  // Old methods (still used):
  parent.appendChild(child);              // add at end
  parent.insertBefore(child, refNode);    // before refNode
  parent.removeChild(child);             // remove child
  parent.replaceChild(newNode, oldNode);  // replace

  // Modern methods (more readable):
  parent.append(child, "text");          // add at end (multiple nodes/strings)
  parent.prepend(child);                 // add at start
  child.before(newNode);                 // insert before child
  child.after(newNode);                  // insert after child
  child.replaceWith(newNode);            // replace child
  child.remove();                        // remove self (no parent ref needed)

  // DocumentFragment for batch inserts (single reflow):
  const frag = document.createDocumentFragment();
  for (let i = 0; i < 1000; i++) {
    const li = document.createElement("li");
    li.textContent = `Item ${i}`;
    frag.appendChild(li);
  }
  document.getElementById("list").appendChild(frag);  // one DOM operation

3. ATTRIBUTES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  el.getAttribute("href")              // get attribute value (string | null)
  el.setAttribute("href", "/home")     // set/update attribute
  el.removeAttribute("disabled")       // remove attribute
  el.hasAttribute("data-id")           // boolean check

  // Properties vs Attributes:
  // Attributes = HTML (string values, initial state)
  // Properties  = JavaScript object (current state, typed)
  input.setAttribute("value", "hello");  // sets HTML attribute
  input.value = "world";                 // sets JS property (current value)
  input.getAttribute("value");           // "hello" (original)
  input.value;                           // "world" (current)

  // data-* attributes:
  el.dataset.userId = "123";          // sets data-user-id="123"
  el.dataset.userId;                  // "123"
  el.getAttribute("data-user-id");    // "123"

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Inline styles via element.style (camelCase):
  el.style.backgroundColor = "red";
  el.style.fontSize = "16px";
  el.style.display = "none";
  el.style.cssText = "color: red; font-size: 16px;";  // bulk set

  // Remove inline style:
  el.style.removeProperty("background-color");
  el.style.backgroundColor = "";  // also removes

  // Read COMPUTED styles (final rendered values):
  const computed = getComputedStyle(el);
  computed.color;         // "rgb(0, 0, 0)"
  computed.fontSize;      // "16px"
  computed.display;       // "block"
  // getComputedStyle is read-only!

  // Prefer CSS classes over inline styles for maintainability.

6. innerHTML vs textContent vs innerText
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // innerHTML:
  // Gets/sets HTML markup inside element
  // XSS risk with user data! Use with sanitized content only.
  el.innerHTML = "<strong>Bold</strong>";  // parses and renders
  const markup = el.innerHTML;             // serializes DOM to HTML string

  // textContent:
  // Gets/sets ALL text content (including hidden elements)
  // Faster than innerHTML; no XSS risk; treats content as raw text
  el.textContent = "<script>alert(1)</script>";  // safely shown as text
  const text = el.textContent;  // all text, including hidden elements

  // innerText:
  // Gets/sets VISIBLE text (CSS-aware; hides display:none, scripts, styles)
  // Slower (triggers reflow to check visibility)
  const visible = el.innerText;  // only visible text

  // Summary:
  // Security: textContent > innerHTML
  // Read all text: textContent
  // Read visible text: innerText
  // Set HTML structure: innerHTML (sanitize first)

7. DIMENSIONS AND POSITION
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  el.offsetWidth / el.offsetHeight    // incl. padding + border (no margin)
  el.clientWidth / el.clientHeight    // incl. padding, excl. border + scrollbar
  el.scrollWidth / el.scrollHeight    // full scrollable content size

  el.offsetTop / el.offsetLeft        // position relative to offsetParent
  el.getBoundingClientRect()          // {top, left, right, bottom, width, height}
                                      // relative to viewport

  window.innerWidth / window.innerHeight  // viewport dimensions
  document.documentElement.scrollTop     // page scroll position

8. COMMON INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What is the difference between innerHTML and textContent?
A: innerHTML parses and renders HTML markup — it can execute scripts if
   inserted carelessly (XSS risk). textContent treats everything as plain
   text and is safe. Use textContent for user-generated content.

Q: What is the difference between attribute and property?
A: Attributes are defined in HTML (always strings, represent initial state).
   Properties are part of the JS DOM object (typed, represent current state).
   They often sync but aren't always equal — e.g. input.value property
   tracks current user input; input.getAttribute("value") returns initial HTML.

Q: Why use DocumentFragment when inserting many elements?
A: Appending to a fragment doesn't trigger reflow. Inserting all elements
   at once via one appendChild(fragment) call causes only one reflow instead
   of one per element — much better performance.

Q: What does classList.toggle do?
A: If the class is present, it removes it; if absent, it adds it. With a
   second boolean argument, it forces add (true) or remove (false).

Q: How do you get the computed/rendered style of an element?
A: Using window.getComputedStyle(element). This returns the final CSS values
   after all stylesheets, inline styles, and inheritance are applied.
   element.style only returns inline styles.
