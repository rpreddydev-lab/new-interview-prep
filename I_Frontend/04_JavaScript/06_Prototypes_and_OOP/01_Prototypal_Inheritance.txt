1. WHAT IS PROTOTYPAL INHERITANCE?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  JavaScript inheritance is prototype-based, not class-based.
  Every object has an internal [[Prototype]] link to another object.
  When a property isn't found on an object, JS walks UP the prototype chain.

  object → [[Prototype]] → [[Prototype]] → ... → Object.prototype → null

  // Every function has a .prototype property:
  function Animal(name) { this.name = name; }
  Animal.prototype.speak = function() { console.log(`${this.name} speaks`); };

  const dog = new Animal("Rex");
  dog.speak();         // "Rex speaks" — found on Animal.prototype
  dog.hasOwnProperty("name");   // true — own property
  dog.hasOwnProperty("speak");  // false — inherited

2. PROTOTYPE CHAIN
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  const obj = {};
  Object.getPrototypeOf(obj) === Object.prototype  // true
  Object.getPrototypeOf(Object.prototype)          // null (end of chain)

  const arr = [];
  Object.getPrototypeOf(arr) === Array.prototype   // true
  Object.getPrototypeOf(Array.prototype) === Object.prototype  // true
  // So arrays have: Array.prototype → Object.prototype → null

  // __proto__ (legacy, non-standard but widely supported):
  obj.__proto__ === Object.prototype  // true
  // Prefer Object.getPrototypeOf() in production code.

3. CONSTRUCTOR FUNCTIONS AND new
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // What `new` does (4 steps):
  // 1. Creates a new empty object
  // 2. Sets its [[Prototype]] to Constructor.prototype
  // 3. Calls the constructor with `this` = the new object
  // 4. Returns the new object (unless constructor returns a non-primitive)

  function Person(name, age) {
    this.name = name;
    this.age = age;
  }
  Person.prototype.greet = function() {
    return `Hi, I'm ${this.name}`;
  };

  const alice = new Person("Alice", 30);
  alice.greet();  // "Hi, I'm Alice"
  alice instanceof Person;  // true
  alice.constructor === Person;  // true

  // instanceof checks the prototype chain:
  alice instanceof Object;  // true (Person.prototype → Object.prototype)

4. INHERITING WITH CONSTRUCTORS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  function Animal(name) {
    this.name = name;
  }
  Animal.prototype.speak = function() {
    return `${this.name} makes a sound.`;
  };

  function Dog(name, breed) {
    Animal.call(this, name);  // call parent constructor
    this.breed = breed;
  }

  // Set up prototype chain:
  Dog.prototype = Object.create(Animal.prototype);
  Dog.prototype.constructor = Dog;  // restore constructor reference

  // Override method:
  Dog.prototype.speak = function() {
    return `${this.name} barks.`;
  };

  const d = new Dog("Rex", "Labrador");
  d.speak();           // "Rex barks."
  d instanceof Dog;    // true
  d instanceof Animal; // true — chain works

5. Object.create()
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Creates an object with a specific prototype:
  const proto = {
    greet() { return `Hello, ${this.name}`; }
  };
  const obj = Object.create(proto);
  obj.name = "Alice";
  obj.greet();  // "Hello, Alice"

  // Create object with no prototype (pure hash map):
  const map = Object.create(null);
  map.key = "value";
  map.hasOwnProperty;  // undefined — no Object.prototype!

6. PROPERTY LOOKUP & hasOwnProperty
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  const parent = { x: 1 };
  const child = Object.create(parent);
  child.y = 2;

  "x" in child                        // true (chain)
  "y" in child                        // true (own)
  child.hasOwnProperty("x")           // false (inherited)
  child.hasOwnProperty("y")           // true (own)
  Object.hasOwn(child, "x")           // false (ES2022 static version — safer)

  // for...in iterates ALL enumerable props including inherited:
  for (const key in child) {
    if (Object.hasOwn(child, key)) {
      console.log(key);  // only "y"
    }
  }

7. COMMON INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What is the prototype chain?
A: Every object has a [[Prototype]] link to another object. When a property
   isn't found on the object, JS follows the chain until it reaches null.
   This chain implements JavaScript's inheritance mechanism.

Q: What is the difference between __proto__ and prototype?
A: __proto__ is an instance's link to its prototype object (the actual chain).
   prototype is a property on FUNCTION objects — it becomes the [[Prototype]]
   of objects created with new. __proto__ is informal; use Object.getPrototypeOf().

Q: What does the new keyword do?
A: Creates an empty object, sets its [[Prototype]] to Constructor.prototype,
   calls the constructor with this = new object, and returns the object.

Q: What is Object.create(null) used for?
A: Creates an object with NO prototype (no Object.prototype methods like
   hasOwnProperty, toString). Useful as a pure dictionary/hash map to avoid
   prototype pollution or key collisions with inherited property names.

Q: What does instanceof check?
A: Whether Constructor.prototype appears anywhere in the object's prototype
   chain. It does NOT check the constructor itself — the chain is what matters.

Q: What is the difference between own property and inherited property?
A: An own property is defined directly on the object. An inherited property
   comes from the prototype chain. hasOwnProperty() or Object.hasOwn() distinguishes them.
