2. CLASS SYNTAX
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  class Animal {
    // Static field (ES2022):
    static count = 0;

    // Instance field (ES2022):
    #sound = "...";  // private field

    constructor(name) {
      this.name = name;
      Animal.count++;
    }

    // Prototype method:
    speak() {
      return `${this.name} says ${this.#sound}`;
    }

    // Static method:
    static create(name) {
      return new Animal(name);
    }
  }

  const a = new Animal("Cat");
  a.speak();           // "Cat says ..."
  Animal.create("Dog"); // static method
  Animal.count;         // 2

3. INHERITANCE WITH extends AND super
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  class Animal {
    constructor(name) { this.name = name; }
    speak() { return `${this.name} makes a sound.`; }
  }

  class Dog extends Animal {
    constructor(name, breed) {
      super(name);       // MUST call super() before using `this`
      this.breed = breed;
    }

    speak() {             // override parent method
      return `${this.name} barks.`;
    }

    parentSpeak() {
      return super.speak();  // call parent method explicitly
    }
  }

  const d = new Dog("Rex", "Labrador");
  d.speak();           // "Rex barks."
  d.parentSpeak();     // "Rex makes a sound."
  d instanceof Dog;    // true
  d instanceof Animal; // true

  // Rules:
  // - Derived class constructor MUST call super() before using `this`
  // - super() invokes the parent constructor
  // - super.method() calls a parent prototype method

4. PRIVATE FIELDS AND METHODS (#)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  class BankAccount {
    #balance = 0;          // truly private — not accessible outside class
    #owner;

    constructor(owner, initialBalance) {
      this.#owner = owner;
      this.#balance = initialBalance;
    }

    deposit(amount) {
      if (amount <= 0) throw new Error("Positive amounts only");
      this.#balance += amount;
    }

    get balance() { return this.#balance; }  // controlled access

    #validate(amount) {  // private method
      return amount > 0 && amount <= this.#balance;
    }

    withdraw(amount) {
      if (!this.#validate(amount)) throw new Error("Invalid withdrawal");
      this.#balance -= amount;
    }
  }

  const account = new BankAccount("Alice", 1000);
  account.balance;        // 1000 (via getter)
  account.#balance;       // SyntaxError — truly private
  account.deposit(500);
  account.balance;        // 1500

5. STATIC FIELDS AND METHODS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  class MathHelper {
    static PI = 3.14159;
    static #instanceCount = 0;

    constructor() { MathHelper.#instanceCount++; }

    static circleArea(r) { return MathHelper.PI * r * r; }
    static getCount() { return MathHelper.#instanceCount; }
  }

  MathHelper.circleArea(5);  // 78.53...
  MathHelper.PI;             // 3.14159
  // Static members belong to the CLASS, not instances.

8. COMMON INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: Are JS classes real classes like in Java or Python?
A: No. They are syntactic sugar over prototype-based inheritance. The object
   model is still prototypal — "classes" just provide a cleaner syntax.

Q: What does super() do?
A: In a derived (child) class constructor, super() calls the parent constructor.
   It must be called before any reference to `this`. super.method() calls a
   parent method from the prototype chain.

Q: What are private class fields (#)?
A: Fields/methods prefixed with # are truly private — they cannot be accessed
   or modified outside the class body. Unlike "convention private" (_name),
   they throw a SyntaxError on external access.

Q: Are class methods enumerable?
A: No. Methods defined in a class body are added to the prototype as
   non-enumerable, unlike constructor-function prototype methods which are
   enumerable by default. This means they won't show up in for...in loops.

Q: Are classes hoisted?
A: The declaration is hoisted but NOT initialized (TDZ). Accessing a class
   before its declaration throws a ReferenceError.
