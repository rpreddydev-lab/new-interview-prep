1. WHAT ARE GETTERS AND SETTERS?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Getters and setters (accessor properties) allow you to define computed
  or controlled properties that LOOK like regular properties to callers.

  - getter: runs a function when the property is READ
  - setter: runs a function when the property is WRITTEN

  // Object literal:
  const person = {
    firstName: "John",
    lastName: "Doe",
    get fullName() {
      return `${this.firstName} ${this.lastName}`;
    },
    set fullName(value) {
      [this.firstName, this.lastName] = value.split(" ");
    }
  };

  person.fullName;           // "John Doe" — calls getter
  person.fullName = "Alice Smith";  // calls setter
  person.firstName;          // "Alice"

2. GETTERS AND SETTERS IN CLASSES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  class Temperature {
    #celsius;

    constructor(celsius) {
      this.#celsius = celsius;
    }

    get fahrenheit() {
      return (this.#celsius * 9) / 5 + 32;
    }
    set fahrenheit(f) {
      this.#celsius = ((f - 32) * 5) / 9;
    }

    get celsius() { return this.#celsius; }
    set celsius(c) {
      if (c < -273.15) throw new RangeError("Below absolute zero");
      this.#celsius = c;
    }
  }

  const t = new Temperature(100);
  t.fahrenheit;        // 212 (computed)
  t.fahrenheit = 32;   // triggers setter
  t.celsius;           // 0 (recalculated)
  t.celsius = -300;    // throws RangeError

3. Object.defineProperty()
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  const obj = { _name: "Alice" };

  Object.defineProperty(obj, "name", {
    get() { return this._name.toUpperCase(); },
    set(v) { this._name = v.trim(); },
    enumerable: true,
    configurable: true
  });

  obj.name;           // "ALICE"
  obj.name = "  Bob  ";
  obj._name;          // "Bob"

  // ACCESSOR properties vs DATA properties:
  // Data:     { value, writable, enumerable, configurable }
  // Accessor: { get, set, enumerable, configurable }
  // Cannot mix value/writable with get/set — throws TypeError.

4. USE CASES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // 1. COMPUTED / DERIVED PROPERTIES:
  class Circle {
    constructor(radius) { this.radius = radius; }
    get area()        { return Math.PI * this.radius ** 2; }
    get circumference() { return 2 * Math.PI * this.radius; }
  }
  const c = new Circle(5);
  c.area;  // 78.53... (no need to store, derived on the fly)

  // 2. VALIDATION:
  class User {
    #age;
    set age(v) {
      if (typeof v !== "number" || v < 0 || v > 150)
        throw new RangeError("Invalid age");
      this.#age = v;
    }
    get age() { return this.#age; }
  }

  // 3. LAZY INITIALIZATION (compute once on first access):
  class HeavyComputation {
    get expensiveResult() {
      const result = doExpensiveWork();
      Object.defineProperty(this, "expensiveResult", {
        value: result, writable: false
      });
      return result;  // next access is cached data property
    }
  }

  // 4. BACKWARD COMPATIBILITY (rename internal storage):
  class Config {
    #timeout = 5000;
    get timeout() { return this.#timeout; }
    set timeout(ms) { this.#timeout = ms; }
    // Old code: config.timeout — still works
    // New code: can change storage without breaking callers
  }

5. PITFALLS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Infinite recursion — setter calling itself:
  class Bad {
    set name(v) { this.name = v; }  // calls setter again — RangeError!
  }
  // Fix: use a backing field or private field:
  class Good {
    #name;
    set name(v) { this.#name = v; }
    get name() { return this.#name; }
  }

  // Getter without setter — silently ignores writes in non-strict:
  const obj = {
    get x() { return 42; }
  };
  obj.x = 99;    // no error in non-strict but does nothing
  obj.x;         // 42

6. COMMON INTERVIEW QUESTIONS

Q: What are getters and setters used for?
A: To define computed properties, validate data on assignment, control
   access to private/backing fields, and maintain a clean public API while
   hiding internal implementation details.

Q: What is the difference between a data property and an accessor property?
A: A data property has value and writable descriptors. An accessor property
   has get/set functions instead. They cannot be mixed — combining value and
   get on the same property throws a TypeError.

Q: How do you avoid infinite recursion in a setter?
A: Use a private field (#name) or a prefixed backing field (_name) as the
   actual storage. The setter should write to that field, not to the
   same property name (which would re-trigger the setter).

Q: Can getters slow down performance?
A: Yes, if a getter does expensive computation every access. Use lazy
   initialization (compute once, replace with data property) or memoization
   inside the getter to cache results.

Q: Are getters enumerable?
A: By default in object literals: yes. In class bodies: no (non-enumerable).
   You can control this with Object.defineProperty's enumerable flag.
