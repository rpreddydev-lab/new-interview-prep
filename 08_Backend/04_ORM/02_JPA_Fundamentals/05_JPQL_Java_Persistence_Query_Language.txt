1. WHAT IS JPQL?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

JPQL is an object-oriented query language defined by the JPA spec.
It is similar to SQL but operates on ENTITIES and their fields,
not on tables and columns.

Key Differences from SQL:
  SQL:   SELECT * FROM users WHERE email = 'alice@example.com'
  JPQL:  SELECT u FROM User u WHERE u.email = :email

  - Use entity name (User), not table name (users)
  - Use field names (email), not column names (email_address)
  - Navigate relationships: u.orders.status (no JOIN needed for navigation)
  - Case-sensitive for entity/field names
  - Keywords (SELECT, FROM, WHERE) are case-insensitive

2. BASIC SELECT QUERIES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

// All users
em.createQuery("SELECT u FROM User u", User.class).getResultList();

// With WHERE
em.createQuery("SELECT u FROM User u WHERE u.active = true", User.class)
  .getResultList();

// Named parameters (preferred)
em.createQuery("SELECT u FROM User u WHERE u.email = :email", User.class)
  .setParameter("email", "alice@example.com")
  .getSingleResult();

// Positional parameters (legacy, avoid)
em.createQuery("SELECT u FROM User u WHERE u.email = ?1", User.class)
  .setParameter(1, "alice@example.com")
  .getSingleResult();

// Select specific fields (returns Object[])
em.createQuery("SELECT u.email, u.firstName FROM User u")
  .getResultList();

// DTO projection using constructor expression
em.createQuery(
    "SELECT new com.example.dto.UserDto(u.id, u.email, u.firstName) " +
    "FROM User u WHERE u.active = true", UserDto.class)
  .getResultList();

3. JOINS IN JPQL
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

INNER JOIN (filters out unmatched):
  SELECT u FROM User u INNER JOIN u.orders o WHERE o.total > 100

LEFT JOIN (keeps users with no orders):
  SELECT u FROM User u LEFT JOIN u.orders o WHERE o.status = 'PENDING'

JOIN FETCH (N+1 fix — eagerly load association):
  SELECT u FROM User u JOIN FETCH u.orders WHERE u.id = :id
  // Single SQL with JOIN; loads user AND orders in one query

LEFT JOIN FETCH (include users with no orders):
  SELECT u FROM User u LEFT JOIN FETCH u.orders

Multiple JOIN FETCH:
  SELECT u FROM User u
  JOIN FETCH u.orders o
  JOIN FETCH o.items
  WHERE u.id = :id

WARNING — Distinct with JOIN FETCH:
  SELECT DISTINCT u FROM User u LEFT JOIN FETCH u.orders
  // Without DISTINCT: user appears N times (once per order)
  // Hibernate 6: use @QueryHints or QueryHint PASS_DISTINCT_THROUGH

4. AGGREGATE FUNCTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  COUNT(u)         → row count
  SUM(o.total)     → sum
  AVG(o.total)     → average
  MIN(u.age)       → minimum
  MAX(u.age)       → maximum

SELECT COUNT(u) FROM User u WHERE u.active = true

SELECT AVG(o.total), MAX(o.total), MIN(o.total)
FROM Order o WHERE o.user.id = :userId

SELECT u.department, COUNT(u), AVG(u.salary)
FROM User u
GROUP BY u.department
HAVING COUNT(u) > 5
ORDER BY AVG(u.salary) DESC

5. WHERE CLAUSE OPERATORS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Comparison:  =, <>, <, >, <=, >=
Logical:     AND, OR, NOT
Null checks: IS NULL, IS NOT NULL
Range:       BETWEEN ... AND ...
Pattern:     LIKE '%alice%', LIKE 'al_ce' (_ = one char, % = many)
             ESCAPE: LIKE '10\%' ESCAPE '\'
Collection:  IN (:emails), NOT IN (:ids), MEMBER OF, IS EMPTY
Subquery:    EXISTS (SELECT ...), ALL, ANY, SOME

Examples:
  WHERE u.age BETWEEN 18 AND 65
  WHERE u.email LIKE '%@gmail.com'
  WHERE u.id IN :ids                  -- list parameter
  WHERE o.items IS EMPTY              -- empty collection check
  WHERE :product MEMBER OF o.items    -- item in collection
  WHERE EXISTS (SELECT 1 FROM Order o WHERE o.user = u AND o.total > 500)

6. BULK UPDATE AND DELETE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

IMPORTANT: Bulk DML (UPDATE/DELETE in JPQL) bypasses the persistence
context. The DB is updated but in-memory entities are NOT refreshed.
Always call em.clear() after bulk operations to avoid stale state.

Bulk UPDATE:
  @Modifying
  @Query("UPDATE User u SET u.active = false WHERE u.lastLogin < :date")
  int deactivateOldUsers(@Param("date") LocalDate date);

  // Or via EntityManager:
  int updated = em.createQuery(
      "UPDATE User u SET u.active = false WHERE u.lastLogin < :date")
      .setParameter("date", date)
      .executeUpdate();
  em.clear();  // clear stale entities from persistence context

Bulk DELETE:
  @Modifying
  @Query("DELETE FROM Order o WHERE o.status = 'CANCELLED' AND o.createdAt < :before")
  int deleteCancelledOrders(@Param("before") LocalDate before);

7. ORDERING AND PAGINATION
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

ORDER BY:
  SELECT u FROM User u ORDER BY u.lastName ASC, u.firstName ASC
  SELECT u FROM User u ORDER BY u.createdAt DESC NULLS LAST

Pagination via EntityManager:
  List<User> page = em.createQuery("SELECT u FROM User u ORDER BY u.id", User.class)
      .setFirstResult(20)    // OFFSET: skip first 20 (page 2, size 20)
      .setMaxResults(20)     // LIMIT 20
      .getResultList();

Via Spring Data Pageable:
  Page<User> findByActive(boolean active, Pageable pageable);

  // Call:
  PageRequest.of(0, 20, Sort.by(Sort.Direction.DESC, "createdAt"))

8. JPQL FUNCTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

String:   CONCAT, SUBSTRING, TRIM, LOWER, UPPER, LENGTH, LOCATE
Math:     ABS, SQRT, MOD, SIZE (collection), INDEX
Date:     CURRENT_DATE, CURRENT_TIME, CURRENT_TIMESTAMP
Type:     TYPE(e) = SubclassName (for inheritance queries)

SELECT UPPER(u.firstName), LENGTH(u.email) FROM User u

SELECT u FROM User u
WHERE TYPE(u) = PremiumUser    -- Inheritance type filter

SELECT u FROM User u
WHERE SIZE(u.orders) > 3      -- users with more than 3 orders

9. SUBQUERIES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

-- Users who placed at least one order over $500
SELECT u FROM User u
WHERE EXISTS (
    SELECT o FROM Order o WHERE o.user = u AND o.total > 500
)

-- Users whose total spending is above average
SELECT u FROM User u
WHERE (SELECT SUM(o.total) FROM Order o WHERE o.user = u) >
      (SELECT AVG(o.total) FROM Order o)

-- Users who have ALL orders in COMPLETED status
SELECT u FROM User u
WHERE 'PENDING' <> ALL
      (SELECT o.status FROM Order o WHERE o.user = u)

10. JPQL IN SPRING DATA JPA
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

@Repository
public interface OrderRepository extends JpaRepository<Order, Long> {

    // Derived query method (auto-parsed from name)
    List<Order> findByStatusAndCreatedAtAfter(String status, LocalDate date);

    // JPQL with @Query
    @Query("SELECT o FROM Order o WHERE o.user.id = :userId AND o.total > :min")
    List<Order> findByUserAndMinTotal(
        @Param("userId") Long userId,
        @Param("min") BigDecimal min
    );

    // Projection (select subset of fields)
    @Query("SELECT o.id as id, o.total as total, o.status as status FROM Order o")
    List<OrderSummary> findOrderSummaries();

    // Modifying JPQL (bulk update)
    @Modifying
    @Transactional
    @Query("UPDATE Order o SET o.status = 'EXPIRED' WHERE o.createdAt < :before AND o.status = 'PENDING'")
    int expireOldOrders(@Param("before") LocalDateTime before);

    // Pagination with JPQL
    @Query("SELECT o FROM Order o WHERE o.user.id = :userId ORDER BY o.createdAt DESC")
    Page<Order> findRecentByUser(@Param("userId") Long userId, Pageable pageable);
}

11. INTERVIEW QUICK HITS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What is JPQL and how does it differ from SQL?
A: JPQL is an object-oriented query language that works on entity objects
   and their fields, not on tables and columns. It's DB-agnostic; Hibernate
   translates JPQL to the DB-specific SQL at runtime.

Q: What is JOIN FETCH and when do you use it?
A: JOIN FETCH is a JPQL extension that eagerly loads an associated collection
   in the same SQL query, solving the N+1 problem.
   "SELECT u FROM User u JOIN FETCH u.orders WHERE u.active = true"
   → single SQL with JOIN; no extra queries for orders.

Q: What happens when you use DISTINCT with JOIN FETCH?
A: Without DISTINCT, a user with 3 orders appears 3 times in the result list.
   Use SELECT DISTINCT u FROM User u JOIN FETCH u.orders to deduplicate.
   In Hibernate 6, use QueryHints.HINT_PASS_DISTINCT_THROUGH = false.

Q: Can JPQL do subqueries?
A: Yes, in WHERE and HAVING clauses. SELECT and FROM subqueries are NOT
   supported in standard JPQL (use native SQL for those).

Q: What is the constructor expression in JPQL?
A: SELECT new com.example.UserDto(u.id, u.email) FROM User u
   This calls the UserDto(Long, String) constructor directly, returning
   a typed list instead of Object[]. Great for read-only DTO projections.

Q: What is the difference between getResultList() and getSingleResult()?
A: getResultList() returns a List (empty if no results, no exception).
   getSingleResult() throws NoResultException (no results) or
   NonUniqueResultException (more than one result). Use Optional to wrap.
