===================================================================
TRAP 1: INTEGER OVERFLOW
===================================================================

// WRONG — silently overflows if a and b are large
int sum = a + b;
int product = a * b;

// RIGHT — cast before the operation
long sum = (long) a + b;
long product = (long) a * b;

// WRONG comparator — overflows when a is very negative, b is positive
(a, b) -> a - b

// RIGHT
(a, b) -> Integer.compare(a, b)
Comparator.comparingInt(x -> x)

// Example silent overflow:
int a = Integer.MIN_VALUE;   // -2147483648
int b = 1;
System.out.println(a - b);   // prints 2147483647 (WRONG — wrapped around)
System.out.println(Long.compare(a, b));  // correct

// Safe midpoint in binary search
int mid = lo + (hi - lo) / 2;   // CORRECT
int mid = (lo + hi) / 2;         // WRONG if lo+hi overflows

===================================================================
TRAP 2: == vs .equals() FOR STRINGS AND WRAPPERS
===================================================================

// For String content comparison — ALWAYS use .equals()
String a = new String("hello");
String b = new String("hello");
System.out.println(a == b);       // FALSE — different objects
System.out.println(a.equals(b));  // TRUE

// Integer cache: JVM caches Integer objects for -128 to 127
Integer x = 127;
Integer y = 127;
System.out.println(x == y);   // TRUE  (cached same object)

Integer p = 128;
Integer q = 128;
System.out.println(p == q);   // FALSE (outside cache range, different objects)
System.out.println(p.equals(q)); // TRUE — ALWAYS use .equals()

// Map uses .equals() for key lookup, but if you use Integer keys:
Map<Integer, String> map = new HashMap<>();
map.put(200, "value");
Integer key = 200;
map.get(key);   // works fine (.equals() used internally)
map.get(key == 200);  // compile error anyway, just be aware

// Collections.sort comparator with Integer:
List<Integer> list = ...;
list.sort((a, b) -> a - b);   // UNSAFE: overflow risk
list.sort(Integer::compare);   // SAFE

===================================================================
TRAP 3: AUTOBOXING / UNBOXING SURPRISES
===================================================================

// NullPointerException from unboxing null Integer
Integer val = null;
int x = val;   // NullPointerException!

// Safe pattern
int x = (val != null) ? val : 0;

// Unboxing in arithmetic
Map<String, Integer> freq = new HashMap<>();
freq.get("missing") + 1   // NullPointerException if key absent!
// RIGHT:
freq.getOrDefault("missing", 0) + 1

// Autoboxing in PriorityQueue comparison
PriorityQueue<Integer> pq = new PriorityQueue<>();
// Calling pq.remove(3) calls remove(Object o) — removes VALUE 3
// Calling pq.remove() calls remove() — removes the TOP element
// These are DIFFERENT methods — easy to confuse

===================================================================
TRAP 4: ARRAYS.ASLIST FIXED-SIZE TRAP
===================================================================

// Returns FIXED-SIZE list — NO add() or remove()
List<Integer> list = Arrays.asList(1, 2, 3);
list.add(4);    // UnsupportedOperationException!
list.set(0, 9); // OK — set is fine (size doesn't change)

// RIGHT — wrap in ArrayList if you need to add/remove
List<Integer> list = new ArrayList<>(Arrays.asList(1, 2, 3));

// List.of() is even stricter — also no null, no set
List<Integer> imm = List.of(1, 2, 3);
imm.set(0, 9);  // UnsupportedOperationException!

===================================================================
TRAP 5: CONCURRENTMODIFICATIONEXCEPTION
===================================================================

// WRONG — removing from list while iterating with for-each
for (Integer x : list) {
    if (x % 2 == 0) list.remove(x);  // ConcurrentModificationException!
}

// RIGHT option 1 — use Iterator.remove()
Iterator<Integer> it = list.iterator();
while (it.hasNext()) {
    if (it.next() % 2 == 0) it.remove();
}

// RIGHT option 2 — iterate backwards (for index-based removal)
for (int i = list.size() - 1; i >= 0; i--)
    if (list.get(i) % 2 == 0) list.remove(i);

// RIGHT option 3 — collect and remove after
List<Integer> toRemove = new ArrayList<>();
for (int x : list) if (x % 2 == 0) toRemove.add(x);
list.removeAll(toRemove);

// RIGHT option 4 — Java 8+ removeIf
list.removeIf(x -> x % 2 == 0);

// Same trap with Map:
for (Map.Entry<String,Integer> e : map.entrySet()) {
    if (e.getValue() == 0) map.remove(e.getKey()); // ConcurrentModificationException!
}
// RIGHT:
map.entrySet().removeIf(e -> e.getValue() == 0);

TRAP 6: STRING SPLIT EDGE CASES
===================================================================

// "." in regex means ANY character — must escape
"a.b.c".split(".")    // returns [] (all chars match, nothing left)
"a.b.c".split("\\.")  // returns ["a", "b", "c"] — CORRECT

// Split on "|" — must escape (| is regex OR)
"a|b|c".split("|")    // splits on every char!
"a|b|c".split("\\|")  // ["a", "b", "c"]

// Trailing empty strings are discarded by default
"a,,b,,".split(",")      // ["a", "", "b"]  — trailing empties dropped
"a,,b,,".split(",", -1)  // ["a", "", "b", "", ""]  — keep all

// Empty string split
"".split(",")    // returns [""] (array of length 1, not 0)
"".isEmpty()     // true — check this first

TRAP 7: CHAR ARITHMETIC PITFALLS
===================================================================

// char + char = int (auto-promoted to int!)
char a = 'A', b = 'B';
char c = a + b;    // COMPILE ERROR: int cannot be converted to char
char c = (char)(a + b);  // OK with cast

// char + int = int
char c = 'a' + 1;  // COMPILE ERROR
char c = (char)('a' + 1);  // 'b' — correct

// Comparing chars
'a' < 'b'   // true (compares int values)
'a' == 97   // true (char vs int)

// Building string from chars
char[] arr = {'h','e','l','l','o'};
String s = new String(arr);          // correct
String s = String.valueOf(arr);      // also correct
String s = arr.toString();           // WRONG — prints [C@hashcode

TRAP 8: ARRAY INITIALIZATION DEFAULTS
===================================================================

int[]     arr = new int[5];      // all 0
boolean[] arr = new boolean[5];  // all false
double[]  arr = new double[5];   // all 0.0
String[]  arr = new String[5];   // all null   ← common source of NPE
int[][]   grid = new int[3][3];  // all 0

// Object arrays: elements are null until assigned
List<List<Integer>> list = new ArrayList<>();
// WRONG — list.get(0) is null
list.add(null);
list.get(0).add(1);  // NullPointerException!

// RIGHT — initialize inner lists
for (int i = 0; i < n; i++) list.add(new ArrayList<>());
// OR use computeIfAbsent for Maps

TRAP 9: MODIFYING PARAMETERS (Pass by Value)
===================================================================

// Java passes REFERENCES by value — you can mutate object state
// but you CANNOT reassign the reference itself
void badReset(int[] arr) {
    arr = new int[5];  // reassigns LOCAL variable — caller's arr unchanged!
}

void goodMutate(int[] arr) {
    Arrays.fill(arr, 0);  // mutates the actual array — works!
}

// Same with List:
void badAdd(List<Integer> list) {
    list = new ArrayList<>();  // caller's list unchanged!
}
void goodAdd(List<Integer> list) {
    list.add(1);  // mutates caller's list — works!
}

TRAP 10: INTEGER.MIN_VALUE AND MATH.ABS
===================================================================

Math.abs(Integer.MIN_VALUE)   // returns Integer.MIN_VALUE! (overflow)
Math.abs(Long.MIN_VALUE)      // same trap for Long

// Safe absolute value for problem solving:

// -Integer.MIN_VALUE == Integer.MIN_VALUE (!)
// This bites you in "reverse integer" problems:
// Always check: result > Integer.MAX_VALUE || result < Integer.MIN_VALUE

TRAP 12: COLLECTIONS WITH NULL VALUES
===================================================================

// HashMap allows one null KEY and multiple null VALUES
Map<String, Integer> map = new HashMap<>();
map.put(null, 1);   // OK
map.put("key", null); // OK
map.get("missing")    // returns null — same as null VALUE — ambiguous!

// Use containsKey() to distinguish:
if (map.containsKey("key")) { int v = map.get("key"); ... }

// TreeMap / TreeSet do NOT allow null keys (NullPointerException)
new TreeMap<>().put(null, 1);   // NullPointerException!

// PriorityQueue does NOT allow null elements
new PriorityQueue<>().offer(null);  // NullPointerException!

// ArrayDeque does NOT allow null elements
new ArrayDeque<>().push(null);  // NullPointerException!

// HashSet allows null
new HashSet<>().add(null);  // OK

===================================================================
TRAP 13: REMOVE ON LIST BY INDEX vs BY VALUE
===================================================================

List<Integer> list = new ArrayList<>(Arrays.asList(1, 2, 3));

list.remove(1);                   // removes INDEX 1 → list is [1, 3]
list.remove(Integer.valueOf(1));  // removes VALUE 1 → list is [2, 3]

// This is the most common silent bug with Integer lists!
// Always use Integer.valueOf(x) when removing by value.

===================================================================
TRAP 14: MODULO WITH NEGATIVE NUMBERS
===================================================================

// Java % returns negative result if dividend is negative
-7 % 3   →  -1   (NOT 2 like in math)
-7 % -3  →  -1

// Fix: add MOD to ensure non-negative
int result = ((a % MOD) + MOD) % MOD;

// Useful when computing modular arithmetic in DP
(a - b + MOD) % MOD  // safe subtraction

TRAP 15: FOR-EACH ON NULL COLLECTION
===================================================================

// NullPointerException if list is null
for (int x : list) { ... }  // NPE if list == null

// Safe check
if (list != null) for (int x : list) { ... }
// OR initialize defensively: List<X> list = new ArrayList<>();
