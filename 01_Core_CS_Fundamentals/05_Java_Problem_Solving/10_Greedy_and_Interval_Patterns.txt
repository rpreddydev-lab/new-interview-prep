===================================================================
PART 1: GREEDY STRATEGY
===================================================================
WHEN to use Greedy:
  1. Problem asks for min/max of something
  2. At each step, a locally optimal choice leads to global optimum
  3. No need to reconsider past choices (no backtracking)

HOW to verify greedy works:
  - Exchange argument: show swapping any two adjacent choices only worsens result
  - Greedy stays ahead: show greedy solution is always ≥ (or ≤) any other

GREEDY vs DP:
  - Greedy: make one irrevocable choice per step (no recalculation)
  - DP: consider all choices, keep optimal (requires overlapping subproblems)

KEY INSIGHT: Almost all interval problems start with SORTING by start time.

// Standard sort by start time
int[][] intervals = {{1,3},{2,6},{8,10}};
Arrays.sort(intervals, (a, b) -> a[0] - b[0]);

// Sort by end time (for activity selection / non-overlapping)
Arrays.sort(intervals, (a, b) -> a[1] - b[1]);

--- MERGE OVERLAPPING INTERVALS — LeetCode 56 ---
// Sort by start, merge if current start ≤ prev end
int[][] merge(int[][] intervals) {
    Arrays.sort(intervals, (a, b) -> a[0] - b[0]);
    List<int[]> merged = new ArrayList<>();
    for (int[] iv : intervals) {
        if (merged.isEmpty() || merged.get(merged.size()-1)[1] < iv[0]) {
            merged.add(iv);                                              // no overlap
        } else {
            merged.get(merged.size()-1)[1] = Math.max(
                merged.get(merged.size()-1)[1], iv[1]);                  // extend
        }
    }
    return merged.toArray(new int[0][]);
}

--- INSERT INTERVAL — LeetCode 57 ---
// Merge a new interval into a sorted non-overlapping list
int[][] insert(int[][] intervals, int[] newInterval) {
    List<int[]> result = new ArrayList<>();
    int i = 0, n = intervals.length;
    // Add all intervals that end before newInterval starts
    while (i < n && intervals[i][1] < newInterval[0]) result.add(intervals[i++]);
    // Merge all overlapping intervals with newInterval
    while (i < n && intervals[i][0] <= newInterval[1]) {
        newInterval[0] = Math.min(newInterval[0], intervals[i][0]);
        newInterval[1] = Math.max(newInterval[1], intervals[i][1]);
        i++;
    }
    result.add(newInterval);
    // Add remaining
    while (i < n) result.add(intervals[i++]);
    return result.toArray(new int[0][]);
}

--- NON-OVERLAPPING INTERVALS (min removals) — LeetCode 435 ---
// Greedy: sort by END time, greedily keep intervals that end earliest
int eraseOverlapIntervals(int[][] intervals) {
    Arrays.sort(intervals, (a, b) -> a[1] - b[1]);   // sort by END
    int count = 0, end = Integer.MIN_VALUE;
    for (int[] iv : intervals) {
        if (iv[0] >= end) {     // no overlap: keep this interval
            end = iv[1];
        } else {                // overlap: remove this interval (count removal)
            count++;
        }
    }
    return count;
}

--- MEETING ROOMS I (can attend all?) — LeetCode 252 ---
boolean canAttendMeetings(int[][] intervals) {
    Arrays.sort(intervals, (a, b) -> a[0] - b[0]);
    for (int i = 1; i < intervals.length; i++)
        if (intervals[i][0] < intervals[i-1][1]) return false;
    return true;
}

--- MEETING ROOMS II (min rooms needed) — LeetCode 253 ---
// Approach 1: Min-Heap of end times
int minMeetingRooms(int[][] intervals) {
    Arrays.sort(intervals, (a, b) -> a[0] - b[0]);
    PriorityQueue<Integer> endTimes = new PriorityQueue<>();   // min-heap
    for (int[] iv : intervals) {
        if (!endTimes.isEmpty() && endTimes.peek() <= iv[0])
            endTimes.poll();    // reuse the room that ends earliest
        endTimes.offer(iv[1]);
    }
    return endTimes.size();     // number of rooms still in use
}

// Approach 2: Sweep line (separate start/end events)
int minMeetingRoomsSweep(int[][] intervals) {
    int[] starts = new int[intervals.length];
    int[] ends   = new int[intervals.length];
    for (int i = 0; i < intervals.length; i++) {
        starts[i] = intervals[i][0];
        ends[i]   = intervals[i][1];
    }
    Arrays.sort(starts); Arrays.sort(ends);
    int rooms = 0, maxRooms = 0, e = 0;
    for (int s = 0; s < starts.length; s++) {
        if (starts[s] < ends[e]) rooms++;
        else { rooms--; e++; }  // a meeting ended before this one started
        maxRooms = Math.max(maxRooms, rooms);
    }
    return maxRooms;
}

--- TASK SCHEDULER — LeetCode 621 ---
// Greedy: schedule most frequent tasks first, idle when needed
int leastInterval(char[] tasks, int n) {
    int[] freq = new int[26];
    for (char c : tasks) freq[c - 'A']++;
    Arrays.sort(freq);
    int maxFreq = freq[25];
    int maxCount = 0;  // number of tasks with maxFreq
    for (int f : freq) if (f == maxFreq) maxCount++;
    // Formula: (maxFreq - 1) * (n + 1) + maxCount
    return Math.max(tasks.length, (maxFreq - 1) * (n + 1) + maxCount);
}

--- JUMP GAME I (can reach end?) — LeetCode 55 ---
boolean canJump(int[] nums) {
    int maxReach = 0;
    for (int i = 0; i < nums.length; i++) {
        if (i > maxReach) return false;      // can't reach here
        maxReach = Math.max(maxReach, i + nums[i]);
    }
    return true;
}

--- JUMP GAME II (min jumps to reach end) — LeetCode 45 ---
int jump(int[] nums) {
    int jumps = 0, curEnd = 0, farthest = 0;
    for (int i = 0; i < nums.length - 1; i++) {
        farthest = Math.max(farthest, i + nums[i]);
        if (i == curEnd) {        // reached boundary of current jump
            jumps++;
            curEnd = farthest;    // extend boundary
        }
    }
    return jumps;
}

--- GAS STATION — LeetCode 134 ---
// If total gas >= total cost, a solution always exists
// Start from position after last deficit
int canCompleteCircuit(int[] gas, int[] cost) {
    int total = 0, tank = 0, start = 0;
    for (int i = 0; i < gas.length; i++) {
        int net = gas[i] - cost[i];
        total += net;
        tank  += net;
        if (tank < 0) { start = i + 1; tank = 0; }  // can't start before i+1
    }
    return total >= 0 ? start : -1;
}

--- CANDY — LeetCode 135 ---
// Each child must have 1; higher rating than neighbor → more candy
int candy(int[] ratings) {
    int n = ratings.length;
    int[] candy = new int[n];
    Arrays.fill(candy, 1);
    // Left pass: if rating[i] > rating[i-1], give one more
    for (int i = 1; i < n; i++)
        if (ratings[i] > ratings[i-1]) candy[i] = candy[i-1] + 1;
    // Right pass: if rating[i] > rating[i+1], ensure more candy
    for (int i = n - 2; i >= 0; i--)
        if (ratings[i] > ratings[i+1]) candy[i] = Math.max(candy[i], candy[i+1] + 1);
    int sum = 0;
    for (int c : candy) sum += c;
    return sum;
}

--- LARGEST NUMBER — LeetCode 179 ---
// Custom sort: compare concatenations
String largestNumber(int[] nums) {
    String[] strs = new String[nums.length];
    for (int i = 0; i < nums.length; i++) strs[i] = String.valueOf(nums[i]);
    Arrays.sort(strs, (a, b) -> (b + a).compareTo(a + b));
    if (strs[0].equals("0")) return "0";
    return String.join("", strs);
}

--- BOAT TO SAVE PEOPLE — LeetCode 881 ---
// Sort, use two pointers: pair heaviest with lightest if possible
int numRescueBoats(int[] people, int limit) {
    Arrays.sort(people);
    int left = 0, right = people.length - 1, boats = 0;
    while (left <= right) {
        if (people[left] + people[right] <= limit) left++;
        right--;
        boats++;
    }
    return boats;
}

--- PARTITION LABELS — LeetCode 763 ---
// Greedy: track last occurrence of each character
List<Integer> partitionLabels(String s) {
    int[] last = new int[26];
    for (int i = 0; i < s.length(); i++) last[s.charAt(i) - 'a'] = i;
    List<Integer> result = new ArrayList<>();
    int start = 0, end = 0;
    for (int i = 0; i < s.length(); i++) {
        end = Math.max(end, last[s.charAt(i) - 'a']);
        if (i == end) { result.add(end - start + 1); start = i + 1; }
    }
    return result;
}

--- MINIMUM NUMBER OF ARROWS TO BURST BALLOONS — LeetCode 452 ---
// Same logic as non-overlapping intervals (sort by end)
int findMinArrowShots(int[][] points) {
    Arrays.sort(points, (a, b) -> Integer.compare(a[1], b[1]));
    int arrows = 1, end = points[0][1];
    for (int i = 1; i < points.length; i++) {
        if (points[i][0] > end) { arrows++; end = points[i][1]; }
    }
    return arrows;
}

--- TWO CITY SCHEDULING — LeetCode 1029 ---
// Sort by (costA - costB); send first half to A, second to B
int twoCitySchedCost(int[][] costs) {
    Arrays.sort(costs, (a, b) -> (a[0]-a[1]) - (b[0]-b[1]));
    int total = 0, n = costs.length / 2;
    for (int i = 0; i < n; i++)      total += costs[i][0];    // city A
    for (int i = n; i < 2*n; i++)    total += costs[i][1];    // city B
    return total;
}

// "Find overlap count at each point" — e.g., max concurrent meetings
int maxOverlap(int[][] intervals) {
    int max = 0;
    for (int[] iv : intervals) max = Math.max(max, iv[1]);
    int[] diff = new int[max + 2];
    for (int[] iv : intervals) { diff[iv[0]]++; diff[iv[1]+1]--; }
    int cur = 0, peak = 0;
    for (int d : diff) { cur += d; peak = Math.max(peak, cur); }
    return peak;
}

// "Apply range updates, query final values" (Difference Array)
// Add val to arr[l..r] for many updates, then prefix sum to get result
int[] applyRangeUpdates(int n, int[][] updates) {
    int[] diff = new int[n + 1];
    for (int[] u : updates) { diff[u[0]] += u[2]; diff[u[1]+1] -= u[2]; }
    int[] result = new int[n];
    int running = 0;
    for (int i = 0; i < n; i++) { running += diff[i]; result[i] = running; }
    return result;
}

PART 5: PATTERN RECOGNITION
===================================================================

Interval Problem Clue                    → Approach
-----------------------------------------|------------------------------
"Merge overlapping intervals"            → Sort by start + linear scan
"Insert new interval"                    → 3-phase: before/overlap/after
"Min removals for non-overlap"          → Sort by END + greedy keep
"Min rooms / resources needed"          → Sort by start + min-heap of ends
"Can attend all meetings?"              → Sort by start, check consecutive
"Point coverage / arrow bursting"       → Sort by END + greedy arrow
"Max overlap at any point"             → Sweep line / difference array

Greedy Problem Clue                      → Approach
-----------------------------------------|------------------------------
"Can reach the end?"                    → Track max reachable index
"Min jumps to end"                      → Level BFS on jump ranges
"Complete a circuit?"                   → Track total surplus + restart
"Assign more to higher rated"          → Two-pass left+right
"Distribute to minimize max"            → Binary search on answer
"Build largest number"                  → Custom string comparator sort
