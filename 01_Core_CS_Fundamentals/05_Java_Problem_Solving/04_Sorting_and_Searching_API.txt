===================================================================
PART 1: SORTING PRIMITIVES — Arrays.sort
===================================================================

// Sorts in-place, ASCENDING
int[]    arr = {3,1,4,1,5};   Arrays.sort(arr);           // [1,1,3,4,5]
long[]   arr = {…};            Arrays.sort(arr);
double[] arr = {…};            Arrays.sort(arr);
char[]   arr = "hello".toCharArray(); Arrays.sort(arr);   // [e,h,l,l,o]

// Sort a RANGE [fromIndex, toIndex)  — toIndex is EXCLUSIVE
Arrays.sort(arr, 1, 4);       // only elements [1..3]

// IMPORTANT: You CANNOT pass a Comparator to sort primitive arrays.
// To sort descending, convert to Integer[] first:
Integer[] boxed = Arrays.stream(arr).boxed().toArray(Integer[]::new);
Arrays.sort(boxed, Comparator.reverseOrder());

===================================================================
PART 2: SORTING OBJECTS — Arrays.sort with Comparator
===================================================================

String[] strs = {"banana","apple","cherry"};

// Natural order (Comparable)
Arrays.sort(strs);

// Reverse order
Arrays.sort(strs, Comparator.reverseOrder());

// By length
Arrays.sort(strs, Comparator.comparingInt(String::length));

// By length then alphabetical
Arrays.sort(strs, Comparator.comparingInt(String::length)
                             .thenComparing(Comparator.naturalOrder()));

// Custom object sort
Arrays.sort(people, Comparator.comparingInt(Person::getAge));
Arrays.sort(people, (a, b) -> a.age - b.age);    // ascending (be careful with overflow)
Arrays.sort(people, (a, b) -> Integer.compare(a.age, b.age)); // SAFE

// 2D array sort — by first element
int[][] intervals = {{2,6},{1,3},{8,10}};
Arrays.sort(intervals, (a, b) -> a[0] - b[0]);   // by start time
Arrays.sort(intervals, (a, b) -> Integer.compare(a[0], b[0])); // safe

// Sort by multiple fields
Arrays.sort(intervals, (a, b) -> {
    if (a[0] != b[0]) return Integer.compare(a[0], b[0]);
    return Integer.compare(a[1], b[1]);
});

===================================================================
PART 3: SORTING LISTS — Collections.sort / List.sort
===================================================================

List<Integer> list = new ArrayList<>(Arrays.asList(3,1,4,1,5));

Collections.sort(list);                           // natural ascending
Collections.sort(list, Comparator.reverseOrder()); // descending
list.sort(Comparator.naturalOrder());             // same, fluent
list.sort((a, b) -> b - a);                      // descending inline

List<String> words = …;
words.sort(Comparator.comparingInt(String::length)
           .thenComparing(Comparator.naturalOrder()));

// Sort and return new list without modifying original (Java 8+)
List<Integer> sorted = list.stream().sorted().collect(Collectors.toList());
List<Integer> sortedDesc = list.stream()
    .sorted(Comparator.reverseOrder())
    .collect(Collectors.toList());

===================================================================
PART 4: COMPARATOR — FULL API
===================================================================

// Static factory methods
Comparator.naturalOrder()                   // Comparable's ordering
Comparator.reverseOrder()                   // reverse of natural
Comparator.comparing(keyExtractor)          // by key (Comparable)
Comparator.comparingInt(keyExtractor)       // by int key  ← prefer for int
Comparator.comparingLong(keyExtractor)      // by long key
Comparator.comparingDouble(keyExtractor)    // by double key
Comparator.nullsFirst(cmp)                 // nulls before non-nulls
Comparator.nullsLast(cmp)                  // nulls after non-nulls

// Instance methods
cmp.reversed()                              // flip order
cmp.thenComparing(nextCmp)                 // tie-break
cmp.thenComparingInt(keyFn)               // tie-break by int

// Building complex comparators
Comparator<String> cmp = Comparator
    .comparingInt(String::length)          // 1st: by length
    .thenComparing(Comparator.naturalOrder()) // 2nd: alphabetical
    .reversed();                            // overall reverse

// Lambda vs method reference
(a, b) -> a.length() - b.length()            // lambda (watch overflow)
Comparator.comparingInt(String::length)       // method ref (preferred)

// Overflow-safe integer comparator
Integer.compare(a, b)       // returns negative/0/positive safely
// a - b can overflow if a is very negative and b is very positive!

===================================================================
PART 5: BINARY SEARCH — Arrays.binarySearch
===================================================================

// PREREQUISITE: array must be sorted in ascending order
int[] arr = {1, 3, 5, 7, 9};

int idx = Arrays.binarySearch(arr, 5);    // returns 2
int idx = Arrays.binarySearch(arr, 4);    // returns negative (-(insertionPoint)-1)

// Decode negative result
if (idx < 0) {
    int insertionPoint = -(idx + 1);      // where it WOULD be inserted
}

// Binary search in a range
Arrays.binarySearch(arr, fromIndex, toIndex, key);

// For List
Collections.binarySearch(sortedList, key);   // same return convention

// Objects (must implement Comparable or provide Comparator)
Arrays.binarySearch(strs, "cherry");
Arrays.binarySearch(strs, "cherry", Comparator.naturalOrder());

PART 6: CUSTOM BINARY SEARCH (write your own — very common)
===================================================================

// Find leftmost position where arr[mid] >= target
int lowerBound(int[] arr, int target) {
    int lo = 0, hi = arr.length;         // hi = arr.length (not len-1)
    while (lo < hi) {
        int mid = lo + (hi - lo) / 2;    // avoid overflow
        if (arr[mid] < target) lo = mid + 1;
        else                   hi = mid;
    }
    return lo;    // insertion point; arr[lo] >= target if lo < arr.length
}

// Find rightmost position where arr[mid] <= target
int upperBound(int[] arr, int target) {
    int lo = 0, hi = arr.length;
    while (lo < hi) {
        int mid = lo + (hi - lo) / 2;
        if (arr[mid] <= target) lo = mid + 1;
        else                    hi = mid;
    }
    return lo;    // first position where arr[lo] > target
}

// Count of target in sorted array
int count(int[] arr, int target) {
    return upperBound(arr, target) - lowerBound(arr, target);
}

// Binary search on ANSWER (search space is on result, not array index)
// Template: "find minimum X such that condition(X) is true"
int lo = minPossible, hi = maxPossible;
while (lo < hi) {
    int mid = lo + (hi - lo) / 2;
    if (condition(mid)) hi = mid;     // mid might be answer, look left
    else                lo = mid + 1; // mid too small, look right
}
return lo;

// Example: find minimum days to ship packages
// condition(days) = can we ship all packages in 'days' days?
int lo = maxWeight, hi = totalWeight;
while (lo < hi) {
    int mid = lo + (hi - lo) / 2;
    if (canShip(weights, mid, D)) hi = mid;
    else lo = mid + 1;
}

PART 7: SORTING ALGORITHM COMPLEXITY (know this for interviews)
===================================================================

Algorithm       | Best    | Average | Worst   | Space  | Stable
----------------|---------|---------|---------|--------|-------
Bubble Sort     | O(n)    | O(n²)   | O(n²)   | O(1)   | Yes
Selection Sort  | O(n²)   | O(n²)   | O(n²)   | O(1)   | No
Insertion Sort  | O(n)    | O(n²)   | O(n²)   | O(1)   | Yes
Merge Sort      | O(nlogn)| O(nlogn)| O(nlogn)| O(n)   | Yes ✓
Quick Sort      | O(nlogn)| O(nlogn)| O(n²)   | O(logn)| No
Heap Sort       | O(nlogn)| O(nlogn)| O(nlogn)| O(1)   | No
Counting Sort   | O(n+k)  | O(n+k)  | O(n+k)  | O(k)   | Yes
Radix Sort      | O(nk)   | O(nk)   | O(nk)   | O(n+k) | Yes
Tim Sort        | O(n)    | O(nlogn)| O(nlogn)| O(n)   | Yes ✓

Java's Arrays.sort for primitives → Dual-Pivot Quicksort (not stable)
Java's Arrays.sort for objects    → TimSort (stable, guaranteed O(n log n))
Java's Collections.sort           → TimSort (stable)

Why does stability matter?
→ Stable sort preserves relative order of equal elements.
→ Important when sorting by multiple keys sequentially.

PART 8: COMMON SORTING PROBLEMS & PATTERNS
===================================================================

--- CUSTOM SORT: largest number from array ---
// E.g., [3,30,34,5,9] → "9534330"
String[] strs = Arrays.stream(nums)
    .mapToObj(String::valueOf)
    .toArray(String[]::new);
Arrays.sort(strs, (a, b) -> (b + a).compareTo(a + b));
return String.join("", strs);

--- SORT BY FREQUENCY ---
Map<Integer, Integer> freq = new HashMap<>();
for (int n : nums) freq.merge(n, 1, Integer::sum);
Integer[] arr = Arrays.stream(nums).boxed().toArray(Integer[]::new);
Arrays.sort(arr, (a, b) -> {
    int cmp = freq.get(b) - freq.get(a);  // higher freq first
    return cmp != 0 ? cmp : a - b;         // then lower value
});

--- SORT INTERVALS ---
// Sort by start, merge overlapping
Arrays.sort(intervals, (a, b) -> a[0] - b[0]);
List<int[]> merged = new ArrayList<>();
for (int[] iv : intervals) {
    if (merged.isEmpty() || merged.get(merged.size()-1)[1] < iv[0])
        merged.add(iv);
    else
        merged.get(merged.size()-1)[1] = Math.max(merged.get(merged.size()-1)[1], iv[1]);
}

--- K CLOSEST TO ORIGIN ---
// Sort by distance squared (avoid sqrt)
Arrays.sort(points, (a, b) -> (a[0]*a[0]+a[1]*a[1]) - (b[0]*b[0]+b[1]*b[1]));
return Arrays.copyOfRange(points, 0, k);

--- MEETING ROOMS ---
// Can a person attend all meetings?
Arrays.sort(intervals, (a, b) -> a[0] - b[0]);
for (int i = 1; i < intervals.length; i++)
    if (intervals[i][0] < intervals[i-1][1]) return false;
return true;

===================================================================
PART 9: KEY TRAPS TO AVOID
===================================================================

1. OVERFLOW in Comparator:
   BAD:  (a, b) -> a - b        // if a = MIN_VALUE, b = 1 → overflows!
   GOOD: (a, b) -> Integer.compare(a, b)
   GOOD: Comparator.comparingInt(x -> x)

2. Binary search on UNSORTED array returns wrong result silently.

3. Arrays.binarySearch returns NEGATIVE for missing, NOT -1.

4. Arrays.asList returns FIXED-SIZE list (no add/remove):
   List<Integer> fixed = Arrays.asList(1,2,3);
   fixed.add(4);  // throws UnsupportedOperationException!
   SOLUTION: new ArrayList<>(Arrays.asList(1,2,3))

5. Sorting in-place mutates the original array.
   Clone first if you need the original: arr.clone() or Arrays.copyOf(arr, arr.length)

6. String[] cannot use Comparator.reverseOrder() without explicit cast.
   Use: Arrays.sort(strs, Comparator.reverseOrder())  ← works for Object[]
