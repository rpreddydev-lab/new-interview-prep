2. STRING  (java.lang.String)  — IMMUTABLE
===================================================================
String s = "hello";

// Info
s.length()                  → int
s.isEmpty()                 → true if len == 0
s.isBlank()                 → true if only whitespace (Java 11)
s.charAt(i)                 → char
s.indexOf('l')              → first index, -1 if not found
s.indexOf('l', 3)           → search from index 3
s.lastIndexOf('l')          → last index
s.contains("ell")           → boolean

// Extraction
s.substring(2)              → "llo"
s.substring(1, 4)           → "ell"  (end exclusive)
s.toCharArray()             → char[]
s.split(",")                → String[]
s.split(",", 2)             → limit to 2 parts
s.chars()                   → IntStream of char values

// Comparison
s.equals("hello")           → boolean  (NEVER use == for content)
s.equalsIgnoreCase("HELLO") → boolean
s.compareTo("world")        → negative / 0 / positive
s.startsWith("he")          → boolean
s.endsWith("lo")            → boolean
s.matches("h.*o")           → boolean (full regex match)

// Transformation
s.toLowerCase()             → "hello"
s.toUpperCase()             → "HELLO"
s.trim()                    → removes leading/trailing whitespace
s.strip()                   → Unicode-aware trim (Java 11)
s.replace('l', 'x')        → "hexxo"
s.replace("ll", "xx")      → "hexxo"
s.replaceAll("[aeiou]", "*") → regex replace
s.replaceFirst("l", "x")   → "hexlo"

// Building / joining
String.valueOf(42)          → "42"
String.valueOf('c')         → "c"
String.join(", ", "a","b","c")  → "a, b, c"
String.join("-", list)      → joins a List<String>
"abc".repeat(3)             → "abcabcabc"  (Java 11)

// Numeric parsing
Integer.parseInt("42")      → 42
Long.parseLong("42")        → 42L
Double.parseDouble("3.14")  → 3.14
Integer.toBinaryString(5)   → "101"
Integer.toHexString(255)    → "ff"
Integer.bitCount(7)         → 3

11. COMPARABLE vs COMPARATOR
===================================================================
// Comparable: class implements its own ordering
class Person implements Comparable<Person> {
    public int compareTo(Person other) {
        return this.age - other.age;  // ascending by age
        // WARNING: use Integer.compare(this.age, other.age) to avoid overflow
    }
}

// Comparator: external ordering
Comparator<Person> byName = Comparator.comparing(p -> p.name);
Comparator<Person> byAgeThenName = Comparator
    .comparingInt(Person::getAge)
    .thenComparing(Person::getName);
Comparator<Person> reversed = byName.reversed();

// Sort tricks
list.sort(Comparator.naturalOrder());
list.sort(Comparator.reverseOrder());
list.sort((a, b) -> b.val - a.val);          // descending int
list.sort((a, b) -> Integer.compare(b, a));  // safe descending
