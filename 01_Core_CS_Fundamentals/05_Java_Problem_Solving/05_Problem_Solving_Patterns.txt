PATTERN 1: FREQUENCY MAP
===================================================================
When to use: Count occurrences, find majority element, anagrams,
             top-K frequent, valid anagram check.

// Basic frequency map
Map<Character, Integer> freq = new HashMap<>();
for (char c : s.toCharArray())
    freq.merge(c, 1, Integer::sum);
// OR
    freq.put(c, freq.getOrDefault(c, 0) + 1);

// Frequency using int[26] (lowercase letters only — FASTER)
int[] count = new int[26];
for (char c : s.toCharArray()) count[c - 'a']++;

// Frequency using int[128] (all ASCII)
int[] count = new int[128];
for (char c : s.toCharArray()) count[c]++;

// Top-K frequent elements
Map<Integer, Integer> freq = new HashMap<>();
for (int n : nums) freq.merge(n, 1, Integer::sum);
PriorityQueue<Integer> minH = new PriorityQueue<>(Comparator.comparingInt(freq::get));
for (int key : freq.keySet()) {
    minH.offer(key);
    if (minH.size() > k) minH.poll();
}

// Bucket sort for frequency (O(n))
int[] buckets = new int[n + 1];    // bucket[i] = list of elements with freq i
List<Integer>[] bucket = new List[nums.length + 1];
for (int i = 0; i < bucket.length; i++) bucket[i] = new ArrayList<>();
for (Map.Entry<Integer,Integer> e : freq.entrySet())
    bucket[e.getValue()].add(e.getKey());

PATTERN 2: TWO POINTERS
===================================================================
When to use: Sorted array problems, pair sum, palindrome, remove
             duplicates, container with most water, partition.

// Template: opposite ends converge
int left = 0, right = n - 1;
while (left < right) {
    if (satisfiesCondition(arr[left], arr[right])) {
        // process
        left++; right--;
    } else if (needBigger) {
        left++;
    } else {
        right--;
    }
}

// Two Sum in sorted array
while (left < right) {
    int sum = arr[left] + arr[right];
    if (sum == target) return new int[]{left, right};
    else if (sum < target) left++;
    else right--;
}

// Remove duplicates from sorted array (in-place)
int slow = 0;
for (int fast = 1; fast < n; fast++)
    if (arr[fast] != arr[slow]) arr[++slow] = arr[fast];
return slow + 1;

// Partition (Dutch National Flag — 3-way partition)
int lo = 0, mid = 0, hi = n - 1;
while (mid <= hi) {
    if (arr[mid] == 0)      swap(arr, lo++, mid++);
    else if (arr[mid] == 1) mid++;
    else                    swap(arr, mid, hi--);
}

// Fast & Slow pointers (cycle detection)
int slow = 0, fast = 0;
while (fast < n && fast + 1 < n) {
    slow = arr[slow];
    fast = arr[arr[fast]];
    if (slow == fast) return true;  // cycle
}

PATTERN 3: SLIDING WINDOW
===================================================================
When to use: Substring/subarray of fixed or variable length,
             max/min in window, longest without repeating chars.

// FIXED window size k
int windowSum = 0;
for (int i = 0; i < k; i++) windowSum += arr[i];   // seed
int maxSum = windowSum;
for (int i = k; i < n; i++) {
    windowSum += arr[i] - arr[i - k];               // slide
    maxSum = Math.max(maxSum, windowSum);
}

// VARIABLE window — longest valid subarray/substring
int left = 0, maxLen = 0;
Map<Character, Integer> window = new HashMap<>();   // or int[26]
for (int right = 0; right < n; right++) {
    // EXPAND: add s.charAt(right) to window
    window.merge(s.charAt(right), 1, Integer::sum);

    // SHRINK: while window is invalid
    while (windowIsInvalid(window)) {
        char lc = s.charAt(left++);
        window.merge(lc, -1, Integer::sum);
        if (window.get(lc) == 0) window.remove(lc);
    }
    maxLen = Math.max(maxLen, right - left + 1);
}

// Longest substring without repeating characters
int left = 0, maxLen = 0;
int[] last = new int[128];  // last seen index + 1
Arrays.fill(last, 0);
for (int right = 0; right < n; right++) {
    left = Math.max(left, last[s.charAt(right)]);   // jump left past repeat
    maxLen = Math.max(maxLen, right - left + 1);
    last[s.charAt(right)] = right + 1;
}

// Minimum window substring (LeetCode 76)
int[] need = new int[128], have = new int[128];
for (char c : t.toCharArray()) need[c]++;
int formed = 0, required = (int) t.chars().distinct().count(); // unique chars
int left = 0, minLen = Integer.MAX_VALUE, start = 0;
for (int right = 0; right < s.length(); right++) {
    char c = s.charAt(right);
    have[c]++;
    if (need[c] > 0 && have[c] == need[c]) formed++;
    while (formed == required) {
        if (right - left + 1 < minLen) { minLen = right - left + 1; start = left; }
        char lc = s.charAt(left++);
        if (need[lc] > 0 && have[lc] == need[lc]) formed--;
        have[lc]--;
    }
}

PATTERN 5: MONOTONIC STACK
===================================================================
When to use: Next greater element, previous smaller, histogram,
             daily temperatures, largest rectangle, trapping water.

// Next Greater Element (to the right)
int[] result = new int[n];
Arrays.fill(result, -1);
Deque<Integer> stack = new ArrayDeque<>();  // stores INDICES
for (int i = 0; i < n; i++) {
    while (!stack.isEmpty() && arr[stack.peek()] < arr[i]) {
        result[stack.pop()] = arr[i];
    }
    stack.push(i);
}

// Previous Smaller Element (to the left)
int[] prevSmall = new int[n];
Deque<Integer> stack = new ArrayDeque<>();
for (int i = 0; i < n; i++) {
    while (!stack.isEmpty() && arr[stack.peek()] >= arr[i]) stack.pop();
    prevSmall[i] = stack.isEmpty() ? -1 : stack.peek();
    stack.push(i);
}

// Largest Rectangle in Histogram
int maxArea = 0;
Deque<Integer> stack = new ArrayDeque<>();
int[] h = Arrays.copyOf(heights, heights.length + 1); // sentinel
for (int i = 0; i < h.length; i++) {
    while (!stack.isEmpty() && h[stack.peek()] > h[i]) {
        int height = h[stack.pop()];
        int width = stack.isEmpty() ? i : i - stack.peek() - 1;
        maxArea = Math.max(maxArea, height * width);
    }
    stack.push(i);
}

// Monotonic DECREASING deque (sliding window maximum)
Deque<Integer> deque = new ArrayDeque<>();  // stores indices
int[] result = new int[n - k + 1];
for (int i = 0; i < n; i++) {
    while (!deque.isEmpty() && deque.peekFirst() < i - k + 1)
        deque.pollFirst();                          // remove out-of-window
    while (!deque.isEmpty() && arr[deque.peekLast()] < arr[i])
        deque.pollLast();                           // remove smaller elements
    deque.offerLast(i);
    if (i >= k - 1) result[i - k + 1] = arr[deque.peekFirst()];
}

PATTERN 6: BFS (Breadth-First Search) with Queue
===================================================================
When to use: Shortest path in unweighted graph, level-order traversal,
             multi-source BFS, grid problems (0/1 matrix, walls & gates).

// Graph BFS template
void bfs(int start, List<List<Integer>> adj) {
    boolean[] visited = new boolean[n];
    Queue<Integer> queue = new ArrayDeque<>();
    queue.offer(start);
    visited[start] = true;
    while (!queue.isEmpty()) {
        int node = queue.poll();
        // process node
        for (int neighbor : adj.get(node)) {
            if (!visited[neighbor]) {
                visited[neighbor] = true;
                queue.offer(neighbor);
            }
        }
    }
}

// Level-order BFS (record level)
int level = 0;
while (!queue.isEmpty()) {
    int size = queue.size();      // nodes at current level
    for (int i = 0; i < size; i++) {
        int node = queue.poll();
        // process node at this level
        for (int next : adj.get(node)) {
            if (!visited[next]) { visited[next] = true; queue.offer(next); }
        }
    }
    level++;
}

// Grid BFS (4-directional)
int[][] dirs = {{0,1},{0,-1},{1,0},{-1,0}};
boolean[][] visited = new boolean[m][n];
Queue<int[]> queue = new ArrayDeque<>();
queue.offer(new int[]{startRow, startCol});
visited[startRow][startCol] = true;
while (!queue.isEmpty()) {
    int[] cur = queue.poll();
    for (int[] d : dirs) {
        int nr = cur[0] + d[0], nc = cur[1] + d[1];
        if (nr >= 0 && nr < m && nc >= 0 && nc < n && !visited[nr][nc]) {
            visited[nr][nc] = true;
            queue.offer(new int[]{nr, nc});
        }
    }
}

PATTERN 7: DFS (Depth-First Search) with Stack or Recursion
===================================================================
When to use: Connected components, cycle detection, topological sort,
             permutations, combinations, backtracking.

// Iterative DFS with explicit stack
void dfs(int start, List<List<Integer>> adj) {
    boolean[] visited = new boolean[n];
    Deque<Integer> stack = new ArrayDeque<>();
    stack.push(start);
    while (!stack.isEmpty()) {
        int node = stack.pop();
        if (visited[node]) continue;
        visited[node] = true;
        // process node
        for (int neighbor : adj.get(node)) stack.push(neighbor);
    }
}

// Recursive DFS
void dfs(int node, boolean[] visited, List<List<Integer>> adj) {
    visited[node] = true;
    // process node
    for (int next : adj.get(node)) {
        if (!visited[next]) dfs(next, visited, adj);
    }
}

// Grid DFS (flood fill / number of islands)
void dfs(char[][] grid, int r, int c) {
    if (r < 0 || r >= m || c < 0 || c >= n || grid[r][c] != '1') return;
    grid[r][c] = '0';    // mark visited
    dfs(grid, r+1, c); dfs(grid, r-1, c);
    dfs(grid, r, c+1); dfs(grid, r, c-1);
}

PATTERN 8: BACKTRACKING
===================================================================
When to use: Subsets, permutations, combinations, N-Queens,
             word search, Sudoku.

// Subsets
void backtrack(int[] nums, int start, List<Integer> curr, List<List<Integer>> res) {
    res.add(new ArrayList<>(curr));       // add current state
    for (int i = start; i < nums.length; i++) {
        curr.add(nums[i]);                // choose
        backtrack(nums, i + 1, curr, res);// explore
        curr.remove(curr.size() - 1);    // un-choose
    }
}

// Permutations
void backtrack(int[] nums, boolean[] used, List<Integer> curr, List<List<Integer>> res) {
    if (curr.size() == nums.length) { res.add(new ArrayList<>(curr)); return; }
    for (int i = 0; i < nums.length; i++) {
        if (used[i]) continue;
        used[i] = true;
        curr.add(nums[i]);
        backtrack(nums, used, curr, res);
        curr.remove(curr.size() - 1);
        used[i] = false;
    }
}

// Combinations of k from n
void backtrack(int start, int k, List<Integer> curr, List<List<Integer>> res) {
    if (curr.size() == k) { res.add(new ArrayList<>(curr)); return; }
    for (int i = start; i <= n - (k - curr.size()) + 1; i++) { // pruning
        curr.add(i);
        backtrack(i + 1, k, curr, res);
        curr.remove(curr.size() - 1);
    }
}

PATTERN 10: UNION-FIND (Disjoint Set Union)
===================================================================
When to use: Connected components, cycle detection, Kruskal's MST,
             number of islands (alternative).

class UnionFind {
    int[] parent, rank;

    UnionFind(int n) {
        parent = new int[n];
        rank = new int[n];
        for (int i = 0; i < n; i++) parent[i] = i;
    }

    int find(int x) {
        if (parent[x] != x) parent[x] = find(parent[x]); // path compression
        return parent[x];
    }

    boolean union(int x, int y) {
        int px = find(x), py = find(y);
        if (px == py) return false;          // already same component
        if (rank[px] < rank[py]) { int t = px; px = py; py = t; }
        parent[py] = px;
        if (rank[px] == rank[py]) rank[px]++;
        return true;
    }

    boolean connected(int x, int y) { return find(x) == find(y); }
}

PATTERN 11: TRIE (Prefix Tree)
===================================================================
When to use: Word search, autocomplete, longest common prefix,
             word dictionary, IP routing.

class TrieNode {
    TrieNode[] children = new TrieNode[26];
    boolean isEnd;
}

class Trie {
    TrieNode root = new TrieNode();

    void insert(String word) {
        TrieNode node = root;
        for (char c : word.toCharArray()) {
            int i = c - 'a';
            if (node.children[i] == null) node.children[i] = new TrieNode();
            node = node.children[i];
        }
        node.isEnd = true;
    }

    boolean search(String word) {
        TrieNode node = root;
        for (char c : word.toCharArray()) {
            int i = c - 'a';
            if (node.children[i] == null) return false;
            node = node.children[i];
        }
        return node.isEnd;
    }

    boolean startsWith(String prefix) {
        TrieNode node = root;
        for (char c : prefix.toCharArray()) {
            int i = c - 'a';
            if (node.children[i] == null) return false;
            node = node.children[i];
        }
        return true;
    }
}
