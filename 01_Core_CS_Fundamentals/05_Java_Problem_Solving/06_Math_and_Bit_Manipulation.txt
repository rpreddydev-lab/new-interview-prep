// GCD (Euclidean algorithm)
int gcd(int a, int b) { return b == 0 ? a : gcd(b, a % b); }
// LCM
long lcm(long a, long b) { return a / gcd(a, b) * b; }  // divide first to avoid overflow

Math.floorMod(-7, 2) →  1   (always non-negative if b > 0)

===================================================================
PART 2: BIT MANIPULATION FUNDAMENTALS
===================================================================

Operator   | Meaning          | Example (5=0101, 3=0011)
-----------|------------------|------------------------
a & b      | AND              | 5 & 3 = 0001 = 1
a | b      | OR               | 5 | 3 = 0111 = 7
a ^ b      | XOR              | 5 ^ 3 = 0110 = 6
~a         | NOT (complement) | ~5 = -6 (two's complement)
a << n     | left shift       | 5 << 1 = 1010 = 10
a >> n     | signed right     | -8 >> 1 = -4
a >>> n    | unsigned right   | -8 >>> 1 = 2147483644

// Key identities
a ^ a = 0          // anything XOR itself = 0
a ^ 0 = a          // anything XOR 0 = itself
a & 0 = 0
a | 0 = a
a & a = a
a & ~a = 0
a | ~a = -1 (all 1s)

===================================================================
PART 3: BIT TRICKS — MEMORIZE THESE
===================================================================

--- Check if bit i is set ---
(n >> i) & 1 == 1    // true if bit i is 1
(n & (1 << i)) != 0  // equivalent

--- Set bit i ---
n | (1 << i)

--- Clear bit i ---
n & ~(1 << i)

--- Toggle bit i ---
n ^ (1 << i)

--- Get lowest set bit ---
n & (-n)             // isolates the rightmost 1-bit
n & (~n + 1)         // same thing

--- Clear lowest set bit ---
n & (n - 1)          // removes the rightmost 1-bit

--- Check power of 2 ---
n > 0 && (n & (n - 1)) == 0

--- Count set bits (popcount) ---
Integer.bitCount(n)  // built-in, O(1)
// Manual:
int count = 0;
while (n != 0) { count += (n & 1); n >>= 1; }  // O(bits)
// Brian Kernighan:
int count = 0;
while (n != 0) { n &= (n - 1); count++; }       // O(set bits)

--- Check if even / odd ---
(n & 1) == 0    // even
(n & 1) == 1    // odd

--- Multiply / divide by power of 2 ---
n << k   // n * 2^k
n >> k   // n / 2^k  (signed, rounds toward negative infinity for negative)

--- Swap two integers without temp ---
a ^= b; b ^= a; a ^= b;   // only safe if a != b in memory

--- Absolute value (branchless, tricky) ---
// Don't do this in interviews — use Math.abs(n)

--- Sign of a number ---
(n >> 31)   // -1 if negative, 0 if non-negative (signed 32-bit)

--- Turn off rightmost consecutive 1-bits ---
n & (n + 1)   // e.g. 0111 → 0000

--- Isolate trailing 0-bits ---
~n & (n - 1)

PART 4: COMMON BIT MANIPULATION PROBLEMS
===================================================================

--- Single Number (all others appear twice) ---
// XOR of same numbers cancels out
int result = 0;
for (int n : nums) result ^= n;
return result;

--- Missing Number (0..n, one missing) ---
int xor = n;
for (int i = 0; i < nums.length; i++) xor ^= i ^ nums[i];
return xor;
// OR: sum formula
return n*(n+1)/2 - Arrays.stream(nums).sum();

--- Number of 1 Bits (Hamming weight) ---
return Integer.bitCount(n);

--- Reverse Bits ---
int result = 0;
for (int i = 0; i < 32; i++) {
    result = (result << 1) | (n & 1);
    n >>= 1;
}
return result;

--- Power of Two ---
return n > 0 && (n & (n - 1)) == 0;

--- Power of Four ---
return n > 0 && (n & (n-1)) == 0 && (n & 0xAAAAAAAA) == 0;
// 0xAAAAAAAA = ...10101010  (marks positions that are powers of 2 but NOT powers of 4)

--- Counting Bits (0..n all at once) ---
int[] dp = new int[n + 1];
for (int i = 1; i <= n; i++)
    dp[i] = dp[i >> 1] + (i & 1);  // dp[i] = dp[i/2] + (i is odd ? 1 : 0)
return dp;

--- Subsets (bitmask) ---
// Generate all 2^n subsets of an array
List<List<Integer>> subsets = new ArrayList<>();
for (int mask = 0; mask < (1 << n); mask++) {
    List<Integer> sub = new ArrayList<>();
    for (int i = 0; i < n; i++)
        if ((mask >> i & 1) == 1) sub.add(nums[i]);
    subsets.add(sub);
}

--- Two numbers appearing once (rest appear twice) ---
int xor = 0;
for (int n : nums) xor ^= n;          // xor = a ^ b
int bit = xor & (-xor);               // lowest differing bit
int a = 0, b = 0;
for (int n : nums) {
    if ((n & bit) == 0) a ^= n;
    else b ^= n;
}

--- Sum of two integers without + operator ---
int getSum(int a, int b) {
    while (b != 0) {
        int carry = (a & b) << 1;
        a = a ^ b;
        b = carry;
    }
    return a;
}

PART 5: MODULAR ARITHMETIC
===================================================================

// Always apply mod at each step to prevent overflow
(a + b) % MOD
(a * b) % MOD            // but: (a * b) might overflow before % if a,b are int
((long)a * b) % MOD      // cast to long FIRST

long MOD = 1_000_000_007L;

// Modular exponentiation (fast power)
long power(long base, long exp, long mod) {
    long result = 1;
    base %= mod;
    while (exp > 0) {
        if ((exp & 1) == 1) result = result * base % mod;
        base = base * base % mod;
        exp >>= 1;
    }
    return result;
}

// Is prime?
boolean isPrime(int n) {
    if (n < 2) return false;
    for (int i = 2; i * i <= n; i++)   // only check up to sqrt(n)
        if (n % i == 0) return false;
    return true;
}

// Sieve of Eratosthenes — all primes up to n
boolean[] sieve(int n) {
    boolean[] notPrime = new boolean[n + 1];
    notPrime[0] = notPrime[1] = true;
    for (int i = 2; i * i <= n; i++)
        if (!notPrime[i])
            for (int j = i * i; j <= n; j += i)
                notPrime[j] = true;
    return notPrime;   // notPrime[i] == false means i is prime
}

// All prime factors of n
List<Integer> primeFactors(int n) {
    List<Integer> factors = new ArrayList<>();
    for (int i = 2; i * i <= n; i++) {
        while (n % i == 0) { factors.add(i); n /= i; }
    }
    if (n > 1) factors.add(n);
    return factors;
}

// All divisors of n
List<Integer> divisors(int n) {
    List<Integer> divs = new ArrayList<>();
    for (int i = 1; i * i <= n; i++) {
        if (n % i == 0) {
            divs.add(i);
            if (i != n / i) divs.add(n / i);
        }
    }
    return divs;
}
