TEMPLATE 4: SLIDING WINDOW (VARIABLE SIZE)
===================================================================
/*
 * Problem: ___________________________________________
 * Approach: Sliding Window
 * Time:    O(n)    Space: O(  )
 */
class Solution {
    public int solve(String s) {
        int n = s.length();
        int left = 0, result = 0;
        Map<Character, Integer> window = new HashMap<>();

        for (int right = 0; right < n; right++) {
            char rc = s.charAt(right);
            // [ADD rc to window]
            window.merge(rc, 1, Integer::sum);

            // [SHRINK while invalid]
            while (/* window is invalid */) {
                char lc = s.charAt(left++);
                window.merge(lc, -1, Integer::sum);
                if (window.get(lc) == 0) window.remove(lc);
            }

            // [UPDATE result]
            result = Math.max(result, right - left + 1);
        }

        return result;
    }
}

TEMPLATE 6: BFS — SHORTEST PATH IN GRID
===================================================================
/*
 * Problem: ___________________________________________
 * Approach: BFS
 * Time:    O(m * n)    Space: O(m * n)
 */
class Solution {
    int[][] dirs = {{0,1},{0,-1},{1,0},{-1,0}};

    public int solve(int[][] grid) {
        int m = grid.length, n = grid[0].length;
        boolean[][] visited = new boolean[m][n];
        Queue<int[]> queue = new ArrayDeque<>();
        int steps = 0;

        // [ADD starting cell(s)]
        queue.offer(new int[]{0, 0});
        visited[0][0] = true;

        while (!queue.isEmpty()) {
            int size = queue.size();
            for (int i = 0; i < size; i++) {
                int[] cur = queue.poll();
                int r = cur[0], c = cur[1];

                // [CHECK if reached goal]
                if (r == m-1 && c == n-1) return steps;

                for (int[] d : dirs) {
                    int nr = r + d[0], nc = c + d[1];
                    if (nr >= 0 && nr < m && nc >= 0 && nc < n
                            && !visited[nr][nc] /* && valid cell */) {
                        visited[nr][nc] = true;
                        queue.offer(new int[]{nr, nc});
                    }
                }
            }
            steps++;
        }
        return -1;  // not reachable
    }
}

/*
 * Problem: ___________________________________________
 * Approach: Backtracking
 * Time:    O(2^n)    Space: O(n)
 */
class Solution {
    List<List<Integer>> result = new ArrayList<>();

    public List<List<Integer>> solve(int[] nums) {
        backtrack(nums, 0, new ArrayList<>());
        return result;
    }

    void backtrack(int[] nums, int start, List<Integer> curr) {
        result.add(new ArrayList<>(curr));   // or: check base case here

        for (int i = start; i < nums.length; i++) {
            // [SKIP DUPLICATES if needed] if (i > start && nums[i] == nums[i-1]) continue;
            curr.add(nums[i]);               // choose
            backtrack(nums, i + 1, curr);   // explore (use i for reuse, i+1 for no-reuse)
            curr.remove(curr.size() - 1);   // un-choose
        }
    }
}

TEMPLATE 8: TOP-K USING PRIORITY QUEUE
===================================================================
/*
 * Problem: ___________________________________________
 * Approach: Min-Heap of size K
 * Time:    O(n log k)    Space: O(k)
 */
class Solution {
    public int[] topKFrequent(int[] nums, int k) {
        // Step 1: build frequency map
        Map<Integer, Integer> freq = new HashMap<>();
        for (int n : nums) freq.merge(n, 1, Integer::sum);

        // Step 2: maintain min-heap of size k (keyed on frequency)
        PriorityQueue<Integer> minH = new PriorityQueue<>(
            Comparator.comparingInt(freq::get)
        );

        for (int key : freq.keySet()) {
            minH.offer(key);
            if (minH.size() > k) minH.poll();
        }

        // Step 3: extract results
        int[] result = new int[k];
        for (int i = k - 1; i >= 0; i--) result[i] = minH.poll();
        return result;
    }
}
