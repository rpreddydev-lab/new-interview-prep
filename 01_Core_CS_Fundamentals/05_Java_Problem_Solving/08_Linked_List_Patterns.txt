===================================================================
  LINKED LIST PATTERNS IN JAVA
  (Node structure, pointer tricks, all common interview patterns)
===================================================================
Key fact: Java has NO built-in singly-linked list for interviews.
LeetCode defines ListNode — you must know this structure cold.

===================================================================
PART 1: NODE DEFINITIONS (memorize these)
===================================================================

// Singly Linked List Node (LeetCode standard)
class ListNode {
    int val;
    ListNode next;
    ListNode() {}
    ListNode(int val) { this.val = val; }
    ListNode(int val, ListNode next) { this.val = val; this.next = next; }
}

// Doubly Linked List Node
class DListNode {
    int val;
    DListNode prev, next;
    DListNode(int val) { this.val = val; }
}

// Build a list from array (helper for testing)
ListNode build(int[] arr) {
    ListNode dummy = new ListNode(0);
    ListNode cur = dummy;
    for (int v : arr) { cur.next = new ListNode(v); cur = cur.next; }
    return dummy.next;
}

// Print a list (helper)
void print(ListNode head) {
    while (head != null) { System.out.print(head.val + " -> "); head = head.next; }
    System.out.println("null");
}

PART 3: TRAVERSAL & BASIC OPERATIONS
===================================================================

// Length of list
int length(ListNode head) {
    int len = 0;
    while (head != null) { len++; head = head.next; }
    return len;
}

// Get kth node (1-indexed)
ListNode getKth(ListNode head, int k) {
    while (--k > 0) head = head.next;
    return head;
}

// Get kth node from END (1-indexed)
// Use two pointers: advance fast by k, then move both
ListNode getKthFromEnd(ListNode head, int k) {
    ListNode fast = head, slow = head;
    while (k-- > 0) fast = fast.next;
    while (fast != null) { fast = fast.next; slow = slow.next; }
    return slow;
}

// Find MIDDLE node (slow/fast pointers)
// If even length, returns SECOND middle (standard)
ListNode findMiddle(ListNode head) {
    ListNode slow = head, fast = head;
    while (fast != null && fast.next != null) {
        slow = slow.next;
        fast = fast.next.next;
    }
    return slow;   // slow is at middle
}

// Find MIDDLE for merge sort (returns FIRST middle for even)
ListNode findMiddleLeft(ListNode head) {
    ListNode slow = head, fast = head.next;  // fast starts one ahead
    while (fast != null && fast.next != null) {
        slow = slow.next;
        fast = fast.next.next;
    }
    return slow;
}

===================================================================
PART 4: REVERSE A LINKED LIST
===================================================================

// Iterative — O(n) time, O(1) space
ListNode reverse(ListNode head) {
    ListNode prev = null, cur = head;
    while (cur != null) {
        ListNode next = cur.next;   // save next
        cur.next = prev;            // reverse pointer
        prev = cur;                 // advance prev
        cur = next;                 // advance cur
    }
    return prev;   // new head
}

// Recursive
ListNode reverseRecursive(ListNode head) {
    if (head == null || head.next == null) return head;
    ListNode newHead = reverseRecursive(head.next);
    head.next.next = head;
    head.next = null;
    return newHead;
}

// Reverse a sublist [left, right] (1-indexed) — LeetCode 92
ListNode reverseBetween(ListNode head, int left, int right) {
    ListNode dummy = new ListNode(0);
    dummy.next = head;
    ListNode prev = dummy;
    for (int i = 1; i < left; i++) prev = prev.next;
    ListNode cur = prev.next;
    for (int i = 0; i < right - left; i++) {
        ListNode next = cur.next;
        cur.next = next.next;
        next.next = prev.next;
        prev.next = next;
    }
    return dummy.next;
}

// Reverse in groups of k — LeetCode 25
ListNode reverseKGroup(ListNode head, int k) {
    ListNode dummy = new ListNode(0);
    dummy.next = head;
    ListNode groupPrev = dummy;
    while (true) {
        ListNode kth = getKthNode(groupPrev, k);
        if (kth == null) break;
        ListNode groupNext = kth.next;
        // reverse the group
        ListNode prev = groupNext, cur = groupPrev.next;
        while (cur != groupNext) {
            ListNode next = cur.next;
            cur.next = prev;
            prev = cur;
            cur = next;
        }
        ListNode tmp = groupPrev.next;
        groupPrev.next = kth;
        groupPrev = tmp;
    }
    return dummy.next;
}
ListNode getKthNode(ListNode node, int k) {
    while (node != null && k > 0) { node = node.next; k--; }
    return node;
}

===================================================================
PART 5: CYCLE DETECTION (Floyd's Algorithm)
===================================================================

// Does cycle exist?
boolean hasCycle(ListNode head) {
    ListNode slow = head, fast = head;
    while (fast != null && fast.next != null) {
        slow = slow.next;
        fast = fast.next.next;
        if (slow == fast) return true;   // cycle detected
    }
    return false;
}

// Find ENTRY POINT of cycle
ListNode detectCycle(ListNode head) {
    ListNode slow = head, fast = head;
    while (fast != null && fast.next != null) {
        slow = slow.next;
        fast = fast.next.next;
        if (slow == fast) {
            // Reset one pointer to head, advance both at same speed
            slow = head;
            while (slow != fast) { slow = slow.next; fast = fast.next; }
            return slow;   // entry point
        }
    }
    return null;   // no cycle
}

PART 6: MERGE OPERATIONS

// Merge two sorted lists — LeetCode 21
ListNode mergeTwoLists(ListNode l1, ListNode l2) {
    ListNode dummy = new ListNode(0);
    ListNode cur = dummy;
    while (l1 != null && l2 != null) {
        if (l1.val <= l2.val) { cur.next = l1; l1 = l1.next; }
        else                  { cur.next = l2; l2 = l2.next; }
        cur = cur.next;
    }
    cur.next = (l1 != null) ? l1 : l2;
    return dummy.next;
}

// Merge K sorted lists — LeetCode 23
ListNode mergeKLists(ListNode[] lists) {
    PriorityQueue<ListNode> pq = new PriorityQueue<>(
        Comparator.comparingInt(n -> n.val)
    );
    for (ListNode l : lists) if (l != null) pq.offer(l);
    ListNode dummy = new ListNode(0), cur = dummy;
    while (!pq.isEmpty()) {
        ListNode node = pq.poll();
        cur.next = node;
        cur = cur.next;
        if (node.next != null) pq.offer(node.next);
    }
    return dummy.next;
}

PART 7: REMOVE OPERATIONS
===================================================================

// Remove nth node from end — LeetCode 19
ListNode removeNthFromEnd(ListNode head, int n) {
    ListNode dummy = new ListNode(0);
    dummy.next = head;
    ListNode fast = dummy, slow = dummy;
    for (int i = 0; i <= n; i++) fast = fast.next;  // advance n+1 steps
    while (fast != null) { fast = fast.next; slow = slow.next; }
    slow.next = slow.next.next;   // remove
    return dummy.next;
}

// Remove all occurrences of val — LeetCode 203
ListNode removeElements(ListNode head, int val) {
    ListNode dummy = new ListNode(0);
    dummy.next = head;
    ListNode cur = dummy;
    while (cur.next != null) {
        if (cur.next.val == val) cur.next = cur.next.next;
        else cur = cur.next;
    }
    return dummy.next;
}

// Remove duplicates from sorted list (keep one) — LeetCode 83
ListNode deleteDuplicates(ListNode head) {
    ListNode cur = head;
    while (cur != null && cur.next != null) {
        if (cur.val == cur.next.val) cur.next = cur.next.next;
        else cur = cur.next;
    }
    return head;
}

// Remove ALL duplicates from sorted list — LeetCode 82
ListNode deleteDuplicatesAll(ListNode head) {
    ListNode dummy = new ListNode(0);
    dummy.next = head;
    ListNode prev = dummy;
    while (prev.next != null) {
        ListNode cur = prev.next;
        if (cur.next != null && cur.val == cur.next.val) {
            int val = cur.val;
            while (prev.next != null && prev.next.val == val)
                prev.next = prev.next.next;
        } else {
            prev = prev.next;
        }
    }
    return dummy.next;
}

PART 8: REORDER & REARRANGEMENT
===================================================================

// Reorder list: L0→L1→…→Ln → L0→Ln→L1→Ln-1→… — LeetCode 143
void reorderList(ListNode head) {
    if (head == null || head.next == null) return;
    // Step 1: find middle
    ListNode slow = head, fast = head;
    while (fast.next != null && fast.next.next != null) {
        slow = slow.next; fast = fast.next.next;
    }
    // Step 2: reverse second half
    ListNode second = reverse(slow.next);
    slow.next = null;   // cut
    // Step 3: merge two halves
    ListNode first = head;
    while (second != null) {
        ListNode tmp1 = first.next, tmp2 = second.next;
        first.next = second;
        second.next = tmp1;
        first = tmp1;
        second = tmp2;
    }
}

// Odd-Even list (odds first, then evens by position) — LeetCode 328
ListNode oddEvenList(ListNode head) {
    if (head == null) return null;
    ListNode odd = head, even = head.next, evenHead = even;
    while (even != null && even.next != null) {
        odd.next = even.next;
        odd = odd.next;
        even.next = odd.next;
        even = even.next;
    }
    odd.next = evenHead;
    return head;
}

// Partition list around x — LeetCode 86
ListNode partition(ListNode head, int x) {
    ListNode lessD = new ListNode(0), greaterD = new ListNode(0);
    ListNode less = lessD, greater = greaterD;
    while (head != null) {
        if (head.val < x) { less.next = head; less = less.next; }
        else              { greater.next = head; greater = greater.next; }
        head = head.next;
    }
    greater.next = null;   // IMPORTANT: terminate greater list
    less.next = greaterD.next;
    return lessD.next;
}

PART 9: INTERSECTION & PALINDROME
===================================================================

// Intersection of two linked lists — LeetCode 160
ListNode getIntersectionNode(ListNode headA, ListNode headB) {
    ListNode a = headA, b = headB;
    while (a != b) {
        a = (a == null) ? headB : a.next;
        b = (b == null) ? headA : b.next;
    }
    return a;   // null if no intersection
}

// Palindrome linked list — LeetCode 234
boolean isPalindrome(ListNode head) {
    // Step 1: find middle
    ListNode slow = head, fast = head;
    while (fast != null && fast.next != null) {
        slow = slow.next; fast = fast.next.next;
    }
    // Step 2: reverse second half
    ListNode rev = reverse(slow);
    // Step 3: compare
    ListNode p1 = head, p2 = rev;
    while (p2 != null) {
        if (p1.val != p2.val) return false;
        p1 = p1.next; p2 = p2.next;
    }
    return true;
}

===================================================================
PART 10: CRITICAL POINTER DISCIPLINE
===================================================================

GOLDEN RULES:
1. Always save `cur.next` BEFORE re-wiring: ListNode next = cur.next;
2. Use dummy head to avoid null-checking on head insertion/deletion.
3. With two pointers, decide if fast starts at head or head.next.
4. Always check `fast != null && fast.next != null` before fast/fast.next.
5. Terminate list ends explicitly: last.next = null (avoids cycles).
6. Draw the state on paper before coding pointer manipulations.

COMMON BUGS:
- Forgetting to set last.next = null after splitting a list → infinite loop
- Off-by-one in "advance k steps" → check carefully with small examples
- Using == to compare node VALUES (should compare node.val, not nodes)
- Not handling single-node or two-node edge cases

PATTERN RECOGNITION TABLE:
Problem Hint                              → Technique
-----------------------------------------|--------------------------------
"Find middle"                            → Slow/Fast pointers
"Detect/find cycle"                      → Floyd's algorithm
"kth from end"                           → Two pointers (gap of k)
"Merge sorted"                           → Dummy head + compare
"Reverse entire / sublist / groups"      → Prev/Cur/Next pointer dance
"Palindrome check"                       → Find middle + reverse + compare
"Intersection of two lists"              → Two pointer (swap on null)
"Reorder / interleave"                   → Split + reverse + merge
"Remove nth from end"                    → Two pointers (gap of n+1)
"Sort linked list"                       → Merge sort on list
