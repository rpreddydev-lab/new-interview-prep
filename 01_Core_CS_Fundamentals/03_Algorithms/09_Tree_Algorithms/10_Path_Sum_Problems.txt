PATH SUM I (LC 112) -- Does path exist?
----------------------------------------------------------------
  boolean hasPathSum(TreeNode root, int targetSum) {
    if (root == null) return false;
    if (root.left == null && root.right == null) return root.val == targetSum;
    return hasPathSum(root.left, targetSum - root.val)
        || hasPathSum(root.right, targetSum - root.val);
  }

----------------------------------------------------------------
PATH SUM II (LC 113) -- Find all root-to-leaf paths
----------------------------------------------------------------
  List<List<Integer>> pathSum(TreeNode root, int targetSum) {
    List<List<Integer>> res = new ArrayList<>();
    dfs(root, targetSum, new ArrayList<>(), res);
    return res;
  }
  void dfs(TreeNode node, int remain, List<Integer> path, List<List<Integer>> res) {
    if (node == null) return;
    path.add(node.val);
    if (node.left == null && node.right == null && remain == node.val)
      res.add(new ArrayList<>(path));
    dfs(node.left, remain - node.val, path, res);
    dfs(node.right, remain - node.val, path, res);
    path.remove(path.size() - 1);  // backtrack
  }

----------------------------------------------------------------
PATH SUM III (LC 437) -- Count paths summing to target
----------------------------------------------------------------
Paths don't have to start at root or end at leaf.
Use PREFIX SUM + HASHMAP (same pattern as subarray sum equals k).

  int pathSum(TreeNode root, int targetSum) {
    Map<Long, Integer> prefixCount = new HashMap<>();
    prefixCount.put(0L, 1);  // empty prefix
    return dfsCount(root, 0, targetSum, prefixCount);
  }
  int dfsCount(TreeNode node, long currSum, int target, Map<Long, Integer> map) {
    if (node == null) return 0;
    currSum += node.val;
    int count = map.getOrDefault(currSum - target, 0);  // paths ending here
    map.merge(currSum, 1, Integer::sum);
    count += dfsCount(node.left, currSum, target, map);
    count += dfsCount(node.right, currSum, target, map);
    map.merge(currSum, -1, Integer::sum);  // backtrack: remove this prefix
    return count;
  }
  // Use long to avoid int overflow when values accumulate

----------------------------------------------------------------
SUM ROOT TO LEAF NUMBERS (LC 129)
----------------------------------------------------------------
Each root-to-leaf path represents a number. Sum all numbers.

  int sumNumbers(TreeNode root) { return dfsSum(root, 0); }
  int dfsSum(TreeNode node, int curr) {
    if (node == null) return 0;
    curr = curr * 10 + node.val;
    if (node.left == null && node.right == null) return curr;  // leaf
    return dfsSum(node.left, curr) + dfsSum(node.right, curr);
  }

----------------------------------------------------------------
INTERVIEW TIPS
----------------------------------------------------------------
  1. Path Sum I/II: subtract target down the path, check at leaves
  2. Path Sum III: prefix sum HashMap -- same as subarray sum equals K
     (mention this connection to interviewer)
  3. CRITICAL for Path Sum III: map.merge(currSum, -1, ...) MUST run
     after both children (backtracking) to avoid counting paths across branches
  4. Use long for prefix sums to avoid integer overflow
  5. Leaf = both left and right are null (not just one null)
