================================================================
  FLOOD FILL AND ISLAND PROBLEMS
================================================================

----------------------------------------------------------------
CORE CONCEPT
----------------------------------------------------------------
Flood fill: visit all connected cells of the same type in a 2D grid.
Foundation for many grid problems: islands, regions, areas.

  Direction arrays:
    4-directional: int[] dr = {0,0,1,-1}; int[] dc = {1,-1,0,0};
    8-directional: also add diagonals (dr = {-1,-1,-1,0,0,1,1,1})

----------------------------------------------------------------
FLOOD FILL (LC 733)
----------------------------------------------------------------
Change color of connected region starting at (sr, sc).

  int[][] floodFill(int[][] image, int sr, int sc, int color) {
    int orig = image[sr][sc];
    if (orig == color) return image;  // avoid infinite loop
    dfs(image, sr, sc, orig, color);
    return image;
  }
  void dfs(int[][] img, int r, int c, int orig, int newColor) {
    if (r<0||r>=img.length||c<0||c>=img[0].length||img[r][c]!=orig) return;
    img[r][c] = newColor;   // in-place modification
    dfs(img, r+1, c, orig, newColor); dfs(img, r-1, c, orig, newColor);
    dfs(img, r, c+1, orig, newColor); dfs(img, r, c-1, orig, newColor);
  }

----------------------------------------------------------------
NUMBER OF ISLANDS (LC 200)
----------------------------------------------------------------
Count connected components of '1's.

  int numIslands(char[][] grid) {
    int count = 0;
    for (int r = 0; r < grid.length; r++)
      for (int c = 0; c < grid[0].length; c++)
        if (grid[r][c] == '1') { dfsIsland(grid, r, c); count++; }
    return count;
  }
  void dfsIsland(char[][] grid, int r, int c) {
    if (r<0||r>=grid.length||c<0||c>=grid[0].length||grid[r][c]!='1') return;
    grid[r][c] = '0';   // mark visited by modifying grid (restore if needed)
    dfsIsland(grid, r+1, c); dfsIsland(grid, r-1, c);
    dfsIsland(grid, r, c+1); dfsIsland(grid, r, c-1);
  }

  If grid modification not allowed: use boolean[][] visited.

----------------------------------------------------------------
SURROUNDED REGIONS (LC 130)
----------------------------------------------------------------
Capture all 'O' regions not connected to border.
  1. DFS from every border 'O', mark as 'S' (safe).
  2. Flip remaining 'O' to 'X' (captured).
  3. Flip 'S' back to 'O'.

  void solve(char[][] board) {
    int m = board.length, n = board[0].length;
    for (int r = 0; r < m; r++) {  // left and right borders
      dfsCapture(board, r, 0); dfsCapture(board, r, n-1);
    }
    for (int c = 0; c < n; c++) {  // top and bottom borders
      dfsCapture(board, 0, c); dfsCapture(board, m-1, c);
    }
    for (int r = 0; r < m; r++) for (int c = 0; c < n; c++) {
      if (board[r][c] == 'O') board[r][c] = 'X';
      else if (board[r][c] == 'S') board[r][c] = 'O';
    }
  }

----------------------------------------------------------------
INTERVIEW TIPS
----------------------------------------------------------------
  1. In-place marking (change grid value) avoids visited[][] overhead
  2. Surrounded Regions: reverse thinking -- DFS from BORDERS first
  3. For large grids: BFS avoids recursion stack overflow
  4. Pacific/Atlantic Water Flow (LC 417): reverse DFS from both oceans;
     answer = intersection of two sets
  5. Max Area of Island: DFS returning count
     (count = 1 + dfs(all 4 neighbors))
