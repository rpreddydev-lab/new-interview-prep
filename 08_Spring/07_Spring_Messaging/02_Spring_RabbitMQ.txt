  1. RABBITMQ CORE CONCEPTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Producer     → sends messages to an Exchange (never directly to a queue)
Exchange     → routes messages to queues based on routing rules
Binding      → link between exchange and queue (with optional routing key)
Queue        → durable buffer where messages wait for consumers
Consumer     → reads from a queue; ACKs or NACKs each message
Virtual Host → isolated namespace (multi-tenancy)

Exchange Types:
  Direct  → routes by exact routing key match
  Topic   → routes by routing key pattern (* = one word, # = zero or more)
  Fanout  → broadcasts to ALL bound queues (ignores routing key)
  Headers → routes by message headers (not routing key)

  4. EXCHANGE, QUEUE, AND BINDING DECLARATIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

@Configuration
public class RabbitConfig {

    public static final String EXCHANGE     = "orders.exchange";
    public static final String QUEUE        = "orders.queue";
    public static final String ROUTING_KEY  = "order.created";
    public static final String DLX          = "orders.dlx";
    public static final String DLQ          = "orders.dlq";

    // ── Dead Letter Setup ──────────────────────────────────
    @Bean
    public DirectExchange dlx() {
        return new DirectExchange(DLX, true, false);
    }

    @Bean
    public Queue dlq() {
        return QueueBuilder.durable(DLQ).build();
    }

    @Bean
    public Binding dlqBinding() {
        return BindingBuilder.bind(dlq()).to(dlx()).with(ROUTING_KEY);
    }

    // ── Main Exchange, Queue, Binding ─────────────────────
    @Bean
    public TopicExchange ordersExchange() {
        return new TopicExchange(EXCHANGE, true, false);
    }

    @Bean
    public Queue ordersQueue() {
        return QueueBuilder.durable(QUEUE)
            .withArgument("x-dead-letter-exchange", DLX)     // failed msgs → DLX
            .withArgument("x-dead-letter-routing-key", ROUTING_KEY)
            .withArgument("x-message-ttl", 300000)           // 5 min TTL
            .build();
    }

    @Bean
    public Binding ordersBinding() {
        return BindingBuilder
            .bind(ordersQueue())
            .to(ordersExchange())
            .with("order.*");   // matches order.created, order.updated, etc.
    }

    // ── JSON Message Converter ────────────────────────────
    @Bean
    public MessageConverter jsonMessageConverter() {
        return new Jackson2JsonMessageConverter();
    }

    @Bean
    public RabbitTemplate rabbitTemplate(ConnectionFactory cf) {
        RabbitTemplate template = new RabbitTemplate(cf);
        template.setMessageConverter(jsonMessageConverter());
        return template;
    }
}

  5. PRODUCER — RabbitTemplate
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

@Service
public class OrderProducer {

    private final RabbitTemplate rabbitTemplate;

    public OrderProducer(RabbitTemplate rabbitTemplate) {
        this.rabbitTemplate = rabbitTemplate;
    }

    // Send to exchange with routing key
    public void sendOrderCreated(OrderEvent event) {
        rabbitTemplate.convertAndSend(
            RabbitConfig.EXCHANGE,
            "order.created",        // routing key
            event
        );
    }

    // Send with message properties (TTL, priority, headers)
    public void sendWithProperties(OrderEvent event) {
        rabbitTemplate.convertAndSend(
            RabbitConfig.EXCHANGE,
            "order.created",
            event,
            message -> {
                MessageProperties props = message.getMessageProperties();
                props.setExpiration("60000");           // 60s TTL
                props.setPriority(5);
                props.setHeader("source", "checkout");
                return message;
            }
        );
    }

    // Direct to queue (bypass exchange)
    public void sendDirectToQueue(OrderEvent event) {
        rabbitTemplate.convertAndSend(RabbitConfig.QUEUE, event);
    }

    // Request-Reply (RPC style — blocks for response)
    public OrderResponse requestReply(OrderRequest request) {
        return (OrderResponse) rabbitTemplate.convertSendAndReceive(
            RabbitConfig.EXCHANGE,
            "order.process",
            request
        );
    }
}

  6. CONSUMER — @RabbitListener
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

@Component
public class OrderConsumer {

    // Basic listener
    @RabbitListener(queues = RabbitConfig.QUEUE)
    public void handleOrder(OrderEvent event) {
        log.info("Received: {}", event);
    }

    // Access full AMQP Message + metadata
    @RabbitListener(queues = RabbitConfig.QUEUE)
    public void handleWithMessage(Message message) {
        byte[] body = message.getBody();
        MessageProperties props = message.getMessageProperties();
        log.info("RoutingKey: {}", props.getReceivedRoutingKey());
    }

    // Manual acknowledgment
    @RabbitListener(queues = RabbitConfig.QUEUE,
                    ackMode = "MANUAL")
    public void handleManual(OrderEvent event, Channel channel,
                              @Header(AmqpHeaders.DELIVERY_TAG) long deliveryTag) {
        try {
            processOrder(event);
            channel.basicAck(deliveryTag, false);     // ACK: remove from queue
        } catch (RecoverableException ex) {
            channel.basicNack(deliveryTag, false, true);  // NACK + requeue
        } catch (Exception ex) {
            channel.basicNack(deliveryTag, false, false); // NACK + DLQ
        }
    }

    // Inline queue/exchange declaration (no @Bean required)
    @RabbitListener(bindings = @QueueBinding(
        value = @Queue(value = "notifications.queue", durable = "true"),
        exchange = @Exchange(value = "notifications", type = ExchangeTypes.TOPIC),
        key = "notification.#"
    ))
    public void handleNotification(NotificationEvent event) {
        notificationService.send(event);
    }
}

  7. PUBLISHER CONFIRMS AND RETURNS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Publisher confirms: broker ACKs that the message was received.
Publisher returns: broker notifies when a message couldn't be routed to any queue.

# application.yml
spring:
  rabbitmq:
    publisher-confirm-type: correlated   # NONE, SIMPLE, CORRELATED
    publisher-returns: true

@Bean
public RabbitTemplate reliableRabbitTemplate(ConnectionFactory cf) {
    RabbitTemplate template = new RabbitTemplate(cf);
    template.setMessageConverter(new Jackson2JsonMessageConverter());
    template.setMandatory(true);  // enables returns

    // Return callback: fires if message not routed to any queue
    template.setReturnsCallback(returned -> {
        log.error("Message returned: exchange={} routingKey={} reply={}",
            returned.getExchange(),
            returned.getRoutingKey(),
            returned.getReplyText());
    });

    // Confirm callback: fires after broker ACKs or NACKs
    template.setConfirmCallback((correlationData, ack, cause) -> {
        if (!ack) {
            log.error("Publisher NACK: cause={}", cause);
        }
    });

    return template;
}

  8. DEAD LETTER QUEUE (DLQ)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Messages are moved to DLQ when:
  - Explicitly NACK'd with requeue=false
  - Message TTL expires
  - Queue length limit exceeded

The DLX (Dead Letter Exchange) routes dead messages to the DLQ.
Configured per-queue via x-dead-letter-exchange argument (see section 4).

@RabbitListener(queues = RabbitConfig.DLQ)
public void handleDeadLetter(Message message) {
    MessageProperties props = message.getMessageProperties();
    // x-death header contains rejection reason, count, source queue
    List<Map<String, ?>> xDeath =
        (List<Map<String, ?>>) props.getHeaders().get("x-death");
    log.error("Dead letter received from queue: {}",
        xDeath.get(0).get("queue"));
    // alert, store for manual reprocessing
}

Q: What is the difference between Direct, Topic, and Fanout exchanges?
A: Direct routes messages to queues where the binding key exactly matches
   the routing key. Topic uses pattern matching (* = one word, # = zero+
   words). Fanout ignores routing keys and sends to ALL bound queues.

Q: How does RabbitMQ guarantee message delivery?
A: Durable queues + persistent messages survive broker restart. Publisher
   confirms (ACK from broker) guarantee the broker received the message.
   Consumer ACKs guarantee the message was processed. Combining all three
   gives strong delivery guarantees.

Q: What happens to a message when the consumer throws an exception?
A: With auto ACK (default), the message is ACKed regardless — it's lost.
   With manual ACK, you control it: basicNack(tag, false, true) requeues,
   basicNack(tag, false, false) sends to DLQ (if configured).

Q: How do you implement delayed message processing in RabbitMQ?
A: Use the RabbitMQ Delayed Message Exchange plugin (x-delayed-message).
   Set the x-delay header to milliseconds. Alternatively, use a TTL queue
   with DLX as a delay mechanism: messages expire after TTL and land in
   the DLX, which routes to the actual processing queue.

Q: When would you choose RabbitMQ over Kafka?
A: RabbitMQ is better for: complex routing (topic/headers exchanges),
   request-reply patterns, tasks that need per-message TTL or priority,
   small-to-medium throughput. Kafka is better for: high-throughput
   streaming, replay, ordered event log, event sourcing.
