@Component
public class MaintenanceTasks {

    private final DataRepository dataRepository;
    private final ReportService reportService;
    private final CacheManager cacheManager;

    // Every 30 seconds (measured from start of previous run)
    @Scheduled(fixedRate = 30_000)
    public void pollExternalApi() {
        log.debug("Polling at {}", LocalDateTime.now());
        externalClient.fetchLatestRates();
    }

    // 10 seconds after previous run finishes
    @Scheduled(fixedDelay = 10_000)
    public void processQueue() {
        List<Task> tasks = taskQueue.poll(50);
        tasks.forEach(this::process);
    }

    // Delay first run by 60s (wait for app fully started)
    @Scheduled(fixedRate = 60_000, initialDelay = 60_000)
    public void healthPing() {
        healthClient.ping();
    }

    // Every night at midnight
    @Scheduled(cron = "0 0 0 * * *")
    public void midnightCleanup() {
        dataRepository.deleteExpired();
    }

    // Every Monday at 9am
    @Scheduled(cron = "0 0 9 * * MON")
    public void weeklyReport() {
        reportService.generateWeeklyReport();
    }

    // Externalize schedule to configuration (recommended for production)
    @Scheduled(cron = "${app.tasks.cleanup.cron:0 0 2 * * *}")
    public void configuredCleanup() {
        dataRepository.deleteOldRecords(30);
    }

    // Scheduling with fixedRateString from property
    @Scheduled(fixedRateString = "${app.tasks.poll.interval:30000}")
    public void configurablePoller() {
        // interval configurable via application.yml
    }
}

  2. CRON EXPRESSION SYNTAX
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Spring uses 6-field cron (not 5-field UNIX cron):

 ┌─────────── second (0–59)
 │ ┌───────── minute (0–59)
 │ │ ┌─────── hour (0–23)
 │ │ │ ┌───── day of month (1–31)
 │ │ │ │ ┌─── month (1–12 or JAN–DEC)
 │ │ │ │ │ ┌─ day of week (0–7 or SUN–SAT; 0 and 7 = Sunday)
 │ │ │ │ │ │
 * * * * * *

Special characters:
  *    — any value
  ,    — list: "0,15,30,45" = every 15 mins (0, 15, 30, 45)
  -    — range: "9-17" = 9am to 5pm
  /    — step: "*/15" = every 15, "0/30" = at 0 and 30
  ?    — no specific value (day-of-month or day-of-week only)
  L    — last: "L" in day-of-month = last day of month
  W    — nearest weekday: "15W" = nearest weekday to 15th
  #    — nth weekday: "3#2" = second Wednesday

  3. COMMON CRON EXPRESSIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Expression              | Description
------------------------|-----------------------------------------------
0 * * * * *             | Every minute (at second 0)
0 */5 * * * *           | Every 5 minutes
0 0 * * * *             | Every hour
0 0 0 * * *             | Every midnight
0 0 12 * * *            | Every day at noon
0 0 9-17 * * MON-FRI    | Every hour 9am-5pm, weekdays only
0 0 0 * * MON           | Every Monday at midnight
0 0 0 1 * *             | First day of each month at midnight
0 0 0 L * *             | Last day of each month at midnight
0 30 8 ? * MON-FRI      | Weekdays at 8:30am
0 0 0 ? * 6L            | Last Saturday of each month
0 0 10 15 * ?           | 15th of every month at 10am
0 0 2 * * *             | Every night at 2am

  4. TIMEZONE IN CRON EXPRESSIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

@Scheduled cron expressions use the JVM's default timezone unless specified.
Always explicitly set timezone for predictable production behavior:

@Scheduled(cron = "0 0 9 * * MON-FRI", zone = "America/New_York")
public void marketOpenTask() {
    // runs at 9am Eastern, not server timezone
}

@Scheduled(cron = "0 0 2 * * *", zone = "UTC")
public void utcMidnightTask() {
    // always at 2am UTC regardless of server timezone
}

  5. PROGRAMMATIC SCHEDULING — TaskScheduler
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

When you need to schedule tasks dynamically at runtime:

@Service
public class DynamicSchedulerService {

    private final TaskScheduler taskScheduler;
    private final Map<String, ScheduledFuture<?>> scheduledTasks = new ConcurrentHashMap<>();

    // Schedule a task dynamically
    public void schedule(String taskId, Runnable task, String cronExpression) {
        cancel(taskId);  // cancel existing if any
        ScheduledFuture<?> future = taskScheduler.schedule(
            task,
            new CronTrigger(cronExpression));
        scheduledTasks.put(taskId, future);
    }

    // Cancel a task
    public void cancel(String taskId) {
        ScheduledFuture<?> future = scheduledTasks.remove(taskId);
        if (future != null) {
            future.cancel(false);  // false = don't interrupt if running
        }
    }

    // Reschedule (e.g., admin changes interval)
    public void reschedule(String taskId, Runnable task, String newCron) {
        schedule(taskId, task, newCron);
    }
}

  6. INTERVIEW Q&A
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What is the 6-field Spring cron format and how does it differ from UNIX cron?
A: Spring uses a 6-field cron: [second] [minute] [hour] [day-of-month] [month] [day-of-week].
   UNIX cron uses 5 fields (no seconds): [minute] [hour] [day-of-month] [month] [day-of-week].
   Spring adds the seconds field as the first position, so "0 0 2 * * *" means
   "at 2:00:00am every day." In Spring, you can use 0 for second or */30 for every 30 seconds.

Q: How do you run a scheduled task only on weekdays?
A: Use day-of-week field: "0 0 9 * * MON-FRI" runs at 9am Monday through Friday.
   Alternatively use "0 0 9 ? * 2-6" (numeric day-of-week, where 1=Monday in Quartz style).

Q: How do you externalize cron expressions from code?
A: Use ${property.name} in the cron attribute with an optional default:
   @Scheduled(cron = "${app.cleanup.cron:0 0 2 * * *}")
   This allows overriding the schedule via application.yml or environment variables
   without redeploying the application. DevOps teams can adjust schedules via config only.

Q: What happens if a scheduled task's execution time exceeds its fixedRate interval?
A: With the default single-threaded scheduler (pool size 1), tasks are queued —
   the next execution waits for the current one to finish. There is no overlap.
   If you configure a multi-threaded TaskScheduler (pool size > 1), multiple
   executions can run concurrently. Use fixedDelay to always wait for completion
   before scheduling the next run.

Q: How do you dynamically change a scheduled task's interval at runtime?
A: Use TaskScheduler and ScheduledFuture programmatically. Cancel the existing
   ScheduledFuture, then schedule a new one with the updated cron or fixed rate.
   Store ScheduledFutures in a map keyed by task name. Expose an admin endpoint
   that calls the rescheduling service when operators want to change intervals.
