  1. FIELD-LEVEL CUSTOM CONSTRAINT
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Two parts:
  1. Constraint annotation (defines the contract)
  2. ConstraintValidator (implements the logic)

// 1. Annotation
@Documented
@Constraint(validatedBy = PhoneNumberValidator.class)
@Target({ElementType.FIELD, ElementType.PARAMETER})
@Retention(RetentionPolicy.RUNTIME)
public @interface ValidPhoneNumber {
    String message() default "{com.example.ValidPhoneNumber.message}";
    Class<?>[] groups() default {};
    Class<? extends Payload>[] payload() default {};
    // custom attribute
    boolean allowInternational() default true;
}

// 2. Validator implementation
public class PhoneNumberValidator implements ConstraintValidator<ValidPhoneNumber, String> {

    private boolean allowInternational;

    @Override
    public void initialize(ValidPhoneNumber annotation) {
        this.allowInternational = annotation.allowInternational();
    }

    @Override
    public boolean isValid(String value, ConstraintValidatorContext context) {
        if (value == null) return true;   // let @NotNull handle null

        String regex = allowInternational
            ? "^\\+?[1-9]\\d{7,14}$"
            : "^[0-9]{10}$";

        boolean valid = value.matches(regex);

        if (!valid) {
            // Custom violation message (overrides annotation default)
            context.disableDefaultConstraintViolation();
            context.buildConstraintViolationWithTemplate(
                "Invalid phone format. Expected: " + (allowInternational ? "+1234567890" : "1234567890"))
                .addConstraintViolation();
        }
        return valid;
    }
}

// Usage:
public class UserRequest {
    @ValidPhoneNumber(allowInternational = false)
    private String phone;
}

  2. CLASS-LEVEL CONSTRAINT (CROSS-FIELD VALIDATION)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

@Documented
@Constraint(validatedBy = DateRangeValidator.class)
@Target(ElementType.TYPE)          // class-level
@Retention(RetentionPolicy.RUNTIME)
public @interface ValidDateRange {
    String message() default "End date must be after start date";
    Class<?>[] groups() default {};
    Class<? extends Payload>[] payload() default {};
    String startField() default "startDate";
    String endField() default "endDate";
}

public class DateRangeValidator implements ConstraintValidator<ValidDateRange, Object> {

    private String startField;
    private String endField;

    @Override
    public void initialize(ValidDateRange annotation) {
        this.startField = annotation.startField();
        this.endField = annotation.endField();
    }

    @Override
    public boolean isValid(Object object, ConstraintValidatorContext context) {
        try {
            BeanWrapper wrapper = new BeanWrapperImpl(object);
            LocalDate start = (LocalDate) wrapper.getPropertyValue(startField);
            LocalDate end   = (LocalDate) wrapper.getPropertyValue(endField);

            if (start == null || end == null) return true;  // let @NotNull handle

            boolean valid = end.isAfter(start);

            if (!valid) {
                // Attach error to the end date field, not the class
                context.disableDefaultConstraintViolation();
                context.buildConstraintViolationWithTemplate(context.getDefaultConstraintMessageTemplate())
                    .addPropertyNode(endField)
                    .addConstraintViolation();
            }
            return valid;
        } catch (Exception ex) {
            return false;
        }
    }
}

// Usage on class:
@ValidDateRange(startField = "startDate", endField = "endDate",
    message = "End date must be after start date")
public class ReportRequest {
    @NotNull private LocalDate startDate;
    @NotNull private LocalDate endDate;
}

Hibernate Validator auto-detects Spring beans in ConstraintValidators
when Spring MVC / Boot is used — no extra config needed.

@Documented
@Constraint(validatedBy = UniqueEmailValidator.class)
@Target(ElementType.FIELD)
@Retention(RetentionPolicy.RUNTIME)
public @interface UniqueEmail {
    String message() default "Email is already registered";
    Class<?>[] groups() default {};
    Class<? extends Payload>[] payload() default {};
}

@Component   // Spring manages this bean — can inject any Spring bean
public class UniqueEmailValidator implements ConstraintValidator<UniqueEmail, String> {

    private final UserRepository userRepository;

    // Spring auto-injects because this is a @Component
    public UniqueEmailValidator(UserRepository userRepository) {
        this.userRepository = userRepository;
    }

    @Override
    public boolean isValid(String email, ConstraintValidatorContext context) {
        if (email == null) return true;
        return !userRepository.existsByEmail(email);
    }
}

// Usage on DTO:
public class RegisterRequest {
    @NotBlank
    @Email
    @UniqueEmail
    private String email;
}

Combine multiple existing constraints into one annotation:

@NotBlank
@Size(min = 8, max = 50)
@Pattern(regexp = "^(?=.*[a-z])(?=.*[A-Z])(?=.*\\d)(?=.*[@$!%*?&]).+$",
         message = "Password must contain uppercase, lowercase, digit, and special character")
@Documented
@Constraint(validatedBy = {})  // no custom validator — composed constraints do the work
@Target({ElementType.FIELD, ElementType.PARAMETER})
@Retention(RetentionPolicy.RUNTIME)
public @interface StrongPassword {
    String message() default "Password does not meet complexity requirements";
    Class<?>[] groups() default {};
    Class<? extends Payload>[] payload() default {};
}

// Usage:
public class PasswordChangeRequest {
    @StrongPassword
    private String newPassword;
}

  5. TESTING CUSTOM VALIDATORS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

@SpringBootTest
class UniqueEmailValidatorTest {

    @Autowired
    private Validator validator;

    @MockBean
    private UserRepository userRepository;

    @Test
    void shouldFailWhenEmailAlreadyExists() {
        given(userRepository.existsByEmail("taken@example.com")).willReturn(true);

        RegisterRequest request = new RegisterRequest("taken@example.com");
        Set<ConstraintViolation<RegisterRequest>> violations = validator.validate(request);

        assertThat(violations)
            .anyMatch(v -> v.getPropertyPath().toString().equals("email")
                && v.getMessage().equals("Email is already registered"));
    }

    @Test
    void shouldPassWhenEmailIsUnique() {
        given(userRepository.existsByEmail("new@example.com")).willReturn(false);

        RegisterRequest request = new RegisterRequest("new@example.com");
        Set<ConstraintViolation<RegisterRequest>> violations = validator.validate(request);

        assertThat(violations).isEmpty();
    }
}

Q: How do you create a custom constraint in Spring?
A: Create a custom annotation meta-annotated with @Constraint(validatedBy = MyValidator.class),
   and implement ConstraintValidator<MyAnnotation, TargetType> with isValid() logic.
   Declare the validator as a @Component to enable Spring dependency injection.
   Apply the annotation to fields, parameters, or classes.

Q: How does a class-level constraint report errors on a specific field?
A: By default, class-level violations are attached to the class itself, showing
   no field in the error response. To attach to a specific field, call
   context.disableDefaultConstraintViolation() then
   context.buildConstraintViolationWithTemplate(msg).addPropertyNode("fieldName")
   .addConstraintViolation(). This appears as a field error in BindingResult.

Q: Can ConstraintValidator inject Spring beans?
A: Yes, if the validator is declared as a @Component (or @Bean in config) and
   Spring Boot's Hibernate Validator configuration is active (default with
   spring-boot-starter-validation). Spring auto-wires it via the ValidatorFactory
   configured by Spring Boot. This allows validators to call repositories or services.

Q: What is a composed constraint?
A: A composed constraint is a custom annotation that is itself annotated with
   one or more existing constraint annotations. It acts as a shorthand for
   combining multiple constraints. The annotation has @Constraint(validatedBy = {})
   (no custom validator needed). All composed constraints are evaluated together.

Q: What is the difference between cross-field validation with a class-level
   constraint vs using @AssertTrue on a method?
A: @AssertTrue on a boolean getter method (e.g., isEndAfterStart()) validates
   the method's return value. It shows up with the getter method's name, not
   a field name — less intuitive for API consumers. Class-level constraints
   explicitly control which field the violation is attached to via addPropertyNode(),
   producing cleaner field-level error responses.
