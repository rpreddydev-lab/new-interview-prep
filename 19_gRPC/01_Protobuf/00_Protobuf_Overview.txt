PROTOCOL BUFFERS - COMPREHENSIVE OVERVIEW
================================================================

----------------------------------------------------------------
1. WHAT IS PROTOBUF?
----------------------------------------------------------------

Protocol Buffers (protobuf) is a binary serialization format and
Interface Definition Language (IDL) created by Google. It is the
default serialization mechanism for gRPC.

Two roles of protobuf:
  1. IDL: Defines the structure of data (like a schema)
  2. Serialization format: Encodes data to compact binary bytes

History:
  - Google created protobuf internally in early 2000s
  - Used to store/serialize structured data in Google's systems
  - Used as the basis for Google's internal RPC system (Stubby)
  - Open-sourced in 2008
  - Proto3 released in 2016 (simplified, added JSON mapping)
  - Proto editions released in 2023 (unifying proto2/proto3)

Key Files:
  - .proto file    → Schema definition (you write this)
  - protoc          → Compiler that processes .proto files
  - Generated code  → Language-specific classes (you don't write this)

----------------------------------------------------------------
2. THE PROTOBUF WORKFLOW
----------------------------------------------------------------

Step 1: Write a .proto file
  syntax = "proto3";
  package com.example.order.v1;
  
  option java_package = "com.example.order.v1";
  option java_multiple_files = true;
  
  message Order {
    string order_id    = 1;
    string customer_id = 2;
    int64  amount_cents = 3;
    OrderStatus status = 4;
  }
  
  enum OrderStatus {
    ORDER_STATUS_UNSPECIFIED = 0;
    ORDER_STATUS_PENDING     = 1;
    ORDER_STATUS_CONFIRMED   = 2;
  }

Step 2: Compile with protoc
  protoc \
    --java_out=./src/main/java \
    --grpc-java_out=./src/main/java \
    --proto_path=./src/main/proto \
    order.proto

Step 3: Use generated code
  Order order = Order.newBuilder()
      .setOrderId("ORD-001")
      .setCustomerId("CUST-123")
      .setAmountCents(4999L)
      .setStatus(OrderStatus.ORDER_STATUS_PENDING)
      .build();
  
  // Serialize to bytes
  byte[] bytes = order.toByteArray();
  
  // Deserialize from bytes
  Order decoded = Order.parseFrom(bytes);
  System.out.println(decoded.getOrderId()); // "ORD-001"

3. FULL PROTO FILE ANATOMY
----------------------------------------------------------------

// syntax must be the first non-comment line
syntax = "proto3";

// package: prevents name collisions, maps to namespace/package
package com.example.ecommerce.v1;

// file-level options
option java_package         = "com.example.ecommerce.v1";
option java_outer_classname = "EcommerceProto";
option java_multiple_files  = true;

// import well-known types
import "google/protobuf/timestamp.proto";
import "google/protobuf/empty.proto";

// ── Enums ────────────────────────────────────────────────────
enum ProductCategory {
  PRODUCT_CATEGORY_UNSPECIFIED = 0;
  PRODUCT_CATEGORY_ELECTRONICS = 1;
  PRODUCT_CATEGORY_CLOTHING    = 2;
  PRODUCT_CATEGORY_BOOKS       = 3;
}

// ── Messages ─────────────────────────────────────────────────
message Money {
  int64  amount_cents = 1;
  string currency_code = 2;  // ISO 4217: "USD", "EUR"
}

message Product {
  string          product_id   = 1;
  string          name         = 2;
  string          description  = 3;
  Money           price        = 4;   // nested message
  ProductCategory category     = 5;   // enum
  repeated string tags         = 6;   // repeated (list)
  google.protobuf.Timestamp created_at = 7;  // well-known type
  
  // nested message within Product
  message Dimensions {
    float length_cm = 1;
    float width_cm  = 2;
    float height_cm = 3;
  }
  Dimensions dimensions = 8;
  
  // map: key must be scalar (not float/double/bytes)
  map<string, string> attributes = 9;
  
  // reserved field numbers and names
  reserved 10, 11;
  reserved "legacy_sku", "old_category";
}

// ── Service ──────────────────────────────────────────────────
service ProductService {
  rpc GetProduct (GetProductRequest) returns (Product);
  rpc ListProducts (ListProductsRequest) returns (stream Product);
  rpc CreateProduct (CreateProductRequest) returns (Product);
  rpc DeleteProduct (DeleteProductRequest) returns (google.protobuf.Empty);
}

message GetProductRequest {
  string product_id = 1;
}

message ListProductsRequest {
  ProductCategory category  = 1;
  int32           page_size = 2;
  string          page_token = 3;
}

message CreateProductRequest {
  Product product = 1;
}

message DeleteProductRequest {
  string product_id = 1;
}

4. DATA TYPES REFERENCE
----------------------------------------------------------------

SCALAR TYPES:
  Proto Type  | Java Type  | Notes
  ------------|------------|--------------------------------------
  double      | double     | 64-bit IEEE 754
  float       | float      | 32-bit IEEE 754
  int32       | int        | Use sint32 for negative numbers
  int64       | long       | Use sint64 for negative numbers
  uint32      | int        | Unsigned; Java treats as unsigned
  uint64      | long       | Unsigned; Java treats as unsigned
  sint32      | int        | ZigZag encoded, efficient negatives
  sint64      | long       | ZigZag encoded, efficient negatives
  fixed32     | int        | Always 4 bytes; use if value > 2^28
  fixed64     | long       | Always 8 bytes; use if value > 2^56
  sfixed32    | int        | Signed, always 4 bytes
  sfixed64    | long       | Signed, always 8 bytes
  bool        | boolean    | Encoded as 0 or 1
  string      | String     | UTF-8 text
  bytes       | ByteString | Arbitrary binary data

DEFAULT VALUES IN PROTO3:
  Type           | Default
  ---------------|--------
  int32/64/uint  | 0
  float/double   | 0.0
  bool           | false
  string         | ""
  bytes          | empty byte array
  enum           | first defined value (must be 0)
  message        | null (in Java: default instance)
  repeated       | empty list
  map            | empty map

IMPORTANT: In proto3, you cannot distinguish between "field not set"
and "field set to default value" for scalar types without using
the optional keyword.

5. PROTO3 OPTIONAL KEYWORD
----------------------------------------------------------------

Problem:
  message User {
    int32 age = 1;  // Is age=0 because user didn't set it, or because user is 0?
  }

Solution — proto3 optional (added 2020):
  message User {
    optional int32 age = 1;  // now has presence tracking
  }

In Java, this generates:
  user.hasAge()   // true if age was explicitly set
  user.getAge()   // returns 0 if not set

Use wrapper types as alternative:
  import "google/protobuf/wrappers.proto";
  
  message User {
    google.protobuf.Int32Value age = 1;  // can be null
  }
  
  // Java usage:
  if (user.hasAge()) {
    int age = user.getAge().getValue();
  }

6. ONEOF - DISCRIMINATED UNION
----------------------------------------------------------------

message PaymentMethod {
  string customer_id = 1;
  
  oneof payment {
    CreditCard  credit_card  = 2;
    BankAccount bank_account = 3;
    Crypto      crypto       = 4;
  }
}

message CreditCard {
  string card_number = 1;
  string expiry      = 2;
  string cvv         = 3;
}

Java generated code:
  PaymentMethod.PaymentCase getPaymentCase()
  // Returns one of: CREDIT_CARD, BANK_ACCOUNT, CRYPTO, PAYMENT_NOT_SET
  
  switch (payment.getPaymentCase()) {
    case CREDIT_CARD:
      CreditCard cc = payment.getCreditCard();
      break;
    case BANK_ACCOUNT:
      BankAccount ba = payment.getBankAccount();
      break;
    default:
      throw new IllegalStateException("Unknown payment type");
  }

Oneof Rules:
  - Only one field can be set at a time
  - Setting field A clears field B
  - Cannot have repeated fields in oneof
  - Cannot span multiple oneofs with same field number

7. MAPS
----------------------------------------------------------------

message Config {
  map<string, string> settings     = 1;
  map<string, int32>  retry_counts = 2;
  map<int64, Feature> features     = 3;
}

Valid key types: bool, any int type, string
Invalid key types: float, double, bytes, enum, message

Map is equivalent to:
  message ConfigEntry {
    string key   = 1;
    string value = 2;
  }
  repeated ConfigEntry settings = 1;

Java generated code:
  Map<String, String> settings = config.getSettingsMap();
  config.getSettingsOrDefault("timeout", "30s");
  config.getSettingsOrThrow("required_key");
  
  Config.Builder b = Config.newBuilder();
  b.putSettings("timeout", "30s");
  b.putSettings("retries", "3");

Note: map field order is NOT guaranteed in the serialized form.

8. WELL-KNOWN TYPES
----------------------------------------------------------------

Google provides pre-defined proto files for common types:

8.1 Timestamp (google/protobuf/timestamp.proto)
  message Timestamp {
    int64 seconds = 1;  // seconds since Unix epoch
    int32 nanos   = 2;  // nanoseconds [0, 999999999]
  }
  
  Java conversion:
    Timestamp ts = Timestamps.fromMillis(System.currentTimeMillis());
    Instant instant = Instant.ofEpochSecond(ts.getSeconds(), ts.getNanos());

8.2 Duration (google/protobuf/duration.proto)
  message Duration {
    int64 seconds = 1;
    int32 nanos   = 2;
  }

8.3 FieldMask (google/protobuf/field_mask.proto)
  message FieldMask {
    repeated string paths = 1;  // ["name", "address.street"]
  }
  
  Used for partial updates (similar to PATCH in REST):
    rpc UpdateProduct(UpdateProductRequest) returns (Product);
    
    message UpdateProductRequest {
      Product   product    = 1;
      FieldMask field_mask = 2;  // which fields to update
    }

8.4 Any (google/protobuf/any.proto)
  message Any {
    string type_url = 1;  // "type.googleapis.com/com.example.Product"
    bytes  value    = 2;  // serialized protobuf bytes
  }
  
  Usage:
    Any any = Any.pack(myProduct);
    Product p = any.unpack(Product.class);

8.5 Struct (google/protobuf/struct.proto)
  Represents a dynamic JSON-like structure.
  Useful when you don't know the schema in advance.

8.6 Empty (google/protobuf/empty.proto)
  Used as a return type for RPCs that don't return data:
    rpc DeleteOrder(DeleteOrderRequest) returns (google.protobuf.Empty);

8.7 Wrapper types (google/protobuf/wrappers.proto)
  StringValue, Int32Value, Int64Value, FloatValue, DoubleValue,
  BoolValue, BytesValue — all contain a single `value` field.
  Use these for nullable scalars.

9. NAMING CONVENTIONS (GOOGLE STYLE GUIDE)
----------------------------------------------------------------

Files:        lower_snake_case.proto
Packages:     lowercase dot-separated, include version:
              com.company.product.v1

Messages:     PascalCase            → OrderItem, CustomerProfile
Fields:       snake_case            → order_id, created_at
Enums:        PascalCase            → OrderStatus
Enum values:  UPPER_SNAKE_CASE with type prefix:
              ORDER_STATUS_PENDING, ORDER_STATUS_CONFIRMED
              Always have a zero value: ORDER_STATUS_UNSPECIFIED = 0
Services:     PascalCase            → OrderService
RPC methods:  PascalCase            → CreateOrder, GetOrder, ListOrders

Common mistake — missing prefix on enum values:
  // BAD: value names clash across enums
  enum Color { RED = 0; GREEN = 1; }
  enum Fruit { RED = 0; YELLOW = 1; }  // ERROR: RED already defined
  
  // GOOD: prefixed
  enum Color { COLOR_UNSPECIFIED = 0; COLOR_RED = 1; COLOR_GREEN = 2; }
  enum Fruit { FRUIT_UNSPECIFIED = 0; FRUIT_RED = 1; FRUIT_YELLOW = 2; }

10. PROTO FILE ORGANIZATION BEST PRACTICES
----------------------------------------------------------------

1. One service per file (or closely related services)
2. Common/shared messages in a separate file (common.proto, types.proto)
3. Use package versioning: order.v1, order.v2
4. Keep request and response messages in the same file as the service
5. Import only what you need
6. Use reserved to protect deleted field numbers/names

File structure example:
  protos/
  ├── common/
  │   ├── money.proto           # Money, Currency types
  │   ├── address.proto         # Address type
  │   └── pagination.proto      # PageRequest, PageResponse
  ├── order/
  │   ├── v1/
  │   │   ├── order.proto       # Order message
  │   │   └── order_service.proto # OrderService definition
  │   └── v2/
  │       └── order_service.proto # Breaking-change v2
  └── product/
      └── v1/
          ├── product.proto
          └── product_service.proto

11. PROTOBUF JSON MAPPING
----------------------------------------------------------------

Proto3 defines a canonical JSON encoding:
  - message → JSON object
  - repeated → JSON array
  - bool → JSON bool
  - string → JSON string
  - int32/int64 → JSON number (int64 as string to avoid precision loss)
  - bytes → base64-encoded string
  - enum → string name ("ORDER_STATUS_PENDING") or number
  - Timestamp → RFC 3339 string ("2024-01-15T10:30:00Z")
  - FieldMask → string with comma-separated paths ("name,address.street")

Java JSON serialization:
  // Requires protobuf-java-util dependency
  import com.google.protobuf.util.JsonFormat;
  
  // Serialize to JSON
  String json = JsonFormat.printer()
      .includingDefaultValueFields()  // include zeros/empty strings
      .print(order);
  
  // Parse from JSON
  Order.Builder builder = Order.newBuilder();
  JsonFormat.parser().merge(jsonString, builder);
  Order order = builder.build();

When to use JSON format:
  - REST endpoints wrapping gRPC services
  - Debugging: log protobuf messages as JSON
  - gRPC-Gateway transcoding
  - Storing in human-readable config files

12. COMMON MISTAKES TO AVOID
----------------------------------------------------------------

❌ Reusing field numbers
  // BROKEN - field 2 was used for "email", now for "phone_number"
  message Contact {
    string name         = 1;
    // string email     = 2;  // DELETED - but 2 is now "occupied"
    string phone_number = 2;  // WRONG - old clients will misinterpret!
  }
  
  ✅ FIX:
  message Contact {
    string name         = 1;
    reserved 2;                // mark 2 as unavailable
    reserved "email";          // mark name as unavailable
    string phone_number = 3;   // new number
  }

❌ Using float/double for monetary values
  float price = 1;  // WRONG - floating point precision loss!
  
  ✅ FIX:
  int64 price_cents = 1;  // store as integer cents

❌ Adding required fields (proto2 anti-pattern)
  required string id = 1;  // BAD - breaks rolling deployments
  
  ✅ FIX: Never use required in proto2; use proto3

❌ Not starting enums with zero value
  enum Status {
    ACTIVE   = 1;  // BAD - proto3 default is 0, this has no 0 value
    INACTIVE = 2;
  }
  
  ✅ FIX:
  enum Status {
    STATUS_UNSPECIFIED = 0;
    STATUS_ACTIVE      = 1;
    STATUS_INACTIVE    = 2;
  }

❌ Changing field type
  int32 count = 1;  → string count = 1;  // BREAKS wire compatibility
