1. NEXT.JS CACHING OVERVIEW
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Next.js App Router has 4 caching layers that work together:

  Layer 1: Request Memoization
    - Per-request deduplication of identical fetch() calls within one render
    - Scope: single server render (not persistent across requests)

  Layer 2: Data Cache
    - Persistent server-side cache (survives across requests/deployments)
    - Configured via fetch() options: next: { revalidate, tags }
    - Stored in the file system or cache store

  Layer 3: Full Route Cache
    - Caches the rendered HTML + RSC payload of a route at build time
    - Static routes served from this cache until revalidated

  Layer 4: Router Cache
    - Client-side in-memory cache of RSC payloads
    - Avoids re-fetching pages during client-side navigation

2. REQUEST MEMOIZATION (LAYER 1)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Same fetch URL called multiple times in one render → only ONE actual request:

  // layout.tsx calls getUser()
  // page.tsx also calls getUser()
  // Result: only ONE database/API call made

  async function getUser(id: string) {
    // React memoizes this: same args → same cached result within a render
    const user = await fetch(`/api/users/${id}`);
    return user.json();
  }

  // For non-fetch requests (database, ORM), use React.cache():
  import { cache } from 'react';

  export const getUser = cache(async (id: string) => {
    return prisma.user.findUnique({ where: { id } });
  });

  // Now layout.tsx and page.tsx can both call getUser(id)
  // Only one DB query is made per request (memoized within render)

3. DATA CACHE (LAYER 2) — fetch() OPTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // FORCE CACHE (SSG behavior) — cached indefinitely until revalidated
  const data = await fetch('https://api.example.com/data', {
    cache: 'force-cache',  // default in App Router
  });

  // NO STORE (SSR behavior) — never cached, always fresh
  const data = await fetch('https://api.example.com/data', {
    cache: 'no-store',
  });

  // TIME-BASED REVALIDATION (ISR behavior) — cache for N seconds
  const data = await fetch('https://api.example.com/data', {
    next: { revalidate: 3600 }, // cache for 1 hour
  });

  // TAG-BASED REVALIDATION — cache with tags for on-demand invalidation
  const data = await fetch('https://api.example.com/posts', {
    next: { tags: ['posts'] },
  });
  // Later, invalidate with: revalidateTag('posts')

  // IMPORTANT: In Next.js 15, fetch() defaults to no-store (was force-cache in 14)
  // Always be explicit about caching intent

Control caching at the route/layout level via exports:

  // Route-level exports in page.tsx or layout.tsx:

  export const dynamic = 'auto';           // Default — inferred from data fetching
  export const dynamic = 'force-static';  // Always static, never dynamic
  export const dynamic = 'force-dynamic'; // Always dynamic, never cached
  export const dynamic = 'error';         // Error if dynamic functions used

  export const revalidate = 3600;          // ISR: revalidate every 1 hour
  export const revalidate = false;         // Cache forever (same as Infinity)
  export const revalidate = 0;             // No caching (same as force-dynamic)

  export const runtime = 'nodejs';         // Default Node.js runtime
  export const runtime = 'edge';           // Edge runtime

  export const dynamicParams = true;       // Allow paths not in generateStaticParams
  export const dynamicParams = false;      // 404 for ungenerated paths

  // How dynamic/static is determined (when dynamic = 'auto'):
  // Static if: no dynamic functions (cookies, headers, searchParams), cache: 'force-cache'
  // Dynamic if: uses cookies(), headers(), useSearchParams(), cache: 'no-store'

5. ON-DEMAND REVALIDATION
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // revalidatePath — invalidate cached page after mutation
  import { revalidatePath } from 'next/cache';

  export async function createPost(formData: FormData) {
    'use server';
    await prisma.post.create({ data: { title: formData.get('title') as string } });
    revalidatePath('/posts');            // invalidate /posts page
    revalidatePath('/posts/[id]', 'page'); // invalidate dynamic post page
    revalidatePath('/blog', 'layout');   // invalidate a layout
  }

  // revalidateTag — invalidate by cache tag
  import { revalidateTag } from 'next/cache';

  export async function updatePost(id: string, data: Partial<Post>) {
    'use server';
    await prisma.post.update({ where: { id }, data });
    revalidateTag('posts');              // invalidate all 'posts' tagged fetches
    revalidateTag(`post-${id}`);         // invalidate specific post tag
  }

  // Tag fetches when fetching:
  const posts = await fetch('/api/posts', {
    next: { tags: ['posts'] },
  });
  const post = await fetch(`/api/posts/${id}`, {
    next: { tags: ['posts', `post-${id}`] },
  });

  // Route Handler for on-demand revalidation via webhook:
  // app/api/revalidate/route.ts
  import { revalidateTag } from 'next/cache';
  import { NextRequest } from 'next/server';

  export async function POST(request: NextRequest) {
    const secret = request.headers.get('x-revalidate-secret');
    if (secret !== process.env.REVALIDATE_SECRET) {
      return Response.json({ error: 'Unauthorized' }, { status: 401 });
    }
    const { tag } = await request.json();
    revalidateTag(tag);
    return Response.json({ revalidated: true, now: Date.now() });
  }

6. ROUTER CACHE (CLIENT-SIDE, LAYER 4)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

The Router Cache is an in-memory client-side cache of RSC payloads.
It persists for the duration of the browser session.

Behavior:
  - Prefetched routes are cached (Link component auto-prefetches on hover)
  - Visited routes are cached (back/forward navigation is instant)
  - Static routes: cached for 5 minutes
  - Dynamic routes: cached for 30 seconds (Next.js 14+)

Invalidating the Router Cache:
  - router.refresh() — clears and refetches current route
  - Server Action (revalidatePath/revalidateTag) — invalidates client cache too
  - Navigating away and back after cache TTL expires

  // Force refresh of current route data:
  'use client';
  import { useRouter } from 'next/navigation';

  export function RefreshButton() {
    const router = useRouter();
    return (
      <button onClick={() => router.refresh()}>
        Refresh data
      </button>
    );
  }

7. CACHING STRATEGY DECISION MATRIX
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Data type                      | Strategy        | Implementation
  ─────────────────────────────  | ──────────────  | ───────────────────────────
  Marketing page (never changes) | Static (SSG)    | force-cache or revalidate=false
  Blog posts (update weekly)     | ISR             | revalidate: 86400 (24h)
  Product pages (restock anytime)| ISR + on-demand | revalidate: 3600 + revalidateTag
  User dashboard (personalized)  | No cache (SSR)  | cache: 'no-store' or cookies()
  Real-time prices               | No cache (SSR)  | cache: 'no-store'
  Public API response (shared)   | Data cache      | next: { revalidate, tags }
  Session/auth data              | No cache        | cookies() → auto dynamic

8. COMMON INTERVIEW QUESTIONS & ANSWERS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q1: What are the 4 caching layers in Next.js App Router?
A:  (1) Request Memoization: deduplicates identical fetch calls within a single
    render cycle (per-request, uses React.cache).
    (2) Data Cache: persistent server cache for fetch() results, survives
    deployments, configured via cache: 'force-cache' / next: { revalidate }.
    (3) Full Route Cache: caches rendered HTML/RSC at build time for static routes.
    (4) Router Cache: client-side in-memory cache of RSC payloads, enables
    instant back/forward navigation.

Q2: What is React.cache() and when do you need it?
A:  React.cache() memoizes the result of a function call within a render tree.
    Use it for non-fetch data sources (Prisma/ORM queries, file reads) to prevent
    duplicate database queries when multiple components in the same render call
    the same function. fetch() is automatically memoized by Next.js; React.cache()
    is the equivalent for custom functions.

Q3: What's the difference between revalidatePath and revalidateTag?
A:  revalidatePath('/posts') invalidates the cached output of a specific URL path.
    revalidateTag('posts') invalidates all fetch() calls that were tagged with
    'posts' across any paths. Tags are more granular — one tag can span multiple
    routes, and you can tag individual records (post-42) for surgical invalidation.

Q4: What changed in Next.js 15 regarding fetch caching?
A:  In Next.js 14 and earlier, fetch() defaulted to force-cache (static).
    In Next.js 15, the default changed to no-store (dynamic). This means routes
    opt into caching explicitly. You must add next: { revalidate } or
    cache: 'force-cache' to cache fetch results in Next.js 15+.

Q5: How does the Router Cache work and how do you invalidate it?
A:  The Router Cache is a client-side in-memory cache of RSC payloads. Visited
    routes are cached, enabling instant back/forward navigation. It invalidates
    automatically after 30 seconds (dynamic routes) or 5 minutes (static routes).
    Force invalidation: router.refresh() re-fetches the current route. Server
    Actions that call revalidatePath/revalidateTag also invalidate the Router Cache.

Q6: What makes a route "dynamic" in the App Router?
A:  A route becomes dynamic when it uses: cookies() or headers() (reads request
    headers), useSearchParams() (reads URL query), fetch with cache: 'no-store',
    or when dynamic = 'force-dynamic' is exported. If any of these is present,
    the route renders on every request (no caching).

================================================================================
  MENTAL MODEL:
  Static = pre-rendered, served from CDN, blazing fast
  ISR    = static + background refresh, best of both worlds
  Dynamic = rendered per-request, always fresh, higher latency
  Cache tags = surgical cache invalidation (prefer over revalidatePath)
