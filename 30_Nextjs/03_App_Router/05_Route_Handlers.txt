1. WHAT ARE ROUTE HANDLERS?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Route Handlers are the App Router replacement for Pages Router API Routes.
They use Web Standard APIs (Request, Response) instead of Node.js-specific types.

  Location: app/api/.../route.ts  (or route.js)
  Convention: file named route.ts (NOT page.tsx — they cannot coexist in same dir)

Key differences from Pages Router API Routes:
  - Use Web API: Request, Response, Headers, URL
  - Support ALL HTTP methods as named exports
  - Built-in caching support via fetch cache config
  - Can run on Edge Runtime
  - No built-in body parser — use req.json() / req.formData() manually

  // app/api/posts/route.ts
  import { NextRequest, NextResponse } from 'next/server';

  // GET /api/posts
  export async function GET(request: NextRequest) {
    const { searchParams } = new URL(request.url);
    const page = Number(searchParams.get('page') ?? '1');
    const limit = Number(searchParams.get('limit') ?? '10');

    const posts = await prisma.post.findMany({
      skip: (page - 1) * limit,
      take: limit,
      orderBy: { createdAt: 'desc' },
    });

    return NextResponse.json({ posts, page, limit });
  }

  // POST /api/posts
  export async function POST(request: NextRequest) {
    const body = await request.json();

    const post = await prisma.post.create({
      data: { title: body.title, content: body.content },
    });

    return NextResponse.json(post, { status: 201 });
  }

  // Supported HTTP method exports:
  // GET | POST | PUT | PATCH | DELETE | HEAD | OPTIONS

  // app/api/posts/[id]/route.ts
  import { NextRequest, NextResponse } from 'next/server';

  type Params = { params: Promise<{ id: string }> }; // Next.js 15: params is a Promise

  // GET /api/posts/:id
  export async function GET(request: NextRequest, { params }: Params) {
    const { id } = await params; // MUST await in Next.js 15
    const post = await prisma.post.findUnique({ where: { id } });
    if (!post) return NextResponse.json({ error: 'Not found' }, { status: 404 });
    return NextResponse.json(post);
  }

  // PUT /api/posts/:id
  export async function PUT(request: NextRequest, { params }: Params) {
    const { id } = await params;
    const body = await request.json();
    const post = await prisma.post.update({ where: { id }, data: body });
    return NextResponse.json(post);
  }

  // DELETE /api/posts/:id
  export async function DELETE(request: NextRequest, { params }: Params) {
    const { id } = await params;
    await prisma.post.delete({ where: { id } });
    return new Response(null, { status: 204 });
  }

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // app/api/protected/route.ts
  import { auth } from '@/auth'; // NextAuth v5
  import { NextResponse } from 'next/server';

  export async function GET() {
    const session = await auth();
    if (!session?.user) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }
    const data = await prisma.userData.findMany({
      where: { userId: session.user.id },
    });
    return NextResponse.json(data);
  }

  // JWT-based auth:
  import { verifyToken } from '@/lib/jwt';

  export async function GET(request: NextRequest) {
    const authHeader = request.headers.get('Authorization');
    if (!authHeader?.startsWith('Bearer ')) {
      return NextResponse.json({ error: 'No token' }, { status: 401 });
    }
    const token = authHeader.slice(7);
    const payload = verifyToken(token);
    if (!payload) {
      return NextResponse.json({ error: 'Invalid token' }, { status: 401 });
    }
    return NextResponse.json({ user: payload });
  }

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Streaming a response with ReadableStream
  export async function GET() {
    const encoder = new TextEncoder();

    const stream = new ReadableStream({
      async start(controller) {
        const messages = ['Hello', ' ', 'World', '!'];
        for (const msg of messages) {
          controller.enqueue(encoder.encode(msg));
          await new Promise(resolve => setTimeout(resolve, 100));
        }
        controller.close();
      },
    });

    return new Response(stream, {
      headers: {
        'Content-Type': 'text/event-stream',
        'Cache-Control': 'no-cache',
        Connection: 'keep-alive',
      },
    });
  }

  // Server-Sent Events (SSE):
  export async function GET() {
    const stream = new TransformStream();
    const writer = stream.writable.getWriter();
    const encoder = new TextEncoder();

    (async () => {
      for (let i = 0; i < 10; i++) {
        await writer.write(encoder.encode(`data: ${JSON.stringify({ count: i })}\n\n`));
        await new Promise(r => setTimeout(r, 500));
      }
      writer.close();
    })();

    return new Response(stream.readable, {
      headers: { 'Content-Type': 'text/event-stream' },
    });
  }

  // app/api/webhooks/stripe/route.ts
  import Stripe from 'stripe';
  import { headers } from 'next/headers';

  const stripe = new Stripe(process.env.STRIPE_SECRET_KEY!);
  const webhookSecret = process.env.STRIPE_WEBHOOK_SECRET!;

  export async function POST(request: Request) {
    const body = await request.text(); // raw text for signature verification
    const headersList = await headers();
    const signature = headersList.get('stripe-signature')!;

    let event: Stripe.Event;
    try {
      event = stripe.webhooks.constructEvent(body, signature, webhookSecret);
    } catch (err) {
      return new Response(`Webhook Error: ${err}`, { status: 400 });
    }

    switch (event.type) {
      case 'payment_intent.succeeded': {
        const pi = event.data.object as Stripe.PaymentIntent;
        await handlePaymentSuccess(pi);
        break;
      }
      case 'customer.subscription.deleted': {
        const sub = event.data.object as Stripe.Subscription;
        await handleSubscriptionCanceled(sub);
        break;
      }
    }

    return new Response(null, { status: 200 });
  }

Q1: What is the difference between App Router Route Handlers and Pages Router API Routes?
A:  Route Handlers use Web Standard APIs (Request, Response) while API Routes use
    Node.js-specific NextApiRequest/NextApiResponse. Route Handlers export named
    functions per HTTP method (GET, POST, etc.) instead of a single default handler
    with if/else. Route Handlers support Edge Runtime natively and have built-in
    caching semantics. Route Handlers live in route.ts, API Routes in pages/api/*.ts.

Q2: Can route.ts and page.tsx coexist in the same directory?
A:  No. A directory can have either route.ts (API endpoint) OR page.tsx (UI page),
    not both. This is because both would conflict on the same URL path.

Q3: How do you handle CORS in Route Handlers?
A:  Add CORS headers to responses manually. Handle OPTIONS preflight requests with
    an OPTIONS export. For global CORS, use next.config.ts headers() function to
    avoid repeating headers in every handler.

Q4: Are GET Route Handlers cached by default?
A:  By default they are NOT cached. To cache, export dynamic = 'force-static'.
    GET handlers automatically become dynamic when they use: request object,
    cookies(), headers(), or any dynamic function. POST/PUT/DELETE/PATCH are
    always dynamic (never cached).

Q5: When would you use a Route Handler vs a Server Action?
A:  Route Handlers are for public APIs, mobile clients, webhooks, and external
    services that need a URL endpoint. Server Actions are for internal mutations
    from your own UI — they are called directly from React components without
    needing a URL. Server Actions give better type safety and progressive
    enhancement for forms.

Q6: How do you read the raw request body for webhook signature verification?
A:  Use await request.text() (not request.json()). Signature verification
    (e.g., Stripe) requires the raw body bytes. request.json() parses and
    re-serializes, which may alter whitespace and break the signature.

================================================================================
  KEY PATTERN: Route Handlers = public HTTP API endpoints.
  Use for external consumers (mobile, 3rd parties, webhooks).
  For your own UI mutations, prefer Server Actions.
