1. THE PROBLEM WITH DEEP INHERITANCE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Deep class hierarchies cause:
  - Fragile Base Class problem: changes to a parent break all children
  - Tight coupling: subclasses depend on parent internals
  - The Gorilla-Banana problem: "you wanted a banana but got a gorilla"

  Solution: favor COMPOSITION over INHERITANCE.
  "HAS-A" (composition) is often better than "IS-A" (inheritance).

2. MIXINS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  A mixin is an object (or function) that provides methods to be added to
  a class without extending it. It avoids multiple inheritance issues.

  // Mixin objects:
  const Serializable = {
    serialize() { return JSON.stringify(this); },
    deserialize(str) { return JSON.parse(str); }
  };

  const Validatable = {
    validate() {
      return Object.keys(this).every(k => this[k] !== null);
    }
  };

  // Apply mixins to a class:
  class User {
    constructor(name, email) {
      this.name = name;
      this.email = email;
    }
  }
  Object.assign(User.prototype, Serializable, Validatable);

  const user = new User("Alice", "alice@example.com");
  user.serialize();   // '{"name":"Alice","email":"alice@example.com"}'
  user.validate();    // true

3. FUNCTIONAL MIXIN PATTERN
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Mixin as a function (allows private state per usage):
  const withTimestamps = (superclass) => class extends superclass {
    constructor(...args) {
      super(...args);
      this.createdAt = new Date();
      this.updatedAt = new Date();
    }
    touch() { this.updatedAt = new Date(); }
  };

  const withSoftDelete = (superclass) => class extends superclass {
    delete() { this.deletedAt = new Date(); }
    isDeleted() { return !!this.deletedAt; }
  };

  class BaseModel {
    constructor(data) { Object.assign(this, data); }
  }

  class Post extends withSoftDelete(withTimestamps(BaseModel)) {
    publish() { this.publishedAt = new Date(); }
  }

  const post = new Post({ title: "Hello" });
  post.createdAt;    // set by withTimestamps
  post.delete();     // from withSoftDelete
  post.isDeleted();  // true

4. COMPOSITION OVER INHERITANCE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Instead of inheriting, compose behavior from small functions:
  const canEat = (state) => ({
    eat: (food) => `${state.name} eats ${food}`
  });

  const canFly = (state) => ({
    fly: () => `${state.name} is flying`
  });

  const canSwim = (state) => ({
    swim: () => `${state.name} is swimming`
  });

  // Create objects by mixing capabilities:
  function createDuck(name) {
    const state = { name };
    return {
      ...canEat(state),
      ...canFly(state),
      ...canSwim(state),
      quack: () => "Quack!"
    };
  }

  function createFish(name) {
    const state = { name };
    return {
      ...canEat(state),
      ...canSwim(state)
    };
  }

  const duck = createDuck("Donald");
  duck.fly();    // "Donald is flying"
  duck.swim();   // "Donald is swimming"
  duck.quack();  // "Quack!"

  // No class hierarchy, no fragile base class — just pure composition.

5. INHERIT ONCE, MIXIN MANY
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Good rule of thumb:
  // Use `extends` for one core IS-A relationship
  // Use mixins for additional capabilities

  class EventEmitter {
    #listeners = {};
    on(event, fn) { (this.#listeners[event] ??= []).push(fn); }
    emit(event, ...args) { this.#listeners[event]?.forEach(fn => fn(...args)); }
  }

  const withLogger = {
    log(msg) { console.log(`[${this.constructor.name}] ${msg}`); }
  };

  class UserService extends EventEmitter {
    createUser(data) {
      const user = { id: Date.now(), ...data };
      this.log(`Created user: ${user.id}`);
      this.emit("user:created", user);
      return user;
    }
  }
  Object.assign(UserService.prototype, withLogger);

Q: What is a mixin in JavaScript?
A: A mixin is an object or function that provides reusable methods that can be
   copied into a class or object without using inheritance. Typically applied
   with Object.assign(SomeClass.prototype, mixin).

Q: What is "composition over inheritance"?
A: Instead of creating complex class hierarchies, you build objects from small,
   reusable behavior units (functions or objects). It reduces coupling and
   avoids fragile base class problems.

Q: How is a functional mixin different from an object mixin?
A: A functional mixin is a function that takes a superclass and returns a new
   class that extends it with additional methods. This allows each mixin to
   have its own constructor logic and private state. Object mixins just copy
   methods into a prototype — no private state per mixin.

Q: What are the problems with deep inheritance chains?
A: Fragile base class (parent changes break children), tight coupling, hard
   to follow execution (method resolution order), difficulty composing
   unrelated functionality (JS only supports single inheritance).

Q: Does JavaScript support multiple inheritance?
A: Not natively via classes (single extends). But you can achieve similar
   results through mixins, composition, or chained functional mixins.
