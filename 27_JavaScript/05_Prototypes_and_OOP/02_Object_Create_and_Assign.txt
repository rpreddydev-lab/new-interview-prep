1. Object.create()
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Creates a new object with the specified [[Prototype]].

  Syntax: Object.create(proto, propertiesObject?)

  // Basic usage:
  const proto = {
    greet() { return `Hello, I'm ${this.name}`; }
  };
  const alice = Object.create(proto);
  alice.name = "Alice";
  alice.greet();  // "Hello, I'm Alice"

  Object.getPrototypeOf(alice) === proto;  // true

  // Create with property descriptors:
  const obj = Object.create(proto, {
    name: {
      value: "Bob",
      writable: true,
      enumerable: true,
      configurable: true
    }
  });

  // Object.create(null) — no prototype:
  const dict = Object.create(null);
  dict.toString;           // undefined (no Object.prototype)
  dict.hasOwnProperty;     // undefined
  "key" in dict;           // works — in operator doesn't use prototype

2. Object.create() FOR INHERITANCE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  function Animal(name) { this.name = name; }
  Animal.prototype.speak = function() { return `${this.name} speaks`; };

  function Dog(name, breed) {
    Animal.call(this, name);  // own properties
    this.breed = breed;
  }

  // Key step: set up prototype chain without calling Animal():
  Dog.prototype = Object.create(Animal.prototype);
  Dog.prototype.constructor = Dog;  // repair constructor reference

  Dog.prototype.bark = function() { return "Woof!"; };

  const d = new Dog("Rex", "Lab");
  d.speak();            // "Rex speaks" (inherited)
  d.bark();             // "Woof!" (own)
  d instanceof Dog;     // true
  d instanceof Animal;  // true

3. Object.assign()
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Copies own enumerable properties from source(s) to target.
  Returns the target object (mutates it!).

  Syntax: Object.assign(target, ...sources)

  // Merge objects:
  const defaults = { color: "red", size: 10, visible: true };
  const overrides = { color: "blue", extra: "yes" };

  const config = Object.assign({}, defaults, overrides);
  // { color: "blue", size: 10, visible: true, extra: "yes" }
  // Later sources override earlier ones.

  // Clone (SHALLOW):
  const original = { a: 1, b: { c: 2 } };
  const clone = Object.assign({}, original);
  clone.a = 99;           // original.a still 1
  clone.b.c = 99;         // original.b.c also 99! (shallow reference)

  // Add properties to existing object:
  const target = { a: 1 };
  Object.assign(target, { b: 2 }, { c: 3 });
  // target is { a: 1, b: 2, c: 3 }

4. Object.assign() — WHAT IT COPIES AND DOESN'T
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  ✓ Copies: own enumerable string-keyed properties
  ✗ Does NOT copy: inherited properties
  ✗ Does NOT copy: non-enumerable properties
  ✗ Does NOT copy: Symbol-keyed properties? → actually IT DOES copy Symbols
  ✗ Does NOT set up prototypes (no [[Prototype]] link)

  const src = {};
  Object.defineProperty(src, "hidden", {
    value: 42,
    enumerable: false   // non-enumerable
  });
  Object.assign({}, src);  // {} — "hidden" not copied

5. Object.assign() vs SPREAD
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Spread is usually preferred — cleaner and non-mutating:
  const merged = { ...defaults, ...overrides };  // new object
  const clone = { ...original };                 // shallow clone

  // Key difference: Object.assign() triggers setters on target;
  // spread uses [[Get]] and creates new properties.
  const target = {
    set x(v) { console.log("setter!"); this._x = v; }
  };
  Object.assign(target, { x: 1 });  // triggers setter — "setter!"
  ({ ...target, x: 1 });             // does NOT trigger setter

7. COMMON INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What does Object.create(proto) do?
A: Creates a new empty object whose [[Prototype]] is set to proto. This is
   the most direct way to set up inheritance without invoking a constructor.

Q: What is the difference between Object.create(null) and {}?
A: {} has Object.prototype in its chain (has toString, hasOwnProperty, etc.).
   Object.create(null) has no prototype — a "pure" object used as a safe
   dictionary to avoid key collisions with inherited property names.

Q: Does Object.assign do a deep copy?
A: No. It performs a SHALLOW copy. Primitive values are copied, but object
   references are shared. Modifying a nested object on the copy also modifies
   the original.

Q: What is the difference between Object.assign and spread?
A: Both do shallow merging. Key difference: Object.assign mutates the target
   and triggers setters. Spread syntax always creates a new object and uses
   [[Get]] rather than triggering setters.

Q: How do you create a deep clone in modern JS?
A: structuredClone(obj) — works for most built-ins (Date, Map, Set, etc.)
   but doesn't clone functions or preserve class prototypes.
