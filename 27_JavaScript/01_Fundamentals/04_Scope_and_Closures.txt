1. SCOPE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Scope defines the accessibility (visibility) of variables.
JS uses LEXICAL SCOPE (also called static scope): the scope of a variable
is determined by WHERE it is written in the source code, not where it's
called from.

Types of scope:
  1. Global Scope
  2. Function (Local) Scope
  3. Block Scope (ES6+, let/const)
  4. Module Scope

2. GLOBAL SCOPE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  - Variables declared outside any function/block
  - Accessible from everywhere
  - var globals become properties of the global object (window/global)
  - let/const globals do NOT become properties of global object

  var globalVar = "global var";
  let globalLet = "global let";

  function test() {
    console.log(globalVar);    // accessible
    console.log(globalLet);    // accessible
  }

3. FUNCTION SCOPE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  - var variables are scoped to the enclosing function
  - Each function call creates a new scope

  function outer() {
    var outerVar = "outer";
    
    function inner() {
      var innerVar = "inner";
      console.log(outerVar);  // "outer" ← accessible via scope chain
    }
    inner();
    console.log(innerVar);    // ReferenceError: innerVar not defined
  }

4. BLOCK SCOPE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  - let and const are block-scoped (any {} creates a block)
  - Blocks: if, for, while, switch, bare {}

  {
    let blockScoped = "inside block";
    const alsoBlock = "const";
    var functionScoped = "escapes block!";
  }
  console.log(blockScoped);      // ReferenceError
  console.log(alsoBlock);        // ReferenceError
  console.log(functionScoped);   // "escapes block!" (var ignores block)

  // for loop with let:
  for (let i = 0; i < 3; i++) {
    setTimeout(() => console.log(i), 0);
  }
  // 0, 1, 2 (each iteration has its own i binding)

5. SCOPE CHAIN
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  When JS looks up a variable, it starts at the current scope and walks
  UP the scope chain to the global scope. First match wins.

  const a = 1;       // Global scope

  function outer() {
    const b = 2;     // outer scope

    function middle() {
      const c = 3;   // middle scope

      function inner() {
        const d = 4; // inner scope
        console.log(a, b, c, d);  // 1, 2, 3, 4 — all accessible
      }
      inner();
    }
    middle();
  }
  outer();

  // Variable SHADOWING:
  const x = "global";
  function shadow() {
    const x = "local";     // shadows global x
    console.log(x);        // "local"
  }
  shadow();
  console.log(x);          // "global" (global unchanged)

6. CLOSURES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

A CLOSURE is a function that REMEMBERS its lexical environment even after
the outer function has returned. The inner function keeps a reference to
variables in the outer scope.

  function makeCounter() {
    let count = 0;             // lives in makeCounter's scope
    
    return function() {        // closure — has access to count
      count++;
      return count;
    };
  }

  const counter = makeCounter();  // makeCounter has returned
  counter();  // 1 — count still accessible!
  counter();  // 2
  counter();  // 3

  // Each call to makeCounter() creates a NEW closure with its OWN count:
  const c1 = makeCounter();
  const c2 = makeCounter();
  c1();  // 1
  c1();  // 2
  c2();  // 1 (independent)

7. PRACTICAL CLOSURE PATTERNS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

A) DATA PRIVACY / ENCAPSULATION:
  function createBankAccount(initialBalance) {
    let balance = initialBalance;  // private

    return {
      deposit(amount) { balance += amount; },
      withdraw(amount) {
        if (amount > balance) throw new Error("Insufficient funds");
        balance -= amount;
      },
      getBalance() { return balance; }
    };
  }

  const account = createBankAccount(100);
  account.deposit(50);
  account.getBalance();  // 150
  account.balance;       // undefined — balance is private

B) FUNCTION FACTORIES:
  function multiply(factor) {
    return (number) => number * factor;  // closure over factor
  }

  const double = multiply(2);
  const triple = multiply(3);
  double(5);   // 10
  triple(5);   // 15

C) MEMOIZATION:
  function memoize(fn) {
    const cache = {};
    return function(n) {
      if (n in cache) return cache[n];
      cache[n] = fn(n);
      return cache[n];
    };
  }

  const slowSquare = (n) => { /* expensive */ return n * n; };
  const fastSquare = memoize(slowSquare);
  fastSquare(10);  // computed
  fastSquare(10);  // from cache

D) PARTIAL APPLICATION / CURRYING:
  function add(a) {
    return function(b) {
      return a + b;
    };
  }
  const add5 = add(5);
  add5(3);   // 8
  add5(10);  // 15

E) EVENT LISTENERS:
  function setupButton(label) {
    let clicks = 0;
    document.getElementById("btn").addEventListener("click", function() {
      clicks++;
      console.log(`${label} clicked ${clicks} times`);
    });
  }

8. CLOSURE GOTCHAS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

A) LOOP WITH var (classic gotcha):
  var funcs = [];
  for (var i = 0; i < 3; i++) {
    funcs.push(function() { console.log(i); });
  }
  funcs[0]();  // 3
  funcs[1]();  // 3
  funcs[2]();  // 3
  // All closures share the SAME var i. After loop, i === 3.

  // Fix 1: use let (creates new binding each iteration):
  for (let i = 0; i < 3; i++) {
    funcs.push(function() { console.log(i); });
  }
  // Fix 2: IIFE to capture i by value:
  for (var i = 0; i < 3; i++) {
    funcs.push((function(j) { return function() { console.log(j); }; })(i));
  }

B) MEMORY LEAKS via closures:
  function leaky() {
    const bigData = new Array(1e6).fill("heavy");
    return () => bigData[0];  // bigData can't be GC'd as long as fn lives
  }
  const fn = leaky();
  // fn holds reference to bigData — potential leak if fn kept alive too long

9. MODULE SCOPE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  In ES6 modules, each file has its own module scope.
  Variables are NOT added to the global object.
  Only exported symbols are accessible from other modules.

  // math.js
  const PI = 3.14;          // module scope — not global
  export const area = (r) => PI * r * r;  // exported

  // app.js
  import { area } from "./math.js";
  console.log(PI);           // ReferenceError — not exported

10. COMMON INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What is a closure?
A: A closure is a function paired with its lexical environment. It gives
   an inner function access to the outer function's variables even after
   the outer function has returned.

Q: What is lexical scope?
A: Lexical scope means scope is determined by WHERE a function is defined
   in the source code, not where it's called. This is how JS works — scope
   is determined at author time (write time), not at call time.

Q: What is the difference between scope and closure?
A: Scope determines variable visibility in a specific block/function.
   Closure is the mechanism by which a function retains access to its
   outer scope even after that outer scope has exited.

Q: How do closures enable data privacy?
A: By keeping variables in an outer function's scope where they are
   accessible to returned inner functions but not directly from outside.
   This is the module pattern / factory function pattern.

Q: Explain the classic var loop closure problem and solutions.
A: With var, all loop iterations share one variable. By the time callbacks
   execute, the loop is done and the variable has its final value.
   Solutions: use let (each iteration gets new binding), or use IIFE to
   capture current value by creating a new scope per iteration.

Q: What is variable shadowing?
A: When an inner scope declares a variable with the same name as an outer
   scope variable, the inner one "shadows" the outer one inside that scope.
   The outer variable still exists, just inaccessible within the inner scope.
