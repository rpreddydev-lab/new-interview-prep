1. DECLARATION KEYWORDS OVERVIEW
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  ┌─────────┬──────────────┬─────────────┬──────────┬────────────┬────────────┐
  │ Keyword │ Scope        │ Hoisted     │ Init val │ Reassign   │ Redeclare  │
  ├─────────┼──────────────┼─────────────┼──────────┼────────────┼────────────┤
  │ var     │ Function     │ Yes (undef) │ undefined│ Yes        │ Yes        │
  ├─────────┼──────────────┼─────────────┼──────────┼────────────┼────────────┤
  │ let     │ Block        │ Yes (TDZ)   │ none     │ Yes        │ No         │
  ├─────────┼──────────────┼─────────────┼──────────┼────────────┼────────────┤
  │ const   │ Block        │ Yes (TDZ)   │ none     │ No         │ No         │
  └─────────┴──────────────┴─────────────┴──────────┴────────────┴────────────┘

  TDZ = Temporal Dead Zone (see section 4)

2. var
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  - Function-scoped (or global if declared outside a function)
  - Hoisted to top of function/global with value `undefined`
  - Can be re-declared in the same scope (no error)
  - Creates property on the global object (window in browsers)

  // Hoisting example:
  console.log(x);  // undefined (NOT ReferenceError)
  var x = 5;
  console.log(x);  // 5

  // Equivalent to:
  var x;           // hoisted
  console.log(x);  // undefined
  x = 5;
  console.log(x);  // 5

  // Function scope:
  function test() {
    var a = 1;
    if (true) {
      var a = 2;          // SAME variable (var is function-scoped)
      console.log(a);     // 2
    }
    console.log(a);       // 2 (leaked from if block!)
  }

  // Classic bug with var in loops:
  for (var i = 0; i < 3; i++) {
    setTimeout(() => console.log(i), 100);
  }
  // Output: 3, 3, 3  (because i is shared, loop done before callbacks fire)

  // Fix 1: use let
  for (let i = 0; i < 3; i++) {
    setTimeout(() => console.log(i), 100);
  }
  // Output: 0, 1, 2

  // Fix 2: IIFE with var
  for (var i = 0; i < 3; i++) {
    ((j) => setTimeout(() => console.log(j), 100))(i);
  }

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
3. let
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  - Block-scoped (inside {})
  - NOT initialized on hoist (Temporal Dead Zone)
  - Cannot be re-declared in same scope
  - Can be reassigned

  // Block scope:
  function test() {
    let a = 1;
    if (true) {
      let a = 2;          // NEW variable (block-scoped)
      console.log(a);     // 2
    }
    console.log(a);       // 1 (original, unaffected)
  }

  // TDZ example:
  console.log(y);  // ReferenceError: Cannot access 'y' before initialization
  let y = 5;

  // Redeclaration error:
  let z = 1;
  let z = 2;  // SyntaxError: Identifier 'z' has already been declared

  // Reassignment (valid):
  let count = 0;
  count++;  // fine

4. const
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  - Block-scoped like let
  - MUST be initialized at declaration
  - Cannot be reassigned (binding is constant, not the value itself)
  - Objects/arrays declared with const can still be mutated!

  const PI = 3.14;
  PI = 3;  // TypeError: Assignment to constant variable

  // const with objects — BINDING is constant, VALUE is mutable:
  const user = { name: "Alice" };
  user.name = "Bob";    // VALID — object is mutable
  user = {};            // TypeError — can't reassign the binding

  const arr = [1, 2, 3];
  arr.push(4);          // VALID — array is mutable
  arr = [5, 6];         // TypeError

  // To make object immutable:
  const frozen = Object.freeze({ x: 1 });
  frozen.x = 99;       // Silently fails (non-strict) or TypeError (strict)
  // Note: freeze is SHALLOW — nested objects are still mutable

  const deep = Object.freeze({ nested: { val: 1 } });
  deep.nested.val = 99; // WORKS — nested not frozen

5. TEMPORAL DEAD ZONE (TDZ)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  The TDZ is the period from the start of the block until the variable
  declaration is reached. During TDZ, accessing the variable throws
  ReferenceError (even though it's been hoisted).

  {
    // TDZ for x starts here
    console.log(x);  // ReferenceError: Cannot access 'x' before initialization
    let x = 5;       // TDZ ends here, x is initialized to 5
    console.log(x);  // 5
  }

  // TDZ with function parameters:
  function test(a = b, b = 2) {}  // ReferenceError: b not defined when a evaluated

  // Why TDZ exists:
  // Enforces "declare before use" discipline
  // Makes let/const variables predictable and debuggable
  // Prevents hoisting-related bugs that plagued var

6. GLOBAL SCOPE DIFFERENCE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  var at top level creates a property on the global object:
    var globalVar = "I'm global";
    console.log(window.globalVar);  // "I'm global" (browser)

  let/const at top level do NOT:
    let blockLet = "not global";
    console.log(window.blockLet);   // undefined

7. BEST PRACTICES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  1. Use const by default
  2. Use let when you need to reassign
  3. Never use var in modern code (except when working with legacy code)
  4. Always declare variables at the top of their scope
  5. Use const for arrays and objects even if mutating them
     (protects against accidental reassignment of the reference)

8. COMMON INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What is the difference between var, let, and const?
A: var is function-scoped and hoisted to undefined. let and const are
   block-scoped and in the TDZ until declared. const cannot be reassigned
   but its value (if an object) can still be mutated.

Q: What is the Temporal Dead Zone?
A: The period between the start of a block and the let/const declaration.
   Accessing the variable in TDZ throws a ReferenceError.

Q: What does const guarantee?
A: const guarantees the BINDING (variable name → memory address) is
   constant. It does NOT make the value immutable. Objects assigned to
   const can still have properties modified. Use Object.freeze() for
   shallow immutability.

Q: Why does the for loop with var produce unexpected output?
A: var is function-scoped, so there's only ONE `i` variable shared across
   all iterations. By the time the callbacks fire, the loop has finished
   and `i` equals 3. With let, each iteration gets its own block-scoped `i`.

Q: Can you redeclare let in an inner block?
A: Yes — because it's a different block scope:
   let x = 1;
   { let x = 2; console.log(x); }  // 2 (shadow)
   console.log(x);                  // 1 (outer)

Q: What happens when const holds a reference to an object vs a primitive?
A: Primitive: truly immutable — can't change value at all.
   Object: reference is locked but object's properties can change freely.
   To deeply freeze, recursively apply Object.freeze().
