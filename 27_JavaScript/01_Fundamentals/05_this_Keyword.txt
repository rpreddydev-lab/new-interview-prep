THE `this` KEYWORD — INTERVIEW PREP
================================================================================

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. WHAT IS `this`?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

`this` refers to the EXECUTION CONTEXT — the object that is currently
"owning" or "calling" the function. Unlike most languages, `this` in JS is
NOT determined by where a function is defined (not lexical), but by HOW
the function is CALLED (dynamic binding).

Exception: Arrow functions — `this` is lexically bound (captured from
the surrounding scope at definition time).

2. RULES FOR `this` BINDING (in order of priority)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Priority:  1 (highest) — new binding
             2 — explicit binding (call/apply/bind)
             3 — implicit binding (method call)
             4 — default binding (lowest)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
3. RULE 1: DEFAULT BINDING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Standalone function call — no context.
  Non-strict: this = global object (window in browser, global in Node)
  Strict: this = undefined

  function greet() {
    console.log(this);
  }
  greet();            // window (non-strict) or undefined (strict)

  // Nested function loses this:
  const obj = {
    name: "Alice",
    greet() {
      function inner() {
        console.log(this.name);  // undefined (inner lost obj context)
      }
      inner();  // default binding
    }
  };
  obj.greet();

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
4. RULE 2: IMPLICIT BINDING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  When a function is called as a METHOD of an object, this = that object.

  const person = {
    name: "Bob",
    greet() {
      console.log(this.name);  // "Bob"
    }
  };
  person.greet();   // this = person

  // Implicit binding LOST (common pitfall):
  const { greet } = person;
  greet();          // this = undefined/window — context lost!

  // Callback pitfall:
  setTimeout(person.greet, 0);  // this = undefined/window — lost!

  // Solutions:
  // 1. Arrow function:
  setTimeout(() => person.greet(), 0);  // this = person
  // 2. bind:
  setTimeout(person.greet.bind(person), 0);  // this = person

5. RULE 3: EXPLICIT BINDING — call, apply, bind
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  A) call — invokes immediately, args passed individually:
     function greet(greeting, punct) {
       console.log(`${greeting}, ${this.name}${punct}`);
     }
     const user = { name: "Alice" };
     greet.call(user, "Hello", "!");   // "Hello, Alice!"

  B) apply — invokes immediately, args passed as array:
     greet.apply(user, ["Hi", "?"]);   // "Hi, Alice?"
     // apply = "array"

  C) bind — returns NEW function with this permanently set:
     const boundGreet = greet.bind(user, "Hey");
     boundGreet(".");   // "Hey, Alice."
     boundGreet("!");   // "Hey, Alice."  (this always user)

     // bind is partial application:
     function multiply(a, b) { return a * b; }
     const double = multiply.bind(null, 2);  // first arg pre-filled
     double(5);   // 10
     double(10);  // 20

  call vs apply vs bind:
    call:  fn.call(ctx, arg1, arg2)    → invokes immediately
    apply: fn.apply(ctx, [arg1, arg2]) → invokes immediately, array args
    bind:  fn.bind(ctx, arg1)          → returns new function, doesn't call

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
6. RULE 4: new BINDING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  When a function is called with `new`, this = the newly created object.

  What `new` does:
    1. Creates a new empty object {}
    2. Sets the object's [[Prototype]] to Constructor.prototype
    3. Executes the constructor with this = new object
    4. Returns the new object (unless constructor explicitly returns an object)

  function Person(name) {
    this.name = name;    // this = the new object
  }

  const alice = new Person("Alice");
  alice.name;            // "Alice"

  // new always takes priority over bind:
  function Foo() { this.x = 1; }
  const BoundFoo = Foo.bind({ x: 99 });
  const obj = new BoundFoo();
  obj.x;                 // 1 (new overrides bind)

7. ARROW FUNCTIONS AND `this`
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Arrow functions do NOT have their own `this`. They inherit `this` from
  the ENCLOSING LEXICAL SCOPE at the time of definition.
  Cannot be changed with call/apply/bind (ignored).

  const obj = {
    name: "Alice",
    greet: function() {
      const arrow = () => {
        console.log(this.name);  // "Alice" — inherits this from greet
      };
      arrow();
    }
  };
  obj.greet();   // "Alice"

  // Compare: regular function loses this:
  const obj2 = {
    name: "Bob",
    greet: function() {
      function inner() {
        console.log(this.name);  // undefined — lost context
      }
      inner();
    }
  };
  obj2.greet();   // undefined

  // Arrow function as method — PITFALL:
  const timer = {
    count: 0,
    start: () => {          // arrow method
      setInterval(() => {
        this.count++;       // this = outer (window/undefined), NOT timer!
      }, 1000);
    }
  };
  // Use regular function as method, arrow inside:
  const timer2 = {
    count: 0,
    start() {               // regular method — this = timer2
      setInterval(() => {
        this.count++;       // this = timer2 ✓
      }, 1000);
    }
  };

8. `this` IN CLASSES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  class Counter {
    constructor() {
      this.count = 0;
      // Solution: bind in constructor:
      this.increment = this.increment.bind(this);
    }

    increment() {
      this.count++;
    }

    // Or use class field arrow function (no bind needed):
    decrement = () => {
      this.count--;
    };
  }

  const c = new Counter();
  const inc = c.increment;
  inc();             // works — bound in constructor
  c.decrement();     // works — arrow function

  // Without bind, method used as callback loses this:
  const btn = document.getElementById("btn");
  btn.addEventListener("click", c.increment);  // this = btn element (lost!)
  btn.addEventListener("click", c.increment.bind(c));  // this = c ✓

9. `this` IN SPECIAL CONTEXTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Global scope (browser):
  console.log(this);           // window

  // Global scope (Node.js):
  console.log(this);           // {} (module-level `this` in CommonJS)
  console.log(globalThis);     // global object (works everywhere, ES2020)

  // Event handlers (DOM):
  button.addEventListener("click", function() {
    console.log(this);         // the button element
  });
  button.addEventListener("click", () => {
    console.log(this);         // outer `this` (not the button!)
  });

10. COMMON INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What is `this` in JavaScript?
A: `this` refers to the current execution context — the object that "owns"
   the function call. It's dynamically determined by HOW the function is
   called, not where it's defined.

Q: What is the difference between call, apply, and bind?
A: All three explicitly set `this`. call() invokes immediately with
   individual args. apply() invokes immediately with args as an array.
   bind() returns a new function with this permanently set (doesn't invoke).

Q: Why does `this` become undefined in a callback?
A: When you pass a method as a callback, it loses its object context.
   The function is called as a standalone function → default binding →
   undefined (strict) or global (non-strict). Fix with .bind() or arrow fn.

Q: What `this` does an arrow function have?
A: Arrow functions have no own `this`. They inherit `this` from the
   enclosing lexical scope. call/apply/bind cannot change arrow's this.

Q: When should you use arrow functions vs regular functions as methods?
A: Regular functions as object methods (so `this` = the object).
   Arrow functions inside methods (so `this` = method's context, not lost).

Q: What is the order of `this` binding priority?
A: 1. new binding (new Foo)
   2. Explicit binding (call/apply/bind)
   3. Implicit binding (obj.method())
   4. Default binding (standalone call)
