2. FUNCTION DECLARATION
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  function greet(name) {
    return `Hello, ${name}!`;
  }

  - Fully hoisted (name AND body available before declaration)
  - Has its own `this` and `arguments`
  - Can be called before it appears in code

3. FUNCTION EXPRESSION
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  const greet = function(name) {
    return `Hello, ${name}!`;
  };

  - NOT hoisted (only the variable is hoisted, not the function)
  - Good for conditional assignment
  - Function name is optional (anonymous expression above)

  // Named function expression:
  const factorial = function fact(n) {
    return n <= 1 ? 1 : n * fact(n - 1);  // can reference self by name
  };
  factorial(5);   // 120
  // fact(5);     // ReferenceError — name not in outer scope

4. ARROW FUNCTIONS (ES6)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  const greet = (name) => `Hello, ${name}!`;
  const add = (a, b) => a + b;
  const double = n => n * 2;       // single param, no parens needed
  const getObj = () => ({ x: 1 }); // return object literal: wrap in ()

  Differences from regular functions:
  - No own `this` (inherits from enclosing lexical scope)
  - No `arguments` object (use rest parameters: ...args)
  - Cannot be used as constructor (no `new`)
  - No `prototype` property
  - Cannot be generator functions
  - Cannot use `yield`

  // No arguments object:
  const fn = () => {
    console.log(arguments);  // ReferenceError in strict, outer arguments otherwise
  };
  const fn2 = (...args) => {
    console.log(args);       // works — rest params
  };

  // Cannot use as constructor:
  const Arrow = () => {};
  new Arrow();  // TypeError: Arrow is not a constructor

5. FUNCTION PARAMETERS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  A) DEFAULT PARAMETERS (ES6):
  function greet(name = "World") {
    return `Hello, ${name}!`;
  }
  greet();          // "Hello, World!"
  greet("Alice");   // "Hello, Alice!"
  greet(undefined); // "Hello, World!" — undefined triggers default
  greet(null);      // "Hello, null!"  — null does NOT trigger default

  // Defaults can be expressions:
  function createId(prefix = "ID", ts = Date.now()) { ... }

  B) REST PARAMETERS:
  function sum(...nums) {
    return nums.reduce((a, b) => a + b, 0);
  }
  sum(1, 2, 3, 4);  // 10

  // Must be last parameter:
  function first(a, b, ...rest) { ... }  // OK
  // function bad(...rest, last) {}      // SyntaxError

  C) arguments OBJECT:
  function test() {
    console.log(arguments);          // Arguments [1, 2, 3]
    console.log(arguments[0]);       // 1
    console.log(Array.from(arguments)); // [1, 2, 3] — convert to array
    const arr = [...arguments];      // [1, 2, 3]
  }
  test(1, 2, 3);

  - arguments is array-like (has length, indices) but NOT a real array
  - Arrow functions don't have `arguments`
  - Prefer rest parameters over arguments in modern code

6. HIGHER-ORDER FUNCTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  A function that takes a function as argument OR returns a function.

  // Takes function as argument:
  [1,2,3].map(x => x * 2)        // [2,4,6]
  [1,2,3].filter(x => x > 1)     // [2,3]
  [1,2,3].reduce((acc,x) => acc + x, 0) // 6

  // Returns a function (factory):
  function multiplier(factor) {
    return x => x * factor;
  }
  const triple = multiplier(3);
  triple(4);   // 12

7. PURE FUNCTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  A pure function:
    1. Returns the same output given the same inputs (deterministic)
    2. Has no side effects (doesn't modify external state)

  // Pure:
  const add = (a, b) => a + b;  // same inputs → same output, no side effects

  // Impure (modifies external state):
  let count = 0;
  function increment() { count++; }  // side effect!

  // Impure (depends on external state):
  function getTime() { return Date.now(); }  // different output each call

8. IMMEDIATELY INVOKED FUNCTION EXPRESSION (IIFE)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  An IIFE is a function that is defined and called immediately.

  Syntax:
  (function() {
    // code
  })();

  // Arrow IIFE:
  (() => {
    // code
  })();

  // With parameters:
  (function(name) {
    console.log("Hello, " + name);
  })("World");  // "Hello, World"

  Why use IIFEs?

  A) SCOPE ISOLATION — avoid polluting global scope:
  (function() {
    var privateVar = "only here";
    // privateVar not accessible outside
  })();
  // console.log(privateVar);  // ReferenceError

  B) CREATE BLOCK SCOPE (pre-ES6):
  var result = (function() {
    var secret = "hidden";
    return secret.length;
  })();

  C) AVOID var LOOP BUG:
  for (var i = 0; i < 3; i++) {
    (function(j) {
      setTimeout(() => console.log(j), 100);
    })(i);
  }
  // 0, 1, 2

  D) MODULE PATTERN (pre-ES6 modules):
  const counter = (function() {
    let count = 0;
    return {
      increment() { count++; },
      decrement() { count--; },
      value() { return count; }
    };
  })();
  counter.increment();
  counter.value();  // 1
  // count is private, not accessible directly

  E) INITIALIZE ONE-TIME SETUP:
  const config = (function() {
    // complex initialization
    return Object.freeze({ apiUrl: "https://api.example.com" });
  })();

10. GENERATOR FUNCTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  function* range(start, end) {
    for (let i = start; i <= end; i++) {
      yield i;          // pauses and returns value
    }
  }

  const gen = range(1, 3);
  gen.next();    // { value: 1, done: false }
  gen.next();    // { value: 2, done: false }
  gen.next();    // { value: 3, done: false }
  gen.next();    // { value: undefined, done: true }

  // Spread a generator:
  [...range(1, 5)]   // [1, 2, 3, 4, 5]

  // Infinite generator:
  function* infiniteIds() {
    let id = 1;
    while (true) {
      yield id++;
    }
  }
  const ids = infiniteIds();
  ids.next().value;  // 1
  ids.next().value;  // 2

11. COMMON INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What is an IIFE and why is it useful?
A: An IIFE (Immediately Invoked Function Expression) is a function that
   executes immediately after definition. Use cases: scope isolation
   (avoid global namespace pollution), module pattern for private state,
   avoiding var loop bugs (pre-let), one-time initialization.

Q: What is the difference between function declaration and function expression?
A: Declaration is hoisted fully (callable before it appears). Expression
   is not hoisted as callable (only the variable is hoisted). Declarations
   cannot be conditional (unreliable behavior). Expressions can be
   conditionally assigned.

Q: What is the difference between arguments and rest parameters?
A: arguments: array-like object, available in regular functions, contains
   all passed args. Rest (...args): a real array, can use array methods,
   works in arrow functions, more explicit.

Q: When can't you use an arrow function?
A: 1. As an object method when you need `this` to refer to the object.
   2. As an event handler when you need `this` = the DOM element.
   3. As a constructor (cannot use with `new`).
   4. When you need the `arguments` object.
   5. As a generator function (no `yield`).

Q: What is a higher-order function?
A: A function that takes another function as a parameter or returns a
   function. Examples: map, filter, reduce, setTimeout, addEventListener.

Q: What does Function.prototype.length return?
A: The number of expected parameters (formal parameters). Rest parameters
   and parameters with defaults are NOT counted.

Q: What is a pure function?
A: A function that always returns the same output for same inputs and has
   no side effects (doesn't modify external state or interact with outside
   world). Easier to test, debug, and reason about.
