1. TYPES OF ERRORS IN JAVASCRIPT
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  ┌──────────────────┬────────────────────────────────────────────────────────┐
  │ Error Type       │ When it occurs                                         │
  ├──────────────────┼────────────────────────────────────────────────────────┤
  │ SyntaxError      │ Invalid JavaScript syntax (parse time)                 │
  │                  │ let 1x = 5;  /  { key: }                               │
  ├──────────────────┼────────────────────────────────────────────────────────┤
  │ ReferenceError   │ Accessing undeclared variable / TDZ access             │

3. try / catch / finally
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  try {
    // code that might throw
    const result = JSON.parse('invalid json');
  } catch (err) {
    // handles any thrown value
    console.error(err.name);     // "SyntaxError"
    console.error(err.message);  // "Unexpected token..."
  } finally {
    // ALWAYS runs (cleanup) — even if try returns or throws
    console.log("cleanup");
  }

  // catch is optional (if you only want finally):
  try { doSomething(); }
  finally { cleanup(); }

  // Finally runs even with return:
  function test() {
    try {
      return "from try";
    } finally {
      console.log("finally runs!");  // prints BEFORE return
    }
  }
  test();  // logs "finally runs!", returns "from try"

  // Finally overrides try/catch return:
  function test2() {
    try { return 1; }
    finally { return 2; }  // this wins!
  }
  test2();  // 2

4. THROWING ERRORS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // You can throw anything (but throwing Error objects is best practice):
  throw new Error("message");        // recommended
  throw new TypeError("bad type");   // be specific
  throw "oops";                      // string (not recommended — no stack)
  throw 42;                          // number (not recommended)
  throw { code: 404, msg: "Not Found" }; // object (no stack)

  // Best practice: always throw Error instances (stack trace + name + message)

  // Custom errors:
  class ValidationError extends Error {
    constructor(message, field) {
      super(message);
      this.name = "ValidationError";
      this.field = field;
    }
  }

  class NotFoundError extends Error {
    constructor(resource) {
      super(`${resource} not found`);
      this.name = "NotFoundError";
      this.statusCode = 404;
    }
  }

  function validateAge(age) {
    if (typeof age !== "number") throw new ValidationError("Must be a number", "age");
    if (age < 0 || age > 150) throw new ValidationError("Invalid range", "age");
    return true;
  }

5. CATCHING SPECIFIC ERROR TYPES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  try {
    someOperation();
  } catch (err) {
    if (err instanceof ValidationError) {
      console.log(`Validation failed on field: ${err.field}`);
    } else if (err instanceof TypeError) {
      console.log("Type error:", err.message);
    } else {
      throw err;  // re-throw unknown errors!
    }
  }

  // Re-throwing is important: catch only what you can handle.

6. ERROR HANDLING IN ASYNC CODE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  A) Promises — .catch():
  fetch("https://api.example.com/data")
    .then(res => res.json())
    .then(data => console.log(data))
    .catch(err => console.error("Fetch failed:", err))
    .finally(() => console.log("done"));

  B) async/await — try/catch:
  async function fetchData() {
    try {
      const res = await fetch("https://api.example.com/data");
      if (!res.ok) throw new Error(`HTTP ${res.status}`);
      const data = await res.json();
      return data;
    } catch (err) {
      console.error("Error:", err.message);
      throw err;  // re-throw if caller needs to handle
    }
  }

  C) Promise.all — fails fast:
  try {
    const [a, b] = await Promise.all([fetch(url1), fetch(url2)]);
  } catch (err) {
    // Catches first rejection — others may still be running
  }

  D) Promise.allSettled — no fail-fast:
  const results = await Promise.allSettled([fetch(url1), fetch(url2)]);
  results.forEach(result => {
    if (result.status === "fulfilled") console.log(result.value);
    else console.error(result.reason);
  });

7. GLOBAL ERROR HANDLERS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Browser — uncaught synchronous errors:
  window.onerror = function(message, source, lineno, colno, error) {
    console.error("Uncaught error:", message);
    return true;  // prevents default browser error handling
  };

  // Browser — unhandled promise rejections:
  window.addEventListener("unhandledrejection", (event) => {
    console.error("Unhandled rejection:", event.reason);
    event.preventDefault();  // prevent console error
  });

  // Node.js:
  process.on("uncaughtException", (err) => {
    console.error("Uncaught:", err);
    process.exit(1);  // always exit after uncaught exceptions!
  });

  process.on("unhandledRejection", (reason, promise) => {
    console.error("Unhandled rejection:", reason);
  });

8. ERROR HANDLING BEST PRACTICES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  1. Always throw Error objects (not strings/numbers)
  2. Create custom error classes for domain-specific errors
  3. Only catch errors you can meaningfully handle; re-throw the rest
  4. Never swallow errors silently: catch(e) {} ← BAD
  5. Use finally for cleanup (close connections, release locks)
  6. Always handle promise rejections (.catch or try/catch with await)
  7. Log errors with enough context (message, stack, relevant data)
  8. In production, use error monitoring (Sentry, Datadog)

9. COMMON INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: Does finally always run?
A: Almost always. It runs whether the try succeeded, threw, or returned.
   Exceptions: process.exit(), infinite loop, power failure.
   If both try and finally have return, finally's return wins.

Q: How do you handle errors in async/await?
A: Wrap await calls in try/catch blocks. The rejected promise is converted
   into a thrown exception that catch can handle.

Q: What is the difference between throw new Error("msg") and throw "msg"?
A: Error objects have .name, .message, and .stack (stack trace) which
   are essential for debugging. Throwing a string loses all that info.

Q: How do you create custom error classes?
A: Extend Error: class MyError extends Error { constructor(msg) {
   super(msg); this.name = "MyError"; } }
   Check type with: err instanceof MyError

Q: What happens to an unhandled promise rejection?
A: In browsers: triggers unhandledrejection event, logs warning.
   In Node.js (modern versions): terminates the process by default.
   Always handle promise rejections.

Q: Can you catch a SyntaxError?
A: Not if it's in the file itself (parse-time error, not runtime). But
   JSON.parse throws SyntaxError at runtime, which CAN be caught.

Q: What is the difference between error types (TypeError vs ReferenceError)?
A: ReferenceError: accessing an undeclared variable or TDZ variable.
   TypeError: operating on the wrong type (calling non-function, accessing
   property of null/undefined).
