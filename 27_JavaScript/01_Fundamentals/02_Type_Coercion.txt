1. WHAT IS TYPE COERCION?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Type coercion is JavaScript's automatic (implicit) conversion of values from
one type to another when an operator or operation expects a different type.

  Implicit coercion (automatic):   "5" + 3   → "53"
  Explicit coercion (manual):      Number("5") → 5

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
2. ABSTRACT EQUALITY (==) vs STRICT EQUALITY (===)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  === (Strict): No coercion. Both type AND value must match.
  ==  (Abstract / Loose): Applies Abstract Equality Comparison algorithm.

  The == Algorithm (simplified ECMAScript spec):
  1. If same type → use === comparison
  2. null == undefined → true (only these two)
  3. number == string → convert string to number, then compare
  4. boolean == x → convert boolean to number, then compare
  5. object == string/number/symbol → call ToPrimitive on object

  Classic == traps:
  1 == "1"           // true  (string → number)
  1 == true          // true  (boolean → number: true → 1)
  0 == false         // true  (false → 0)
  0 == ""            // true  ("" → 0)
  0 == "0"           // true  ("0" → 0)
  "" == false        // true  ("" → 0, false → 0)
  null == undefined  // true  (special rule)
  null == 0          // false (null only == undefined)
  null == ""         // false
  NaN == NaN         // false (NaN never equals anything)
  [] == 0            // true  ([] → "" → 0)
  [] == ""           // true  ([] → "")
  [] == false        // true  ([] → "" → 0, false → 0)
  [0] == false       // true  ([0] → "0" → 0, false → 0)
  {} == "[object Object]" // false (but may vary)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
3. ToPrimitive CONVERSION
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

When an object is coerced to a primitive, ToPrimitive is called:
  1. If hint is "number" (most cases): try valueOf(), then toString()
  2. If hint is "string" (template literals, explicit): try toString(), then valueOf()
  3. Arrays: valueOf() returns the array itself (not primitive), 
     so toString() is called: [1,2,3].toString() → "1,2,3"
  4. Objects: {}.toString() → "[object Object]"
             {}.valueOf() → the object itself (not primitive)

  // Custom ToPrimitive via Symbol.toPrimitive:
  const obj = {
    [Symbol.toPrimitive](hint) {
      if (hint === "number") return 42;
      if (hint === "string") return "hello";
      return true; // default
    }
  };
  +obj           // 42 (number hint)
  `${obj}`       // "hello" (string hint)
  obj + ""       // "true" (default hint)

4. COERCION WITH + OPERATOR
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  + is overloaded: numeric addition OR string concatenation.
  If EITHER operand is a string → concatenation.

  1 + 1             // 2
  "1" + 1           // "11"
  1 + "1"           // "11"
  1 + 2 + "3"       // "33"  (1+2=3, then 3+"3"="33")
  "1" + 2 + 3       // "123" ("1"+2="12", "12"+3="123")
  1 + true          // 2    (true → 1)
  1 + false         // 1    (false → 0)
  1 + null          // 1    (null → 0)
  1 + undefined     // NaN  (undefined → NaN)
  1 + []            // "1"  ([] → "" → "1")
  1 + {}            // "1[object Object]"
  {} + []           // 0 in Chrome console ({}parsed as block, +[] = 0)
  [] + {}           // "[object Object]"

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
5. COERCION WITH - * / % OPERATORS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  These operators always try to convert operands to numbers:

  "5" - 3        // 2   (string "5" → 5)
  "5" * 2        // 10
  "6" / "2"      // 3
  true - 1       // 0   (true → 1)
  null - 1       // -1  (null → 0)
  undefined - 1  // NaN (undefined → NaN)
  "abc" - 1      // NaN
  [] - 1         // -1  ([] → "" → 0)
  [3] - 1        // 2   ([3] → "3" → 3)
  [1,2] - 1      // NaN ([1,2] → "1,2" → NaN)

6. EXPLICIT CONVERSION METHODS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  TO NUMBER:
  Number("42")          // 42
  Number("")            // 0
  Number(" ")           // 0
  Number("3.14")        // 3.14
  Number(true)          // 1
  Number(false)         // 0
  Number(null)          // 0
  Number(undefined)     // NaN
  Number("abc")         // NaN
  Number([])            // 0  ([] → "" → 0)
  Number([3])           // 3  ([3] → "3" → 3)
  Number([1,2])         // NaN
  +"42"                 // 42 (unary + shorthand)
  parseInt("42.5")      // 42 (integer part)
  parseFloat("42.5")    // 42.5
  parseInt("0xFF", 16)  // 255 (hex)

  TO STRING:
  String(42)            // "42"
  String(null)          // "null"
  String(undefined)     // "undefined"
  String(true)          // "true"
  (42).toString()       // "42"
  (255).toString(16)    // "ff" (hex)
  (7).toString(2)       // "111" (binary)
  "" + 42               // "42" (concatenation coercion)

  TO BOOLEAN:
  Boolean(0)            // false
  Boolean("")           // false
  Boolean(null)         // false
  Boolean(undefined)    // false
  Boolean(NaN)          // false
  Boolean(false)        // false
  // Everything else → true
  !!0                   // false (double negation shorthand)
  !!""                  // false
  !!"0"                 // true (non-empty string)
  !![]                  // true (all objects)

7. COMPARISON COERCION (<, >, <=, >=)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  If both sides are strings → lexicographic comparison
  Otherwise → convert both to numbers

  "10" > "9"     // false! (lexicographic: "1" < "9")
  10 > 9         // true
  "10" > 9       // true  (string → number)
  null > 0       // false
  null == 0      // false
  null >= 0      // true  (!) ← bizarre edge case
  // Explanation: >= uses numeric comparison (null → 0), 0 >= 0 is true
  //              == uses special null/undefined rule, null only == undefined

8. COMMON INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What is the output of [] + [] and [] + {}?
A: [] + [] → "" (both arrays convert to empty string "")
   [] + {} → "[object Object]" ([] → "", {} → "[object Object]")
   {} + [] → 0 when {} is interpreted as an empty block; +[] = 0
           → "[object Object]" when {} is an object expression

Q: What is the result of false == ""?
A: true. false → 0, "" → 0, so 0 == 0 → true.

Q: Why does null >= 0 return true but null == 0 return false?
A: >= uses numeric coercion: null → 0, so 0 >= 0 is true.
   == applies the Abstract Equality Algorithm which has a special rule:
   null only equals null or undefined, not 0.

Q: What is the difference between implicit and explicit coercion?
A: Implicit: JS engine converts types automatically (e.g., "1" + 2 = "12").
   Explicit: Developer converts manually (e.g., Number("1") + 2 = 3).

Q: What does !!"" evaluate to?
A: false. "" is falsy, !("") = true, !!("") = false.

Q: What is the result of 1 + 2 + "3" vs "1" + 2 + 3?
A: 1 + 2 + "3" → 3 + "3" → "33" (numeric first, then string concat)
   "1" + 2 + 3  → "12" + 3 → "123" (string concat from left)

Q: When should you use == vs ===?
A: Almost always use ===. Only useful == case: null check:
   value == null  // true for both null AND undefined (intentional loose check)
   Equivalent to: value === null || value === undefined
