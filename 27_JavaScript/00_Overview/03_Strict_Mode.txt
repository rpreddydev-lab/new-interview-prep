1. WHAT IS STRICT MODE?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Strict mode was introduced in ES5 (2009). It opts into a restricted variant
of JavaScript that:
  - Catches common coding mistakes and throws errors for them
  - Prevents use of some potentially problematic language features
  - Makes it easier for JS engines to perform optimizations

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
2. HOW TO ENABLE STRICT MODE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  a) Global scope (top of file):
     "use strict";
     // entire script is in strict mode

  b) Function scope:
     function strictFn() {
       "use strict";
       // only this function is strict
     }

  c) ES6 Modules:
     // Modules are ALWAYS in strict mode, no directive needed
     // <script type="module"> → automatically strict

  d) Classes:
     // Class bodies are ALWAYS in strict mode

Note: Cannot un-strict a module or class. "use strict" is a directive
      prologue — must be the very first statement.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
3. WHAT STRICT MODE DISALLOWS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

A) UNDECLARED VARIABLES
  // Non-strict: creates global variable (silent bug)
  x = 5;

  // Strict: throws ReferenceError
  "use strict";
  x = 5;  // ReferenceError: x is not defined

B) DELETING VARIABLES, FUNCTIONS, OR ARGUMENTS
  "use strict";
  let y = 5;
  delete y;        // SyntaxError: Delete of an unqualified identifier

C) DUPLICATE PARAMETER NAMES
  // Non-strict: last value wins silently
  function dup(a, a) { return a; }  // ok but confusing

  // Strict: syntax error
  "use strict";
  function dup(a, a) { ... }  // SyntaxError

D) WRITING TO NON-WRITABLE PROPERTIES
  "use strict";
  const obj = {};
  Object.defineProperty(obj, "x", { value: 1, writable: false });
  obj.x = 2;  // TypeError: Cannot assign to read-only property

  // Non-strict: silently fails

E) WRITING TO GETTER-ONLY PROPERTIES
  "use strict";
  const obj = { get x() { return 1; } };
  obj.x = 2;  // TypeError

F) WRITING TO NON-EXTENSIBLE OBJECTS
  "use strict";
  const obj = Object.preventExtensions({});
  obj.newProp = "hi";  // TypeError

G) DELETING NON-DELETABLE PROPERTIES
  "use strict";
  delete Object.prototype;  // TypeError

H) WITH STATEMENT (forbidden entirely)
  // Non-strict: allowed but causes performance & scope confusion
  with (obj) { ... }

  // Strict: SyntaxError
  "use strict";
  with (obj) { ... }  // SyntaxError: Strict mode code may not include...

I) `arguments` OBJECT CHANGES
  // Non-strict: arguments tracks parameter value changes
  function test(a) {
    a = 2;
    console.log(arguments[0]); // 2 (non-strict)
  }

  // Strict: arguments is a static snapshot
  "use strict";
  function test(a) {
    a = 2;
    console.log(arguments[0]); // original value passed (strict)
  }

J) `this` IN REGULAR FUNCTIONS
  // Non-strict: this === window (browser) or global (Node)
  function fn() { console.log(this); }
  fn();  // Window {}

  // Strict: this === undefined
  "use strict";
  function fn() { console.log(this); }
  fn();  // undefined

K) OCTAL NUMERIC LITERALS
  "use strict";
  const n = 0777;  // SyntaxError (use 0o777 instead — ES6 octal syntax)

L) RESERVED WORDS AS IDENTIFIERS
  "use strict";
  // Can't use: implements, interface, let, package, private, protected,
  //            public, static, yield as variable names
  let static = 1;  // SyntaxError

4. WHY USE STRICT MODE?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  1. Catches bugs early (typos become errors, not silent global variables)
  2. Prevents unsafe actions (eval/arguments safer)
  3. JS engines can optimize better (no with, stable bindings)
  4. Prepares code for future ECMAScript versions
  5. Required groundwork for ES6+ modules (always strict)

5. STRICT MODE & eval
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  In strict mode, eval() does NOT introduce variables into surrounding scope:

  "use strict";
  eval("var x = 2");
  console.log(typeof x);  // "undefined" (x is scoped to eval block)

  // Non-strict:
  eval("var x = 2");
  console.log(x);  // 2 (leaks into surrounding scope — dangerous!)

6. STRICT MODE IN CLASSES & MODULES (AUTOMATIC)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // module.js — no "use strict" needed:
  export function greet(name) {
    undeclaredVar = "bug";  // ReferenceError automatically
  }

  // class — automatically strict:
  class MyClass {
    method() {
      undeclaredVar = "bug";  // ReferenceError automatically
    }
  }

7. COMMON INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What is strict mode in JavaScript?
A: "use strict" opts a script/function into a restricted subset of JS. It
   makes previously silent errors throw exceptions, prevents certain unsafe
   syntax, and helps JS engines optimize code.

Q: How does `this` behave differently in strict mode?
A: In non-strict functions called without a receiver, `this` defaults to the
   global object (window/global). In strict mode, `this` is undefined.
   This prevents accidental global object mutation.

Q: Are ES6 modules automatically in strict mode?
A: Yes. Any file loaded as an ES6 module (<script type="module"> or .mjs)
   is implicitly in strict mode. Same for class bodies.

Q: What happens when you accidentally omit 'var' in non-strict mode?
A: Without "use strict", assigning to an undeclared variable creates an
   implicit global variable — a major source of bugs. In strict mode,
   this throws a ReferenceError immediately.

Q: Can strict mode be applied partially?
A: Yes, per-function. A non-strict script can contain strict functions.
   This enables incremental adoption in legacy codebases.
   
   var legacy = function() { /* non-strict */ };
   var modern  = function() { "use strict"; /* strict */ };

Q: Does strict mode affect performance?
A: Yes, positively. Strict mode removes certain dynamic behaviors (like
   with statement, arguments aliasing) that prevent V8 from making
   static analysis optimizations.
