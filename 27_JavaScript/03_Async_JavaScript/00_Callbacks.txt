1. WHAT IS A CALLBACK?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  A callback is a function passed as an argument to another function, which
  invokes it at some point in the future. This is the foundational pattern
  for async programming in JavaScript.

  function greet(name, callback) {
    console.log("Hello, " + name);
    callback();
  }
  greet("Alice", () => console.log("Callback called!"));

2. SYNCHRONOUS CALLBACKS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Synchronous callbacks run immediately within the calling function:

  [1, 2, 3].forEach(n => console.log(n));  // runs synchronously
  [1, 2, 3].map(n => n * 2);              // synchronous
  [1, 2, 3].filter(n => n > 1);           // synchronous
  [1, 2, 3].sort((a, b) => a - b);        // synchronous

3. ASYNCHRONOUS CALLBACKS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Async callbacks are invoked LATER, after an async operation completes:

  // setTimeout / setInterval:
  setTimeout(() => console.log("2 seconds later"), 2000);

  // DOM events:
  document.getElementById("btn").addEventListener("click", () => {
    console.log("clicked!");
  });

  // Node.js file system:
  const fs = require("fs");
  fs.readFile("file.txt", "utf8", (err, data) => {
    if (err) { console.error(err); return; }
    console.log(data);
  });

4. THE NODE.JS ERROR-FIRST CALLBACK CONVENTION
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Standard Node.js pattern: first argument is error (null if success):

  function readData(id, callback) {
    db.find(id, (err, data) => {
      if (err) return callback(err);
      callback(null, data);    // null = no error
    });
  }

  readData(1, (err, data) => {
    if (err) return console.error("Error:", err);
    console.log("Data:", data);
  });

5. CALLBACK HELL (PYRAMID OF DOOM)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Deep nesting when dependent async operations are chained:

  getUserById(1, (err, user) => {
    if (err) return handleError(err);
    getPostsByUser(user.id, (err, posts) => {
      if (err) return handleError(err);
      getCommentsByPost(posts[0].id, (err, comments) => {
        if (err) return handleError(err);
        getAuthorByComment(comments[0].id, (err, author) => {
          if (err) return handleError(err);
          console.log(author);  // finally!
        });
      });
    });
  });

  Problems:
  - Hard to read and maintain
  - Error handling is repetitive and verbose
  - Difficult to run operations in parallel
  - Hard to debug and test

  Solutions:
  1. Named functions (flatten the pyramid):
  function handleAuthor(err, author) { if (err) return handleError(err); console.log(author); }
  function handleComments(err, comments) { if (err) return handleError(err); getAuthorByComment(comments[0].id, handleAuthor); }
  function handlePosts(err, posts) { if (err) return handleError(err); getCommentsByPost(posts[0].id, handleComments); }
  function handleUser(err, user) { if (err) return handleError(err); getPostsByUser(user.id, handlePosts); }
  getUserById(1, handleUser);

  2. Promises (chain .then())
  3. async/await (modern, cleanest)

6. PROMISIFYING CALLBACKS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Convert callback-based functions to Promise-based:

  // Manual promisify:
  function readFilePromise(path, encoding) {
    return new Promise((resolve, reject) => {
      fs.readFile(path, encoding, (err, data) => {
        if (err) reject(err);
        else resolve(data);
      });
    });
  }

  // Node.js built-in util.promisify:
  const { promisify } = require("util");
  const readFile = promisify(fs.readFile);
  const data = await readFile("file.txt", "utf8");

7. COMMON INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What is a callback function?
A: A function passed as an argument to another function, meant to be
   called back at some point. Can be synchronous (called immediately like
   in array.map()) or asynchronous (called after I/O, timeout, event).

Q: What is callback hell and how do you solve it?
A: Callback hell is deeply nested callbacks making code hard to read/maintain.
   Solutions: named functions (flatten nesting), Promises (chain .then()),
   async/await (write async code synchronously).

Q: What is the error-first callback convention?
A: A Node.js convention where the first parameter of a callback is reserved
   for an error (null if no error). Callers must check err first before
   using data.

Q: What is the difference between synchronous and asynchronous callbacks?
A: Synchronous callbacks execute immediately within the calling function
   (like Array.map callbacks). Asynchronous callbacks are queued and
   called later, after an async operation completes (setTimeout, I/O, events).
