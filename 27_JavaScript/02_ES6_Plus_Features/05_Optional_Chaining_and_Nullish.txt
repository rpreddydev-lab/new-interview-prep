1. OPTIONAL CHAINING (?.)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Optional chaining (?.) short-circuits and returns undefined if the
  left side is null or undefined, instead of throwing a TypeError.

  // Without optional chaining — defensive, verbose:
  const city = user && user.address && user.address.city;

  // With optional chaining — clean:
  const city = user?.address?.city;

  // All forms:
  obj?.property            // property access
  obj?.["property"]        // bracket notation
  arr?.[0]                 // array element
  fn?.()                   // function/method call
  obj?.method?.()          // method that might not exist

  // Examples:
  const user = { profile: { name: "Alice" } };
  user?.profile?.name       // "Alice"
  user?.settings?.theme     // undefined (no settings, no throw)
  user?.greet?.()           // undefined (greet doesn't exist, no throw)

  const nullUser = null;
  nullUser?.name             // undefined (not TypeError!)

  // With arrays:
  const arr = null;
  arr?.[0]                   // undefined (not TypeError)
  arr?.map(x => x * 2)       // undefined

  // With function calls:
  const callback = null;
  callback?.()               // undefined (not TypeError)

  // Optional chaining in assignment (cannot use on left side):
  obj?.prop = value;         // SyntaxError — can't assign through ?.

2. SHORT-CIRCUIT EVALUATION
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // If any part is null/undefined, the entire chain returns undefined:
  const user = null;
  user?.address?.city?.toLowerCase();  // undefined (doesn't throw)

  // Side effects after ?. are skipped:
  let x = 0;
  null?.foo(x++);  // x is still 0 — x++ not evaluated due to short-circuit

3. NULLISH COALESCING (??)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  ?? returns the RIGHT side only when the left is null OR undefined.
  (Unlike || which returns right side for ANY falsy value)

  null ?? "default"        // "default"
  undefined ?? "default"   // "default"
  0 ?? "default"           // 0      ← 0 is NOT null/undefined
  "" ?? "default"          // ""     ← empty string is NOT null/undefined
  false ?? "default"       // false  ← false is NOT null/undefined
  NaN ?? "default"         // NaN    ← NaN is NOT null/undefined

  // Compare with ||:
  null || "default"        // "default"
  undefined || "default"   // "default"
  0 || "default"           // "default" ← 0 is falsy, returns right!
  "" || "default"          // "default" ← "" is falsy, returns right!
  false || "default"       // "default" ← false is falsy, returns right!

  // Real-world impact:
  function greet(name) {
    return `Hello, ${name || "Guest"}`;  // WRONG: greet("") → "Hello, Guest"
  }
  function greet2(name) {
    return `Hello, ${name ?? "Guest"}`;  // CORRECT: greet("") → "Hello, "
  }

  function getPort(config) {
    return config.port || 3000;   // WRONG: port=0 → 3000 (falsy 0 overridden!)
    return config.port ?? 3000;   // CORRECT: port=0 → 0 (zero is valid)
  }

4. COMBINING ?. AND ??
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  const user = null;

  // Chain with fallback:
  const city = user?.address?.city ?? "Unknown";  // "Unknown"

  // Method with fallback:
  const name = user?.profile?.getName?.() ?? "Guest";

  // Config access:
  const timeout = config?.network?.timeout ?? 5000;

5. NULLISH ASSIGNMENT (??=) — ES2021
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // ??= assigns only if the left side is null/undefined:
  let a = null;
  a ??= "default";
  a;  // "default"

  let b = 0;
  b ??= "default";
  b;  // 0 (not null/undefined, not assigned)

  // Related logical assignments:
  a ||= "fallback";  // assigns if a is falsy
  a &&= "newval";    // assigns if a is truthy

6. COMMON INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What is optional chaining and why was it introduced?
A: ?. short-circuits property/method access when the left side is null
   or undefined, returning undefined instead of throwing TypeError. It
   replaces verbose guard chains like a && a.b && a.b.c.

Q: What is the difference between ?? and ||?
A: || returns the right side for ANY falsy value (0, "", false, NaN, null, undefined).
   ?? returns the right side ONLY for null and undefined.
   Use ?? when 0, "", or false are valid values you want to preserve.

Q: What does user?.greet?.() return if greet exists but throws?
A: It THROWS. ?. only protects against null/undefined — if the method
   exists but throws during execution, the error propagates normally.

Q: Can you use ?. on the left side of assignment?
A: No. obj?.prop = value throws SyntaxError. Optional chaining only works
   for reads, not for assignments.

Q: What is ??= (nullish assignment)?
A: Assigns a value only if the left side is null or undefined.
   a ??= "default" is shorthand for a = a ?? "default".
