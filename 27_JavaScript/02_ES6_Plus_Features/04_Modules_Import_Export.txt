1. WHY MODULES?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Before ES6 modules, all scripts shared global scope → name collisions,
  pollution, dependency management nightmares.

  ES6 Modules (ESM) solve:
  - Encapsulation (module-level scope)
  - Explicit dependencies (import declares what you need)
  - Static analysis (tree shaking, bundling optimizations)
  - Strict mode by default
  - Single instance (modules evaluated once and cached)

2. NAMED EXPORTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // math.js — named exports:
  export const PI = 3.14159;
  export function add(a, b) { return a + b; }
  export class Calculator { ... }

  // Or export at the end:
  const PI = 3.14159;
  function add(a, b) { return a + b; }
  export { PI, add };

  // Export with rename:
  export { add as sum, PI as pi };

  // Named imports:
  import { PI, add } from "./math.js";
  import { PI as pi, add as sum } from "./math.js";  // rename on import

  // Import all named exports as namespace:
  import * as Math from "./math.js";
  Math.PI;    // 3.14159
  Math.add(1, 2);  // 3

3. DEFAULT EXPORT
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Only ONE default export per module:
  // calculator.js:
  export default class Calculator {
    add(a, b) { return a + b; }
  }

  // Or:
  function multiply(a, b) { return a * b; }
  export default multiply;

  // Default import — no curly braces, any name you choose:
  import Calculator from "./calculator.js";
  import multiply from "./calculator.js";
  import MyCalc from "./calculator.js";    // can name it anything

4. MIXING DEFAULT AND NAMED EXPORTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // utils.js:
  export default function main() { ... }
  export const helper1 = () => { ... };
  export const helper2 = () => { ... };

  // Import both:
  import main, { helper1, helper2 } from "./utils.js";

5. RE-EXPORTING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // index.js — barrel file pattern:
  export { add, subtract } from "./math.js";
  export { default as Calculator } from "./calculator.js";
  export * from "./utils.js";                  // re-export all named
  export * as utils from "./utils.js";         // re-export as namespace

  // Then consumers import from one place:
  import { add, Calculator, utils } from "./index.js";

6. DYNAMIC IMPORTS (ES2020)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // import() returns a Promise — enables code splitting & lazy loading:
  async function loadModule() {
    const { add } = await import("./math.js");
    console.log(add(1, 2));
  }

  // Conditional import:
  if (user.isAdmin) {
    const { AdminPanel } = await import("./adminPanel.js");
    AdminPanel.show();
  }

  // Dynamic path:
  const lang = "fr";
  const translations = await import(`./locales/${lang}.js`);

7. MODULE CHARACTERISTICS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  1. Strict mode by default (no "use strict" needed)
  2. Module-level scope (variables don't pollute global)
  3. Executed once — modules are cached after first load
     (importing same module from two places → same instance)
  4. Import bindings are LIVE (reflect changes to exported values)
  5. Static imports must be at top level (not inside if/try/function)
     Dynamic import() can be anywhere
  6. this at module top level is undefined (not window)

  // Live bindings:
  // counter.js:
  export let count = 0;
  export function increment() { count++; }

  // app.js:
  import { count, increment } from "./counter.js";
  count;        // 0
  increment();
  count;        // 1 — live binding! reflects the change

8. ESM vs CommonJS (Quick Comparison)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  ┌─────────────────────┬────────────────────────┬──────────────────────────┐
  │                     │ ESM (import/export)    │ CommonJS (require)       │
  ├─────────────────────┼────────────────────────┼──────────────────────────┤
  │ Syntax              │ import/export          │ require/module.exports   │
  ├─────────────────────┼────────────────────────┼──────────────────────────┤
  │ Load time           │ Static (compile time)  │ Dynamic (runtime)        │
  ├─────────────────────┼────────────────────────┼──────────────────────────┤
  │ Asynchronous        │ Yes (in browsers)      │ No (synchronous)         │
  ├─────────────────────┼────────────────────────┼──────────────────────────┤
  │ Tree shaking        │ Yes (static analysis)  │ No (dynamic)             │
  ├─────────────────────┼────────────────────────┼──────────────────────────┤
  │ Bindings            │ Live                   │ Copied (snapshot)        │
  ├─────────────────────┼────────────────────────┼──────────────────────────┤
  │ Conditional import  │ import() only          │ require() anywhere       │
  ├─────────────────────┼────────────────────────┼──────────────────────────┤
  │ this at top level   │ undefined              │ module.exports           │
  ├─────────────────────┼────────────────────────┼──────────────────────────┤
  │ Used in             │ Browser + Node (.mjs)  │ Node.js (legacy)         │
  └─────────────────────┴────────────────────────┴──────────────────────────┘

9. COMMON INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What is the difference between named and default exports?
A: Named: multiple per module, must use exact name (or alias with as).
   Default: one per module, imported without curly braces with any name.

Q: What is a barrel file?
A: An index.js that re-exports from multiple modules, providing a single
   entry point for consumers. Reduces import paths complexity.

Q: What makes ESM imports static?
A: import statements must be at the top level and cannot be conditional.
   This allows bundlers to analyze dependencies at build time and do tree
   shaking. Use dynamic import() for runtime conditional loading.

Q: What are live bindings in ES modules?
A: Imported variables are live read-only views of the exported bindings.
   When the exporting module changes the value, importers see the update.
   CommonJS provides a copy of the value at import time.

Q: What is dynamic import()?
A: import() is a function-like syntax that returns a Promise resolving to
   the module namespace. Enables code splitting (lazy loading) and
   conditional imports. Works at runtime, unlike static import.

Q: Are ES6 modules in strict mode?
A: Yes, always. No "use strict" directive needed. Also, `this` at the top
   level of a module is undefined (not the global object).
