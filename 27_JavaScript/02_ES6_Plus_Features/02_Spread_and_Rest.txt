1. REST vs SPREAD — OVERVIEW
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Both use the ... syntax but do OPPOSITE things:

  REST    – COLLECTS multiple elements INTO an array/object
            Used in: function parameters, destructuring

  SPREAD  – EXPANDS an iterable/object OUT into individual elements
            Used in: function calls, array literals, object literals

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
2. REST PARAMETERS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Collect all remaining arguments into an array:
  function sum(...nums) {
    return nums.reduce((a, b) => a + b, 0);
  }
  sum(1, 2, 3, 4);   // 10

  // Collect "the rest" after named params:
  function log(level, ...messages) {
    console.log(`[${level}]`, ...messages);
  }
  log("INFO", "server", "started", "on port 3000");
  // [INFO] server started on port 3000

  // Rules:
  // - Must be the LAST parameter
  // - Only ONE rest parameter per function
  // - Creates a real array (unlike arguments object)

  // Rest in destructuring:
  const [first, ...others] = [1, 2, 3, 4];
  first;   // 1
  others;  // [2, 3, 4]

  const { a, ...remaining } = { a: 1, b: 2, c: 3 };
  a;         // 1
  remaining; // { b: 2, c: 3 }

3. SPREAD OPERATOR — ARRAYS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  const a = [1, 2, 3];
  const b = [4, 5, 6];

  // Combine arrays:
  const combined = [...a, ...b];        // [1,2,3,4,5,6]

  // Copy array (shallow):
  const copy = [...a];                   // [1,2,3] (new array)
  copy === a;                            // false — different reference

  // Insert in middle:
  const extended = [...a, 10, ...b];    // [1,2,3,10,4,5,6]

  // Spread into function arguments:
  Math.max(...a);     // 3
  Math.max(...[5, 3, 8, 1]); // 8

  // Convert iterable to array:
  const str = "hello";
  [...str];           // ["h","e","l","l","o"]
  [...new Set([1,2,1,3])]; // [1,2,3]
  [...new Map([["a",1]])];  // [["a",1]]
  [...document.querySelectorAll("div")]; // Array from NodeList

4. SPREAD OPERATOR — OBJECTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  const defaults = { theme: "light", lang: "en", fontSize: 14 };
  const userPrefs = { theme: "dark", fontSize: 16 };

  // Merge objects (later properties override earlier):
  const config = { ...defaults, ...userPrefs };
  // { theme: "dark", lang: "en", fontSize: 16 }

  // Shallow copy:
  const copy = { ...defaults };  // new object
  copy === defaults;              // false

  // Override specific properties:
  const updated = { ...user, age: 31, role: "admin" };

  // Remove a property (without mutation):
  const { password, ...safeUser } = user;
  // safeUser has everything except password

  // Add computed key:
  const key = "dynamicProp";
  const obj = { ...base, [key]: "value" };

  // Spread only works on enumerable own properties:
  const proto = { inherited: true };
  const child = Object.create(proto);
  child.own = true;
  { ...child };   // { own: true } — inherited NOT included

5. SPREAD IS SHALLOW COPY
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  const original = { a: 1, nested: { b: 2 } };
  const copy = { ...original };

  copy.a = 99;
  original.a;         // 1 — primitive: not affected

  copy.nested.b = 99;
  original.nested.b;  // 99 — nested object is still shared reference!

  // Deep copy:
  const deepCopy = structuredClone(original);  // ES2022
  const deepCopy2 = JSON.parse(JSON.stringify(original));  // limited

6. PRACTICAL PATTERNS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Immutable array updates (for React/Redux):
  const addItem = (arr, item) => [...arr, item];
  const removeItem = (arr, i) => [...arr.slice(0, i), ...arr.slice(i + 1)];
  const updateItem = (arr, i, val) =>
    arr.map((item, idx) => idx === i ? val : item);

  // Immutable object updates:
  const updateUser = (user, changes) => ({ ...user, ...changes });
  const deleteKey = (obj, key) => {
    const { [key]: _, ...rest } = obj;
    return rest;
  };

  // Merge with default values:
  function createUser(overrides = {}) {
    return {
      name: "Guest",
      role: "user",
      active: true,
      ...overrides
    };
  }
  createUser({ name: "Alice", role: "admin" });

  // Spread with apply (legacy style):
  function add(a, b) { return a + b; }
  const args = [1, 2];
  add.apply(null, args);  // old way
  add(...args);           // modern way

7. COMMON INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What is the difference between rest and spread?
A: Same syntax (...) but opposite purpose. Rest collects multiple elements
   into a single array/object (used in params/destructuring). Spread
   expands an iterable/object into individual elements (used in calls/literals).

Q: How does spread differ from Object.assign?
A: Spread ({ ...obj }) and Object.assign({}, obj) both do shallow copies of
   own enumerable properties. Difference: Object.assign calls setters on
   the target; spread does not. Spread is more readable.

Q: What happens when two spread objects have the same key?
A: The LAST one wins:
   { ...{ a: 1 }, ...{ a: 2 } }  →  { a: 2 }

Q: Can you spread non-iterable objects?
A: In function calls and arrays, you can only spread iterables (array, string,
   Set, Map, generators). Spreading plain objects only works in object literals.
   Spreading a non-iterable in an array/call throws TypeError.

Q: What does rest parameter return vs arguments?
A: Rest returns a real Array with all array methods. arguments is array-like
   (has length/indices but no .map/.filter). Also, arrow functions have no
   arguments but DO support rest params.

Q: How do you clone an object immutably?
A: const clone = { ...obj };    // shallow (ES6)
   const deep = structuredClone(obj);  // deep (ES2022)
