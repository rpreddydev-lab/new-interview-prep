  // Use backticks `` instead of quotes:
  const name = "Alice";
  const greeting = `Hello, ${name}!`;   // "Hello, Alice!"

  // Before ES6 (string concatenation):
  const old = "Hello, " + name + "!";

  // Expressions in ${}: any valid JS expression works
  const a = 5, b = 10;
  `Sum: ${a + b}`;           // "Sum: 15"
  `${a > b ? "a" : "b"}`;   // "b"
  `${name.toUpperCase()}`;   // "ALICE"

4. TAGGED TEMPLATE LITERALS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Tagged templates let you process a template literal with a function.
  The tag function receives the raw strings and interpolated values.

  function tag(strings, ...values) {
    console.log(strings);  // array of string parts
    console.log(values);   // array of interpolated values
    return strings.reduce((result, str, i) => {
      return result + str + (values[i] !== undefined ? values[i] : "");
    }, "");
  }

  const name = "Alice";
  const age = 30;
  tag`Hello ${name}, you are ${age} years old`;
  // strings: ["Hello ", ", you are ", " years old"]
  // values: ["Alice", 30]

  // Real-world use case: SQL sanitization:
  function sql(strings, ...values) {
    return {
      text: strings.reduce((q, s, i) => q + s + (values[i] !== undefined ? `$${i}` : ""), ""),
      values
    };
  }
  const query = sql`SELECT * FROM users WHERE id = ${userId}`;
  // Prevents SQL injection

  // Real-world: CSS-in-JS (styled-components):
  const Button = styled.button`
    background: ${props => props.primary ? "blue" : "white"};
    color: ${props => props.primary ? "white" : "blue"};
  `;

  // Highlighting/escaping:
  function highlight(strings, ...values) {
    return strings.reduce((result, str, i) => {
      const value = values[i] !== undefined ? `<mark>${values[i]}</mark>` : "";
      return result + str + value;
    }, "");
  }
  highlight`Hello ${name}, welcome!`;
  // "Hello <mark>Alice</mark>, welcome!"

5. String.raw TAG
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // String.raw — built-in tag that returns raw string (no escape processing):
  String.raw`Hello\nWorld`;    // "Hello\\nWorld" (literal backslash + n)
  `Hello\nWorld`;              // "Hello\nWorld"  (actual newline)

  // Useful for Windows paths and regex:
  const path = String.raw`C:\Users\Alice\Documents`;  // no double backslash needed
  const pattern = String.raw`\d+\.\d+`;               // regex pattern

Q: What are template literals and how are they different from regular strings?
A: Template literals use backticks and support embedded expressions (${...}),
   multi-line strings natively, and tagged template processing. Regular
   strings use quotes and require concatenation + explicit \n for newlines.

Q: What is a tagged template literal?
A: A tagged template applies a function to the template. The tag function
   receives an array of string parts and the interpolated values separately,
   allowing custom processing (sanitization, formatting, localization).

Q: What does String.raw do?
A: It's a built-in tag that returns the raw string without processing
   escape sequences. String.raw`\n` returns "\n" (two chars), not a newline.

Q: Can you put any expression in ${}?
A: Yes — any valid JS expression: variables, function calls, ternaries,
   arithmetic, method calls, even nested template literals.

Q: What is the difference between ${} and ${}?
A: In template literals, ${expr} evaluates expr and calls .toString() on
   the result. Objects: "[object Object]", null: "null", undefined: "undefined".
