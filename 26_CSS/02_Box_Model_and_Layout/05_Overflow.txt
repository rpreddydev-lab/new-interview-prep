CSS OVERFLOW — In-Depth Interview Preparation Guide
================================================================================

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. WHAT IS OVERFLOW?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Overflow occurs when an element's content is larger than its box (exceeds
its defined width/height). The overflow property controls what happens to
that excess content.

  overflow: visible;  /* default — content overflows visibly outside box */
  overflow: hidden;   /* content is clipped — excess hidden */
  overflow: scroll;   /* always show scrollbars (even if not needed) */
  overflow: auto;     /* show scrollbars only when content overflows */
  overflow: clip;     /* clips like hidden, but does NOT create BFC */

overflow is shorthand for overflow-x and overflow-y:
  overflow: hidden auto;  /* overflow-x: hidden; overflow-y: auto */
  overflow-x: hidden;
  overflow-y: auto;

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
2. overflow: visible (DEFAULT)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  • Content renders outside the box
  • Does NOT create a BFC
  • Useful for dropdowns, tooltips that need to overflow their container

  .container {
    width: 200px;
    height: 100px;
    overflow: visible;  /* default */
    border: 1px solid;
  }
  /* Content that exceeds 200×100 still renders outside the box */

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
3. overflow: hidden
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  • Content is clipped at the box boundary
  • Creates a BFC (Block Formatting Context)
    → Contains floats (clearfix alternative)
    → Prevents margin collapse
  • No scrollbars
  • Clipping can hide things you want visible (dropdowns, tooltips)

  .card {
    overflow: hidden;         /* clips content, creates BFC */
    border-radius: 8px;       /* pairs with overflow:hidden to clip child images */
  }

  .image-crop {
    width: 200px;
    height: 200px;
    overflow: hidden;
  }
  .image-crop img {
    width: 100%;
    height: 100%;
    object-fit: cover;
  }

  /* Classic use: contain floats */
  .clearfix { overflow: hidden; }

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
4. overflow: scroll
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  • ALWAYS shows scrollbars (even if content fits)
  • Creates a BFC
  • Causes layout jumps when content changes (scrollbar appears/disappears)
  • Useful when you always want scrollbars visible for consistent layout

  .code-block {
    overflow: scroll;
    max-height: 400px;
  }

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
5. overflow: auto
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  • Shows scrollbars ONLY when needed (content overflows)
  • Creates a BFC
  • Preferred over scroll for most use cases

  .chat-messages {
    overflow-y: auto;
    max-height: 400px;
  }

  .horizontal-scroll {
    overflow-x: auto;
    white-space: nowrap;  /* prevents wrapping so horizontal scroll works */
  }

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
6. overflow: clip (CSS 2021+)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  • Clips content like hidden
  • Does NOT create a BFC (unlike hidden)
  • Does NOT allow programmatic scrolling
  • overflow-clip-margin: 10px — allows content to overflow by 10px before clipping

  .no-bfc-clip { overflow: clip; }
  /* Use when you want clipping behavior without BFC side effects */

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
7. TEXT OVERFLOW
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

text-overflow controls how overflowing text is displayed when it doesn't fit.
REQUIRES overflow: hidden (or scroll/auto) AND white-space: nowrap to work.

  text-overflow: clip;      /* default — text is cut off */
  text-overflow: ellipsis;  /* shows "..." at the end */
  text-overflow: "...";     /* custom string (limited browser support) */

Single-line text truncation:
  .truncate {
    white-space: nowrap;      /* 1. prevent wrapping */
    overflow: hidden;         /* 2. hide overflow */
    text-overflow: ellipsis;  /* 3. show ... */
    /* Usually combined with a max-width: */
    max-width: 200px;
  }

Multi-line text truncation (line-clamp):
  .line-clamp-3 {
    display: -webkit-box;
    -webkit-line-clamp: 3;          /* show max 3 lines */
    -webkit-box-orient: vertical;
    overflow: hidden;
  }

  /* Modern (still needs -webkit- prefix for broad support as of 2024): */
  .line-clamp-3 {
    overflow: hidden;
    display: -webkit-box;
    -webkit-box-orient: vertical;
    -webkit-line-clamp: 3;
    line-clamp: 3;
  }

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
8. SCROLL BEHAVIOR AND SNAPPING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Smooth scrolling:
  html { scroll-behavior: smooth; }
  /* Anchor links and JS scrollIntoView will animate smoothly */

  /* Respect user motion preferences: */
  @media (prefers-reduced-motion: no-preference) {
    html { scroll-behavior: smooth; }
  }

CSS Scroll Snapping:
  Snaps scroll position to defined points.

  /* Parent container */
  .carousel {
    scroll-snap-type: x mandatory;
    /* x = horizontal axis; y = vertical */
    /* mandatory = always snap; proximity = snap if close */
    overflow-x: auto;
    display: flex;
  }

  /* Each child */
  .slide {
    scroll-snap-align: start;  /* snap to start | center | end */
    flex: 0 0 100%;            /* each slide = 100% width */
    scroll-snap-stop: always;  /* don't skip snaps */
  }

  Full-page vertical snap:
  .sections-container {
    scroll-snap-type: y mandatory;
    overflow-y: scroll;
    height: 100vh;
  }
  .section {
    scroll-snap-align: start;
    height: 100vh;
  }

9. OVERFLOW AND BFC SUMMARY
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

overflow value | Creates BFC | Scrollable | Clipped
───────────────┼─────────────┼────────────┼────────
visible        | NO          | NO         | NO
hidden         | YES         | NO         | YES
scroll         | YES         | YES        | YES
auto           | YES         | When needed| YES
clip           | NO          | NO         | YES

10. KEY INTERVIEW QUESTIONS — OVERFLOW
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What are the values of overflow and what do they do?
A: visible (default): content renders outside the box.
   hidden: clips content, creates BFC.
   scroll: always shows scrollbars, creates BFC.
   auto: shows scrollbars only when content overflows, creates BFC.
   clip: clips content like hidden but does NOT create a BFC.

Q: How do you truncate text with an ellipsis?
A: Three CSS properties are required together:
   white-space: nowrap;      (prevent text wrapping)
   overflow: hidden;         (hide overflow)
   text-overflow: ellipsis;  (show "..." at cut-off point)
   Usually combined with max-width or width.

Q: How do you clamp multi-line text?
A: Use -webkit-line-clamp:
   display: -webkit-box;
   -webkit-box-orient: vertical;
   -webkit-line-clamp: 3;
   overflow: hidden;
   This shows a maximum of 3 lines with "..." at the end. Still needs -webkit-
   prefix for full browser support as of 2024.

Q: What is scroll snapping in CSS?
A: scroll-snap-type on the container and scroll-snap-align on the children
   make the scroll position snap to defined points. Useful for carousels,
   full-page sections, and image galleries. Values: x/y mandatory/proximity.

Q: Does overflow: hidden affect fixed-positioned children?
A: No. A fixed element is positioned relative to the viewport, so overflow:hidden
   on an ancestor does NOT clip it. However, if the ancestor has transform/filter/
   perspective, then the fixed child is re-parented to that element and WILL be clipped.

Q: What is the difference between overflow: hidden and overflow: clip?
A: Both clip overflowing content. The key difference: overflow:hidden creates a
   Block Formatting Context (BFC) which contains floats and prevents margin collapse.
   overflow:clip does NOT create a BFC — it's purely visual clipping without layout
   side effects.
