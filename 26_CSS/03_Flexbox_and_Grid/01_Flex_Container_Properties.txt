COMPLETE FLEX CONTAINER PROPERTIES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  display: flex | inline-flex
  flex-direction: row | row-reverse | column | column-reverse
  flex-wrap: nowrap | wrap | wrap-reverse
  flex-flow: <direction> <wrap>     (shorthand)
  justify-content: flex-start | flex-end | center | space-between |
                   space-around | space-evenly | start | end | left | right
  align-items: stretch | flex-start | flex-end | center | baseline |
               first baseline | last baseline
  align-content: stretch | flex-start | flex-end | center | space-between |
                 space-around | space-evenly
  gap: <row-gap> <column-gap>
  row-gap: <length>
  column-gap: <length>

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. display: flex vs inline-flex
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  display: flex
    → Container is BLOCK-LEVEL (takes full width)
    → Children are flex items

  display: inline-flex
    → Container is INLINE-LEVEL (only as wide as content)
    → Children still behave as flex items
    → Useful for: buttons with icons, tag/badge components

  .tag {
    display: inline-flex;
    align-items: center;
    gap: 4px;
    padding: 4px 8px;
    border-radius: 9999px;
    background: #e5e7eb;
  }

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
2. flex-direction
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  flex-direction: row;             /* → (LTR default) */
  flex-direction: row-reverse;     /* ← */
  flex-direction: column;          /* ↓ */
  flex-direction: column-reverse;  /* ↑ */

  This sets the MAIN AXIS and therefore:
  • Changes which axis justify-content/align-items operate on
  • Changes the "flow" direction of items

  .sidebar-layout {
    display: flex;
    flex-direction: row;    /* sidebar and content side by side */
  }

  .card {
    display: flex;
    flex-direction: column;  /* stacked layout: image, title, body, footer */
  }

  /* Push footer to bottom of card: */
  .card {
    display: flex;
    flex-direction: column;
    min-height: 300px;
  }
  .card__footer {
    margin-top: auto;   /* auto margin pushes footer to bottom */
  }

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
3. flex-wrap
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  flex-wrap: nowrap;        /* all on one line (may overflow) */
  flex-wrap: wrap;          /* wraps to next line when needed */
  flex-wrap: wrap-reverse;  /* wraps but in reverse direction */

  /* Responsive card grid without media queries: */
  .cards {
    display: flex;
    flex-wrap: wrap;
    gap: 16px;
  }
  .card {
    flex: 1 1 280px;  /* grow, shrink, basis=280px → auto wraps */
  }

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
4. justify-content — MAIN AXIS alignment
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Controls distribution of items ALONG the main axis.

  flex-start    → [ABC           ]
  flex-end      → [           ABC]
  center        → [     ABC     ]
  space-between → [A      B      C]  (edges flush)
  space-around  → [ A   B   C  ]    (half-space at edges)
  space-evenly  → [  A   B   C  ]   (equal space everywhere)

  Real examples:
  /* Navbar: logo left, nav right */
  .navbar {
    display: flex;
    justify-content: space-between;
    align-items: center;
  }

  /* Centered hero text */
  .hero {
    display: flex;
    justify-content: center;
    align-items: center;
  }

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
5. align-items — CROSS AXIS alignment (single line)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Controls alignment of items PERPENDICULAR to main axis (within a single line).

  stretch (default) → items expand to fill the container height (equal-height columns)
  flex-start        → items align to the top (in row direction)
  flex-end          → items align to the bottom (in row direction)
  center            → items vertically centered
  baseline          → items aligned by their text baselines
  first baseline    → align by first line's baseline
  last baseline     → align by last line's baseline

  /* Equal-height columns: default stretch behavior */
  .columns { display: flex; }  /* children automatically equal height */

  /* Align icons with text */
  .menu-item {
    display: flex;
    align-items: center;
    gap: 8px;
  }

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
6. align-content — CROSS AXIS alignment (multi-line)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Only applies when flex-wrap is active and there are multiple lines.
Controls distribution of FLEX LINES in the container.

  stretch (default) → lines stretch to fill container
  flex-start        → lines packed at start
  flex-end          → lines packed at end
  center            → lines centered
  space-between     → equal space between lines
  space-around      → equal space around lines
  space-evenly      → equal space including edges

  .tag-cloud {
    display: flex;
    flex-wrap: wrap;
    align-content: flex-start;  /* lines packed to top */
    height: 300px;
    gap: 8px;
  }

7. gap (row-gap and column-gap)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  gap: 16px;              /* equal gap on both axes */
  gap: 16px 24px;         /* row-gap 16px, column-gap 24px */
  row-gap: 16px;
  column-gap: 24px;

  Before gap: margins on items with negative margin on container (messy)
  With gap: clean, no edge margin problems

  .grid-cards {
    display: flex;
    flex-wrap: wrap;
    gap: 24px;
  }
  .card {
    flex: 1 1 calc(33.333% - 16px);
    /* No margin needed — gap handles spacing */
  }

8. PRACTICAL FLEX CONTAINER PATTERNS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Pattern 1 — Sticky footer layout:
  .app {
    display: flex;
    flex-direction: column;
    min-height: 100vh;
  }
  .app__main { flex: 1; }  /* takes remaining space, pushes footer down */

Pattern 2 — Navigation bar:
  .nav {
    display: flex;
    align-items: center;
    justify-content: space-between;
    padding: 0 24px;
    height: 60px;
  }
  .nav__links {
    display: flex;
    gap: 24px;
    align-items: center;
  }

Pattern 3 — Holy Grail Layout (header, 3-col, footer):
  .page {
    display: flex;
    flex-direction: column;
    min-height: 100vh;
  }
  .page__body {
    display: flex;
    flex: 1;
  }
  .page__sidebar { width: 240px; flex-shrink: 0; }
  .page__main    { flex: 1; }
  .page__aside   { width: 200px; flex-shrink: 0; }

Pattern 4 — Image gallery with auto-wrap:
  .gallery {
    display: flex;
    flex-wrap: wrap;
    gap: 16px;
  }
  .gallery__item {
    flex: 0 1 calc(25% - 12px);  /* 4 columns with gap */
  }

9. KEY INTERVIEW QUESTIONS — FLEX CONTAINER
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What changes when you use flex-direction: column?
A: The main axis becomes vertical and the cross axis becomes horizontal.
   justify-content now controls vertical distribution;
   align-items now controls horizontal alignment.
   This is a common confusion — justify/align are always relative to main/cross axis.

Q: How would you create a sticky footer with flexbox?
A: Set the page wrapper to: display:flex; flex-direction:column; min-height:100vh;
   Then give the main content area: flex:1; — it grows to fill available space,
   pushing the footer to the bottom regardless of content length.

Q: What is the difference between align-items and align-content?
A: align-items: applies to each flex line individually (aligns items within a line).
   align-content: applies to the flex lines as a group (distributes space between lines).
   align-content only works when flex-wrap is enabled and multiple lines exist.

Q: How do you create a responsive card grid with flexbox without media queries?
A: .cards { display: flex; flex-wrap: wrap; gap: 16px; }
   .card { flex: 1 1 280px; } ← flex-basis 280px causes auto-wrapping.
   When the container width is less than 2×280px, cards wrap to a single column.

Q: What is baseline alignment and when would you use it?
A: align-items: baseline aligns flex items so their text baselines line up.
   Useful when items have different font sizes, padding, or contain mixed content —
   the text appears on a consistent line across all items. Common for navigation
   items or inline form elements with labels and inputs.
