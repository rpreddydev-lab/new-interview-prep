1. WHAT ARE MEDIA QUERIES?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Media queries allow CSS to be applied conditionally based on device/viewport
characteristics — screen width, height, orientation, resolution, color scheme, etc.

They are the backbone of responsive web design.

Syntax:
  @media <media-type> and (<media-feature>) {
    /* styles */
  }

  @media screen and (min-width: 768px) {
    .container { max-width: 960px; }
  }

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
2. MEDIA TYPES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  all       — All devices (default if omitted)
  screen    — Screens (monitors, phones, tablets)
  print     — Printers (print preview)
  speech    — Screen readers (rare)

  /* Print styles — hide nav and ads when printing */
  @media print {
    nav, .ads { display: none; }
    body { font-size: 12pt; color: black; }
    a::after { content: " (" attr(href) ")"; }
  }

3. COMMON MEDIA FEATURES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Feature                | Description
  ─────────────────────────────────────────────────────────
  width / min-width /    | Viewport width
  max-width              |
  height / min-height /  | Viewport height
  max-height             |
  orientation            | portrait or landscape
  resolution             | dpi, dpcm (for retina screens)
  prefers-color-scheme   | light or dark (user OS setting)
  prefers-reduced-motion | reduce (user motion preference)
  prefers-contrast       | more, less, forced
  hover                  | hover (pointer device) or none (touch)
  pointer                | fine (mouse), coarse (touch), none
  display-mode           | standalone, fullscreen (for PWAs)
  aspect-ratio           | viewport aspect ratio

  /* Dark mode */
  @media (prefers-color-scheme: dark) {
    body { background: #121212; color: #e0e0e0; }
  }

  /* Reduced motion — disable animations */
  @media (prefers-reduced-motion: reduce) {
    *, *::before, *::after {
      animation-duration: 0.01ms !important;
      transition-duration: 0.01ms !important;
    }
  }

  /* Retina / high-DPI screens */
  @media (-webkit-min-device-pixel-ratio: 2),
         (min-resolution: 192dpi) {
    .logo { background-image: url(logo@2x.png); }
  }

  /* Touch devices — no hover state */
  @media (hover: none) and (pointer: coarse) {
    .tooltip { display: none; }  /* don't show hover tooltips on touch */
    .button { min-height: 44px; }  /* larger touch targets */
  }

4. LOGICAL OPERATORS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  and   — Both conditions must be true
  ,     — OR operator (comma = "or")
  not   — Negates the entire query
  only  — Hides query from old browsers (rarely needed now)

  /* AND — between min and max */
  @media screen and (min-width: 600px) and (max-width: 1200px) { }

  /* OR — comma-separated */
  @media screen and (min-width: 600px), print { }

  /* NOT — negate */
  @media not screen { }  /* everything except screens */
  @media not (prefers-color-scheme: dark) { }

5. RANGE SYNTAX (Modern CSS — Level 4)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

CSS Media Queries Level 4 introduces range syntax:

  /* Old syntax */
  @media (min-width: 600px) and (max-width: 1200px) { }

  /* New range syntax — more readable */
  @media (600px <= width <= 1200px) { }
  @media (width >= 600px) { }
  @media (width < 1200px) { }

  Browser support: modern browsers (2022+)

6. COMMON BREAKPOINTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  There are no "correct" breakpoints — they depend on content.
  Common reference breakpoints used by frameworks:

  Tailwind CSS:
    sm:  640px   (≥ small phones landscape)
    md:  768px   (≥ tablets)
    lg:  1024px  (≥ laptops)
    xl:  1280px  (≥ desktops)
    2xl: 1536px  (≥ large screens)

  Bootstrap 5:
    sm:  576px
    md:  768px
    lg:  992px
    xl:  1200px
    xxl: 1400px

  Best practice: Define breakpoints based on when your content BREAKS,
  not based on specific device sizes. Devices are too varied.

8. CONTAINER QUERIES (@container) — COMPONENT-LEVEL RESPONSIVE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Media queries respond to the VIEWPORT. Container queries respond to a PARENT.

  /* Define the container */
  .card-wrapper {
    container-type: inline-size;
    container-name: card;  /* optional */
  }

  /* Style card based on its container's width */
  @container card (min-width: 400px) {
    .card { flex-direction: row; }
  }

  @container (min-width: 600px) {
    .card-title { font-size: 2rem; }
  }

  Container types:
    inline-size   — responds to container's inline (width) axis
    size          — responds to both inline and block axes
    normal        — creates containment but no container query context

  Container query units:
    cqi  — 1% of container's inline size
    cqb  — 1% of container's block size
    cqw  — 1% of container's width
    cqh  — 1% of container's height

  .card { padding: 5cqi; }  /* padding = 5% of container's inline size */

9. KEY INTERVIEW QUESTIONS — MEDIA QUERIES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What is a media query?
A: A conditional block of CSS that applies styles when device/viewport conditions
   are met. They enable responsive design by letting you write styles for
   different screen sizes, orientations, color schemes, and more.

Q: What is the difference between media queries and container queries?
A: Media queries respond to the viewport size. Container queries respond to
   the size of a parent element. Container queries are better for reusable
   components that may appear in different-sized containers.

Q: What does min-width vs max-width mean in media queries?
A: min-width targets viewports AT OR ABOVE a size (mobile-first: add styles
   as screen gets larger). max-width targets viewports AT OR BELOW a size
   (desktop-first: override styles as screen gets smaller).

Q: What does the "only" keyword do in a media query?
A: It hides the query from old browsers that don't support the media query syntax.
   "@media only screen and ..." — Old browsers ignore this. Modern browsers
   treat "only" as a no-op and apply the query normally.

Q: How do you implement dark mode with CSS?
A: Use @media (prefers-color-scheme: dark) to detect the user's OS preference
   and swap colors/images accordingly. Can also be done with CSS variables:
   :root { --bg: white; --text: black; }
   @media (prefers-color-scheme: dark) { :root { --bg: #121212; --text: white; } }
