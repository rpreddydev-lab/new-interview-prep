1. WHAT IS TAILWIND CSS?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Tailwind CSS is a utility-first CSS framework. Instead of writing custom CSS,
you compose pre-defined utility classes directly in HTML.

  <!-- Traditional approach -->
  <button class="btn-primary">Submit</button>
  /* CSS */
  .btn-primary { background: #3b82f6; color: white; padding: 8px 16px; border-radius: 4px; }

  <!-- Tailwind approach -->
  <button class="bg-blue-500 text-white px-4 py-2 rounded hover:bg-blue-600">
    Submit
  </button>

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
2. UTILITY-FIRST PHILOSOPHY
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Each utility class = one CSS property (or a small group):
    bg-blue-500     → background-color: rgb(59 130 246)
    text-white      → color: rgb(255 255 255)
    px-4            → padding-left: 1rem; padding-right: 1rem
    py-2            → padding-top: 0.5rem; padding-bottom: 0.5rem
    rounded         → border-radius: 0.25rem
    flex            → display: flex
    items-center    → align-items: center
    justify-between → justify-content: space-between
    gap-4           → gap: 1rem
    text-xl         → font-size: 1.25rem; line-height: 1.75rem
    font-bold       → font-weight: 700
    shadow-md       → box-shadow: ...
    hover:bg-blue-600 → :hover pseudo-class modifier

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
3. RESPONSIVE PREFIXES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Tailwind is MOBILE-FIRST. Unprefixed = all screens. Prefixed = min-width up.

  Breakpoint | Prefix | Min Width
  ──────────────────────────────
  Mobile     | (none) | 0px
  sm         | sm:    | 640px
  md         | md:    | 768px
  lg         | lg:    | 1024px
  xl         | xl:    | 1280px
  2xl        | 2xl:   | 1536px

  <!-- Mobile: stacked, tablet: 2 col, desktop: 3 col -->
  <div class="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
    <!-- cards -->
  </div>

  <!-- Hide on mobile, show on desktop -->
  <div class="hidden lg:block">Desktop only content</div>

  <!-- Font size: mobile → tablet → desktop -->
  <h1 class="text-2xl md:text-3xl lg:text-5xl">Heading</h1>

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
4. STATE MODIFIERS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  hover:   — :hover
  focus:   — :focus
  active:  — :active
  visited: — :visited
  disabled:— :disabled
  checked: — :checked
  group-hover: — parent has .group, child responds to parent hover
  peer-focus:  — sibling with .peer focused

  <button class="bg-blue-500 hover:bg-blue-600 active:scale-95 disabled:opacity-50">
    Button
  </button>

  <!-- Group hover: parent hover triggers child change -->
  <div class="group">
    <h3 class="group-hover:text-blue-500">Card Title</h3>
    <p class="opacity-0 group-hover:opacity-100 transition-opacity">Hover reveal</p>
  </div>

  <!-- Peer: label changes when sibling input is focused -->
  <input class="peer" type="text" />
  <label class="peer-focus:text-blue-500">Email</label>

5. DARK MODE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Tailwind supports dark mode via the dark: prefix.

  Two modes (set in tailwind.config.js):
    darkMode: 'media'  — based on prefers-color-scheme OS setting
    darkMode: 'class'  — based on .dark class on <html>

  <!-- class-based dark mode -->
  <div class="bg-white dark:bg-gray-900 text-black dark:text-white">
    Content
  </div>

  Toggling dark mode (class strategy):
    document.documentElement.classList.toggle('dark');

6. TAILWIND CONFIGURATION
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // tailwind.config.js
  module.exports = {
    content: ['./src/**/*.{html,js,ts,jsx,tsx}'],  // files to scan for classes
    theme: {
      extend: {
        colors: {
          brand: {
            50:  '#f0f9ff',
            500: '#0ea5e9',
            900: '#0c4a6e',
          }
        },
        fontFamily: {
          heading: ['Inter', 'sans-serif'],
        },
        spacing: {
          18: '4.5rem',  // adds p-18, m-18, etc.
        },
        screens: {
          xs: '480px',   // custom breakpoint
        },
      },
    },
    plugins: [
      require('@tailwindcss/typography'),   // prose classes
      require('@tailwindcss/forms'),        // form reset
      require('@tailwindcss/aspect-ratio'),
    ],
  }

7. @apply — EXTRACTING COMPONENT CLASSES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  When you find yourself repeating the same utility combinations, extract
  them into a class using @apply:

  /* button.css */
  .btn {
    @apply inline-flex items-center px-4 py-2 rounded font-semibold
           transition-colors duration-200 cursor-pointer border-none;
  }

  .btn-primary {
    @apply btn bg-blue-500 text-white hover:bg-blue-600 active:scale-95;
  }

  .btn-danger {
    @apply btn bg-red-500 text-white hover:bg-red-600;
  }

  Use @apply sparingly — it defeats the purpose of utility-first if overused.
  Prefer component abstractions in JS frameworks (React components, Vue SFCs).

8. JIT (JUST-IN-TIME) COMPILER
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Tailwind v3+ uses JIT by default. JIT generates only the CSS you actually use.

  Benefits:
    • Near-zero unused CSS in production (tiny bundle)
    • Arbitrary value support: bg-[#1a2b3c], mt-[37px], w-[calc(100%-200px)]
    • Instant dev builds (no purging step needed)

  Arbitrary values:
    <div class="bg-[#1da1f2]">Twitter Blue</div>
    <div class="w-[350px] h-[350px]">Custom size</div>
    <div class="grid-cols-[1fr_2fr_1fr]">Custom grid</div>
    <div class="mt-[clamp(1rem,5vw,3rem)]">Fluid margin</div>

9. TAILWIND PROS AND CONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  PROS:
    ✅ No naming things — no more inventing class names for one-offs
    ✅ Extremely fast prototyping
    ✅ Consistent design tokens (spacing, colors, typography)
    ✅ Responsive and hover variants built in
    ✅ Tiny production CSS (JIT generates only used classes)
    ✅ Excellent documentation and tooling (IntelliSense)
    ✅ Works with any framework

  CONS:
    ❌ HTML becomes verbose / "class soup"
    ❌ Learning curve for all utility names
    ❌ Hard to visually distinguish markup from style at a glance
    ❌ Requires PostCSS/build toolchain
    ❌ Not ideal for highly dynamic styles (conditional classes can be complex)
    ❌ Without components/@apply, repetition in multi-file codebases

10. KEY INTERVIEW QUESTIONS — TAILWIND CSS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q: What is Tailwind CSS and how is it different from Bootstrap?
A: Tailwind is a utility-first CSS framework — it provides low-level utilities
   (one class = one CSS rule) and lets you build designs directly in HTML.
   Bootstrap is a component-based framework — it provides pre-built UI
   components (buttons, cards, modals) with opinionated styling. Tailwind
   gives more design freedom but requires more work; Bootstrap gets you
   consistent UI faster but is harder to customize.

Q: How does Tailwind avoid shipping unused CSS?
A: Via its JIT (Just-In-Time) compiler, which scans files specified in the
   content config for used class names and generates ONLY that CSS. This
   results in a very small production CSS file regardless of how many Tailwind
   utilities exist.

Q: What does "utility-first" mean?
A: Utility-first means every CSS class has a single, specific purpose. Instead
   of writing component-scoped CSS, you apply many small utility classes
   directly in HTML to compose the desired style. Benefits: no naming, no file-
   switching, instant design iteration.

Q: What is @apply in Tailwind?
A: @apply lets you use Tailwind utility classes inside regular CSS rules,
   creating named component classes from utilities. Use it to avoid
   repeating class strings when a pattern appears many times. However,
   overusing @apply defeats utility-first — prefer component abstractions
   in JS/templates.

Q: How does Tailwind handle responsive design?
A: Mobile-first: unprefixed utilities apply to all screen sizes. Prefixed
   utilities (sm:, md:, lg:, xl:, 2xl:) apply min-width media queries.
   Example: "text-sm md:text-base lg:text-lg" gives different font sizes
   at each breakpoint.
