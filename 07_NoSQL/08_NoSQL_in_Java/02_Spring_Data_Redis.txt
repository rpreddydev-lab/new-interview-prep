STRINGREDISTEMPLATE -- LOW LEVEL

  @Autowired
  private StringRedisTemplate redisTemplate;

  // String ops:
  ValueOperations<String, String> ops = redisTemplate.opsForValue();
  ops.set("user:session:abc123", "U100");
  ops.set("user:session:abc123", "U100", Duration.ofMinutes(30));
  String userId = ops.get("user:session:abc123");

  // Hash ops:
  HashOperations<String, String, String> hashOps = redisTemplate.opsForHash();
  hashOps.put("user:U100", "name", "Alice");
  hashOps.put("user:U100", "email", "alice@example.com");
  Map<String, String> user = hashOps.entries("user:U100");

  // List ops (queue):
  ListOperations<String, String> listOps = redisTemplate.opsForList();
  listOps.leftPush("queue:emails", "email:task:1");
  String task = listOps.rightPop("queue:emails");

  // Set ops:
  SetOperations<String, String> setOps = redisTemplate.opsForSet();
  setOps.add("product:P1:tags", "wireless", "bluetooth", "sale");
  Set<String> tags = setOps.members("product:P1:tags");

  // Sorted Set (leaderboard):
  ZSetOperations<String, String> zOps = redisTemplate.opsForZSet();
  zOps.add("leaderboard:2026-03", "user:U100", 9850.0);
  zOps.incrementScore("leaderboard:2026-03", "user:U100", 200.0);
  Set<String> top10 = zOps.reverseRange("leaderboard:2026-03", 0, 9);

  // TTL management:
  redisTemplate.expire("user:session:abc123", Duration.ofMinutes(30));
  Long ttl = redisTemplate.getExpire("user:session:abc123", TimeUnit.SECONDS);

SPRING CACHE ABSTRACTION (@CACHEABLE)
================================================================

  @EnableCaching  // on @Configuration class

  @Service
  public class ProductService {

      // Cache result on first call; return cached on subsequent calls:
      @Cacheable(value = "products", key = "#productId")
      public Product getProduct(String productId) {
          return productRepository.findById(productId).orElseThrow();
      }

      // Update the cache entry on write:
      @CachePut(value = "products", key = "#product.id")
      public Product updateProduct(Product product) {
          return productRepository.save(product);
      }

      // Remove from cache on delete:
      @CacheEvict(value = "products", key = "#productId")
      public void deleteProduct(String productId) {
          productRepository.deleteById(productId);
      }

      // Evict all entries in cache:
      @CacheEvict(value = "products", allEntries = true)
      public void clearProductCache() { }
  }

  Redis cache config (application.yml):
  spring:
    cache:
      type: redis
      redis:
        time-to-live: 600000   # 10 minutes in ms
        cache-null-values: false

CUSTOM REDISTEMPLATE WITH SERIALIZATION
================================================================

  @Bean
  public RedisTemplate<String, Object> redisTemplate(RedisConnectionFactory factory) {
      RedisTemplate<String, Object> template = new RedisTemplate<>();
      template.setConnectionFactory(factory);
      Jackson2JsonRedisSerializer<Object> serializer =
          new Jackson2JsonRedisSerializer<>(Object.class);
      template.setValueSerializer(serializer);
      template.setHashValueSerializer(serializer);
      template.setKeySerializer(new StringRedisSerializer());
      template.setHashKeySerializer(new StringRedisSerializer());
      template.afterPropertiesSet();
      return template;
  }

INTERVIEW Q&A
================================================================

  Q: What is the difference between @Cacheable and @CachePut?
  A: @Cacheable: checks cache first; if hit, returns cached value without
     calling the method. If miss, calls method and caches result.
     @CachePut: ALWAYS calls the method and ALWAYS updates the cache.
     Use @CachePut on write operations to keep cache in sync with DB.

  Q: How do you serialize Java objects in Redis?
  A: By default, StringRedisTemplate uses String serialization.
     For objects, configure a RedisTemplate bean with Jackson2JsonRedisSerializer
     to store objects as JSON. Alternatively, use JdkSerializationRedisSerializer
     (binary, Java-only). JSON is preferred for readability and cross-language
     compatibility.
