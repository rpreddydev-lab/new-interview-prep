  import org.springframework.data.annotation.Id;
  import org.springframework.data.mongodb.core.mapping.Document;
  import org.springframework.data.mongodb.core.mapping.Field;
  import org.springframework.data.mongodb.core.index.Indexed;

  @Document(collection = "products")
  public class Product {

      @Id
      private String id;  // maps to MongoDB _id (ObjectId as String)

      @Indexed(unique = true)
      private String skuCode;

      @Field("product_name")
      private String name;

      private double price;
      private String category;

      @Field("created_at")
      private Instant createdAt;

      private List<String> tags = new ArrayList<>();

      // getters, setters, constructors
  }

REPOSITORY
================================================================

  public interface ProductRepository extends MongoRepository<Product, String> {
      List<Product> findByCategory(String category);
      List<Product> findByPriceLessThanAndCategory(double price, String cat);
      Optional<Product> findBySkuCode(String sku);

      @Query("{ 'tags': { $all: ?0 } }")
      List<Product> findByAllTags(List<String> tags);

      @Query(value = "{ 'category': ?0 }", fields = "{ 'name': 1, 'price': 1 }")
      List<Product> findProjectedByCategory(String category);
  }

  @Autowired
  private MongoTemplate mongoTemplate;

  // Dynamic query:
  Query query = new Query();
  query.addCriteria(Criteria.where("category").is("electronics")
                            .and("price").lte(500.0));
  query.with(Sort.by(Sort.Direction.ASC, "price"));
  query.limit(10);
  List<Product> products = mongoTemplate.find(query, Product.class);

  // Update:
  Update update = new Update()
      .set("price", 899.99)
      .inc("stockCount", -1)
      .push("tags").value("sale");
  UpdateResult result = mongoTemplate.updateFirst(
      Query.query(Criteria.where("skuCode").is("SKU001")),
      update, Product.class
  );

  // Upsert:
  mongoTemplate.upsert(query, update, Product.class);

  // Delete:
  mongoTemplate.remove(Query.query(Criteria.where("id").is("abc")), Product.class);

  import org.springframework.data.mongodb.core.aggregation.*;

  Aggregation agg = Aggregation.newAggregation(
      Aggregation.match(Criteria.where("createdAt").gte(startDate)),
      Aggregation.group("category")
          .count().as("orderCount")
          .sum("price").as("totalRevenue"),
      Aggregation.project("category", "orderCount", "totalRevenue")
          .andExpression("totalRevenue / orderCount").as("avgRevenue"),
      Aggregation.sort(Sort.Direction.DESC, "totalRevenue"),
      Aggregation.limit(5)
  );

  AggregationResults<CategoryStats> results =
      mongoTemplate.aggregate(agg, "orders", CategoryStats.class);

  List<CategoryStats> stats = results.getMappedResults();

TRANSACTIONS (MULTI-DOCUMENT, MONGODB 4.0+)
================================================================

  @Autowired
  private MongoTransactionManager txManager;

  @Transactional
  public void transferStock(String fromSku, String toSku, int qty) {
      Update decrement = new Update().inc("stock", -qty);
      Update increment = new Update().inc("stock", qty);
      mongoTemplate.updateFirst(Query.query(Criteria.where("sku").is(fromSku)), decrement, Product.class);
      mongoTemplate.updateFirst(Query.query(Criteria.where("sku").is(toSku)), increment, Product.class);
  }

  // Required bean:
  @Bean
  MongoTransactionManager txManager(MongoDatabaseFactory dbFactory) {
      return new MongoTransactionManager(dbFactory);
  }

INTERVIEW Q&A
================================================================

  Q: How do you run an aggregation pipeline in Spring Data MongoDB?
  A: Use MongoTemplate.aggregate(). Build an Aggregation object with
     static factory methods: Aggregation.match(), Aggregation.group(),
     Aggregation.project(), Aggregation.sort(), etc. Pass the output
     class as target type. For complex pipelines, you can also use
     the @Query annotation with an aggregation pipeline array on the
     repository method, or use reactive templates for non-blocking flows.

  Q: How do you enable multi-document transactions in MongoDB with Spring?
  A: Transactions require a replica set (not standalone). Register a
     MongoTransactionManager bean with the MongoDatabaseFactory. Then
     annotate your service method with @Transactional. All MongoTemplate
     and repository operations within the method participate in the transaction.
