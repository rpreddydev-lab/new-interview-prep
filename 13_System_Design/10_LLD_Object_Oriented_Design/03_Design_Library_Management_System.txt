1. REQUIREMENTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

CLARIFICATIONS:
  - Multiple copies of same book? Yes (e.g., 5 copies of "Clean Code")
  - Members borrow books? Yes, with due dates
  - Reservation/waitlist? Yes, if all copies checked out
  - Fine for late returns? Yes, per day
  - Search by: title, author, ISBN, genre
  - Multiple branches? Single branch initially

ENTITIES:
  Book, BookItem (physical copy), Author, Member, Loan, Reservation,
  Catalog (search), Fine, Librarian

OPERATIONS:
  Member: search, checkout, return, reserve, pay fine, view history
  Librarian: add/remove books, manage members, view reports


2. CLASS DESIGN
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

enum BookStatus { AVAILABLE, CHECKED_OUT, RESERVED, LOST }
enum MemberStatus { ACTIVE, SUSPENDED, EXPIRED }
enum ReservationStatus { WAITING, FULFILLED, CANCELLED }

// --- Author ---

class Author {
    private String authorId;
    private String name;
    private String biography;
    
    Author(String authorId, String name) {
        this.authorId = authorId;
        this.name = name;
    }
    
    String getAuthorId() { return authorId; }
    String getName()     { return name; }
}

// --- Book (logical entity — shared across copies) ---

class Book {
    private String isbn;
    private String title;
    private List<Author> authors;
    private String genre;
    private int publicationYear;
    private String publisher;
    private List<BookItem> copies;    // physical copies
    
    Book(String isbn, String title) {
        this.isbn = isbn;
        this.title = title;
        this.authors = new ArrayList<>();
        this.copies = new ArrayList<>();
    }
    
    void addCopy(BookItem item) { copies.add(item); }
    
    int getAvailableCount() {
        return (int) copies.stream().filter(c -> c.getStatus() == BookStatus.AVAILABLE).count();
    }
    
    Optional<BookItem> getAvailableCopy() {
        return copies.stream().filter(c -> c.getStatus() == BookStatus.AVAILABLE).findFirst();
    }
    
    String getIsbn()    { return isbn; }
    String getTitle()   { return title; }
    List<Author> getAuthors() { return Collections.unmodifiableList(authors); }
}

// --- BookItem (physical copy) ---

class BookItem {
    private String barcode;          // unique per physical copy
    private String isbn;             // links to Book
    private BookStatus status;
    private String location;         // shelf reference "A3-12"
    private LocalDate dueDate;       // set when checked out
    
    BookItem(String barcode, String isbn, String location) {
        this.barcode = barcode;
        this.isbn = isbn;
        this.location = location;
        this.status = BookStatus.AVAILABLE;
    }
    
    void checkout(LocalDate dueDate) {
        this.status = BookStatus.CHECKED_OUT;
        this.dueDate = dueDate;
    }
    
    void returnBook() {
        this.status = BookStatus.AVAILABLE;
        this.dueDate = null;
    }
    
    boolean isOverdue() {
        return status == BookStatus.CHECKED_OUT && LocalDate.now().isAfter(dueDate);
    }
    
    long getDaysOverdue() {
        if (!isOverdue()) return 0;
        return ChronoUnit.DAYS.between(dueDate, LocalDate.now());
    }
    
    BookStatus getStatus() { return status; }
    String getBarcode()    { return barcode; }
    String getIsbn()       { return isbn; }
    LocalDate getDueDate() { return dueDate; }
    void setStatus(BookStatus s) { this.status = s; }
}

// --- Member ---

class Member {
    private String memberId;
    private String name;
    private String email;
    private MemberStatus status;
    private int maxCheckouts;        // typically 5
    private List<Loan> activeLoans;
    private List<Reservation> reservations;
    private double outstandingFine;
    private LocalDate membershipExpiry;
    
    Member(String memberId, String name, String email) {
        this.memberId = memberId;
        this.name = name;
        this.email = email;
        this.status = MemberStatus.ACTIVE;
        this.maxCheckouts = 5;
        this.activeLoans = new ArrayList<>();
        this.reservations = new ArrayList<>();
    }
    
    boolean canCheckout() {
        return status == MemberStatus.ACTIVE
            && activeLoans.size() < maxCheckouts
            && outstandingFine < 10.0      // suspended if fine > $10
            && LocalDate.now().isBefore(membershipExpiry);
    }
    
    void addLoan(Loan loan)           { activeLoans.add(loan); }
    void removeLoan(Loan loan)        { activeLoans.remove(loan); }
    void addReservation(Reservation r) { reservations.add(r); }
    void addFine(double amount)       { outstandingFine += amount; }
    void payFine(double amount)       {
        outstandingFine = Math.max(0, outstandingFine - amount);
    }
    
    String getMemberId()         { return memberId; }
    double getOutstandingFine()  { return outstandingFine; }
    List<Loan> getActiveLoans()  { return Collections.unmodifiableList(activeLoans); }
}

// --- Loan ---

class Loan {
    private String loanId;
    private Member member;
    private BookItem bookItem;
    private LocalDate checkoutDate;
    private LocalDate dueDate;
    private LocalDate returnDate;    // null if not returned
    private boolean isReturned;
    
    Loan(String loanId, Member member, BookItem bookItem, int loanDays) {
        this.loanId = loanId;
        this.member = member;
        this.bookItem = bookItem;
        this.checkoutDate = LocalDate.now();
        this.dueDate = LocalDate.now().plusDays(loanDays);
        this.isReturned = false;
    }
    
    boolean isOverdue() {
        return !isReturned && LocalDate.now().isAfter(dueDate);
    }
    
    double calculateFine(double finePerDay) {
        if (!isOverdue()) return 0;
        long days = ChronoUnit.DAYS.between(dueDate, LocalDate.now());
        return days * finePerDay;
    }
    
    void markReturned() {
        this.isReturned = true;
        this.returnDate = LocalDate.now();
    }
    
    String getLoanId()      { return loanId; }
    Member getMember()      { return member; }
    BookItem getBookItem()  { return bookItem; }
}

// --- Reservation ---

class Reservation {
    private String reservationId;
    private Member member;
    private String isbn;             // which book reserved
    private LocalDateTime reservedAt;
    private ReservationStatus status;
    private LocalDateTime expiryTime;  // hold expires 48 hours after available
    
    Reservation(String reservationId, Member member, String isbn) {
        this.reservationId = reservationId;
        this.member = member;
        this.isbn = isbn;
        this.reservedAt = LocalDateTime.now();
        this.status = ReservationStatus.WAITING;
    }
    
    void fulfil() {
        this.status = ReservationStatus.FULFILLED;
        this.expiryTime = LocalDateTime.now().plusHours(48);  // 48-hour hold
    }
    
    boolean isExpired() {
        return expiryTime != null && LocalDateTime.now().isAfter(expiryTime);
    }
    
    ReservationStatus getStatus() { return status; }
    Member getMember()            { return member; }
    String getIsbn()              { return isbn; }
}

// --- Catalog (search) ---

class Catalog {
    private Map<String, Book> booksByIsbn;
    private Map<String, List<Book>> booksByTitle;    // title → books
    private Map<String, List<Book>> booksByAuthor;   // authorId → books
    private Map<String, List<Book>> booksByGenre;
    
    Catalog() {
        booksByIsbn   = new HashMap<>();
        booksByTitle  = new HashMap<>();
        booksByAuthor = new HashMap<>();
        booksByGenre  = new HashMap<>();
    }
    
    void addBook(Book book) {
        booksByIsbn.put(book.getIsbn(), book);
        booksByTitle.computeIfAbsent(book.getTitle().toLowerCase(), k -> new ArrayList<>()).add(book);
        book.getAuthors().forEach(a ->
            booksByAuthor.computeIfAbsent(a.getAuthorId(), k -> new ArrayList<>()).add(book));
    }
    
    Optional<Book> searchByIsbn(String isbn) {
        return Optional.ofNullable(booksByIsbn.get(isbn));
    }
    
    List<Book> searchByTitle(String title) {
        return booksByTitle.getOrDefault(title.toLowerCase(), Collections.emptyList());
    }
    
    List<Book> searchByAuthorId(String authorId) {
        return booksByAuthor.getOrDefault(authorId, Collections.emptyList());
    }
}

// --- LibraryService (Facade / Orchestrator) ---

class LibraryService {
    private Catalog catalog;
    private Map<String, Member> members;            // memberId → member
    private Queue<Reservation> reservationQueues;   // per ISBN (simplified: global)
    private Map<String, Queue<Reservation>> perBookReservations;  // isbn → FIFO queue
    private int loanDays = 14;
    private double finePerDay = 0.25;  // $0.25/day
    private AtomicInteger loanCounter = new AtomicInteger(0);
    
    LibraryService() {
        catalog = new Catalog();
        members = new HashMap<>();
        perBookReservations = new HashMap<>();
    }
    
    // CHECKOUT: member borrows a book
    Loan checkout(String memberId, String isbn) {
        Member member = getOrThrow(memberId);
        if (!member.canCheckout()) throw new CheckoutException("Member cannot checkout");
        
        Book book = catalog.searchByIsbn(isbn)
                           .orElseThrow(() -> new NotFoundException("Book not found: " + isbn));
        
        BookItem copy = book.getAvailableCopy()
                            .orElseThrow(() -> new NoAvailableCopyException("All copies checked out"));
        
        copy.checkout(LocalDate.now().plusDays(loanDays));
        String loanId = "LOAN-" + loanCounter.incrementAndGet();
        Loan loan = new Loan(loanId, member, copy, loanDays);
        member.addLoan(loan);
        
        return loan;
    }
    
    // RETURN: member returns a book
    void returnBook(String loanId, Member member) {
        Loan loan = member.getActiveLoans().stream()
                          .filter(l -> l.getLoanId().equals(loanId))
                          .findFirst()
                          .orElseThrow(() -> new NotFoundException("Loan not found"));
        
        // Calculate and apply fine
        double fine = loan.calculateFine(finePerDay);
        if (fine > 0) member.addFine(fine);
        
        // Return book
        loan.getBookItem().returnBook();
        loan.markReturned();
        member.removeLoan(loan);
        
        // Check if anyone is waiting for this book
        String isbn = loan.getBookItem().getIsbn();
        Queue<Reservation> queue = perBookReservations.get(isbn);
        if (queue != null && !queue.isEmpty()) {
            Reservation next = queue.poll();
            next.fulfil();  // notify next person in queue (email/notification)
            loan.getBookItem().setStatus(BookStatus.RESERVED);
        }
    }
    
    // RESERVE: add to waitlist if book unavailable
    Reservation reserve(String memberId, String isbn) {
        Member member = getOrThrow(memberId);
        Book book = catalog.searchByIsbn(isbn)
                           .orElseThrow(() -> new NotFoundException("Book not found"));
        
        if (book.getAvailableCount() > 0) {
            throw new IllegalStateException("Book is available, checkout instead");
        }
        
        Reservation reservation = new Reservation("RES-" + System.currentTimeMillis(), member, isbn);
        perBookReservations.computeIfAbsent(isbn, k -> new LinkedList<>()).add(reservation);
        member.addReservation(reservation);
        
        return reservation;
    }
    
    private Member getOrThrow(String memberId) {
        return Optional.ofNullable(members.get(memberId))
                       .orElseThrow(() -> new NotFoundException("Member not found: " + memberId));
    }
}


3. DESIGN PATTERNS USED
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

FACADE: LibraryService provides simple checkout/return/reserve API
  hiding catalog lookups, fine calculations, reservation queue management

OBSERVER: When book returned and reservation queue is non-empty
  notify the next-in-queue member (email/push notification)
  
  Book.addReturnListener(observer) → observer.onBookAvailable(isbn)

COMPOSITE: Book (logical) has many BookItem (physical)
  Book.getAvailableCopy() selects from its copies

FACTORY: LoanFactory.create(member, bookItem) → Loan
  Encapsulates ID generation and date calculation

STRATEGY: FineCalculationStrategy
  StandardFineStrategy: $0.25/day
  HolidayFineStrategy: $0.10/day on holidays


4. KEY DISCUSSION POINTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

CONCURRENCY:
  Race condition: two members check out last copy simultaneously
  Fix: synchronized checkout() or optimistic locking on BookItem.status
  
  Database approach: UPDATE book_items SET status='CHECKED_OUT'
                     WHERE barcode=? AND status='AVAILABLE'
  Check rows updated = 1; if 0, another thread won.

RESERVATION FAIRNESS:
  FIFO queue per ISBN using LinkedList<Reservation>
  When book returned: notify head of queue (not just anyone waiting)

SEARCH PERFORMANCE:
  In-memory Catalog uses HashMap → O(1) ISBN lookup
  For full-text search by title/author: use Elasticsearch in real system
  For LLD interview: simple Map-based index is fine to describe
