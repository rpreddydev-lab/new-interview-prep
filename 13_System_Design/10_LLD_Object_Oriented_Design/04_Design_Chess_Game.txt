2. CLASS DESIGN

enum Color { WHITE, BLACK }

enum GameStatus {
    IN_PROGRESS, WHITE_WINS, BLACK_WINS, DRAW_STALEMATE, DRAW_AGREEMENT
}

enum PieceType {
    KING, QUEEN, ROOK, BISHOP, KNIGHT, PAWN
}

// --- Position / Square ---

class Position {
    private final int row;   // 0-7 (0 = rank 1, 7 = rank 8)
    private final int col;   // 0-7 (0 = file a, 7 = file h)
    
    Position(int row, int col) {
        if (row < 0 || row > 7 || col < 0 || col > 7)
            throw new IllegalArgumentException("Invalid position: " + row + "," + col);
        this.row = row;
        this.col = col;
    }
    
    // Static factory for chess notation "e4" -> Position(3, 4)
    static Position fromNotation(String notation) {
        int col = notation.charAt(0) - 'a';
        int row = notation.charAt(1) - '1';
        return new Position(row, col);
    }
    
    String toNotation() {
        return "" + (char)('a' + col) + (row + 1);
    }
    
    boolean isValid() { return row >= 0 && row <= 7 && col >= 0 && col <= 7; }
    int getRow() { return row; }
    int getCol() { return col; }
    
    @Override
    public boolean equals(Object o) {
        if (!(o instanceof Position)) return false;
        Position p = (Position) o;
        return row == p.row && col == p.col;
    }
}

// --- Abstract Piece ---

abstract class Piece {
    protected Color color;
    protected PieceType type;
    protected boolean hasMoved;  // important for castling, pawn double-move
    
    Piece(Color color, PieceType type) {
        this.color = color;
        this.type = type;
        this.hasMoved = false;
    }
    
    // Returns all valid target positions from current position on current board
    abstract List<Position> getLegalMoves(Position current, Board board);
    
    // Helper: can this piece attack the given position?
    boolean canAttack(Position current, Position target, Board board) {
        return getLegalMoves(current, board).contains(target);
    }
    
    Color getColor()    { return color; }
    PieceType getType() { return type; }
    boolean hasMoved()  { return hasMoved; }
    void setMoved()     { hasMoved = true; }
}

// --- Concrete Pieces ---

class Rook extends Piece {
    Rook(Color color) { super(color, PieceType.ROOK); }
    
    @Override
    List<Position> getLegalMoves(Position current, Board board) {
        List<Position> moves = new ArrayList<>();
        // Rook moves along ranks (rows) and files (cols)
        int[][] directions = {{1,0},{-1,0},{0,1},{0,-1}};
        
        for (int[] dir : directions) {
            int r = current.getRow() + dir[0];
            int c = current.getCol() + dir[1];
            while (r >= 0 && r <= 7 && c >= 0 && c <= 7) {
                Position pos = new Position(r, c);
                Piece occupant = board.getPieceAt(pos);
                if (occupant == null) {
                    moves.add(pos);  // empty square, can move here
                } else {
                    if (occupant.getColor() != this.color) {
                        moves.add(pos);  // can capture opponent
                    }
                    break;  // blocked by piece (own or enemy)
                }
                r += dir[0]; c += dir[1];
            }
        }
        return moves;
    }
}

class Knight extends Piece {
    Knight(Color color) { super(color, PieceType.KNIGHT); }
    
    @Override
    List<Position> getLegalMoves(Position current, Board board) {
        List<Position> moves = new ArrayList<>();
        int[][] jumps = {{2,1},{2,-1},{-2,1},{-2,-1},{1,2},{1,-2},{-1,2},{-1,-2}};
        
        for (int[] jump : jumps) {
            int r = current.getRow() + jump[0];
            int c = current.getCol() + jump[1];
            if (r >= 0 && r <= 7 && c >= 0 && c <= 7) {
                Position pos = new Position(r, c);
                Piece occupant = board.getPieceAt(pos);
                if (occupant == null || occupant.getColor() != this.color) {
                    moves.add(pos);  // knights jump over pieces
                }
            }
        }
        return moves;
    }
}

class Pawn extends Piece {
    Pawn(Color color) { super(color, PieceType.PAWN); }
    
    @Override
    List<Position> getLegalMoves(Position current, Board board) {
        List<Position> moves = new ArrayList<>();
        int direction = (color == Color.WHITE) ? 1 : -1;  // white moves up, black down
        int r = current.getRow();
        int c = current.getCol();
        
        // 1 step forward
        Position oneForward = new Position(r + direction, c);
        if (board.getPieceAt(oneForward) == null) {
            moves.add(oneForward);
            
            // 2 steps forward on first move
            if (!hasMoved) {
                Position twoForward = new Position(r + 2 * direction, c);
                if (board.getPieceAt(twoForward) == null) {
                    moves.add(twoForward);
                }
            }
        }
        
        // Diagonal captures
        for (int dc : new int[]{-1, 1}) {
            if (c + dc >= 0 && c + dc <= 7) {
                Position diag = new Position(r + direction, c + dc);
                Piece occupant = board.getPieceAt(diag);
                if (occupant != null && occupant.getColor() != this.color) {
                    moves.add(diag);  // capture
                }
                // En passant: check board.getEnPassantTarget()
            }
        }
        return moves;
    }
}

// --- Board ---

class Board {
    private Piece[][] grid;          // [row][col]
    private Position enPassantTarget; // set after pawn double-move
    
    Board() {
        grid = new Piece[8][8];
        initializeBoard();
    }
    
    void initializeBoard() {
        // White pieces (row 0 and 1)
        grid[0][0] = new Rook(Color.WHITE);
        grid[0][1] = new Knight(Color.WHITE);
        grid[0][2] = new Bishop(Color.WHITE);
        grid[0][3] = new Queen(Color.WHITE);
        grid[0][4] = new King(Color.WHITE);
        // ... etc
        for (int c = 0; c < 8; c++) grid[1][c] = new Pawn(Color.WHITE);
        // Black pieces (rows 6 and 7) mirror-image
        for (int c = 0; c < 8; c++) grid[6][c] = new Pawn(Color.BLACK);
        // ...
    }
    
    Piece getPieceAt(Position pos) {
        return grid[pos.getRow()][pos.getCol()];
    }
    
    void movePiece(Position from, Position to) {
        Piece piece = grid[from.getRow()][from.getCol()];
        grid[to.getRow()][to.getCol()] = piece;
        grid[from.getRow()][from.getCol()] = null;
        if (piece != null) piece.setMoved();
    }
    
    // Is any opponent piece attacking the given position?
    boolean isUnderAttack(Position pos, Color defendingColor) {
        Color attackingColor = (defendingColor == Color.WHITE) ? Color.BLACK : Color.WHITE;
        for (int r = 0; r < 8; r++) {
            for (int c = 0; c < 8; c++) {
                Piece p = grid[r][c];
                if (p != null && p.getColor() == attackingColor) {
                    if (p.canAttack(new Position(r, c), pos, this)) {
                        return true;
                    }
                }
            }
        }
        return false;
    }
    
    Position findKing(Color color) {
        for (int r = 0; r < 8; r++)
            for (int c = 0; c < 8; c++)
                if (grid[r][c] instanceof King && grid[r][c].getColor() == color)
                    return new Position(r, c);
        throw new RuntimeException("King not found — invalid board state");
    }
    
    boolean isInCheck(Color color) {
        Position kingPos = findKing(color);
        return isUnderAttack(kingPos, color);
    }
}

// --- Game ---

class Game {
    private Board board;
    private Player whitePlayer;
    private Player blackPlayer;
    private Color currentTurn;
    private GameStatus status;
    private List<Move> moveHistory;
    
    Game(Player white, Player black) {
        this.board = new Board();
        this.whitePlayer = white;
        this.blackPlayer = black;
        this.currentTurn = Color.WHITE;  // white moves first
        this.status = GameStatus.IN_PROGRESS;
        this.moveHistory = new ArrayList<>();
    }
    
    boolean makeMove(Position from, Position to) {
        if (status != GameStatus.IN_PROGRESS) return false;
        
        Piece piece = board.getPieceAt(from);
        if (piece == null || piece.getColor() != currentTurn) return false;
        
        // Validate move is legal
        List<Position> legalMoves = piece.getLegalMoves(from, board);
        if (!legalMoves.contains(to)) return false;
        
        // Execute move
        Piece captured = board.getPieceAt(to);
        Move move = new Move(from, to);
        move.setPieceCaptured(captured);
        board.movePiece(from, to);
        
        // Check: did this move put own king in check? (illegal if so)
        if (board.isInCheck(currentTurn)) {
            // Undo move
            board.movePiece(to, from);
            if (captured != null) board.setPieceAt(to, captured);
            return false;
        }
        
        moveHistory.add(move);
        
        // Switch turns
        Color opponent = (currentTurn == Color.WHITE) ? Color.BLACK : Color.WHITE;
        
        // Check for checkmate or stalemate
        if (hasNoLegalMoves(opponent)) {
            if (board.isInCheck(opponent)) {
                status = (currentTurn == Color.WHITE) ? GameStatus.WHITE_WINS : GameStatus.BLACK_WINS;
            } else {
                status = GameStatus.DRAW_STALEMATE;
            }
        }
        
        currentTurn = opponent;
        return true;
    }
    
    private boolean hasNoLegalMoves(Color color) {
        for (int r = 0; r < 8; r++) {
            for (int c = 0; c < 8; c++) {
                Piece p = board.getPieceAt(new Position(r, c));
                if (p != null && p.getColor() == color) {
                    if (!p.getLegalMoves(new Position(r, c), board).isEmpty()) {
                        return false;  // found at least one legal move
                    }
                }
            }
        }
        return true;  // no legal moves
    }
    
    GameStatus getStatus() { return status; }
    List<Move> getMoveHistory() { return Collections.unmodifiableList(moveHistory); }
}


3. DESIGN PATTERNS USED
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

INHERITANCE: Piece → King/Queen/Rook/Bishop/Knight/Pawn
  Each piece knows its own legal moves (polymorphism)
  getLegalMoves() called uniformly regardless of piece type

TEMPLATE METHOD: Board.isUnderAttack() iterates all pieces, calls canAttack()
  Algorithm skeleton in base; piece behavior defined in subclasses

COMMAND PATTERN: Move class encapsulates a move
  Enables: undo (reverse the move), replay (apply from history), notation

STRATEGY (extension): Different game variants
  MoveValidationStrategy → StandardChessRules, Chess960Rules, TimedGameRules


4. INTERVIEW DISCUSSION
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

CHECK DETECTION: After every move, verify opponent's king is not in check
  Board.isInCheck(opponentColor) → find king → isUnderAttack(kingPos)?

CHECKMATE: Opponent is in check AND has no legal moves to escape
  hasNoLegalMoves() + isInCheck() = checkmate

STALEMATE: NOT in check but also has no legal moves
  hasNoLegalMoves() + !isInCheck() = stalemate

UNDO: Move stores captured piece → board.movePiece(to, from) + restore captured

PERFORMANCE NOTE: isUnderAttack() scans entire board O(64) per call
  Called for every piece move check → O(64 × 64) for hasNoLegalMoves
  Real chess engines use bitboards (64-bit integers) for O(1) attack detection
  For an LLD interview, correctness > optimization — mention the optimization exists
