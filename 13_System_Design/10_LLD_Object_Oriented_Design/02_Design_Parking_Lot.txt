// --- Enums ---

enum VehicleType {
    MOTORCYCLE, CAR, TRUCK
}

enum SpotType {
    SMALL,    // motorcycles
    MEDIUM,   // cars
    LARGE     // trucks/buses
}

enum SpotStatus {
    AVAILABLE, OCCUPIED
}

enum PaymentStatus {
    PENDING, COMPLETED, FAILED
}

// --- Vehicle Hierarchy ---

abstract class Vehicle {
    private String licensePlate;
    private VehicleType vehicleType;
    
    Vehicle(String licensePlate, VehicleType vehicleType) {
        this.licensePlate = licensePlate;
        this.vehicleType = vehicleType;
    }
    
    abstract SpotType getRequiredSpotType();
    
    String getLicensePlate() { return licensePlate; }
    VehicleType getVehicleType() { return vehicleType; }
}

class Car extends Vehicle {
    Car(String licensePlate) {
        super(licensePlate, VehicleType.CAR);
    }
    
    @Override
    SpotType getRequiredSpotType() { return SpotType.MEDIUM; }
}

class Motorcycle extends Vehicle {
    Motorcycle(String licensePlate) {
        super(licensePlate, VehicleType.MOTORCYCLE);
    }
    
    @Override
    SpotType getRequiredSpotType() { return SpotType.SMALL; }
}

class Truck extends Vehicle {
    Truck(String licensePlate) {
        super(licensePlate, VehicleType.TRUCK);
    }
    
    @Override
    SpotType getRequiredSpotType() { return SpotType.LARGE; }
}

// --- ParkingSpot ---

class ParkingSpot {
    private String spotId;           // e.g., "F1-A-012"
    private int floorNumber;
    private SpotType spotType;
    private SpotStatus status;
    private Vehicle parkedVehicle;
    
    ParkingSpot(String spotId, int floorNumber, SpotType spotType) {
        this.spotId = spotId;
        this.floorNumber = floorNumber;
        this.spotType = spotType;
        this.status = SpotStatus.AVAILABLE;
    }
    
    // Thread-safe park/unpark operations
    synchronized boolean park(Vehicle vehicle) {
        if (status != SpotStatus.AVAILABLE) return false;
        if (!canFit(vehicle)) return false;
        
        this.parkedVehicle = vehicle;
        this.status = SpotStatus.OCCUPIED;
        return true;
    }
    
    synchronized Vehicle unpark() {
        if (status != SpotStatus.OCCUPIED) return null;
        Vehicle vehicle = this.parkedVehicle;
        this.parkedVehicle = null;
        this.status = SpotStatus.AVAILABLE;
        return vehicle;
    }
    
    boolean isAvailable() { return status == SpotStatus.AVAILABLE; }
    
    private boolean canFit(Vehicle vehicle) {
        return this.spotType == vehicle.getRequiredSpotType();
    }
    
    String getSpotId() { return spotId; }
    SpotType getSpotType() { return spotType; }
}

// --- ParkingFloor ---

class ParkingFloor {
    private int floorNumber;
    private List<ParkingSpot> spots;
    private Map<SpotType, Integer> availableSpotCount;  // thread-safe counters
    
    ParkingFloor(int floorNumber, int smallSpots, int mediumSpots, int largeSpots) {
        this.floorNumber = floorNumber;
        this.spots = new ArrayList<>();
        this.availableSpotCount = new ConcurrentHashMap<>();
        
        initializeSpots(SpotType.SMALL, smallSpots);
        initializeSpots(SpotType.MEDIUM, mediumSpots);
        initializeSpots(SpotType.LARGE, largeSpots);
    }
    
    // Find first available spot for vehicle type
    Optional<ParkingSpot> findAvailableSpot(VehicleType vehicleType) {
        SpotType required = getRequiredSpotType(vehicleType);
        return spots.stream()
                    .filter(s -> s.getSpotType() == required && s.isAvailable())
                    .findFirst();
    }
    
    int getAvailableCount(SpotType spotType) {
        return availableSpotCount.getOrDefault(spotType, 0);
    }
    
    // Called on entry/exit to update counts
    void updateAvailableCount(SpotType spotType, int delta) {
        availableSpotCount.merge(spotType, delta, Integer::sum);
    }
    
    private void initializeSpots(SpotType type, int count) {
        for (int i = 0; i < count; i++) {
            String id = "F" + floorNumber + "-" + type.toString().charAt(0) + "-" + String.format("%03d", i+1);
            spots.add(new ParkingSpot(id, floorNumber, type));
        }
        availableSpotCount.put(type, count);
    }
    
    private SpotType getRequiredSpotType(VehicleType vehicleType) {
        switch (vehicleType) {
            case MOTORCYCLE: return SpotType.SMALL;
            case CAR:        return SpotType.MEDIUM;
            case TRUCK:      return SpotType.LARGE;
            default: throw new IllegalArgumentException("Unknown vehicle type");
        }
    }
}

// --- Pricing Strategy ---

interface PricingStrategy {
    double calculateFee(VehicleType vehicleType, long durationMinutes);
}

class HourlyPricingStrategy implements PricingStrategy {
    private Map<VehicleType, Double> hourlyRates;
    
    HourlyPricingStrategy() {
        hourlyRates = new EnumMap<>(VehicleType.class);
        hourlyRates.put(VehicleType.MOTORCYCLE, 1.0);  // $1/hour
        hourlyRates.put(VehicleType.CAR,         2.0);  // $2/hour
        hourlyRates.put(VehicleType.TRUCK,        4.0);  // $4/hour
    }
    
    @Override
    public double calculateFee(VehicleType vehicleType, long durationMinutes) {
        double hours = Math.ceil(durationMinutes / 60.0);  // round up to next hour
        return hours * hourlyRates.get(vehicleType);
    }
}

// --- Parking Ticket ---

class ParkingTicket {
    private String ticketId;
    private String spotId;
    private Vehicle vehicle;
    private LocalDateTime entryTime;
    private LocalDateTime exitTime;
    private double fee;
    private PaymentStatus paymentStatus;
    
    ParkingTicket(String ticketId, String spotId, Vehicle vehicle) {
        this.ticketId = ticketId;
        this.spotId = spotId;
        this.vehicle = vehicle;
        this.entryTime = LocalDateTime.now();
        this.paymentStatus = PaymentStatus.PENDING;
    }
    
    void processExit() {
        this.exitTime = LocalDateTime.now();
    }
    
    long getDurationMinutes() {
        return Duration.between(entryTime, exitTime).toMinutes();
    }
    
    // Getters
    String getTicketId()     { return ticketId; }
    Vehicle getVehicle()     { return vehicle; }
    String getSpotId()       { return spotId; }
    LocalDateTime getEntry() { return entryTime; }
    double getFee()          { return fee; }
    void setFee(double fee)  { this.fee = fee; }
    void setPaid()           { paymentStatus = PaymentStatus.COMPLETED; }
}

// --- ParkingLot (Singleton + Facade) ---

class ParkingLot {
    private static ParkingLot instance;
    private static final Object lock = new Object();
    
    private String name;
    private List<ParkingFloor> floors;
    private PricingStrategy pricingStrategy;
    private Map<String, ParkingTicket> activeTickets;  // ticketId → ticket
    private AtomicInteger ticketCounter;
    
    private ParkingLot(String name, PricingStrategy pricingStrategy) {
        this.name = name;
        this.pricingStrategy = pricingStrategy;
        this.floors = new ArrayList<>();
        this.activeTickets = new ConcurrentHashMap<>();
        this.ticketCounter = new AtomicInteger(0);
    }
    
    // Thread-safe Singleton
    static ParkingLot getInstance(String name, PricingStrategy strategy) {
        if (instance == null) {
            synchronized (lock) {
                if (instance == null) {
                    instance = new ParkingLot(name, strategy);
                }
            }
        }
        return instance;
    }
    
    void addFloor(ParkingFloor floor) {
        floors.add(floor);
    }
    
    // Entry gate: find spot, issue ticket
    ParkingTicket park(Vehicle vehicle) {
        for (ParkingFloor floor : floors) {
            Optional<ParkingSpot> spot = floor.findAvailableSpot(vehicle.getVehicleType());
            if (spot.isPresent()) {
                boolean parked = spot.get().park(vehicle);  // thread-safe
                if (parked) {
                    String ticketId = "TKT-" + ticketCounter.incrementAndGet();
                    ParkingTicket ticket = new ParkingTicket(ticketId, spot.get().getSpotId(), vehicle);
                    activeTickets.put(ticketId, ticket);
                    
                    SpotType spotType = spot.get().getSpotType();
                    floor.updateAvailableCount(spotType, -1);
                    
                    return ticket;
                }
            }
        }
        throw new ParkingLotFullException("No available spots for " + vehicle.getVehicleType());
    }
    
    // Exit gate: calculate fee, free spot
    double exit(String ticketId) {
        ParkingTicket ticket = activeTickets.get(ticketId);
        if (ticket == null) throw new InvalidTicketException("Unknown ticket: " + ticketId);
        
        ticket.processExit();
        double fee = pricingStrategy.calculateFee(
            ticket.getVehicle().getVehicleType(),
            ticket.getDurationMinutes()
        );
        ticket.setFee(fee);
        ticket.setPaid();
        
        // Free the spot (find floor by spot ID prefix, then free)
        freeSpot(ticket.getSpotId());
        activeTickets.remove(ticketId);
        
        return fee;
    }
    
    boolean isFull(VehicleType vehicleType) {
        return floors.stream().allMatch(
            f -> f.getAvailableCount(getSpotType(vehicleType)) == 0
        );
    }
    
    private void freeSpot(String spotId) {
        // Parse floor from spotId "F1-M-012" → floor 1
        int floorNumber = Integer.parseInt(spotId.substring(1, 2));
        ParkingFloor floor = floors.get(floorNumber - 1);
        // In real impl, floor would have a map by spotId for O(1) lookup
        // Simplified: scan (or use map)
    }
    
    private SpotType getSpotType(VehicleType vt) {
        switch (vt) {
            case MOTORCYCLE: return SpotType.SMALL;
            case CAR:        return SpotType.MEDIUM;
            case TRUCK:      return SpotType.LARGE;
            default: throw new IllegalArgumentException();
        }
    }
}


3. DESIGN PATTERNS USED
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

SINGLETON: ParkingLot — only one lot per system
  Thread-safe via double-checked locking

STRATEGY: PricingStrategy interface
  Swap pricing without modifying ParkingLot: HourlyPricingStrategy, FlatRateStrategy, WeekendPricingStrategy

INHERITANCE: Vehicle → Car, Motorcycle, Truck
  Each subclass knows its required spot type

FACADE: ParkingLot.park() / exit()
  Hides complexity of floor selection, spot locking, ticket management

OBSERVER (extension): When lot becomes full, notify display signs
  ParkingLot extends Observable
  DisplaySign implements Observer → update "FULL" message


4. INTERVIEW DISCUSSION POINTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

CONCURRENCY: How do we prevent two gates from assigning the same spot?
  ParkingSpot.park() is synchronized — only one thread can enter at a time.
  Using tryLock approach: if synchronized block fails (spot taken by other thread),
  the outer loop tries the next available spot.

SCALABILITY: What if the lot has 10,000 spots and we call park() with full floor scan?
  Optimization: Maintain a PriorityQueue or deque of available spots per type per floor.
  Pop from queue on park, push back on exit → O(log N) instead of O(N) scan.

EDGE CASES:
  - Vehicle tries to exit without a ticket → InvalidTicketException
  - Lot is completely full → ParkingLotFullException → caller shows "FULL"
  - Oversized motorcycle in car spot? Design says no: spot type must match
  - Power outage: tickets persisted in DB, not just in-memory map
