2. CLASS DESIGN
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

enum RoomType      { SINGLE, DOUBLE, QUEEN, KING, SUITE, DELUXE }
enum RoomStatus    { AVAILABLE, OCCUPIED, CLEANING, MAINTENANCE }
enum BookingStatus { PENDING, CONFIRMED, CHECKED_IN, CHECKED_OUT, CANCELLED }
enum BedType       { SINGLE, DOUBLE, KING, QUEEN }

// --- Room ---

class Room {
    private String roomNumber;        // "101", "502"
    private RoomType type;
    private BedType bedType;
    private int floor;
    private double pricePerNight;
    private boolean hasOceanView;
    private RoomStatus status;
    private List<Amenity> amenities;  // pool access, AC, WiFi, etc.
    private int maxOccupancy;
    
    Room(String roomNumber, RoomType type, double pricePerNight, int maxOccupancy) {
        this.roomNumber = roomNumber;
        this.type = type;
        this.pricePerNight = pricePerNight;
        this.maxOccupancy = maxOccupancy;
        this.status = RoomStatus.AVAILABLE;
        this.amenities = new ArrayList<>();
    }
    
    boolean isAvailableFor(LocalDate checkIn, LocalDate checkOut, List<Reservation> reservations) {
        // Check no overlapping reservation exists for this room
        return reservations.stream()
            .filter(r -> r.getRoom().equals(this))
            .filter(r -> r.getStatus() != BookingStatus.CANCELLED)
            .noneMatch(r -> datesOverlap(r.getCheckIn(), r.getCheckOut(), checkIn, checkOut));
    }
    
    private boolean datesOverlap(LocalDate existStart, LocalDate existEnd,
                                  LocalDate newStart, LocalDate newEnd) {
        return !newEnd.isBefore(existStart) && !newStart.isAfter(existEnd);
    }
    
    double getPriceForStay(LocalDate checkIn, LocalDate checkOut) {
        long nights = ChronoUnit.DAYS.between(checkIn, checkOut);
        return nights * pricePerNight;
    }
    
    String getRoomNumber()    { return roomNumber; }
    RoomType getType()        { return type; }
    RoomStatus getStatus()    { return status; }
    double getPricePerNight() { return pricePerNight; }
    void setStatus(RoomStatus s) { this.status = s; }
}

// --- Guest ---

class Guest {
    private String guestId;
    private String name;
    private String email;
    private String phoneNumber;
    private String passportNumber;       // required for international guests
    private List<Reservation> history;
    private int loyaltyPoints;
    
    Guest(String guestId, String name, String email) {
        this.guestId = guestId;
        this.name = name;
        this.email = email;
        this.history = new ArrayList<>();
        this.loyaltyPoints = 0;
    }
    
    void addLoyaltyPoints(int points) { loyaltyPoints += points; }
    boolean redeemPoints(int points) {
        if (loyaltyPoints < points) return false;
        loyaltyPoints -= points;
        return true;
    }
    
    String getGuestId() { return guestId; }
    String getName()    { return name; }
    String getEmail()   { return email; }
    int getLoyaltyPoints() { return loyaltyPoints; }
    List<Reservation> getHistory() { return Collections.unmodifiableList(history); }
}

// --- Reservation ---

class Reservation {
    private String reservationId;
    private Guest guest;
    private Room room;
    private LocalDate checkIn;
    private LocalDate checkOut;
    private int numGuests;
    private BookingStatus status;
    private double totalAmount;
    private double depositPaid;
    private List<RoomServiceOrder> orders;
    private LocalDateTime createdAt;
    
    Reservation(String reservationId, Guest guest, Room room,
                LocalDate checkIn, LocalDate checkOut, int numGuests) {
        this.reservationId = reservationId;
        this.guest = guest;
        this.room = room;
        this.checkIn = checkIn;
        this.checkOut = checkOut;
        this.numGuests = numGuests;
        this.status = BookingStatus.PENDING;
        this.totalAmount = room.getPriceForStay(checkIn, checkOut);
        this.orders = new ArrayList<>();
        this.createdAt = LocalDateTime.now();
    }
    
    boolean canCancelFree() {
        // Free cancellation if >24h before check-in
        return LocalDate.now().isBefore(checkIn.minusDays(1));
    }
    
    double calculateCancellationFee() {
        if (canCancelFree()) return 0;
        return totalAmount * 0.5;  // 50% of total
    }
    
    void addRoomServiceOrder(RoomServiceOrder order) {
        orders.add(order);
        totalAmount += order.getPrice();
    }
    
    void confirm()    { this.status = BookingStatus.CONFIRMED; }
    void checkIn()    { this.status = BookingStatus.CHECKED_IN;  room.setStatus(RoomStatus.OCCUPIED); }
    void checkOut()   { this.status = BookingStatus.CHECKED_OUT; room.setStatus(RoomStatus.CLEANING); }
    void cancel()     { this.status = BookingStatus.CANCELLED; }
    
    String getReservationId() { return reservationId; }
    Guest getGuest()          { return guest; }
    Room getRoom()            { return room; }
    LocalDate getCheckIn()    { return checkIn; }
    LocalDate getCheckOut()   { return checkOut; }
    BookingStatus getStatus() { return status; }
    double getTotalAmount()   { return totalAmount; }
}

// --- Pricing Strategy (handles seasonal/weekend rates) ---

interface RoomPricingStrategy {
    double calculatePrice(Room room, LocalDate checkIn, LocalDate checkOut);
}

class StandardPricingStrategy implements RoomPricingStrategy {
    @Override
    public double calculatePrice(Room room, LocalDate checkIn, LocalDate checkOut) {
        long nights = ChronoUnit.DAYS.between(checkIn, checkOut);
        double total = 0;
        LocalDate date = checkIn;
        while (!date.isEqual(checkOut)) {
            double nightRate = room.getPricePerNight();
            // Weekend surcharge: Fri/Sat cost 20% more
            if (date.getDayOfWeek() == DayOfWeek.FRIDAY || date.getDayOfWeek() == DayOfWeek.SATURDAY) {
                nightRate *= 1.2;
            }
            total += nightRate;
            date = date.plusDays(1);
        }
        return total;
    }
}

class SeasonalPricingStrategy implements RoomPricingStrategy {
    private Map<Month, Double> seasonalMultipliers;
    
    SeasonalPricingStrategy() {
        seasonalMultipliers = new EnumMap<>(Month.class);
        seasonalMultipliers.put(Month.DECEMBER, 1.5);  // peak holiday
        seasonalMultipliers.put(Month.JULY, 1.3);       // summer peak
        // ... others default to 1.0
    }
    
    @Override
    public double calculatePrice(Room room, LocalDate checkIn, LocalDate checkOut) {
        long nights = ChronoUnit.DAYS.between(checkIn, checkOut);
        double multiplier = seasonalMultipliers.getOrDefault(checkIn.getMonth(), 1.0);
        return nights * room.getPricePerNight() * multiplier;
    }
}

// --- Hotel ---

class Hotel {
    private String hotelId;
    private String name;
    private String location;
    private List<Room> rooms;
    private List<Reservation> reservations;    // all reservations
    private RoomPricingStrategy pricingStrategy;
    
    Hotel(String hotelId, String name, String location) {
        this.hotelId = hotelId;
        this.name = name;
        this.location = location;
        this.rooms = new ArrayList<>();
        this.reservations = new ArrayList<>();
        this.pricingStrategy = new StandardPricingStrategy();
    }
    
    // Search: find available rooms of given type for given dates
    List<Room> searchAvailableRooms(RoomType type, LocalDate checkIn, LocalDate checkOut, int numGuests) {
        return rooms.stream()
            .filter(r -> r.getType() == type)
            .filter(r -> r.getMaxOccupancy() >= numGuests)
            .filter(r -> r.isAvailableFor(checkIn, checkOut, reservations))
            .collect(Collectors.toList());
    }
    
    // Book a room (create reservation)
    synchronized Reservation bookRoom(Guest guest, String roomNumber,
                                      LocalDate checkIn, LocalDate checkOut, int numGuests) {
        Room room = getRoomByNumber(roomNumber)
            .orElseThrow(() -> new NotFoundException("Room not found: " + roomNumber));
        
        // Double-check availability (inside synchronized block to prevent race condition)
        if (!room.isAvailableFor(checkIn, checkOut, reservations)) {
            throw new RoomNotAvailableException("Room " + roomNumber + " not available for selected dates");
        }
        
        String reservationId = "RES-" + UUID.randomUUID().toString().substring(0, 8).toUpperCase();
        Reservation reservation = new Reservation(reservationId, guest, room, checkIn, checkOut, numGuests);
        
        // Override price with pricing strategy
        double price = pricingStrategy.calculatePrice(room, checkIn, checkOut);
        reservation.setTotalAmount(price);
        
        reservations.add(reservation);
        guest.getHistory().add(reservation);   // simplified: guest history
        
        return reservation;
    }
    
    void checkIn(String reservationId) {
        Reservation res = getReservation(reservationId);
        if (res.getStatus() != BookingStatus.CONFIRMED) throw new IllegalStateException("Must be CONFIRMED");
        if (!LocalDate.now().isEqual(res.getCheckIn())) throw new IllegalStateException("Not check-in date");
        res.checkIn();
    }
    
    void checkOut(String reservationId) {
        Reservation res = getReservation(reservationId);
        if (res.getStatus() != BookingStatus.CHECKED_IN) throw new IllegalStateException("Not checked in");
        res.checkOut();
        // Award loyalty points: 1 point per $1 spent
        int points = (int) res.getTotalAmount();
        res.getGuest().addLoyaltyPoints(points);
    }
    
    double cancelReservation(String reservationId) {
        Reservation res = getReservation(reservationId);
        if (res.getStatus() == BookingStatus.CHECKED_IN) throw new IllegalStateException("Cannot cancel active stay");
        double cancellationFee = res.calculateCancellationFee();
        res.cancel();
        return cancellationFee;
    }
    
    private Optional<Room> getRoomByNumber(String roomNumber) {
        return rooms.stream().filter(r -> r.getRoomNumber().equals(roomNumber)).findFirst();
    }
    
    private Reservation getReservation(String reservationId) {
        return reservations.stream()
            .filter(r -> r.getReservationId().equals(reservationId))
            .findFirst()
            .orElseThrow(() -> new NotFoundException("Reservation not found"));
    }
}

3. CONCURRENCY: PREVENTING DOUBLE BOOKING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

RACE CONDITION SCENARIO:
  Thread A: searches room 101 for Dec 1-5 → available!
  Thread B: searches room 101 for Dec 1-5 → available!
  Thread A: books room 101
  Thread B: books room 101 → DOUBLE BOOKING!

SOLUTION 1 — Java synchronized (single server):
  synchronized Reservation bookRoom(...) { ... check + insert atomically ... }

SOLUTION 2 — Database-level (real system):
  INSERT INTO reservations (room_id, check_in, check_out, ...)
  with a constraint:
  
  Postgres EXCLUSION CONSTRAINT:
    EXCLUDE USING gist (
        room_id WITH =,
        daterange(check_in, check_out, '[)') WITH &&
    )
  
  This prevents overlapping date ranges at the DB level.
  If two transactions try to insert overlapping, one gets UNIQUE VIOLATION.

SOLUTION 3 — Optimistic Locking:
  rooms table has version INT
  SELECT * FROM rooms WHERE id=? → get version=5
  UPDATE rooms SET version=6 WHERE id=? AND version=5
  If 0 rows updated → someone else changed it → retry

5. INTERVIEW ANSWER — WHAT TO SAY
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

"I'll start by identifying key entities: Hotel, Room, Guest, Reservation.

The key distinction is that Room availability is date-dependent — a room
may be available Jan 1-3 and Jan 6-10 but booked Jan 4-5. I model this
by checking all existing CONFIRMED/CHECKED_IN reservations for overlapping
date ranges.

For pricing, I use the Strategy pattern so the hotel can switch between
standard/seasonal/last-minute pricing without changing core booking logic.

The trickiest part is preventing double-booking. In a single JVM I use
synchronized. In production, I'd use a database exclusion constraint on
(room_id, daterange) — Postgres supports this natively using gist indexes.

State machine for a reservation: PENDING → CONFIRMED → CHECKED_IN →
CHECKED_OUT. Cancellation applies from PENDING or CONFIRMED states only.

I'd also add an Observer so that any state change triggers email/SMS
notifications to the guest automatically."
