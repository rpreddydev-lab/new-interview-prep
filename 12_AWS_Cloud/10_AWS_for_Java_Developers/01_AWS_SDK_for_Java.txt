  ── BASIC OPERATIONS ──────────────────────────────────────────
  
  // Upload
  s3.putObject(
      PutObjectRequest.builder()
          .bucket("my-bucket")
          .key("images/photo.jpg")
          .contentType("image/jpeg")
          .serverSideEncryption(ServerSideEncryption.AWS_KMS)
          .ssekmsKeyId("arn:aws:kms:us-east-1:123456789:key/key-id")
          .build(),
      RequestBody.fromBytes(imageBytes)
  );
  
  // Download
  ResponseBytes<GetObjectResponse> objectBytes = s3.getObjectAsBytes(
      GetObjectRequest.builder().bucket("my-bucket").key("images/photo.jpg").build()
  );
  byte[] data = objectBytes.asByteArray();
  
  // Delete
  s3.deleteObject(DeleteObjectRequest.builder()
      .bucket("my-bucket").key("images/photo.jpg").build());
  
  // List objects (paginated)
  ListObjectsV2Paginator paginator = s3.listObjectsV2Paginator(
      ListObjectsV2Request.builder().bucket("my-bucket").prefix("images/").build()

      System.out.println(s3Object.key() + " - " + s3Object.size());
  });

  ── MULTIPART UPLOAD ──────────────────────────────────────────
  
  Use for files > 5 GB (required), recommended for files > 100 MB
  
  // 1. Initiate multipart upload
  CreateMultipartUploadResponse createResponse = s3.createMultipartUpload(
      CreateMultipartUploadRequest.builder()
          .bucket("my-bucket")
          .key("large-file.zip")
          .build()
  );
  String uploadId = createResponse.uploadId();
  
  // 2. Upload parts (each part minimum 5 MB except last)
  List<CompletedPart> completedParts = new ArrayList<>();
  int partNumber = 1;
  
  for (byte[] chunk : chunks) {
      UploadPartResponse partResponse = s3.uploadPart(
          UploadPartRequest.builder()
              .bucket("my-bucket")
              .key("large-file.zip")
              .uploadId(uploadId)
              .partNumber(partNumber)
              .build(),
          RequestBody.fromBytes(chunk)
      );
      
      completedParts.add(CompletedPart.builder()
          .partNumber(partNumber)
          .eTag(partResponse.eTag())
          .build());
      partNumber++;
  }
  
  // 3. Complete multipart upload
  s3.completeMultipartUpload(
      CompleteMultipartUploadRequest.builder()
          .bucket("my-bucket")
          .key("large-file.zip")
          .uploadId(uploadId)
          .multipartUpload(CompletedMultipartUpload.builder()
              .parts(completedParts).build())
          .build()
  );
  
  // 4. Abort on error
  // s3.abortMultipartUpload(AbortMultipartUploadRequest.builder()
  //     .bucket("my-bucket").key("large-file.zip").uploadId(uploadId).build());

  ── PRESIGNED URLS ──────────────────────────────────────────
  
  // Generate a presigned URL for secure, time-limited upload/download
  S3Presigner presigner = S3Presigner.builder().region(Region.US_EAST_1).build();
  
  // GET presigned URL (let browser download without AWS credentials)
  GetObjectPresignRequest getPresignRequest = GetObjectPresignRequest.builder()
      .signatureDuration(Duration.ofMinutes(15))
      .getObjectRequest(GetObjectRequest.builder()
          .bucket("my-bucket")
          .key("reports/report.pdf")
          .build())
      .build();
  
  PresignedGetObjectRequest presignedGet = presigner.presignGetObject(getPresignRequest);
  URL url = presignedGet.url();  // Share this URL — valid for 15 minutes
  
  // PUT presigned URL (let clients upload directly to S3 without hitting your server)
  PutObjectPresignRequest putPresignRequest = PutObjectPresignRequest.builder()
      .signatureDuration(Duration.ofMinutes(10))
      .putObjectRequest(PutObjectRequest.builder()
          .bucket("my-bucket")
          .key("uploads/user-123/profile.jpg")
          .contentType("image/jpeg")
          .build())
      .build();
  
  PresignedPutObjectRequest presignedPut = presigner.presignPutObject(putPresignRequest);
  URL uploadUrl = presignedPut.url();  // Client sends HTTP PUT to this URL

3. DYNAMODB ENHANCED CLIENT
================================================================

  ── SETUP ──────────────────────────────────────────────────
  
  <!-- pom.xml -->
  <dependency>
    <groupId>software.amazon.awssdk</groupId>
    <artifactId>dynamodb-enhanced</artifactId>
  </dependency>
  
  DynamoDbClient dynamoDbClient = DynamoDbClient.builder()
      .region(Region.US_EAST_1).build();
  
  DynamoDbEnhancedClient enhancedClient = DynamoDbEnhancedClient.builder()
      .dynamoDbClient(dynamoDbClient).build();

  ── BEAN ANNOTATIONS ──────────────────────────────────────
  
  @DynamoDbBean                         // Mark as DynamoDB-mappable
  public class Order {
      
      private String orderId;             // Partition key (PK)
      private String userId;              // Sort key (SK) 
      private String status;
      private Instant createdAt;
      private BigDecimal totalAmount;
      
      @DynamoDbPartitionKey
      public String getOrderId() { return orderId; }
      
      @DynamoDbSortKey
      public String getUserId() { return userId; }
      
      // GSI: query by userId
      @DynamoDbSecondaryPartitionKey(indexNames = "userId-status-index")
      public String getUserId() { return userId; }
      
      @DynamoDbSecondarySortKey(indexNames = "userId-status-index")
      public String getStatus() { return status; }
      
      // Custom attribute name
      @DynamoDbAttribute("created_at")
      public Instant getCreatedAt() { return createdAt; }
      
      // Ignore field
      @DynamoDbIgnore
      public String getTransientField() { return "..."; }
      
      // Setters + no-arg constructor required
      public void setOrderId(String orderId) { this.orderId = orderId; }
      // ...
  }

  ── CRUD OPERATIONS ──────────────────────────────────────
  
  DynamoDbTable<Order> orderTable = enhancedClient
      .table("Orders", TableSchema.fromBean(Order.class));
  
  // CREATE
  Order order = new Order();
  order.setOrderId(UUID.randomUUID().toString());
  order.setUserId("user-123");
  order.setStatus("PENDING");
  order.setCreatedAt(Instant.now());
  order.setTotalAmount(new BigDecimal("49.99"));
  
  orderTable.putItem(order);
  
  // Conditional create (fail if exists)
  orderTable.putItem(PutItemEnhancedRequest.builder(Order.class)
      .item(order)
      .conditionExpression(Expression.builder()
          .expression("attribute_not_exists(orderId)")
          .build())
      .build());
  
  // READ
  Order fetchedOrder = orderTable.getItem(Key.builder()
      .partitionValue("order-abc-123")
      .sortValue("user-123")
      .build());
  
  // UPDATE (replaces entire item — use updateItem for partial update)
  order.setStatus("PROCESSING");
  orderTable.putItem(order);
  
  // Partial update with UpdateItemEnhancedRequest
  // Only updates non-null fields in the Order bean
  orderTable.updateItem(UpdateItemEnhancedRequest.builder(Order.class)
      .item(order)
      .ignoreNulls(true)  // Only update provided fields
      .build());
  
  // DELETE
  orderTable.deleteItem(Key.builder()
      .partitionValue("order-abc-123")
      .sortValue("user-123")
      .build());

  ── QUERY AND SCAN ──────────────────────────────────────
  
  // Query by partition key
  PageIterable<Order> pages = orderTable.query(QueryConditional
      .keyEqualTo(Key.builder().partitionValue("order-abc-123").build()));
  
  pages.items().forEach(item -> System.out.println(item.getStatus()));
  
  // Query with filter expression
  QueryEnhancedRequest queryRequest = QueryEnhancedRequest.builder()
      .queryConditional(QueryConditional.keyEqualTo(
          Key.builder().partitionValue("order-abc-123").build()))
      .filterExpression(Expression.builder()
          .expression("#status = :status")
          .putExpressionName("#status", "status")
          .putExpressionValue(":status", AttributeValue.fromS("PENDING"))
          .build())
      .limit(10)
      .build();
  
  // Query GSI
  DynamoDbIndex<Order> gsi = orderTable.index("userId-status-index");
  PageIterable<Order> gsiPages = gsi.query(QueryConditional
      .keyEqualTo(Key.builder().partitionValue("user-123").build()));
  
  // Scan (expensive — avoid in production for large tables)
  PageIterable<Order> allOrders = orderTable.scan();

  ── TRANSACTIONS ──────────────────────────────────────────
  
  // TransactWriteItems: up to 100 operations atomically
  TransactWriteItemsEnhancedRequest transactRequest = 
      TransactWriteItemsEnhancedRequest.builder()
          .addPutItem(orderTable, order)
          .addUpdateItem(inventoryTable, inventoryUpdate)

4. SQS — COMPREHENSIVE OPERATIONS
================================================================

  SqsClient sqsClient = SqsClient.builder().region(Region.US_EAST_1).build();
  String queueUrl = "https://sqs.us-east-1.amazonaws.com/123456789/order-queue";
  
  ── SEND ──────────────────────────────────────────────────
  
  // Single message
  sqsClient.sendMessage(SendMessageRequest.builder()
      .queueUrl(queueUrl)
      .messageBody(objectMapper.writeValueAsString(order))
      .messageAttributes(Map.of(
          "eventType", MessageAttributeValue.builder()
              .dataType("String").stringValue("ORDER_PLACED").build()
      ))
      .delaySeconds(5)         // Delay visibility by 5s
      .build());
  
  // Batch send (up to 10 messages, max 256 KB total)
  List<SendMessageBatchRequestEntry> entries = orders.stream()
      .map(o -> SendMessageBatchRequestEntry.builder()
          .id(o.getOrderId())        // Batch item ID for tracking failures
          .messageBody(serialize(o))
          .build())
      .collect(toList());
  
  SendMessageBatchResponse batchResponse = sqsClient.sendMessageBatch(
      SendMessageBatchRequest.builder()
          .queueUrl(queueUrl)
          .entries(entries)
          .build()
  );
  
  // Check individual failures
  batchResponse.failed().forEach(f -> 
      log.error("Failed to send {}: {}", f.id(), f.message()));

  ── RECEIVE AND DELETE ──────────────────────────────────
  
  // Receive (up to 10 messages per call)
  ReceiveMessageResponse response = sqsClient.receiveMessage(
      ReceiveMessageRequest.builder()
          .queueUrl(queueUrl)
          .maxNumberOfMessages(10)
          .waitTimeSeconds(20)           // Long polling (prevents empty responses)
          .visibilityTimeout(30)         // 30s to process before re-visible
          .messageAttributeNames("All")  // Include message attributes
          .build()
  );
  
  for (Message msg : response.messages()) {
      try {
          Order order = objectMapper.readValue(msg.body(), Order.class);
          processOrder(order);
          
          // Delete after successful processing
          sqsClient.deleteMessage(DeleteMessageRequest.builder()
              .queueUrl(queueUrl)
              .receiptHandle(msg.receiptHandle())
              .build());
          
      } catch (Exception e) {
          // Don't delete — let it retry after visibility timeout
          log.error("Failed to process message {}", msg.messageId(), e);
      }
  }
  
  // Batch delete for efficiency
  List<DeleteMessageBatchRequestEntry> deleteEntries = processedMessages.stream()
      .map(msg -> DeleteMessageBatchRequestEntry.builder()
          .id(msg.messageId())
          .receiptHandle(msg.receiptHandle())
          .build())
      .collect(toList());
  
  sqsClient.deleteMessageBatch(DeleteMessageBatchRequest.builder()
      .queueUrl(queueUrl)
      .entries(deleteEntries)
      .build());

  ── ASYNC SQS ──────────────────────────────────────────
  
  SqsAsyncClient asyncSqsClient = SqsAsyncClient.builder()
      .region(Region.US_EAST_1).build();
  
  // Non-blocking send
  CompletableFuture<SendMessageResponse> future = asyncSqsClient.sendMessage(
      SendMessageRequest.builder()
          .queueUrl(queueUrl)
          .messageBody(serialize(order))
          .build()
  );
  
  future.thenAccept(resp -> log.info("Sent message: {}", resp.messageId()))
        .exceptionally(ex -> { log.error("Failed", ex); return null; });

5. ERROR HANDLING
================================================================

  try {
      s3.getObject(GetObjectRequest.builder()
          .bucket("my-bucket").key("missing-key").build());
          
  } catch (NoSuchKeyException e) {
      // S3-specific: object doesn't exist
      log.warn("Object not found: {}", e.awsErrorDetails().errorMessage());
      
  } catch (S3Exception e) {
      // Any S3 service error (inherits AwsServiceException)
      log.error("S3 error: {} (HTTP {})", 
          e.awsErrorDetails().errorMessage(),
          e.statusCode());
      
  } catch (SdkClientException e) {
      // Client-side: no internet, timeout, request serialization
      log.error("Client error (network/config issue): {}", e.getMessage());
      
  } catch (AwsServiceException e) {
      // Catch-all for any AWS service error
      String errorCode = e.awsErrorDetails().errorCode();
      int httpStatus = e.statusCode();
      boolean isRetryable = e.retryable();
      
      if (isRetryable) {
          // SDK already retried — consider circuit breaker
      }
  }
  
  // Common exceptions per service:
  // S3: NoSuchKeyException, NoSuchBucketException, S3Exception
  // DynamoDB: ConditionalCheckFailedException, TransactionCanceledException
  // SQS: QueueDoesNotExistException, SqsException
  // KMS: KmsKeyNotFoundException, KmsDisabledException, AccessDeniedException
  // General: SdkClientException (network), AwsServiceException (AWS-side errors)

7. INTERVIEW QUESTIONS
================================================================

Q: How do presigned URLs work in S3 and when would you use them?
A: A presigned URL is a time-limited URL that embeds AWS credentials (access key ID, 
   signature) in the URL itself, allowing the holder to perform a specific S3 operation 
   without having AWS credentials.
   
   How it works:
   1. Your backend (with AWS credentials) calls S3Presigner to generate the URL
   2. The URL contains: bucket/key, expiry time, signature (HMAC-SHA256 of request)
   3. You return the URL to the client (browser, mobile app)
   4. Client uses the URL directly — S3 validates the signature and expiry
   
   Use cases:
   - Presigned GET: Let authenticated users download private S3 files without 
     exposing your S3 bucket publicly or proxying through your server
   - Presigned PUT: Let clients upload directly to S3 (bypasses your server for 
     large uploads), reducing your server bandwidth costs
   - Example: User profile picture upload — generate presigned PUT, client uploads 
     directly to S3, you get notified via S3 event notification
   
   Limitations: Maximum expiry is 7 days (604800 seconds). URL is single-use-by-design 
   (though technically it can be used multiple times until expiry — use one-time tokens 
   in your app logic if strict single-use is required).

Q: Explain the DynamoDB Enhanced Client vs the low-level DynamoDbClient. 
   When would you use each?
A: DynamoDbClient (low-level): Uses raw AttributeValue maps. More verbose but full control.
   DynamoDbEnhancedClient: Annotations on POJOs → automatic marshaling/unmarshaling.
   
   Low-level example:
   Map<String, AttributeValue> item = Map.of(
       "orderId", AttributeValue.fromS("abc"),
       "amount",  AttributeValue.fromN("49.99")
   );
   dynamoDbClient.putItem(PutItemRequest.builder()
       .tableName("Orders").item(item).build());
   
   Enhanced example:
   @DynamoDbBean class Order { @DynamoDbPartitionKey String orderId; ... }
   orderTable.putItem(order);
   
   Use Enhanced Client when: Building application code with domain objects.
   Use Low-level Client when: Writing generic utilities, dynamic attribute handling, 
   maximum performance (no reflection), or when working with variable schemas.
   
   Internally, Enhanced Client uses the low-level client and adds a TableSchema 
   layer for marshaling. Both are thread-safe.

Q: What is the difference between SdkClientException and AwsServiceException?
A: Both inherit from SdkException but represent different failure categories:
   
   SdkClientException: Error BEFORE the request reaches AWS.
   - Network timeout, no internet connection
   - Request couldn't be serialized
   - DNS lookup failure
   - SSL/TLS handshake failure
   - These are retried by SDK based on retry policy
   
   AwsServiceException: Error AFTER AWS received the request (HTTP 4xx/5xx).
   - 400 Bad Request: Invalid parameters
   - 403 AccessDeniedException: IAM permission denied
   - 404 NoSuchKeyException: Resource not found
   - 429 ThrottlingException: Rate limit exceeded (SDK retries these)
   - 500 InternalServiceErrorException: AWS-side issue (SDK retries these)
   - Has .awsErrorDetails() with errorCode, errorMessage, httpStatusCode
   - Has .retryable() to indicate if SDK has already retried
   
   In catch blocks, catch specific exceptions first (NoSuchKeyException), 
   then broader ones (S3Exception → AwsServiceException) for fallback handling.
