  import org.springframework.data.cassandra.core.mapping.*;
  import org.springframework.data.annotation.Id;

  @Table("orders_by_user")
  public class OrderByUser {

      @PrimaryKeyColumn(name = "user_id", type = PrimaryKeyType.PARTITIONED)
      private UUID userId;

      @PrimaryKeyColumn(name = "created_at", type = PrimaryKeyType.CLUSTERED,
                        ordering = Ordering.DESCENDING)
      private Instant createdAt;

      @PrimaryKeyColumn(name = "order_id", type = PrimaryKeyType.CLUSTERED)
      private UUID orderId;

      @Column("status")
      private String status;

      @Column("total_amount")
      private BigDecimal totalAmount;

      // getters, setters, constructors
  }

REPOSITORY
================================================================

  public interface OrderByUserRepository
          extends CassandraRepository<OrderByUser, UUID> {

      // Always include partition key:
      List<OrderByUser> findByUserId(UUID userId);

      // Partition key + clustering column range:
      List<OrderByUser> findByUserIdAndCreatedAtBetween(
          UUID userId, Instant from, Instant to);

      // Limit results:
      List<OrderByUser> findTop10ByUserIdOrderByCreatedAtDesc(UUID userId);
  }

LIGHTWEIGHT TRANSACTIONS (PAXOS)
================================================================

  // INSERT IF NOT EXISTS (idempotent registration):
  INSERT INTO users (user_id, email) VALUES (?, ?) IF NOT EXISTS;

  // UPDATE with condition:
  UPDATE orders SET status = 'CONFIRMED'
  WHERE order_id = ?
  IF status = 'PENDING';

  In Spring:
  cassandraTemplate.query(
      "UPDATE orders SET status='CONFIRMED' WHERE order_id=? IF status='PENDING'",
      orderId
  );
  -- Returns ResultSet; check wasApplied() on first Row to know if condition was met

================================================================
INTERVIEW Q&A
================================================================

  Q: Why must you always include the partition key in a Cassandra WHERE clause?
  A: Cassandra routes queries to the correct node(s) using the partition key
     hash. Without the partition key, Cassandra would have to broadcast the
     query to all nodes (ALLOW FILTERING, full table scan), which is O(n)
     and explicitly discouraged in production. Model your tables around the
     queries you need, with the partition key matching your primary query filter.

  Q: What is a lightweight transaction in Cassandra?
  A: A PAXOS-based conditional write: IF NOT EXISTS or IF column = value.
     Provides compare-and-set semantics. Much slower than regular writes
     (requires 4 round trips). Use sparingly: registration (unique email),
     inventory decrement without going negative. SERIAL or LOCAL_SERIAL
     consistency is required.
