PROBLEM 1: Second Highest Salary   (LC #176 — Easy/Medium)
===================================================================
Table: Employee (id INT, salary INT)

-- Q: Find the second highest distinct salary. Return NULL if it doesn't exist.

Solution:
  SELECT MAX(salary) AS SecondHighestSalary
  FROM Employee
  WHERE salary < (SELECT MAX(salary) FROM Employee);
  -- Returns NULL automatically if there's only one distinct salary.

  -- Alternative using DENSE_RANK:
  SELECT salary AS SecondHighestSalary
  FROM (
      SELECT salary, DENSE_RANK() OVER (ORDER BY salary DESC) AS rnk
      FROM Employee
  ) t
  WHERE rnk = 2
  LIMIT 1;

  -- Wrap in subquery to return NULL (not empty set) when missing:
  SELECT (
      SELECT DISTINCT salary FROM Employee
      ORDER BY salary DESC LIMIT 1 OFFSET 1
  ) AS SecondHighestSalary;

PROBLEM 2: Nth Highest Salary   (LC #177 — Medium)
===================================================================
Table: Employee (id INT, salary INT)
-- Q: Create a function / query to find the Nth highest salary.

Solution (as a query for any N):
  SELECT salary AS getNthHighestSalary
  FROM (
      SELECT salary, DENSE_RANK() OVER (ORDER BY salary DESC) AS rnk
      FROM Employee
  ) t
  WHERE rnk = N   -- substitute N
  LIMIT 1;

  -- Or with OFFSET (no ties support):
  SELECT DISTINCT salary
  FROM Employee
  ORDER BY salary DESC
  LIMIT 1 OFFSET N-1;

PROBLEM 3: Rank Scores   (LC #178 — Medium)
===================================================================
Table: Scores (id INT, score DECIMAL)
-- Q: Rank scores in descending order. Ties share the same rank (no gaps).

Solution:
  SELECT
      score,
      DENSE_RANK() OVER (ORDER BY score DESC) AS `rank`
  FROM Scores
  ORDER BY score DESC;

  -- Note: `rank` is a reserved word — backtick-quote it in MySQL.

PROBLEM 4: Consecutive Numbers   (LC #180 — Medium)
===================================================================
Table: Logs (id INT AUTO_INCREMENT, num INT)
-- Q: Find all numbers that appear at least 3 times consecutively.

Solution:
  SELECT DISTINCT l1.num AS ConsecutiveNums
  FROM Logs l1
  JOIN Logs l2 ON l2.id = l1.id + 1 AND l2.num = l1.num
  JOIN Logs l3 ON l3.id = l1.id + 2 AND l3.num = l1.num;

  -- Window function approach:
  SELECT DISTINCT num AS ConsecutiveNums
  FROM (
      SELECT
          num,
          LEAD(num, 1) OVER (ORDER BY id) AS next1,
          LEAD(num, 2) OVER (ORDER BY id) AS next2
      FROM Logs
  ) t
  WHERE num = next1 AND num = next2;

PROBLEM 5: Department Highest Salary   (LC #184 — Medium)
===================================================================
Tables:
  Employee   (id, name, salary, departmentId)
  Department (id, name)
-- Q: Find employees who earn the highest salary in their department.

Solution:
  WITH dept_max AS (
      SELECT departmentId, MAX(salary) AS max_salary
      FROM Employee
      GROUP BY departmentId
  )
  SELECT
      d.name   AS Department,
      e.name   AS Employee,
      e.salary AS Salary
  FROM Employee e
  JOIN Department d  ON d.id = e.departmentId
  JOIN dept_max  dm  ON dm.departmentId = e.departmentId
                     AND e.salary = dm.max_salary;

PROBLEM 6: Department Top Three Salaries   (LC #185 — Hard)
===================================================================
Tables: Employee (id, name, salary, departmentId), Department (id, name)
-- Q: Find employees who have the top 3 DISTINCT salaries in their dept.

Solution:
  WITH ranked AS (
      SELECT
          e.name           AS Employee,
          d.name           AS Department,
          e.salary,
          DENSE_RANK() OVER (PARTITION BY e.departmentId ORDER BY e.salary DESC) AS rnk
      FROM Employee e
      JOIN Department d ON d.id = e.departmentId
  )
  SELECT Department, Employee, salary AS Salary
  FROM ranked
  WHERE rnk <= 3
  ORDER BY Department, salary DESC;

PROBLEM 7: Delete Duplicate Emails   (LC #196 — Easy)
===================================================================
Table: Person (id INT, email VARCHAR)
-- Q: Delete all duplicate emails, keeping only the row with smallest id.

Solution:
  DELETE FROM Person
  WHERE id NOT IN (
      SELECT min_id FROM (
          SELECT MIN(id) AS min_id FROM Person GROUP BY email
      ) t
  );

  -- MySQL workaround (can't reference same table in subquery directly):
  -- Wrap in subquery alias.

  -- CTE approach (PostgreSQL):
  WITH dupes AS (
      SELECT id, ROW_NUMBER() OVER (PARTITION BY email ORDER BY id) AS rn
      FROM Person
  )
  DELETE FROM Person WHERE id IN (SELECT id FROM dupes WHERE rn > 1);

PROBLEM 8: Rising Temperature   (LC #197 — Easy)
===================================================================
Table: Weather (id INT, recordDate DATE, temperature INT)
-- Q: Find all dates where temperature was higher than the previous day.

Solution:
  SELECT w1.id
  FROM Weather w1
  JOIN Weather w2
    ON w2.recordDate = w1.recordDate - INTERVAL 1 DAY
  WHERE w1.temperature > w2.temperature;

  -- Window function approach:
  SELECT id
  FROM (
      SELECT
          id,
          temperature,
          LAG(temperature) OVER (ORDER BY recordDate) AS prev_temp,
          LAG(recordDate)  OVER (ORDER BY recordDate) AS prev_date,
          recordDate
      FROM Weather
  ) t
  WHERE temperature > prev_temp
    AND recordDate = prev_date + INTERVAL 1 DAY;   -- guard against gaps in dates

PROBLEM 9: Employees Earning More Than Their Managers   (LC #181 — Easy)
===================================================================
Table: Employee (id, name, salary, managerId)
-- Q: Find employees who earn more than their direct manager.

Solution:
  SELECT e.name AS Employee
  FROM Employee e
  JOIN Employee m ON m.id = e.managerId
  WHERE e.salary > m.salary;

PROBLEM 10: Customers Who Never Order   (LC #183 — Easy)
===================================================================
Tables: Customers (id, name), Orders (id, customerId)
-- Q: Find customers who never placed any order.

Solution:
  -- LEFT JOIN approach:
  SELECT c.name AS Customers
  FROM Customers c
  LEFT JOIN Orders o ON o.customerId = c.id
  WHERE o.id IS NULL;

  -- NOT IN approach:
  SELECT name AS Customers
  FROM Customers
  WHERE id NOT IN (SELECT DISTINCT customerId FROM Orders);

  -- NOT EXISTS approach:
  SELECT name AS Customers
  FROM Customers c
  WHERE NOT EXISTS (SELECT 1 FROM Orders o WHERE o.customerId = c.id);

PROBLEM 11: Game Play Analysis IV   (LC #550 — Medium)
===================================================================
Table: Activity (player_id, device_id, event_date DATE, games_played INT)
  Primary key: (player_id, event_date)
-- Q: Report the fraction of players that logged in again on the day
--    after their first login (round to 2 decimal places).

Solution:
  WITH first_login AS (
      SELECT player_id, MIN(event_date) AS first_date
      FROM Activity
      GROUP BY player_id
  ),
  next_day_login AS (
      SELECT a.player_id
      FROM Activity a
      JOIN first_login f ON f.player_id = a.player_id
      WHERE a.event_date = f.first_date + INTERVAL 1 DAY
  )
  SELECT
      ROUND(COUNT(DISTINCT n.player_id) * 1.0 / COUNT(DISTINCT f.player_id), 2) AS fraction
  FROM first_login f
  LEFT JOIN next_day_login n ON n.player_id = f.player_id;

PROBLEM 12: Trips and Users   (LC #262 — Hard)
===================================================================
Tables:
  Trips (id, client_id, driver_id, city_id, status, request_at DATE)
  Users (users_id, banned ENUM('Yes','No'), role ENUM('client','driver','partner'))
  status: 'completed','cancelled_by_driver','cancelled_by_client'
-- Q: Find the cancellation rate (rounded to 2 decimals) for unbanned
--    clients/drivers between 2013-10-01 and 2013-10-03.

Solution:
  SELECT
      t.request_at                              AS Day,
      ROUND(
          SUM(CASE WHEN t.status != 'completed' THEN 1 ELSE 0 END)
          * 1.0 / COUNT(*),
          2
      )                                         AS 'Cancellation Rate'
  FROM Trips t
  JOIN Users c ON c.users_id = t.client_id  AND c.banned = 'No' AND c.role = 'client'
  JOIN Users d ON d.users_id = t.driver_id  AND d.banned = 'No' AND d.role = 'driver'
  WHERE t.request_at BETWEEN '2013-10-01' AND '2013-10-03'
  GROUP BY t.request_at
  ORDER BY t.request_at;

PROBLEM 13: Friend Requests II: Who Has the Most Friends (LC #602 — Medium)
===================================================================
Table: RequestAccepted (requester_id INT, accepter_id INT, accept_date DATE)
-- Q: Find the person with the most total friends (as requester or accepter).

Solution:
  WITH all_friendships AS (
      SELECT requester_id AS person FROM RequestAccepted
      UNION ALL
      SELECT accepter_id  AS person FROM RequestAccepted
  )
  SELECT person AS id, COUNT(*) AS num
  FROM all_friendships
  GROUP BY person
  ORDER BY num DESC
  LIMIT 1;

PROBLEM 14: Median Employee Salary   (LC #569 — Hard)
===================================================================
Table: Employee (id INT, company VARCHAR, salary INT)
-- Q: Find the median salary for each company.

Solution:
  WITH ranked AS (
      SELECT
          id,
          company,
          salary,
          ROW_NUMBER() OVER (PARTITION BY company ORDER BY salary, id) AS rn,
          COUNT(*)     OVER (PARTITION BY company)                      AS total
      FROM Employee
  )
  SELECT id, company, salary
  FROM ranked
  WHERE rn BETWEEN total / 2.0 AND total / 2.0 + 1;
  -- For odd count (e.g. 5): rn 2.5..3.5 → matches rn=3
  -- For even count (e.g. 6): rn 3.0..4.0 → matches rn=3 and rn=4

PROBLEM 15: Human Traffic of Stadium   (LC #601 — Hard)
===================================================================
Table: Stadium (id INT, visit_date DATE, people INT)
-- Q: Display the records with 3 or more consecutive id rows
--    where people >= 100.

Solution:
  WITH over100 AS (
      SELECT
          id,
          visit_date,
          people,
          id - ROW_NUMBER() OVER (ORDER BY id) AS grp
      FROM Stadium
      WHERE people >= 100
  ),
  valid_groups AS (
      SELECT grp
      FROM over100
      GROUP BY grp
      HAVING COUNT(*) >= 3
  )
  SELECT id, visit_date, people
  FROM over100
  WHERE grp IN (SELECT grp FROM valid_groups)
  ORDER BY visit_date;

  -- Key trick: id - ROW_NUMBER() is constant for consecutive ids.
  --            A gap in id resets the island → new grp value.

---------------------------------------|-----------------------------
Nth highest / 2nd max                  → DENSE_RANK or nested MAX
Rank with ties                         → DENSE_RANK() OVER

3+ consecutive rows                    → id - ROW_NUMBER() island trick
Delete duplicates                      → ROW_NUMBER CTE + DELETE
Temperature vs previous day            → LAG or self-JOIN on date-1
Never ordered / unmatched              → LEFT JOIN + IS NULL / NOT EXISTS
First login + day-after retention      → MIN group + JOIN on date+1
Cancellation rate                      → JOIN to filter banned + CASE WHEN
