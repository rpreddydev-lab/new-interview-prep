===================================================================
  NULL HANDLING PATTERNS
  (The most common source of silent wrong answers in SQL)
===================================================================

===================================================================
PART 1: HOW NULL BEHAVES — THE RULES
===================================================================

NULL means "unknown". Any comparison with NULL yields UNKNOWN (not TRUE or FALSE).

  NULL = NULL      → UNKNOWN  (not TRUE!)
  NULL != NULL     → UNKNOWN
  NULL > 5         → UNKNOWN
  NULL + 5         → NULL
  NULL OR TRUE     → TRUE     (exception: OR with TRUE short-circuits)
  NULL AND FALSE   → FALSE    (exception: AND with FALSE short-circuits)
  NULL OR FALSE    → NULL
  NULL AND TRUE    → NULL

-- Check for NULL correctly:
  WHERE column IS NULL        -- ✅ correct
  WHERE column = NULL         -- ❌ always returns no rows (= NULL is UNKNOWN)
  WHERE column IS NOT NULL    -- ✅ correct
  WHERE column != NULL        -- ❌ never works

===================================================================
PART 2: NULL IN AGGREGATE FUNCTIONS
===================================================================

-- Aggregate functions IGNORE NULLs (except COUNT(*))
  COUNT(*)              → counts ALL rows including NULLs
  COUNT(column)         → counts only non-NULL values in that column
  SUM(column)           → ignores NULLs (returns NULL if ALL are NULL)
  AVG(column)           → ignores NULLs (denominator = non-NULL count)
  MIN(column)           → ignores NULLs
  MAX(column)           → ignores NULLs

-- Example:
  salary: 50000, 60000, NULL, 70000
  COUNT(*)        → 4
  COUNT(salary)   → 3
  SUM(salary)     → 180000  (NULL ignored)
  AVG(salary)     → 60000   (180000 / 3, not /4!)

-- Q: Count how many employees have no manager assigned.
SELECT COUNT(*) AS no_manager_count
FROM employees
WHERE manager_id IS NULL;

-- Q: What percentage of employees have salary data missing?
SELECT
    COUNT(*)                          AS total,
    COUNT(salary)                     AS has_salary,
    COUNT(*) - COUNT(salary)          AS missing_salary,
    ROUND((COUNT(*) - COUNT(salary)) * 100.0 / COUNT(*), 2) AS pct_missing
FROM employees;

===================================================================
PART 3: COALESCE — REPLACE NULL WITH A DEFAULT
===================================================================

COALESCE(expr1, expr2, ..., exprN)
  → Returns the FIRST non-NULL expression. Short-circuits.

-- Q: Show salary; display 0 if salary is NULL.
SELECT name, COALESCE(salary, 0) AS salary FROM employees;

-- Q: Show manager name, or 'No Manager' if manager_id is NULL.
SELECT
    e.name,
    COALESCE(m.name, 'No Manager') AS manager_name
FROM employees e
LEFT JOIN employees m ON e.manager_id = m.id;

-- Q: Use first available contact: phone, then email, then 'Unknown'.
SELECT
    name,
    COALESCE(phone, email, 'Unknown') AS contact
FROM contacts;

-- COALESCE for arithmetic (avoid NULL propagation):
SELECT
    name,
    base_salary + COALESCE(bonus, 0) AS total_compensation
FROM employees;

===================================================================
PART 4: NULLIF — PRODUCE NULL FROM A SPECIFIC VALUE
===================================================================

NULLIF(expr1, expr2)
  → Returns NULL if expr1 = expr2, otherwise returns expr1.
  → Main use: PREVENT DIVISION BY ZERO.

-- Q: Calculate conversion rate; avoid divide-by-zero.
SELECT
    campaign,
    conversions,
    impressions,
    ROUND(conversions * 100.0 / NULLIF(impressions, 0), 2) AS conversion_rate
FROM campaigns;

-- Q: Replace empty strings with NULL.
SELECT NULLIF(name, '') AS name FROM users;

-- NULLIF + COALESCE together:
SELECT COALESCE(NULLIF(name, ''), 'Anonymous') AS display_name FROM users;

===================================================================
PART 5: NULL IN JOINS — LEFT JOIN TRAP
===================================================================

-- Q: Find customers with no orders (NULL after LEFT JOIN).
SELECT c.id, c.name
FROM customers c
LEFT JOIN orders o ON o.customer_id = c.id
WHERE o.id IS NULL;   -- ← must check a column from the RIGHT table, not c.id

-- Common mistake: checking left-table column for NULL (always non-null if PK):
WHERE c.id IS NULL    -- ❌ wrong — c.id is a PK, never null
WHERE o.id IS NULL    -- ✅ correct — o.id is null when no match

-- Q: Show all employees with their department name.
--    Employees with no department still appear.
SELECT
    e.name,
    COALESCE(d.name, 'Unassigned') AS department
FROM employees e
LEFT JOIN departments d ON e.department = d.name;

===================================================================
PART 6: NULL IN WHERE — IN vs NOT IN TRAP
===================================================================

-- CRITICAL: NOT IN with a subquery that returns NULLs returns NO rows!
-- Because: x NOT IN (1, 2, NULL) → x != 1 AND x != 2 AND x != NULL
--          x != NULL → UNKNOWN → entire expression = UNKNOWN

-- BAD (if subquery can return NULLs):
SELECT * FROM customers
WHERE id NOT IN (SELECT customer_id FROM orders);
-- If ANY customer_id in orders is NULL → returns EMPTY SET!

-- SAFE alternatives:
-- Option 1: Filter NULLs in subquery
WHERE id NOT IN (SELECT customer_id FROM orders WHERE customer_id IS NOT NULL);

-- Option 2: Use NOT EXISTS (null-safe)
WHERE NOT EXISTS (SELECT 1 FROM orders o WHERE o.customer_id = c.id);

-- Option 3: Use LEFT JOIN + IS NULL
LEFT JOIN orders o ON o.customer_id = c.id WHERE o.id IS NULL

===================================================================
PART 7: NULL IN GROUP BY AND ORDER BY
===================================================================

-- GROUP BY: NULLs are grouped together as one group.
SELECT department, COUNT(*) FROM employees GROUP BY department;
-- Employees with NULL department form their own group: (NULL, count)

-- ORDER BY: NULLs sort LAST in ASC, FIRST in DESC (MySQL/PostgreSQL default)
-- To control NULL sort position:
ORDER BY salary DESC NULLS LAST    -- PostgreSQL
ORDER BY salary DESC NULLS FIRST

-- MySQL workaround (no NULLS LAST/FIRST):
ORDER BY salary IS NULL, salary DESC   -- NULLs last
ORDER BY salary IS NOT NULL, salary    -- NULLs first

===================================================================
PART 8: NULL IN CASE WHEN
===================================================================

-- CASE evaluates conditions top to bottom; NULL conditions fall through.
SELECT
    name,
    CASE
        WHEN salary IS NULL   THEN 'No Data'
        WHEN salary >= 90000  THEN 'Senior'
        WHEN salary >= 60000  THEN 'Mid'
        ELSE 'Junior'
    END AS level
FROM employees;

-- CASE WHEN with column = NULL (wrong — never matches):
CASE WHEN salary = NULL THEN 'No Data'   -- ❌ always skipped
CASE WHEN salary IS NULL THEN 'No Data'  -- ✅ correct

===================================================================
PART 9: NULL-SAFE EQUALITY (MySQL specific)
===================================================================

-- <=> operator: NULL-safe equality (returns 1 if both are NULL)
SELECT * FROM employees WHERE manager_id <=> NULL;   -- same as IS NULL
SELECT 1 <=> 1   -- 1 (true)
SELECT 1 <=> NULL -- 0 (false)
SELECT NULL <=> NULL -- 1 (true)

-- PostgreSQL uses IS NOT DISTINCT FROM for null-safe equality:
WHERE manager_id IS NOT DISTINCT FROM NULL

===================================================================
PART 10: COMMON NULL INTERVIEW QUESTIONS
===================================================================

-- Q1: Write a query to show employees whose commission is NULL.
SELECT name FROM employees WHERE commission IS NULL;

-- Q2: Replace NULL commission with 0 and compute total pay.
SELECT name, salary + COALESCE(commission, 0) AS total_pay FROM employees;

-- Q3: Why does this query return no results?
--     SELECT * FROM employees WHERE manager_id = NULL;
-- Answer: NULL comparisons always yield UNKNOWN. Use IS NULL instead.

-- Q4: How many distinct departments are there (excluding NULL)?
SELECT COUNT(DISTINCT department) FROM employees;
-- COUNT(DISTINCT col) automatically excludes NULLs.

-- Q5: Find employees who share the same manager
--     (manager_id not null, find pairs).
SELECT a.name AS emp1, b.name AS emp2, a.manager_id
FROM employees a
JOIN employees b ON a.manager_id = b.manager_id AND a.id < b.id
WHERE a.manager_id IS NOT NULL;

===================================================================
NULL HANDLING QUICK REFERENCE
===================================================================
Scenario                              → Use
--------------------------------------|-----------------------------
Replace NULL with default value        → COALESCE(col, default)
Prevent division by zero              → NULLIF(denominator, 0)
Filter rows where column is NULL      → WHERE col IS NULL
Filter rows where column is not NULL  → WHERE col IS NOT NULL
NOT IN with possible NULLs in subquery→ NOT EXISTS or add IS NOT NULL
Sort NULLs last in PostgreSQL         → ORDER BY col NULLS LAST
Sort NULLs last in MySQL              → ORDER BY col IS NULL, col
Aggregate ignoring NULLs              → SUM/AVG/COUNT(col) (auto)
Count including NULLs                 → COUNT(*)
