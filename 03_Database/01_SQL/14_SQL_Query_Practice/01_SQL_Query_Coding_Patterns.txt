REFERENCE SCHEMA — memorise this, it's used in all examples
------------------------------------------------------------
CREATE TABLE employees (
    id          INT PRIMARY KEY,
    name        VARCHAR(100),
    department  VARCHAR(50),
    salary      DECIMAL(10,2),
    hire_date   DATE,
    manager_id  INT            -- NULL for top-level managers
);

CREATE TABLE departments (
    id       INT PRIMARY KEY,
    name     VARCHAR(50),
    location VARCHAR(100)
);

CREATE TABLE customers (
    id      INT PRIMARY KEY,
    name    VARCHAR(100),
    city    VARCHAR(50),
    country VARCHAR(50)
);

CREATE TABLE orders (
    id          INT PRIMARY KEY,
    customer_id INT,
    order_date  DATE,
    total       DECIMAL(10,2),
    status      VARCHAR(20)   -- 'PENDING','SHIPPED','DELIVERED','CANCELLED'
);

CREATE TABLE products (
    id       INT PRIMARY KEY,
    name     VARCHAR(100),
    category VARCHAR(50),
    price    DECIMAL(10,2)
);

CREATE TABLE order_items (
    id         INT PRIMARY KEY,
    order_id   INT,
    product_id INT,
    qty        INT,
    unit_price DECIMAL(10,2)
);

===================================================================
PATTERN 1: AGGREGATE + GROUP BY + HAVING
===================================================================
Clue words: "count per", "total per", "average by", "more than N"

-- Q: Find departments where the average salary exceeds 70000.
SELECT
    department,
    AVG(salary)  AS avg_salary,
    COUNT(*)     AS headcount
FROM employees
GROUP BY department
HAVING AVG(salary) > 70000
ORDER BY avg_salary DESC;

-- Q: Find customers who placed more than 3 orders.
SELECT
    c.name,
    COUNT(o.id) AS order_count
FROM customers c
JOIN orders o ON o.customer_id = c.id
GROUP BY c.id, c.name
HAVING COUNT(o.id) > 3;

-- Q: Total revenue per product category (only categories > 10000).
SELECT
    p.category,
    SUM(oi.qty * oi.unit_price) AS revenue
FROM order_items oi
JOIN products p ON p.id = oi.product_id
GROUP BY p.category
HAVING SUM(oi.qty * oi.unit_price) > 10000
ORDER BY revenue DESC;

===================================================================
PATTERN 2: TOP N PER GROUP  (most-asked window function pattern)
===================================================================
Clue words: "top N per", "highest salary in each", "latest order per customer"

-- Q: Get the 2 highest-paid employees in each department.
WITH ranked AS (
    SELECT
        name,
        department,
        salary,
        ROW_NUMBER() OVER (PARTITION BY department ORDER BY salary DESC) AS rn
    FROM employees
)
SELECT name, department, salary
FROM ranked
WHERE rn <= 2;

-- ROW_NUMBER vs RANK vs DENSE_RANK — choose carefully:
--   ROW_NUMBER : exactly N rows, tie-broken arbitrarily  ← use for "get N rows"
--   RANK       : ties share rank, next rank SKIPS        ← use for "top N ranks"
--   DENSE_RANK : ties share rank, no gap                 ← use for "Nth distinct salary"

-- Q: Get the latest order for each customer.
WITH latest AS (
    SELECT
        customer_id,
        id          AS order_id,
        order_date,
        total,
        ROW_NUMBER() OVER (PARTITION BY customer_id ORDER BY order_date DESC) AS rn
    FROM orders
)
SELECT customer_id, order_id, order_date, total
FROM latest
WHERE rn = 1;

===================================================================
PATTERN 3: Nth HIGHEST VALUE  (subquery or DENSE_RANK)
===================================================================
Clue words: "second highest", "third largest", "Nth distinct"

-- Q: Find the 2nd highest salary (without window functions).
SELECT MAX(salary) AS second_highest
FROM employees
WHERE salary < (SELECT MAX(salary) FROM employees);

-- Q: Find the Nth highest salary (generalised — works for any N).
SELECT salary
FROM (
    SELECT salary, DENSE_RANK() OVER (ORDER BY salary DESC) AS rnk
    FROM employees
) ranked
WHERE rnk = 2;   -- change 2 to N

-- Q: 3rd highest salary per department.
WITH ranked AS (
    SELECT
        name,
        department,
        salary,
        DENSE_RANK() OVER (PARTITION BY department ORDER BY salary DESC) AS rnk
    FROM employees
)
SELECT name, department, salary
FROM ranked
WHERE rnk = 3;

===================================================================
PATTERN 4: FIND DUPLICATES / DEDUPLICATE
===================================================================
Clue words: "duplicate emails", "duplicate rows", "keep only one"

-- Q: Find all duplicate email addresses.
SELECT email, COUNT(*) AS cnt
FROM users
GROUP BY email
HAVING COUNT(*) > 1;

-- Q: Find employee records that appear more than once (same name + dept).
SELECT name, department, COUNT(*) AS cnt
FROM employees
GROUP BY name, department
HAVING COUNT(*) > 1;

-- Q: Delete duplicates, keep the row with the lowest id.
-- (Works in MySQL, PostgreSQL with CTE)
WITH dupes AS (
    SELECT
        id,
        ROW_NUMBER() OVER (PARTITION BY email ORDER BY id ASC) AS rn
    FROM users
)
DELETE FROM users
WHERE id IN (SELECT id FROM dupes WHERE rn > 1);

-- Q: Select one row per group without deleting (just SELECT dedup).
WITH deduped AS (
    SELECT *, ROW_NUMBER() OVER (PARTITION BY email ORDER BY id) AS rn
    FROM users
)
SELECT id, name, email FROM deduped WHERE rn = 1;

===================================================================
PATTERN 5: SELF JOIN  (compare rows within same table)
===================================================================
Clue words: "employees and their manager", "find employees earning more than manager",
            "pairs of employees in same department"

-- Q: List each employee with their manager's name.
SELECT
    e.name   AS employee,
    m.name   AS manager
FROM employees e
LEFT JOIN employees m ON e.manager_id = m.id;

-- Q: Find employees who earn MORE than their manager.
SELECT
    e.name    AS employee,
    e.salary  AS emp_salary,
    m.name    AS manager,
    m.salary  AS mgr_salary
FROM employees e
JOIN employees m ON e.manager_id = m.id
WHERE e.salary > m.salary;

-- Q: Find all pairs of employees in the same department (no self-pairing, no duplicates).
SELECT
    a.name AS emp1,
    b.name AS emp2,
    a.department
FROM employees a
JOIN employees b ON a.department = b.department AND a.id < b.id;

===================================================================
PATTERN 6: LEFT JOIN + IS NULL  (find missing / unmatched rows)
===================================================================
Clue words: "customers who never ordered", "employees without a manager",
            "products never sold", "not in"

-- Q: Customers who have never placed an order.
SELECT c.id, c.name
FROM customers c
LEFT JOIN orders o ON o.customer_id = c.id
WHERE o.id IS NULL;

-- Q: Products that have never been ordered.
SELECT p.id, p.name
FROM products p
LEFT JOIN order_items oi ON oi.product_id = p.id
WHERE oi.id IS NULL;

-- Q: Top-level managers (employees with no manager above them).
SELECT id, name
FROM employees
WHERE manager_id IS NULL;

-- ALTERNATIVE: NOT EXISTS (often faster on large tables)
SELECT c.id, c.name
FROM customers c
WHERE NOT EXISTS (
    SELECT 1 FROM orders o WHERE o.customer_id = c.id
);

-- ALTERNATIVE: NOT IN (beware NULLs — if subquery can return NULL, use NOT EXISTS)
SELECT id, name FROM customers
WHERE id NOT IN (SELECT DISTINCT customer_id FROM orders WHERE customer_id IS NOT NULL);

PATTERN 7: RUNNING TOTALS & CUMULATIVE AGGREGATES
===================================================================
Clue words: "running total", "cumulative sum", "moving average", "year-to-date"

-- Q: Running total of salary ordered by hire date.
SELECT
    name,
    hire_date,
    salary,
    SUM(salary) OVER (ORDER BY hire_date
                      ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS running_total
FROM employees;

-- Q: Running total per department.
SELECT
    name,
    department,
    salary,
    SUM(salary) OVER (PARTITION BY department ORDER BY hire_date
                      ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS dept_running_total
FROM employees;

-- Q: 3-month moving average of monthly revenue.
SELECT
    order_month,
    monthly_revenue,
    AVG(monthly_revenue) OVER (
        ORDER BY order_month
        ROWS BETWEEN 2 PRECEDING AND CURRENT ROW
    ) AS moving_avg_3m
FROM (
    SELECT
        DATE_FORMAT(order_date, '%Y-%m') AS order_month,
        SUM(total)                       AS monthly_revenue
    FROM orders
    GROUP BY DATE_FORMAT(order_date, '%Y-%m')
) monthly;

-- Frame clause cheatsheet:
--   ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW  → cumulative
--   ROWS BETWEEN 2 PRECEDING AND CURRENT ROW          → 3-row moving window
--   ROWS BETWEEN CURRENT ROW AND UNBOUNDED FOLLOWING  → suffix sum
--   RANGE BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW → cumulative (default for SUM OVER ORDER BY)

PATTERN 8: LAG / LEAD  (compare with previous or next row)
===================================================================
Clue words: "change from previous", "difference from last month",
            "salary growth", "consecutive days"

-- Q: Show each employee's salary and the salary of the previous hire.
SELECT
    name,
    hire_date,
    salary,
    LAG(salary, 1, 0) OVER (ORDER BY hire_date) AS prev_salary,
    salary - LAG(salary, 1, 0) OVER (ORDER BY hire_date) AS diff
FROM employees;

-- Q: Month-over-month revenue change.
SELECT
    order_month,
    monthly_revenue,
    LAG(monthly_revenue) OVER (ORDER BY order_month)  AS prev_month_revenue,
    monthly_revenue
        - LAG(monthly_revenue) OVER (ORDER BY order_month) AS mom_change
FROM (
    SELECT
        DATE_FORMAT(order_date, '%Y-%m') AS order_month,
        SUM(total)                       AS monthly_revenue
    FROM orders
    GROUP BY DATE_FORMAT(order_date, '%Y-%m')
) m;

-- Q: Find employees whose salary is higher than the next hire's salary.
SELECT name, hire_date, salary
FROM (
    SELECT
        name,
        hire_date,
        salary,
        LEAD(salary) OVER (ORDER BY hire_date) AS next_salary
    FROM employees
) t
WHERE salary > next_salary;

PATTERN 9: CONSECUTIVE ROWS / SEQUENCES (Row number trick)
===================================================================
Clue words: "consecutive days", "consecutive logins", "streak",
            "find gaps in a sequence"

-- KEY TRICK: For a consecutive sequence, (value - ROW_NUMBER) is constant.

-- Q: Find employees hired on consecutive dates (group consecutive hire dates).
SELECT
    grp,
    MIN(hire_date) AS start_date,
    MAX(hire_date) AS end_date,
    COUNT(*)       AS streak_length
FROM (
    SELECT
        hire_date,
        hire_date - INTERVAL ROW_NUMBER() OVER (ORDER BY hire_date) DAY AS grp
    FROM employees
) t
GROUP BY grp
HAVING COUNT(*) > 1;

-- Q: Find users who logged in for 3 or more consecutive days.
WITH numbered AS (
    SELECT
        user_id,
        login_date,
        login_date - INTERVAL ROW_NUMBER() OVER (PARTITION BY user_id ORDER BY login_date) DAY AS grp
    FROM logins
),
streaks AS (
    SELECT user_id, grp, COUNT(*) AS streak_len
    FROM numbered
    GROUP BY user_id, grp
)
SELECT DISTINCT user_id
FROM streaks
WHERE streak_len >= 3;

-- Q: Find gaps in an ID sequence.
SELECT
    id + 1           AS gap_start,
    next_id - 1      AS gap_end
FROM (
    SELECT
        id,
        LEAD(id) OVER (ORDER BY id) AS next_id
    FROM orders
) t
WHERE next_id > id + 1;

PATTERN 10: PIVOT — ROWS TO COLUMNS
===================================================================
Clue words: "show months as columns", "transpose", "crosstab"

-- Q: Show total sales per product per quarter as separate columns.
SELECT
    p.name,
    SUM(CASE WHEN QUARTER(o.order_date) = 1 THEN oi.qty * oi.unit_price ELSE 0 END) AS Q1,
    SUM(CASE WHEN QUARTER(o.order_date) = 2 THEN oi.qty * oi.unit_price ELSE 0 END) AS Q2,
    SUM(CASE WHEN QUARTER(o.order_date) = 3 THEN oi.qty * oi.unit_price ELSE 0 END) AS Q3,
    SUM(CASE WHEN QUARTER(o.order_date) = 4 THEN oi.qty * oi.unit_price ELSE 0 END) AS Q4
FROM products p
JOIN order_items oi ON oi.product_id = p.id
JOIN orders o       ON o.id = oi.order_id
GROUP BY p.name;

-- Q: Count employees by department and seniority band (salary range).
SELECT
    department,
    COUNT(CASE WHEN salary < 50000              THEN 1 END) AS junior,
    COUNT(CASE WHEN salary BETWEEN 50000 AND 90000 THEN 1 END) AS mid,
    COUNT(CASE WHEN salary > 90000              THEN 1 END) AS senior
FROM employees
GROUP BY department;

PATTERN 11: RECURSIVE CTE  (hierarchical data)
===================================================================
Clue words: "org chart", "manager hierarchy", "all descendants",
            "path from root", "depth level"

-- Q: Print the full reporting hierarchy under a given manager (id=1).
WITH RECURSIVE org_tree AS (
    -- Anchor: start with the root manager
    SELECT id, name, manager_id, 0 AS depth
    FROM employees
    WHERE id = 1

    UNION ALL

    -- Recursive: find each employee's direct reports
    SELECT e.id, e.name, e.manager_id, ot.depth + 1
    FROM employees e
    JOIN org_tree ot ON e.manager_id = ot.id
)
SELECT depth, id, name, manager_id
FROM org_tree
ORDER BY depth, id;

-- Q: Find all ancestors of a given employee (id=10) up to the root.
WITH RECURSIVE ancestry AS (
    SELECT id, name, manager_id, 0 AS level
    FROM employees
    WHERE id = 10

    UNION ALL

    SELECT e.id, e.name, e.manager_id, a.level + 1
    FROM employees e
    JOIN ancestry a ON e.id = a.manager_id
)
SELECT * FROM ancestry ORDER BY level DESC;

-- Q: Build the path string from root to each node.
WITH RECURSIVE paths AS (
    SELECT id, name, manager_id, CAST(name AS CHAR(500)) AS path
    FROM employees
    WHERE manager_id IS NULL   -- root

    UNION ALL

    SELECT e.id, e.name, e.manager_id,
           CONCAT(p.path, ' > ', e.name)
    FROM employees e
    JOIN paths p ON e.manager_id = p.id
)
SELECT id, name, path FROM paths;

PATTERN 12: CTEs + MULTI-STEP QUERIES (complex data problems)
===================================================================
Clue words: "for each department find...", "customers who...",
            "products where..." — any problem needing multiple steps

-- Q: Find departments where ALL employees earn above 50000.
SELECT department
FROM employees
GROUP BY department
HAVING MIN(salary) > 50000;

-- Q: For each customer, find their most expensive order.
WITH ranked_orders AS (
    SELECT
        customer_id,
        id          AS order_id,
        total,
        RANK() OVER (PARTITION BY customer_id ORDER BY total DESC) AS rnk
    FROM orders
)
SELECT
    c.name,
    ro.order_id,
    ro.total AS max_order_total
FROM customers c
JOIN ranked_orders ro ON ro.customer_id = c.id AND ro.rnk = 1;

-- Q: Show each product's revenue as a percentage of its category total.
WITH category_totals AS (
    SELECT
        p.category,
        SUM(oi.qty * oi.unit_price) AS cat_total
    FROM order_items oi
    JOIN products p ON p.id = oi.product_id
    GROUP BY p.category
),
product_revenue AS (
    SELECT
        p.id,
        p.name,
        p.category,
        SUM(oi.qty * oi.unit_price) AS revenue
    FROM order_items oi
    JOIN products p ON p.id = oi.product_id
    GROUP BY p.id, p.name, p.category
)
SELECT
    pr.name,
    pr.category,
    pr.revenue,
    ct.cat_total,
    ROUND(pr.revenue * 100.0 / ct.cat_total, 2) AS pct_of_category
FROM product_revenue pr
JOIN category_totals ct ON ct.category = pr.category
ORDER BY pr.category, pct_of_category DESC;

-- Q: Customers who ordered in Jan 2025 but NOT in Feb 2025.
WITH jan_customers AS (
    SELECT DISTINCT customer_id FROM orders
    WHERE order_date BETWEEN '2025-01-01' AND '2025-01-31'
),
feb_customers AS (
    SELECT DISTINCT customer_id FROM orders
    WHERE order_date BETWEEN '2025-02-01' AND '2025-02-28'
)
SELECT c.name
FROM customers c
JOIN jan_customers j   ON j.customer_id = c.id
LEFT JOIN feb_customers f ON f.customer_id = c.id
WHERE f.customer_id IS NULL;

===================================================================
PATTERN QUICK LOOKUP TABLE
===================================================================
Problem shape                          → Pattern + key clause
---------------------------------------|----------------------------------
Count/sum/avg per group                → GROUP BY + HAVING
Top N rows per group                   → ROW_NUMBER() + CTE + WHERE rn<=N
Nth highest value                      → DENSE_RANK() + WHERE rnk=N
Duplicates                             → GROUP BY + HAVING COUNT > 1
Delete duplicates, keep one            → ROW_NUMBER CTE + DELETE
Employee and their manager             → Self JOIN on manager_id
Rows with no match in another table    → LEFT JOIN + WHERE right.id IS NULL
Running / cumulative total             → SUM() OVER (ORDER BY ... ROWS...)
Compare to previous/next row           → LAG() / LEAD()
Consecutive sequences / streaks        → value - ROW_NUMBER() = constant trick
Rows to columns (pivot)                → CASE WHEN + SUM + GROUP BY
Hierarchy / org chart                  → Recursive CTE
Percentage within group                → value / SUM() OVER (PARTITION BY)
Customers in period A but not period B → Two CTEs + LEFT JOIN + IS NULL
===================================================================
