1. WHAT ARE ENUMS?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Enums (enumerations) let you define a set of named constants. They create
both a TYPE and a VALUE in TypeScript, making them one of the few constructs
that do produce JavaScript output (unlike interfaces and type aliases).

  enum Direction {
    Up,
    Down,
    Left,
    Right
  }

  function move(dir: Direction): void { ... }
  move(Direction.Up);  // ✅
  move(5);             // ❌ Error: Argument of type '5' is not assignable to 'Direction'

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
2. NUMERIC ENUMS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Default behavior — members auto-increment from 0:

  enum Direction {
    Up,     // 0
    Down,   // 1
    Left,   // 2
    Right   // 3
  }

  console.log(Direction.Up);    // 0
  console.log(Direction[0]);    // "Up" (reverse mapping!)
  console.log(Direction.Down);  // 1

  Custom starting value:
  enum StatusCode {
    OK = 200,
    Created = 201,
    BadRequest = 400,
    NotFound = 404,
    InternalError = 500
  }

  Partial initialization (auto-increments from last set value):
  enum FileAccess {
    None,           // 0
    Read = 1 << 1,  // 2
    Write = 1 << 2, // 4
    ReadWrite = Read | Write, // 6 (computed member — must follow initialized members)
    G = "123".length // 3 — computed
  }

  REVERSE MAPPING (numeric enums only):
  TypeScript generates a reverse lookup object automatically:
  // Compiled to:
  var Direction;
  (function (Direction) {
      Direction[Direction["Up"] = 0] = "Up";
      Direction[Direction["Down"] = 1] = "Down";
      ...
  })(Direction || (Direction = {}));

  // So you can do:
  Direction[0]         // "Up"
  Direction["Up"]      // 0
  Direction.Up         // 0

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
3. STRING ENUMS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Each member has an explicit string value. NO auto-increment and NO reverse mapping.

  enum Direction {
    Up    = "UP",
    Down  = "DOWN",
    Left  = "LEFT",
    Right = "RIGHT"
  }

  console.log(Direction.Up);    // "UP"
  console.log(Direction["UP"]); // ❌ Not supported (no reverse mapping)

  ADVANTAGES of String Enums:
  1. Human-readable values (easier debugging)
  2. Stable serialization (safe to store in DB, send over API)
  3. No unexpected numeric coercion issues
  4. No reverse mapping (which can cause accidental double-entries)

  enum UserRole {
    Admin = "ADMIN",
    User  = "USER",
    Guest = "GUEST"
  }

  function getPermissions(role: UserRole): string[] {
    switch (role) {
      case UserRole.Admin: return ["read", "write", "delete"];
      case UserRole.User:  return ["read", "write"];
      case UserRole.Guest: return ["read"];
    }
  }

4. HETEROGENEOUS ENUMS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Mixing string and number members in one enum. NOT RECOMMENDED — confusing.

  enum Mixed {
    No  = 0,
    Yes = "YES"
  }
  // Avoid in practice

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
5. CONST ENUMS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  const enums are completely INLINED at compile time — no JavaScript object
  is generated at all. This is the most efficient form.

  const enum Direction {
    Up = "UP",
    Down = "DOWN"
  }

  const d = Direction.Up;
  // Compiled to:
  const d = "UP";  // inlined directly — no Direction object exists at runtime!

  BENEFITS:
  • Zero runtime overhead (no enum object)
  • Smaller bundle size

  LIMITATIONS:
  • Cannot be used with computed values
  • Cannot be imported from .d.ts files (ambient modules) when isolatedModules: true
  • Cannot do reverse mapping
  • Cannot be re-exported as a value

  RULE: Use const enum for internal enums within a single codebase.
        Use regular enum if you need runtime access (reverse mapping, runtime checks).

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
6. AMBIENT ENUMS (declare enum)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Used in .d.ts files to describe enums that exist in JavaScript but were
  not written in TypeScript.

  declare enum Status {
    Active,
    Inactive
  }

  // No JavaScript is emitted — TypeScript trusts the declaration.

7. ENUM MEMBER TYPES AND COMPUTED VALUES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Enum members can be:
  1. Constant members (compile-time known):
     - Literal number: 0, 1, -1
     - Literal string: "a"
     - Reference to another constant enum member: A = B
     - Simple expressions: 1 + 1, ~1, 1 << 2

  2. Computed members (runtime expressions):
     - "hello".length
     - Math.floor(3.5)
     - Any JS expression

  enum FileAccess {
    None,
    Read    = 1 << 1,   // 2 — bit shift (constant)
    Write   = 1 << 2,   // 4 — bit shift (constant)
    ReadWrite = Read | Write, // 6 — expression (constant)
    G       = "123".length   // 3 — computed (runtime)
  }

  IMPORTANT: Computed members must come LAST (or after all initialized members).
  Computed members cannot be followed by auto-incremented members.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
8. USING ENUMS AS TYPES (UNION FROM ENUM VALUES)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Each enum member is also a type. You can use members as literal types:

  enum Status { Active = "ACTIVE", Inactive = "INACTIVE" }

  // Using the entire enum as a type:
  let s: Status = Status.Active;  // must be a Status member

  // Using a specific member as a literal type:
  type ActiveStatus = Status.Active; // just "ACTIVE"
  let active: Status.Active = Status.Active; // only Active allowed

  // Numeric enums have a QUIRK — they accept any number:
  enum E { A = 0, B = 1 }
  let e: E = 10; // ✅ No error! TypeScript allows any number for numeric enums
  // This is a known inconsistency. String enums don't have this issue.

9. ITERATING OVER ENUMS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  For numeric enums — be careful of reverse mappings!

  enum Color { Red = 0, Green = 1, Blue = 2 }

  // Iterating includes both forward and reverse entries:
  for (const key in Color) {
    console.log(key);  // "Red", "Green", "Blue", "0", "1", "2"
  }

  // Filter to only string keys (enum names):
  const colorNames = Object.keys(Color).filter(key => isNaN(Number(key)));
  // ["Red", "Green", "Blue"]

  // String enums — clean iteration:
  enum Status { Active = "ACTIVE", Inactive = "INACTIVE" }
  Object.values(Status); // ["ACTIVE", "INACTIVE"] — no reverse entries

  // Get enum values as an array (string enums):
  const statusValues = Object.values(Status) as string[];

10. ENUM ALTERNATIVES — WHEN TO AVOID ENUMS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Some TypeScript developers prefer alternatives to enums:

  a) Union of String Literals (most popular alternative):
     type Direction = "UP" | "DOWN" | "LEFT" | "RIGHT";
     
     Pros:
     • No runtime object generated
     • Works with isolatedModules
     • Cleaner TypeScript idiom
     • Can be used in .d.ts without issues
     
     Cons:
     • No single source of truth for values (can duplicate strings)
     • No reverse mapping

  b) Object with `as const`:
     const Direction = {
       Up: "UP",
       Down: "DOWN",
       Left: "LEFT",
       Right: "RIGHT",
     } as const;
     type Direction = typeof Direction[keyof typeof Direction];
     // "UP" | "DOWN" | "LEFT" | "RIGHT"

     Pros:
     • JavaScript-idiomatic (plain object)
     • Iterable at runtime
     • Works with isolatedModules
     • Type and value from the same declaration

  c) Numeric bit flags (when you need bitwise operations):
     const Permissions = {
       None:      0,
       Read:      1 << 0,  // 1
       Write:     1 << 1,  // 2
       Execute:   1 << 2,  // 4
       ReadWrite: (1 << 0) | (1 << 1),  // 3
     } as const;
     type Permission = typeof Permissions[keyof typeof Permissions];

11. COMPILED OUTPUT COMPARISON
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Regular enum:
    enum Color { Red = "RED", Green = "GREEN" }
    // Compiles to:
    var Color;
    (function (Color) {
        Color["Red"] = "RED";
        Color["Green"] = "GREEN";
    })(Color || (Color = {}));

  const enum:
    const enum Color { Red = "RED", Green = "GREEN" }
    const c = Color.Red;
    // Compiles to:
    const c = "RED"; // completely inlined — no Color object!

  String literal union:
    type Color = "RED" | "GREEN"
    // Compiles to: (nothing — completely erased)

12. COMMON INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q1: What are enums in TypeScript?
    Enums are a set of named constants. They exist at both compile time (as types)
    AND runtime (as JavaScript objects), making them one of the few TypeScript
    features that generate actual JavaScript output.

Q2: What is the difference between numeric and string enums?
    - Numeric enums: auto-increment from 0, support reverse mapping (Direction[0] = "Up")
    - String enums: each member has explicit string value, no reverse mapping,
      more predictable and readable. String enums are generally preferred.

Q3: What is a const enum and when would you use it?
    A const enum is inlined at compile time — no JavaScript object is generated.
    All usages are replaced with their literal values. Use for internal enums
    where you want zero runtime overhead and no need for the runtime object.
    Don't use with isolatedModules or in .d.ts files.

Q4: What is the reverse mapping in numeric enums?
    TypeScript generates a bidirectional mapping for numeric enums:
    Direction.Up = 0 AND Direction[0] = "Up"
    This allows looking up enum names by value. String enums don't have this.

Q5: What is the quirk/pitfall of numeric enums and type safety?
    Numeric enums accept ANY number, not just declared values:
    enum E { A = 0, B = 1 }
    let e: E = 100; // ✅ TypeScript allows this! No error.
    String enums don't have this issue — only enum members are accepted.

Q6: Why might you prefer a string literal union over an enum?
    • No runtime object generated (smaller bundle)
    • Works with isolatedModules
    • More idiomatic TypeScript (recommended by many style guides)
    • Simpler to understand
    • No extra JavaScript emitted
    Downside: no namespace grouping, no reverse mapping.

Q7: How do you iterate over an enum's values safely?
    For string enums: Object.values(MyEnum) — clean, no reverse entries.
    For numeric enums: filter Object.keys to string keys only:
    Object.keys(MyEnum).filter(k => isNaN(Number(k)))

Q8: What is the `as const` object pattern as an enum alternative?
    const ROLE = { Admin: "ADMIN", User: "USER" } as const;
    type Role = typeof ROLE[keyof typeof ROLE]; // "ADMIN" | "USER"
    This is a plain JavaScript object that behaves like an enum but:
    • Works with isolatedModules
    • Iterable at runtime
    • No TypeScript-specific syntax
    • Zero runtime overhead for the type

Q9: Can you use computed values in a const enum?
    No. const enums can only have constant (compile-time known) member values.
    Computed members (like "hello".length) are not allowed.

Q10: What's the difference between enum and declare enum?
     - enum: TypeScript declares AND generates the enum object in JS output
     - declare enum (ambient): TypeScript only adds types — no JS emitted.
       Used in .d.ts files to describe enums that exist in external JS code.
