1. WHAT ARE LITERAL TYPES?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Literal types restrict a value to EXACTLY one specific value.
Instead of saying "any string", a literal type says "exactly this string".

  // Regular type:  any string
  let status: string = "active";  // "pending", "active", "closed" — all allowed

  // Literal type: ONLY this specific string
  let status: "active" = "active";  // ONLY "active" is allowed
  status = "pending"; // ❌ Error: Type '"pending"' is not assignable to type '"active"'

Literal types exist for:
  • String literals:  "active" | "inactive"
  • Number literals:  1 | 2 | 3
  • Boolean literals: true | false (these ARE booleans)
  • Template literals: `item-${number}` (TS 4.1+)

2. STRING LITERAL TYPES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Single literal:
  type Yes = "yes";
  let answer: Yes = "yes";  // only "yes" allowed

  // Union of string literals (most common use case):
  type Status = "pending" | "active" | "closed" | "cancelled";
  type Direction = "north" | "south" | "east" | "west";
  type HttpMethod = "GET" | "POST" | "PUT" | "DELETE" | "PATCH" | "HEAD" | "OPTIONS";
  type Alignment = "left" | "center" | "right" | "justify";
  type Theme = "light" | "dark" | "system";

  function setTheme(theme: Theme): void {
    document.body.dataset.theme = theme;
  }
  setTheme("dark");     // ✅
  setTheme("midnight"); // ❌ Error: "midnight" not in Theme union

  // String literal as function parameter validation:
  type LogLevel = "debug" | "info" | "warn" | "error";

  function log(level: LogLevel, message: string): void {
    console[level](message);
  }
  log("info", "Server started");   // ✅
  log("verbose", "DB query");      // ❌ Error: "verbose" not in LogLevel

3. NUMERIC LITERAL TYPES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  type DiceValue = 1 | 2 | 3 | 4 | 5 | 6;
  type ZeroOrOne = 0 | 1;  // binary
  type Port = 80 | 443 | 3000 | 8080 | 8443;

  function rollDice(): DiceValue {
    return Math.ceil(Math.random() * 6) as DiceValue;
  }

  // Numeric literals for bit flags:
  type Bit = 0 | 1;

  // Map values to types:
  type Level = 1 | 2 | 3 | 4 | 5;
  const permissions: Record<Level, string[]> = {
    1: ["read"],
    2: ["read", "write"],
    3: ["read", "write", "manage"],
    4: ["read", "write", "manage", "publish"],
    5: ["read", "write", "manage", "publish", "admin"],
  };

4. BOOLEAN LITERAL TYPES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  boolean is actually a union of literal types:  true | false

  type True  = true;
  type False = false;
  type Bool  = true | false; // equivalent to `boolean`

  // Boolean literals in conditional types:
  type IsString<T> = T extends string ? true : false;
  type A = IsString<string>; // true
  type B = IsString<number>; // false

  // Dependent types based on boolean param:
  function createConfig<T extends boolean>(
    strict: T
  ): T extends true ? { mode: "strict"; timeout: 5000 } : { mode: "loose"; timeout: 30000 } {
    return strict
      ? { mode: "strict", timeout: 5000 } as any
      : { mode: "loose", timeout: 30000 } as any;
  }
  const cfg = createConfig(true);  // { mode: "strict"; timeout: 5000 }

5. TYPE WIDENING AND LITERAL TYPES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  TypeScript WIDENS literal types for `let` (they can change):

  let s = "hello";  // type: string (widened from "hello")
  let n = 42;       // type: number (widened from 42)
  let b = true;     // type: boolean (widened from true)

  TypeScript keeps literal types for `const` (they can't change):

  const s = "hello";  // type: "hello" (literal — unchanged)
  const n = 42;       // type: 42 (literal)
  const b = true;     // type: true (literal)

  Object properties are WIDENED even with const:
  const obj = { status: "active" };  // status: string (not "active"!)
  // Because properties can be reassigned: obj.status = "inactive"

  Solution: use `as const` to preserve literal types in objects:
  const obj = { status: "active" } as const;
  // status: "active" (readonly literal type)

6. LITERAL TYPES WITH FUNCTIONS — OVERLOADS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Literal types enable precise function overloads:

  function createElement(tag: "a"): HTMLAnchorElement;
  function createElement(tag: "div"): HTMLDivElement;
  function createElement(tag: "input"): HTMLInputElement;
  function createElement(tag: string): HTMLElement {
    return document.createElement(tag);
  }

  const a = createElement("a");       // type: HTMLAnchorElement
  const div = createElement("div");   // type: HTMLDivElement
  const input = createElement("input"); // type: HTMLInputElement

  // Without overloads: would just return HTMLElement (less specific)

7. TEMPLATE LITERAL TYPES (TypeScript 4.1+)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Template literal types generate new string literal types from combinations:

  type Greeting = `Hello, ${string}`;
  let g: Greeting = "Hello, World";    // ✅
  let g2: Greeting = "Goodbye, World"; // ❌ Error

  // Combine union types to generate all combinations:
  type Color = "red" | "green" | "blue";
  type Shade = "light" | "dark";
  type ColorShade = `${Shade}-${Color}`;
  // "light-red" | "light-green" | "light-blue" | "dark-red" | "dark-green" | "dark-blue"

  // CSS property types:
  type CSSUnit = "px" | "em" | "rem" | "vh" | "vw" | "%";
  type CSSLength = `${number}${CSSUnit}`;

  // Event name patterns:
  type ButtonEvents = "click" | "hover" | "focus";
  type ButtonEventHandler = `on${Capitalize<ButtonEvents>}`;
  // "onClick" | "onHover" | "onFocus"

  // Deep key access:
  type Paths<T extends object, K extends keyof T = keyof T> =
    K extends string
      ? T[K] extends object
        ? `${K}.${Paths<T[K]>}` | K
        : K
      : never;

  type User = { name: string; address: { city: string; zip: string } };
  type UserPaths = Paths<User>; // "name" | "address" | "address.city" | "address.zip"

8. TEMPLATE LITERAL UTILITY TYPES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  TypeScript includes built-in string manipulation types:

  type U = Uppercase<"hello">;    // "HELLO"
  type L = Lowercase<"HELLO">;    // "hello"
  type C = Capitalize<"hello">;   // "Hello"
  type U2 = Uncapitalize<"HELLO">; // "hELLO"

  // Practical: generating getter names from property names
  type Getters<T> = {
    [K in keyof T as `get${Capitalize<string & K>}`]: () => T[K];
  };

  type User = { name: string; age: number };
  type UserGetters = Getters<User>;
  // { getName: () => string; getAge: () => number }

a) Exhaustive Config Options:
   type RequestMethod = "GET" | "POST" | "PUT" | "DELETE";
   type ContentType = "application/json" | "text/html" | "multipart/form-data";

   interface Request {
     method: RequestMethod;
     contentType?: ContentType;
     url: string;
   }

b) Type-Safe Event System:
   type UserEvent = "login" | "logout" | "signup";
   type PostEvent = "create" | "delete" | "publish";
   type AppEvent = `user:${UserEvent}` | `post:${PostEvent}`;
   // "user:login" | "user:logout" | "user:signup" | "post:create" | ...

   function emit(event: AppEvent): void { ... }
   emit("user:login");     // ✅
   emit("user:unknown");   // ❌

c) Readonly Literal Properties:
   const ENDPOINTS = {
     users: "/api/users",
     posts: "/api/posts",
     auth: "/api/auth",
   } as const;
   type Endpoint = typeof ENDPOINTS[keyof typeof ENDPOINTS];
   // "/api/users" | "/api/posts" | "/api/auth"

d) Function that narrows return type based on input:
   type Format = "json" | "xml" | "csv";
   type FormatResult<F extends Format> =
     F extends "json" ? object :
     F extends "xml"  ? string :
     string[];

   function parse<F extends Format>(input: string, format: F): FormatResult<F> {
     // implementation
   }
   const obj = parse('{"a":1}', "json"); // type: object
   const xml = parse("<root/>", "xml");   // type: string

10. COMMON INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q1: What are literal types in TypeScript?
    Literal types are types that represent exactly one specific value.
    "active" is a string literal type — only the string "active" satisfies it.
    42 is a number literal type. true and false are boolean literal types.
    They're most useful in unions: type Status = "active" | "inactive".

Q2: Why is `const s = "hello"` different from `let s = "hello"` in terms of types?
    `const s = "hello"` → type is "hello" (literal type — can't change)
    `let s = "hello"` → type is string (widened — because let values can change)
    This is called "type widening." TypeScript is conservative: if you might
    reassign, it infers the wider type.

Q3: What is `as const` and how does it relate to literal types?
    `as const` prevents type widening, ensuring TypeScript infers the narrowest
    possible types (literals) and makes all properties readonly:
    const cfg = { status: "active" } as const;
    // status: "active" (not string) and readonly

Q4: What are template literal types?
    Introduced in TS 4.1, template literal types combine string literals using
    template syntax: type EventName = `on${Capitalize<string>}`.
    They can combine union types to generate all permutations, enabling
    complex string pattern matching at the type level.

Q5: What are the built-in string manipulation utility types?
    Uppercase<S>, Lowercase<S>, Capitalize<S>, Uncapitalize<S>
    These transform string literal types. Used in mapped types with template
    literals to generate property names.

Q6: How do literal types enable better function overloads?
    By using specific string literals as parameters:
    function createElement(tag: "div"): HTMLDivElement;
    function createElement(tag: "a"): HTMLAnchorElement;
    The return type changes based on the literal argument. Without literals,
    all overloads would return HTMLElement.

Q7: What is the difference between `string` and `"active" | "inactive"`?
    string: any string value (wide type)
    "active" | "inactive": only those two specific strings (narrow type)
    The union of literals is a subset of string. You get compile-time
    validation that only valid values are used.

Q8: How do you derive a union type from an array of values?
    const statuses = ["active", "inactive", "pending"] as const;
    type Status = typeof statuses[number]; // "active" | "inactive" | "pending"
    The `as const` makes it readonly string[] with literal types.
    `[number]` indexes into the array type to get the union.

Q9: How do you use template literal types for event naming?
    type EventName<T extends string> = `${T}:change` | `${T}:reset`;
    type FormEvents = EventName<"input" | "select">;
    // "input:change" | "input:reset" | "select:change" | "select:reset"

Q10: What happens when you try to assign a wider type to a literal type?
     type X = "hello";
     let x: X = "hello";  // ✅
     let s: string = "anything";
     x = s;  // ❌ Error: Type 'string' is not assignable to type '"hello"'
     You cannot assign a wider type to a narrower literal type.
     But you CAN assign a literal to its wider type:
     let s: string = x;  // ✅ "hello" is assignable to string
