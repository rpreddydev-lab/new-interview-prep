1. WHAT IS A TYPE ALIAS?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

A type alias creates a NEW NAME for any type — primitive, union, intersection,
tuple, function, object, or complex computed type.

Syntax: type AliasName = <type expression>

  type ID = number | string;       // union type alias
  type Username = string;          // primitive alias
  type Point = { x: number; y: number }; // object type alias
  type Callback = () => void;      // function type alias
  type Pair<T> = [T, T];           // generic type alias

Key fact: Type aliases are ERASED at runtime — no JavaScript is generated.
          They're purely a compile-time construct.

2. BASIC TYPE ALIASES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

a) Primitive Aliases (improve readability):
   ─────────────────────────────────────────
   type UserID = number;
   type Email = string;
   type Percentage = number;

   function getUser(id: UserID): void { ... }  // clearer intent

   Note: These are just aliases — UserID is still number. No nominal safety.
   getUser("not-a-number"); // ❌ Error — "string" not assignable to "number"
   getUser(42 as UserID);   // ✅ but so is just getUser(42)

b) Object Types:
   ──────────────
   type User = {
     id: number;
     name: string;
     email: string;
     role: "admin" | "user" | "guest";
   };

   type ReadonlyUser = {
     readonly id: number;
     readonly name: string;
   };

   type PartialUser = {
     id?: number;
     name?: string;
   };

c) Function Types:
   ─────────────────
   type Transformer<T, U> = (input: T) => U;
   type Predicate<T> = (value: T) => boolean;
   type EventHandler = (event: Event) => void;
   type AsyncFn<T> = () => Promise<T>;

   const toUpperCase: Transformer<string, string> = (s) => s.toUpperCase();
   const isEven: Predicate<number> = (n) => n % 2 === 0;

d) Union Types:
   ─────────────
   type StringOrNumber = string | number;
   type Status = "pending" | "active" | "closed";
   type Nullable<T> = T | null;
   type Maybe<T> = T | null | undefined;

   let status: Status = "pending";
   status = "unknown"; // ❌ Error

e) Intersection Types:
   ─────────────────────
   type Named = { name: string };
   type Aged = { age: number };
   type Person = Named & Aged;
   // Person = { name: string; age: number }

   type AdminUser = User & { adminLevel: number; permissions: string[] };

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
3. GENERIC TYPE ALIASES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Type aliases can be parameterized with generics:

  type Optional<T> = T | undefined;
  type Nullable<T> = T | null;
  type Maybe<T> = T | null | undefined;

  type Pair<T, U = T> = [T, U];   // U defaults to T
  type Result<T, E = Error> = { success: true; data: T } | { success: false; error: E };
  type AsyncResult<T> = Promise<Result<T>>;

  type ApiResponse<T> = {
    data: T;
    status: number;
    message: string;
    timestamp: string;
  };

  type UserResponse = ApiResponse<User>;  // expands to ApiResponse with User
  type ListResponse<T> = ApiResponse<T[]>;

  // Recursive generic types:
  type Tree<T> = {
    value: T;
    left?: Tree<T>;
    right?: Tree<T>;
  };

  type JSONValue =
    | string
    | number
    | boolean
    | null
    | JSONValue[]
    | { [key: string]: JSONValue };

4. SELF-REFERENTIAL (RECURSIVE) TYPE ALIASES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Type aliases can reference themselves (since TypeScript 3.7 with deferred
  resolution — interfaces can too, but types have more flexibility):

  type NestedArray<T> = T | NestedArray<T>[];
  // [1, [2, [3, 4]], 5] satisfies NestedArray<number>

  type TreeNode = {
    id: number;
    children: TreeNode[];
  };

  // Linked list:
  type ListNode<T> = {
    value: T;
    next: ListNode<T> | null;
  };

5. ADVANCED TYPE ALIASES — USING UTILITY TYPES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Type aliases shine when combined with TypeScript's built-in utility types:

  type User = { id: number; name: string; email: string; password: string };

  // Create variations using utility types:
  type CreateUserDTO = Omit<User, "id">;
  // { name: string; email: string; password: string }

  type UpdateUserDTO = Partial<Omit<User, "id">>;
  // { name?: string; email?: string; password?: string }

  type PublicUser = Pick<User, "id" | "name">;
  // { id: number; name: string }

  type ReadonlyUser = Readonly<User>;
  // All properties become readonly

  type UserWithoutPassword = Omit<User, "password">;

  // Combining aliases:
  type APIUser = Readonly<Pick<User, "id" | "name" | "email">>;

6. TYPE ALIAS FOR MAPPED TYPES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  type aliases are REQUIRED for mapped types (interfaces can't do this):

  // Make all properties optional:
  type Partial<T> = { [K in keyof T]?: T[K] };

  // Make all properties required:
  type Required<T> = { [K in keyof T]-?: T[K] };

  // Make all properties readonly:
  type Readonly<T> = { readonly [K in keyof T]: T[K] };

  // Map values to another type:
  type StringValues<T> = { [K in keyof T]: string };

  // Nullable all properties:
  type Nullable<T> = { [K in keyof T]: T[K] | null };

  // Custom: Make only certain keys optional:
  type PartialBy<T, K extends keyof T> = Omit<T, K> & Partial<Pick<T, K>>;
  type UserUpdateInput = PartialBy<User, "name" | "email">;

7. CONDITIONAL TYPE ALIASES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Type aliases can express conditional logic at the type level:

  type IsString<T> = T extends string ? true : false;
  type A = IsString<string>;   // true
  type B = IsString<number>;   // false

  type Flatten<T> = T extends Array<infer Item> ? Item : T;
  type Nums = Flatten<number[]>;  // number
  type Str = Flatten<string>;     // string (not array, so unchanged)

  type NonNullable<T> = T extends null | undefined ? never : T;
  type C = NonNullable<string | null>;  // string

  // Distributive conditional types (auto-distribute over unions):
  type ToArray<T> = T extends unknown ? T[] : never;
  type StrOrNumArray = ToArray<string | number>; // string[] | number[]

8. DIFFERENCES: TYPE ALIAS vs INTERFACE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  ┌───────────────────────────────┬───────────────────┬───────────────────┐
  │ Feature                       │ type              │ interface         │
  ├───────────────────────────────┼───────────────────┼───────────────────┤
  │ Object shapes                 │ ✅                │ ✅                │
  │ Function types                │ ✅                │ ✅                │
  │ Primitive aliases             │ ✅                │ ❌                │
  │ Union types                   │ ✅                │ ❌                │
  │ Intersection (&)              │ ✅                │ extends (similar) │
  │ Tuples                        │ ✅                │ ❌                │
  │ Conditional types             │ ✅                │ ❌                │
  │ Mapped types                  │ ✅                │ ❌                │
  │ Template literal types        │ ✅                │ ❌                │
  │ Declaration merging           │ ❌                │ ✅                │
  │ Class implements              │ ✅ (object shapes)│ ✅                │
  │ extends for inheritance       │ ❌                │ ✅                │
  │ Error messages (readability)  │ Sometimes verbose │ Usually cleaner   │
  └───────────────────────────────┴───────────────────┴───────────────────┘

  Rule of thumb:
  • Use `interface` for object shapes, especially if others will extend them
    or if you publish a library (for augmentation support)
  • Use `type` for unions, intersections, mapped types, conditional types,
    primitives, tuples, and anything that can't be expressed as interface

9. PRACTICAL PATTERNS WITH TYPE ALIASES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

a) Discriminated Union Pattern:
   ──────────────────────────────
   type Loading = { status: "loading" };
   type Success<T> = { status: "success"; data: T };
   type Failure = { status: "failure"; error: string };
   type AsyncState<T> = Loading | Success<T> | Failure;

   function render<T>(state: AsyncState<T>): string {
     switch (state.status) {
       case "loading": return "Loading...";
       case "success": return JSON.stringify(state.data);
       case "failure": return `Error: ${state.error}`;
     }
   }

b) Builder Pattern:
   ──────────────────
   type Builder<T, Required extends keyof T = never> = {
     set<K extends keyof T>(key: K, value: T[K]): Builder<T, Required | K>;
     build(): Pick<T, Required>;
   };

c) Event System Pattern:
   ────────────────────────
   type Events = {
     "user:login": { userId: string; timestamp: Date };
     "user:logout": { userId: string };
     "post:created": { postId: string; title: string };
   };

   type EventHandler<E extends keyof Events> = (payload: Events[E]) => void;

   function on<E extends keyof Events>(event: E, handler: EventHandler<E>) {
     // fully typed event system
   }

   on("user:login", ({ userId, timestamp }) => { ... }); // fully typed!

10. COMMON INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q1: What is a type alias in TypeScript?
    A type alias is a name given to any type expression using the `type` keyword.
    It can alias primitives, unions, intersections, objects, functions, tuples,
    and complex computed types. Type aliases are erased at runtime.

Q2: What can a type alias do that an interface cannot?
    Type aliases can express:
    • Union types: type ID = string | number
    • Intersection types: type AB = A & B
    • Tuples: type Pair = [string, number]
    • Conditional types: type IsString<T> = T extends string ? true : false
    • Mapped types: type Readonly<T> = { readonly [K in keyof T]: T[K] }
    • Template literal types: type EventName = `on${string}`
    • Primitive aliases: type Age = number

Q3: What can an interface do that a type alias cannot?
    Declaration merging: Multiple interface declarations with the same name
    are merged. Type aliases cannot be re-declared.

Q4: Are type aliases and interfaces completely interchangeable?
    For simple object shapes they are mostly equivalent. The key differences:
    1. Interfaces support declaration merging, types don't
    2. Types support union, conditional, mapped, template literal types
    3. Interfaces show cleaner error messages in some cases

Q5: What is a recursive type alias?
    A type alias that references itself:
    type JSON = string | number | boolean | null | JSON[] | { [k: string]: JSON }
    Used for tree structures, linked lists, deeply nested data.

Q6: How do you make a type alias generic?
    type Container<T> = { value: T; label: string };
    type Result<T, E = Error> = { ok: T } | { err: E };

Q7: What is a union type alias and when do you use it?
    A union type alias combines multiple types: type ID = string | number.
    Used when a value can be one of several types, for discriminated unions,
    and for string literal unions (status, event names, etc.).

Q8: What is an intersection type alias and when do you use it?
    type AB = A & B combines two types into one with all properties of both.
    Used for composition (Mixin patterns), combining an entity with audit fields,
    combining base type with role-specific fields.

Q9: How do you create a "partial" version of a type?
    Using the built-in Partial<T>:
    type PartialUser = Partial<User>; // all properties optional
    Or manually: type Partial<T> = { [K in keyof T]?: T[K] }

Q10: Can a class implement a type alias?
     Yes, as long as the type alias is an object type (not a union or primitive):
     type Runnable = { run(): void };
     class Task implements Runnable { run() { ... } }
     But: type StringOrNumber = string | number;
     class Foo implements StringOrNumber {} // ❌ Error — can't implement a union
