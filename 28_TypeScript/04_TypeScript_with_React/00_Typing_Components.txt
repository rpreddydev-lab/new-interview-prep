2. FUNCTION COMPONENTS — PROPS TYPING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  a) Inline props (simplest):
     function Button({ label, onClick }: {
       label: string;
       onClick: () => void;
     }) {
       return <button onClick={onClick}>{label}</button>;
     }

  b) Named interface/type (recommended for reuse and readability):
     interface ButtonProps {
       label: string;
       disabled?: boolean;
       variant?: "primary" | "secondary" | "danger";
       onClick: (event: React.MouseEvent<HTMLButtonElement>) => void;
     }

     function Button({ label, disabled = false, variant = "primary", onClick }: ButtonProps) {
       return (
         <button
           className={`btn btn-${variant}`}
           disabled={disabled}
           onClick={onClick}
         >
           {label}
         </button>
       );
     }

  c) FC / FunctionComponent (less recommended now):
     // React.FC adds children implicitly (pre React 18) — generally avoid
     const Button: React.FC<ButtonProps> = ({ label }) => <button>{label}</button>;
     // React.FC also adds return type React.ReactElement | null

  d) Preferred modern pattern — just annotate the return type:
     function Card({ title, children }: {
       title: string;
       children: React.ReactNode;
     }): React.JSX.Element {
       return <div className="card"><h2>{title}</h2>{children}</div>;
     }

3. CHILDREN TYPES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  React.ReactNode    — most permissive: JSX, string, number, null, undefined, array
  React.ReactElement — only JSX elements (not strings/numbers/null)
  React.JSX.Element  — alias for React.ReactElement (preferred in newer TS)
  React.PropsWithChildren<P> — adds `children?: React.ReactNode` to P

  // Prefer ReactNode for children prop:
  interface LayoutProps {
    children: React.ReactNode;
  }

  // Or use PropsWithChildren<T>:
  type LayoutProps = React.PropsWithChildren<{ title: string }>;

  // Single child only (use React.ReactElement, not ReactNode):
  interface WrapperProps {
    children: React.ReactElement;
  }

4. EXTENDING NATIVE ELEMENT PROPS (SPREADING)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Extend native button props so you can pass `type`, `disabled`, `aria-*`, etc.:
  interface ButtonProps extends React.ButtonHTMLAttributes<HTMLButtonElement> {
    variant?: "primary" | "secondary";
    loading?: boolean;
  }

  function Button({ variant = "primary", loading, children, ...rest }: ButtonProps) {
    return (
      <button
        className={`btn btn-${variant} ${loading ? "loading" : ""}`}
        disabled={loading}
        {...rest}   // ✅ all native button attributes passed through
      >
        {loading ? "Loading..." : children}
      </button>
    );
  }

  // Common HTML attribute interfaces:
  // React.HTMLAttributes<HTMLElement>     — all HTML elements
  // React.InputHTMLAttributes<HTMLInputElement>
  // React.AnchorHTMLAttributes<HTMLAnchorElement>
  // React.SVGAttributes<SVGElement>

  // Extending div:
  type CardProps = React.HTMLAttributes<HTMLDivElement> & {
    elevated?: boolean;
  };

5. CLASS COMPONENTS (LEGACY)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // React.Component<Props, State>:
  interface CounterProps {
    initialCount?: number;
    label: string;
  }

  interface CounterState {
    count: number;
  }

  class Counter extends React.Component<CounterProps, CounterState> {
    state: CounterState = {
      count: this.props.initialCount ?? 0,
    };

    increment = () => this.setState(prev => ({ count: prev.count + 1 }));
    decrement = () => this.setState(prev => ({ count: prev.count - 1 }));

    render() {
      return (
        <div>
          <p>{this.props.label}: {this.state.count}</p>
          <button onClick={this.increment}>+</button>
          <button onClick={this.decrement}>-</button>
        </div>
      );
    }
  }

6. FORWARD REFS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  import React, { forwardRef } from "react";

  interface InputProps extends React.InputHTMLAttributes<HTMLInputElement> {
    label: string;
  }

  // forwardRef<RefType, PropsType>:
  const TextInput = forwardRef<HTMLInputElement, InputProps>(
    ({ label, ...rest }, ref) => (
      <label>
        {label}
        <input ref={ref} {...rest} />
      </label>
    )
  );

  // Usage:
  const inputRef = React.useRef<HTMLInputElement>(null);
  <TextInput label="Name" ref={inputRef} />;
  inputRef.current?.focus();

8. INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q1: What is the difference between React.ReactNode and React.ReactElement?
    ReactNode is the broadest — it includes JSX elements, strings, numbers,
    null, undefined, booleans, and arrays of those. ReactElement is narrower
    — it only represents actual JSX elements (not strings/numbers/null).
    Use ReactNode for children props; ReactElement when you need a JSX value.

Q2: Why is React.FC less preferred in modern TypeScript React?
    React.FC implicitly added `children` to every component in React 17 and
    earlier (removed in React 18 types). It also adds complexity without benefit.
    Modern practice: just type props directly with an interface and optionally
    annotate the return type as React.JSX.Element.

Q3: How do you type a component that accepts all native button attributes?
    Extend React.ButtonHTMLAttributes<HTMLButtonElement>:
    interface MyButtonProps extends React.ButtonHTMLAttributes<HTMLButtonElement> {
      loading?: boolean;
    }
    This allows passing any valid HTML button attribute plus your custom ones.

Q4: How do you type a component's children?
    Use `children: React.ReactNode` in the props interface. For stricter
    single-element requirements: `children: React.ReactElement`.
    React.PropsWithChildren<T> adds children?: React.ReactNode to T automatically.

Q5: What is forwardRef and how do you type it?
    forwardRef allows passing a ref through a component to a DOM element.
    Type it as: forwardRef<RefType, PropsType>
    const Input = forwardRef<HTMLInputElement, InputProps>(({ ...props }, ref) => ...)

Q6: How do you type a component's return value?
    Use React.JSX.Element (or React.ReactElement) for non-null returns.
    Use React.JSX.Element | null if it can return null.
    In practice, you can omit the return type and let TypeScript infer it.

Q7: What is the difference between React.Component and React.PureComponent?
    Both are generic: Component<Props, State>. PureComponent implements
    shouldComponentUpdate with a shallow prop/state comparison, preventing
    unnecessary re-renders. Both are class components. In TypeScript they're
    typed identically — PureComponent<Props, State>.

Q8: How do you pass an event handler as a prop?
    Use the appropriate React event type:
    interface ButtonProps { onClick: (e: React.MouseEvent<HTMLButtonElement>) => void; }
    Or use the shorthand: onClick?: React.MouseEventHandler<HTMLButtonElement>

Q9: What does the `...rest` pattern accomplish with TypeScript?
    When you destructure custom props and spread the rest: `{ loading, ...rest }`
    and the type extends `React.ButtonHTMLAttributes`, the `rest` variable is
    correctly typed as the remaining HTML attributes. This lets consumers pass
    any standard HTML attribute without re-declaring them.

Q10: How do you set defaultProps in modern TypeScript React?
     Use destructuring defaults in the function signature:
     function Badge({ text, color = "blue" }: { text: string; color?: string }) {}
     The legacy static defaultProps still works but is deprecated.
     With function defaults, TypeScript correctly understands the types.
