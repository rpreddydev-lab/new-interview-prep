1. REACT SYNTHETIC EVENTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  React wraps native browser events in SyntheticEvent objects for cross-browser
  compatibility. TypeScript provides specific types for each event kind.

  React.SyntheticEvent<T, E>
    T = the element type (e.g., HTMLInputElement)
    E = the native event type (e.g., Event)

  All specific event types extend React.SyntheticEvent.

3. CLICK EVENTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Inline handler:
  <button onClick={(e: React.MouseEvent<HTMLButtonElement>) => {
    e.preventDefault();
    console.log(e.target, e.currentTarget);
  }}>
    Click me
  </button>

  // Handler function:
  const handleClick: React.MouseEventHandler<HTMLButtonElement> = (e) => {
    e.stopPropagation();
    console.log("Clicked at", e.clientX, e.clientY);
  };
  <button onClick={handleClick}>Click</button>

  // In a prop interface:
  interface CardProps {
    onClick: (event: React.MouseEvent<HTMLDivElement>) => void;
  }

4. INPUT AND CHANGE EVENTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // text input onChange:
  const handleChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    const value: string = e.target.value;
    setName(value);
  };
  <input type="text" onChange={handleChange} />

  // Checkbox:
  const handleCheck = (e: React.ChangeEvent<HTMLInputElement>) => {
    const checked: boolean = e.target.checked;
    setEnabled(checked);
  };
  <input type="checkbox" onChange={handleCheck} />

  // Select:
  const handleSelect = (e: React.ChangeEvent<HTMLSelectElement>) => {
    const value: string = e.target.value;
    setOption(value);
  };
  <select onChange={handleSelect}>...</select>

  // Textarea:
  const handleTextarea = (e: React.ChangeEvent<HTMLTextAreaElement>) => {
    setText(e.target.value);
  };

  // Controlled input component:
  interface InputProps {
    value: string;
    onChange: React.ChangeEventHandler<HTMLInputElement>;
    placeholder?: string;
  }

  function ControlledInput({ value, onChange, placeholder }: InputProps) {
    return <input value={value} onChange={onChange} placeholder={placeholder} />;
  }

6. KEYBOARD EVENTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  const handleKeyDown = (e: React.KeyboardEvent<HTMLInputElement>) => {
    if (e.key === "Enter") {
      console.log("Enter pressed");
      e.preventDefault();
    }
    if (e.ctrlKey && e.key === "s") {
      console.log("Ctrl+S");
    }
    // e.key: string — the key value
    // e.code: string — the physical key code
    // e.keyCode: number — deprecated but still available
    // e.ctrlKey, e.altKey, e.shiftKey, e.metaKey: boolean
  };

  <input onKeyDown={handleKeyDown} />

9. EVENT TARGET NARROWING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // e.target vs e.currentTarget:
  // e.currentTarget — the element with the handler (always typed as T)
  // e.target        — the element that triggered the event (may be child — EventTarget)

  const handleClick = (e: React.MouseEvent<HTMLDivElement>) => {
    // e.currentTarget is HTMLDivElement ✅
    // e.target is EventTarget | null — needs narrowing:
    if (e.target instanceof HTMLButtonElement) {
      console.log(e.target.name); // ✅ narrowed
    }
  };

  // When you need e.target typed — use e.currentTarget instead:
  const handleInput = (e: React.ChangeEvent<HTMLInputElement>) => {
    console.log(e.currentTarget.value); // ✅ always HTMLInputElement
    // OR:
    const input = e.target; // ChangeEvent always gives HTMLInputElement for target too
    console.log(input.value); // ✅ in ChangeEvent, target === currentTarget
  };

10. INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q1: What is React.SyntheticEvent?
    React wraps native DOM events in SyntheticEvent objects for cross-browser
    normalization. All React event types extend it. In TypeScript,
    React.SyntheticEvent<T> has a currentTarget typed as T.

Q2: What is the difference between React.ChangeEvent and React.MouseEvent?
    ChangeEvent<T> fires when an input's value changes (input, select, textarea).
    It provides e.target.value and e.target.checked. MouseEvent<T> fires for
    mouse interactions (click, mouseover, etc.) and provides coordinates.

Q3: How do you type an onClick handler for a button?
    Two ways:
    1. Inline: onClick={(e: React.MouseEvent<HTMLButtonElement>) => { ... }}
    2. Handler shorthand: const handler: React.MouseEventHandler<HTMLButtonElement> = e => { ... }

Q4: What is the difference between e.target and e.currentTarget in React?
    e.currentTarget — the element with the event listener, always typed as T
    e.target — the originating element (may be a child of currentTarget)
    TypeScript types currentTarget as T but target as EventTarget (needs narrowing).

Q5: How do you type a form submit handler?
    (e: React.FormEvent<HTMLFormElement>) => { e.preventDefault(); ... }
    Or use the shorthand: React.FormEventHandler<HTMLFormElement>

Q6: What does React.ChangeEventHandler<HTMLInputElement> mean?
    It's a shorthand type alias for:
    (event: React.ChangeEvent<HTMLInputElement>) => void
    It's useful when passing handlers as props.

Q7: How do you get the input value from an onChange event?
    e.target.value for text inputs, e.target.checked for checkboxes.
    Both are correctly typed when using ChangeEvent<HTMLInputElement>.

Q8: How do you type a keyboard event handler that checks for specific keys?
    (e: React.KeyboardEvent<HTMLInputElement>) => {
      if (e.key === "Enter") { ... }     // e.key is a string
      if (e.code === "Space") { ... }    // physical key
      if (e.ctrlKey && e.key === "s") {} // modifier + key
    }

Q9: Why is `e.currentTarget` more reliable than `e.target`?
    e.currentTarget is always the element that has the event handler attached,
    and TypeScript types it precisely as the element type T. e.target may be
    any element that triggered the event (possibly a child element), and
    TypeScript types it as EventTarget, requiring instanceof narrowing.

Q10: How do you pass an event handler as a prop with TypeScript?
     Use the appropriate React handler type in the props interface:
     interface Props { onSelect: React.ChangeEventHandler<HTMLSelectElement> }
     Or spell it out: onSelect: (e: React.ChangeEvent<HTMLSelectElement>) => void
     Both work — the handler shorthand is more concise.
