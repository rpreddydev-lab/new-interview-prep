1. THE THREE SCENARIOS FOR THIRD-PARTY TYPES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  a) Package ships its own types (best case):
     The package.json has "types" or "typings" pointing to a .d.ts file.
     Install only the package — no extra step needed.
     Examples: axios, zod, @tanstack/react-query, zustand, vite

  b) Types available via DefinitelyTyped (@types/*):
     Community-maintained types at https://github.com/DefinitelyTyped/DefinitelyTyped
     npm install @types/<package> --save-dev
     Examples: @types/react, @types/node, @types/lodash, @types/express

  c) No types available:
     Declare ambient types yourself in a .d.ts file.

2. USING @types/react — KEY TYPES TO KNOW
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  React.ReactNode          — Any renderable value
  React.ReactElement       — JSX element
  React.JSX.Element        — Alias for ReactElement (preferred in new code)
  React.FC<P>              — FunctionComponent with props P
  React.PropsWithChildren<P>  — P & { children?: ReactNode }
  React.Ref<T>             — ref object or callback ref
  React.RefObject<T>       — React.createRef() result
  React.MutableRefObject<T>   — useRef() result
  React.CSSProperties      — React inline style object
  React.ComponentType<P>   — React.FC<P> | React.ComponentClass<P>
  React.ComponentProps<T>  — Props of a component type T
  React.ElementType        — string | ComponentType (for `as` prop)
  React.HTMLAttributes<T>  — All HTML attributes for element T
  React.ButtonHTMLAttributes<HTMLButtonElement>
  React.InputHTMLAttributes<HTMLInputElement>

3. MODULE AUGMENTATION — EXTENDING THIRD-PARTY TYPES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  a) Extending Express Request (JWT / session user):
     // src/types/express.d.ts
     import { User } from "../models/User";
     declare module "express-serve-static-core" {
       interface Request {
         user?: User;
         sessionId?: string;
       }
     }

     // Now in your route handlers:
     app.get("/me", (req, res) => {
       if (!req.user) return res.status(401).send("Unauthorized");
       res.json(req.user); // ✅ User type
     });

  b) Extending Window global:
     // src/types/global.d.ts
     declare global {
       interface Window {
         analytics: {
           track(event: string, properties?: Record<string, unknown>): void;
         };
         __APP_VERSION__: string;
       }
     }
     export {}; // Must be a module

     window.analytics.track("page_view"); // ✅ typed

  c) Augmenting @types/react to add a prop to all JSX elements:
     declare module "react" {
       interface HTMLAttributes<T> {
         "data-testid"?: string; // add to all HTML elements
       }
     }

4. WRITING AMBIENT DECLARATIONS FOR UNTYPED PACKAGES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Quick escape hatch (avoid in real projects):
  declare module "untyped-package";
  // All imports: any

  // Minimal useful declaration:
  declare module "legacy-analytics" {
    export function track(event: string, data?: Record<string, unknown>): void;
    export function identify(userId: string): void;
    export const version: string;
  }

  // Non-JS asset declarations (for bundlers):
  declare module "*.svg" {
    import React from "react";
    const ReactComponent: React.FunctionComponent<React.SVGProps<SVGSVGElement>>;
    export default ReactComponent;
  }

  declare module "*.png" { const src: string; export default src; }
  declare module "*.jpg" { const src: string; export default src; }
  declare module "*.css" { const classes: Record<string, string>; export default classes; }
  declare module "*.module.css" {
    const classes: Record<string, string>;
    export default classes;
  }

5. COMMONLY USED THIRD-PARTY LIBRARY TYPES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  a) React Router v6 (has built-in types):
     import { useNavigate, useParams, Link } from "react-router-dom";
     const { id } = useParams<{ id: string }>(); // id: string | undefined
     const navigate = useNavigate();              // NavigateFunction

  b) React Hook Form:
     import { useForm, SubmitHandler } from "react-hook-form";
     interface FormValues { email: string; password: string; }
     const { register, handleSubmit, formState: { errors } } = useForm<FormValues>();
     const onSubmit: SubmitHandler<FormValues> = data => console.log(data);

  c) Zod for runtime type validation + type inference:
     import { z } from "zod";
     const UserSchema = z.object({
       id:    z.number(),
       name:  z.string().min(1),
       email: z.string().email(),
     });
     type User = z.infer<typeof UserSchema>; // { id: number; name: string; email: string }

     const result = UserSchema.safeParse(data);
     if (result.success) { result.data.name; } // ✅ fully typed

  d) @tanstack/react-query:
     import { useQuery, useMutation } from "@tanstack/react-query";
     const { data, isLoading } = useQuery<User[]>({
       queryKey: ["users"],
       queryFn: () => fetch("/api/users").then(r => r.json()),
     });
     // data: User[] | undefined

6. tsconfig SETTINGS FOR THIRD-PARTY TYPES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // typeRoots — where TypeScript looks for @types packages:
  "typeRoots": ["./node_modules/@types", "./src/types"]

  // types — limit which @types packages are included globally:
  "types": ["node", "react", "react-dom"]
  // Without this, ALL @types/* in node_modules are included.

  // skipLibCheck — skip type checking of all .d.ts files (faster, less strict):
  "skipLibCheck": true  // Recommended for most projects

  // paths — module aliases:
  "paths": { "@/*": ["./src/*"] }

7. INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q1: What is DefinitelyTyped?
    A community-maintained repository of TypeScript type definitions for popular
    JavaScript libraries that don't ship their own types. Types are published as
    @types/<package> on npm. It contains thousands of packages like @types/react,
    @types/lodash, @types/express.

Q2: How do you know if a package ships its own types vs. needs @types?
    Check the package's package.json for "types" or "typings" fields. If present,
    the package ships its own types. If absent, check npm for @types/<package>.
    TypeScript IDE extensions also show a "types" indicator.

Q3: What is module augmentation?
    Adding to an existing module's exported types without modifying the original
    package. Use `declare module "package-name" { ... }` to extend its types.
    Example: adding `user` to Express's Request interface.

Q4: How do you type imports of non-JS files like SVGs or CSS modules?
    Create a declaration file with wildcard module declarations:
    declare module "*.svg" { const src: string; export default src; }
    declare module "*.module.css" { const classes: Record<string, string>; export default classes; }

Q5: What does `declare module "some-package"` without a body do?
    It declares the module as typed as `any` — a quick escape hatch.
    All imports from it will be `any`. Better to add a real type signature
    when possible.

Q6: What is `skipLibCheck: true` in tsconfig?
    It tells TypeScript to skip type checking of all declaration files (.d.ts).
    This speeds up compilation and avoids errors from conflicting type versions
    in node_modules. Recommended for most projects but hides library type errors.

Q7: What is React.CSSProperties used for?
    Typing inline style objects in React:
    const style: React.CSSProperties = { color: "red", fontSize: 16 }
    It maps to all valid CSS property names with appropriate TypeScript types.

Q8: What is React.ComponentProps<T>?
    It extracts the props type from a component:
    type ButtonProps = React.ComponentProps<typeof Button>
    Useful when you want to pass all props of a component to a wrapper without
    duplicating the type definition.

Q9: How does Zod's z.infer<typeof Schema> work?
    z.infer extracts the TypeScript type from a Zod schema. The schema validates
    at runtime and Zod's type inference generates the matching compile-time type.
    This gives you both runtime validation and compile-time type safety from
    one definition.

Q10: Why is React.ComponentType<P> useful?
     It's the union of FunctionComponent<P> | ComponentClass<P>, allowing you to
     write code that accepts either kind of component:
     function wrap(Component: React.ComponentType<UserProps>) { ... }
     This is useful for HOCs and rendering components dynamically.
