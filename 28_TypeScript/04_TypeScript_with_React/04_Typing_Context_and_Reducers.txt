  // Option A: Provide a real default (if a sensible one exists):
  interface ThemeContextType {
    theme: "light" | "dark";
    toggleTheme: () => void;
  }

  const ThemeContext = React.createContext<ThemeContextType>({
    theme: "light",
    toggleTheme: () => {},
  });

  // Option B: Use undefined default + non-null assertion in hook (most common):
  const ThemeContext = React.createContext<ThemeContextType | undefined>(undefined);

  function useTheme(): ThemeContextType {
    const ctx = React.useContext(ThemeContext);
    if (!ctx) {
      throw new Error("useTheme must be used within a ThemeProvider");
    }
    return ctx;  // ✅ ThemeContextType — undefined eliminated
  }

2. COMPLETE CONTEXT PROVIDER PATTERN
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // 1. Define types:
  interface User {
    id: string;
    name: string;
    email: string;
    role: "admin" | "user";
  }

  interface AuthContextType {
    user: User | null;
    login: (email: string, password: string) => Promise<void>;
    logout: () => void;
    isLoading: boolean;
  }

  // 2. Create context:
  const AuthContext = React.createContext<AuthContextType | undefined>(undefined);

  // 3. Create provider:
  interface AuthProviderProps {
    children: React.ReactNode;
  }

  function AuthProvider({ children }: AuthProviderProps) {
    const [user, setUser] = React.useState<User | null>(null);
    const [isLoading, setIsLoading] = React.useState(false);

    const login = React.useCallback(async (email: string, password: string) => {
      setIsLoading(true);
      try {
        const response = await fetch("/api/login", {
          method: "POST",
          body: JSON.stringify({ email, password }),
        });
        const userData: User = await response.json();
        setUser(userData);
      } finally {
        setIsLoading(false);
      }
    }, []);

    const logout = React.useCallback(() => {
      setUser(null);
    }, []);

    const value: AuthContextType = React.useMemo(
      () => ({ user, login, logout, isLoading }),
      [user, login, logout, isLoading]
    );

    return (
      <AuthContext.Provider value={value}>
        {children}
      </AuthContext.Provider>
    );
  }

  // 4. Custom hook:
  function useAuth(): AuthContextType {
    const ctx = React.useContext(AuthContext);
    if (!ctx) throw new Error("useAuth must be inside AuthProvider");
    return ctx;
  }

  // 5. Usage in component:
  function Profile() {
    const { user, logout } = useAuth();
    if (!user) return <p>Not logged in</p>;
    return (
      <div>
        <p>Welcome, {user.name}</p>
        <button onClick={logout}>Logout</button>
      </div>
    );
  }

3. useReducer WITH COMPLEX TYPED STATE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // 1. State type:
  interface CartState {
    items: CartItem[];
    totalCount: number;
    totalPrice: number;
  }

  interface CartItem {
    id: string;
    name: string;
    price: number;
    quantity: number;
  }

  // 2. Action type (discriminated union):
  type CartAction =
    | { type: "ADD_ITEM";    item: CartItem }
    | { type: "REMOVE_ITEM"; id: string }
    | { type: "UPDATE_QTY";  id: string; quantity: number }
    | { type: "CLEAR_CART" };

  // 3. Reducer:
  function cartReducer(state: CartState, action: CartAction): CartState {
    switch (action.type) {
      case "ADD_ITEM": {
        const existing = state.items.find(i => i.id === action.item.id);
        const items = existing
          ? state.items.map(i => i.id === action.item.id ? { ...i, quantity: i.quantity + 1 } : i)
          : [...state.items, action.item];
        return recalculate({ ...state, items });
      }
      case "REMOVE_ITEM":
        return recalculate({ ...state, items: state.items.filter(i => i.id !== action.id) });
      case "UPDATE_QTY":
        return recalculate({
          ...state,
          items: state.items.map(i => i.id === action.id ? { ...i, quantity: action.quantity } : i),
        });
      case "CLEAR_CART":
        return { items: [], totalCount: 0, totalPrice: 0 };
      default:
        const _: never = action; // exhaustiveness
        return state;
    }
  }

  function recalculate(state: CartState): CartState {
    return {
      ...state,
      totalCount: state.items.reduce((sum, i) => sum + i.quantity, 0),
      totalPrice: state.items.reduce((sum, i) => sum + i.price * i.quantity, 0),
    };
  }

  // 4. Context + useReducer:
  interface CartContextType {
    state: CartState;
    dispatch: React.Dispatch<CartAction>;
  }
  const CartContext = React.createContext<CartContextType | undefined>(undefined);

  function CartProvider({ children }: { children: React.ReactNode }) {
    const [state, dispatch] = React.useReducer(cartReducer, { items: [], totalCount: 0, totalPrice: 0 });
    return (
      <CartContext.Provider value={{ state, dispatch }}>
        {children}
      </CartContext.Provider>
    );
  }

  function useCart() {
    const ctx = React.useContext(CartContext);
    if (!ctx) throw new Error("useCart must be inside CartProvider");
    return ctx;
  }

  // 5. Usage:
  function CartButton() {
    const { state, dispatch } = useCart();
    return (
      <div>
        <span>Cart: {state.totalCount} items</span>
        <button onClick={() => dispatch({ type: "CLEAR_CART" })}>Clear</button>
      </div>
    );
  }

4. TYPED useReducer WITH IMMER
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  import { produce } from "immer";

  type ImmerReducer<S, A> = (draft: S, action: A) => void | S;

  function useImmerReducer<S, A>(
    reducer: ImmerReducer<S, A>,
    initialState: S
  ): [S, React.Dispatch<A>] {
    return React.useReducer(
      (state: S, action: A) => produce(state, draft => reducer(draft as S, action)),
      initialState
    );
  }

  const immerCartReducer: ImmerReducer<CartState, CartAction> = (draft, action) => {
    switch (action.type) {
      case "ADD_ITEM":
        const existing = draft.items.find(i => i.id === action.item.id);
        if (existing) { existing.quantity++; }
        else { draft.items.push(action.item); }
        break;
      case "REMOVE_ITEM":
        draft.items = draft.items.filter(i => i.id !== action.id);
        break;
      case "CLEAR_CART":
        draft.items = [];
    }
    // Note: no need to recalculate — can derive totals in selector or memo
  };

6. INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q1: How do you type React.createContext?
    createContext<T>(defaultValue: T) where T is your context value type.
    Common pattern: createContext<MyType | undefined>(undefined) to force
    consumers to check they're inside the provider, via a custom hook that
    throws if the context is undefined.

Q2: Why use `| undefined` as the default context value?
    It ensures consumers explicitly handle the case where they're outside the
    provider. A custom hook checks for undefined and throws a meaningful error,
    eliminating the undefined from the returned type.

Q3: How do you type the dispatch function from useReducer?
    TypeScript infers it as React.Dispatch<ActionType>.
    You can use this type in context:
    interface ContextType { dispatch: React.Dispatch<CartAction> }

Q4: What is the benefit of discriminated union actions with useReducer?
    TypeScript narrows the action type in each case branch, giving type-safe
    access to action-specific properties (like `action.id` only in REMOVE_ITEM).
    The default branch can use `never` for exhaustiveness checking.

Q5: How do you avoid re-renders when providing context values?
    Memoize the context value with useMemo:
    const value = useMemo(() => ({ user, login, logout }), [user, login, logout]);
    Without memoization, a new object is created on every render, causing all
    consumers to re-render even if nothing changed.

Q6: What is the useCallback pattern in context providers?
    Wrap handler functions in useCallback to prevent recreating them on every
    render. This keeps them stable so they don't cause unnecessary context
    re-renders or dependency changes in consumers.

Q7: How do you split context value to prevent unnecessary re-renders?
    Create separate contexts for different parts of the value:
    const StateContext = createContext<State>(...)
    const DispatchContext = createContext<Dispatch<Action>>(...)
    Components that only need dispatch don't re-render when state changes.

Q8: How do you type React.Dispatch?
    React.Dispatch<A> is equivalent to (action: A) => void.
    It's what useReducer returns as the second element.

Q9: What is the difference between useContext and custom hooks for context?
    useContext directly accesses the context — it doesn't validate or enhance the value.
    Custom hooks wrap useContext, throw if context is undefined, and can add
    computed properties or helper methods on top of the raw context value.

Q10: When would you use useReducer + context instead of useState + context?
     useReducer is better for: complex state with many sub-values, multiple
     related updates that should happen together, when next state depends on
     previous state in non-trivial ways, or when you want Redux-like patterns
     (explicit actions, pure reducer, easy testing).
