1. WHY GENERIC COMPONENTS?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Generic components allow you to create reusable, type-safe components that
  adapt to whatever data type is passed to them — without duplicating code or
  losing type information.

  // Without generics — loses type info:
  function List({ items }: { items: any[] }) {
    return <ul>{items.map((item, i) => <li key={i}>{item}</li>)}</ul>;
  }

  // With generics — preserves type:
  function List<T>({ items, renderItem }: {
    items: T[];
    renderItem: (item: T) => React.ReactNode;
  }) {
    return <ul>{items.map((item, i) => <li key={i}>{renderItem(item)}</li>)}</ul>;
  }

  interface SelectProps<T> {
    options: T[];
    value: T;
    onChange: (value: T) => void;
    getLabel: (option: T) => string;
    getValue: (option: T) => string;
  }

  function Select<T>({ options, value, onChange, getLabel, getValue }: SelectProps<T>) {
    return (
      <select
        value={getValue(value)}
        onChange={e => {
          const selected = options.find(o => getValue(o) === e.target.value);
          if (selected) onChange(selected);
        }}
      >
        {options.map(option => (
          <option key={getValue(option)} value={getValue(option)}>
            {getLabel(option)}
          </option>
        ))}
      </select>
    );
  }

  // Usage — TypeScript infers T = { id: number; name: string }:
  type Country = { id: number; name: string };
  const countries: Country[] = [{ id: 1, name: "USA" }, { id: 2, name: "UK" }];

  <Select
    options={countries}
    value={countries[0]}
    onChange={setCountry}          // (country: Country) => void ✅
    getLabel={c => c.name}
    getValue={c => String(c.id)}
  />

3. TSX SYNTAX — AVOIDING `<T>` AMBIGUITY
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  In .tsx files, the < character before a type parameter can be misread as
  JSX. Fix with a trailing comma or constraint:

  // ❌ Ambiguous in .tsx:
  const identity = <T>(value: T): T => value; // looks like JSX

  // ✅ Use a trailing comma:
  const identity = <T,>(value: T): T => value;

  // ✅ Add a constraint:
  const identity = <T extends unknown>(value: T): T => value;

  // ✅ Use a regular function (no ambiguity):
  function identity<T>(value: T): T { return value; }

4. CONSTRAINED GENERIC COMPONENTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Require items to have an `id` property for stable keys:
  interface HasId { id: string | number }

  interface ListProps<T extends HasId> {
    items: T[];
    renderItem: (item: T, index: number) => React.ReactNode;
    emptyMessage?: string;
  }

  function TypedList<T extends HasId>({ items, renderItem, emptyMessage = "No items" }: ListProps<T>) {
    if (items.length === 0) return <p>{emptyMessage}</p>;
    return (
      <ul>
        {items.map((item, i) => (
          <li key={item.id}>{renderItem(item, i)}</li>  // ✅ item.id always exists
        ))}
      </ul>
    );
  }

  // Usage:
  type User = { id: number; name: string; email: string };
  const users: User[] = [...];

  <TypedList
    items={users}
    renderItem={user => <span>{user.name} — {user.email}</span>}
    emptyMessage="No users found"
  />

5. GENERIC TABLE COMPONENT
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  interface ColumnDef<T> {
    key: keyof T;
    header: string;
    render?: (value: T[keyof T], row: T) => React.ReactNode;
  }

  interface TableProps<T extends HasId> {
    data: T[];
    columns: ColumnDef<T>[];
  }

  function Table<T extends HasId>({ data, columns }: TableProps<T>) {
    return (
      <table>
        <thead>
          <tr>{columns.map(col => <th key={String(col.key)}>{col.header}</th>)}</tr>
        </thead>
        <tbody>
          {data.map(row => (
            <tr key={row.id}>
              {columns.map(col => {
                const value = row[col.key];
                return (
                  <td key={String(col.key)}>
                    {col.render ? col.render(value, row) : String(value)}
                  </td>
                );
              })}
            </tr>
          ))}
        </tbody>
      </table>
    );
  }

  // Usage:
  const columns: ColumnDef<User>[] = [
    { key: "name",  header: "Name" },
    { key: "email", header: "Email" },
    { key: "id",    header: "ID", render: val => <code>{String(val)}</code> },
  ];
  <Table data={users} columns={columns} />

6. GENERIC FORM FIELD COMPONENT
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // A Field that works with any form value type:
  interface FieldProps<T, K extends keyof T> {
    label: string;
    fieldKey: K;
    value: T[K];
    onChange: (key: K, value: T[K]) => void;
  }

  function TextField<T extends Record<string, string>, K extends keyof T>({
    label,
    fieldKey,
    value,
    onChange,
  }: FieldProps<T, K>) {
    return (
      <div>
        <label>{label}</label>
        <input
          value={String(value)}
          onChange={e => onChange(fieldKey, e.target.value as T[K])}
        />
      </div>
    );
  }

7. POLYMORPHIC COMPONENTS (AS PROP)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  A polymorphic component renders as a different HTML element based on an `as` prop:

  type AsProp<C extends React.ElementType> = {
    as?: C;
  };

  type PropsToOmit<C extends React.ElementType, P> = keyof (AsProp<C> & P);

  type PolymorphicComponentProp<C extends React.ElementType, Props = {}> =
    React.PropsWithChildren<Props & AsProp<C>> &
    Omit<React.ComponentPropsWithoutRef<C>, PropsToOmit<C, Props>>;

  // Simplified button/link:
  interface ButtonOwnProps {
    variant?: "primary" | "ghost";
    size?: "sm" | "md" | "lg";
  }

  type ButtonProps<C extends React.ElementType = "button"> =
    ButtonOwnProps & { as?: C } & Omit<React.ComponentPropsWithoutRef<C>, keyof ButtonOwnProps | "as">;

  function Button<C extends React.ElementType = "button">({
    as,
    variant = "primary",
    size = "md",
    ...rest
  }: ButtonProps<C>) {
    const Component = as ?? "button";
    return <Component className={`btn btn-${variant} btn-${size}`} {...rest} />;
  }

  // Usage:
  <Button>Click me</Button>                           // renders <button>
  <Button as="a" href="/page">Link</Button>           // renders <a href="/page">
  <Button as={Link} to="/page">Router Link</Button>  // renders <Link to="/page">

8. INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q1: What is a generic React component?
    A component with a type parameter that allows it to work with any type
    without losing type information. Instead of using any[] for items, you use
    T[] and the caller's TypeScript infers T from usage.

Q2: Why do you need a trailing comma in arrow generic components in .tsx files?
    In .tsx files, TypeScript can mistake <T> in `<T>(x: T) => x` for a JSX
    tag. Adding a trailing comma: `<T,>(x: T) => x` disambiguates it.
    Alternatively, use a constraint: <T extends unknown>(x: T) => x.

Q3: How do you constrain a generic component so T must have an `id` property?
    interface HasId { id: string | number }
    function List<T extends HasId>({ items }: { items: T[] }) { ... }
    Now every item is guaranteed to have an id for use as a React key.

Q4: How does TypeScript infer the generic type in a generic component?
    TypeScript infers T from the props you pass. If you call
    <List items={users} />, TypeScript sees items is User[] and infers T = User.
    You can also explicitly specify: <List<User> items={users} />

Q5: What is a polymorphic component and how do you type the `as` prop?
    A polymorphic component renders as different HTML elements or components
    based on an `as` prop. The `as` prop is typed as `React.ElementType` and
    the component spreads the appropriate HTML attributes using
    React.ComponentPropsWithoutRef<C> or React.ComponentPropsWithRef<C>.

Q6: How do you type a renderItem prop in a generic list?
    Use the generic parameter:
    interface ListProps<T> { items: T[]; renderItem: (item: T) => React.ReactNode; }
    The renderItem callback automatically knows the type of each item.

Q7: What is keyof T useful for in generic components?
    It restricts props to valid property names of T. For example, a Table
    component can type its column definitions using keyof T to ensure only
    valid column keys are specified:
    columns: Array<{ key: keyof T; header: string }>

Q8: Can you use a generic component with React.forwardRef?
    Yes, but the syntax is slightly awkward:
    const Input = forwardRef(<T extends HTMLInputElement>(props: InputProps, ref: React.Ref<T>) => ...)
    Or use a regular function component that accepts a ref prop typed manually.

Q9: When does TypeScript fail to infer a component's generic type?
    When the inference is ambiguous or when the generic type can't be determined
    from any prop. In those cases, you need to explicitly specify the type:
    <Select<Country> options={countries} ... />

Q10: How is a generic component different from using `any[]` for items?
     With `any[]`, all type information is lost — you can't type renderItem,
     onChange, or returned values. With a generic T[], TypeScript tracks the
     actual type through the component. If T = User, renderItem receives a
     User and all its properties are available with full type safety.
