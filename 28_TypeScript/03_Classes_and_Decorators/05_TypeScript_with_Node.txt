1. SETTING UP A TYPESCRIPT NODE.JS PROJECT
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  a) Install dependencies:
     npm init -y
     npm install typescript @types/node --save-dev
     npx tsc --init

  b) Recommended tsconfig.json for Node.js:
     {
       "compilerOptions": {
         "target": "ES2022",
         "module": "CommonJS",        // or "NodeNext" for ESM
         "moduleResolution": "Node",  // or "NodeNext"
         "outDir": "dist",
         "rootDir": "src",
         "strict": true,
         "esModuleInterop": true,
         "skipLibCheck": true,
         "declaration": true,
         "sourceMap": true,
         "lib": ["ES2022"]
       },
       "include": ["src/**/*"],
       "exclude": ["node_modules", "dist"]
     }

  c) For ESM (Node 18+):
     "module": "NodeNext",
     "moduleResolution": "NodeNext"
     And add to package.json: "type": "module"

2. @types/node — TYPING NODE.JS BUILT-INS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  npm install @types/node --save-dev

  This provides types for all Node.js built-in modules:
  fs, path, http, https, os, crypto, events, stream, buffer, process, etc.

  import * as fs from "node:fs/promises";  // ✅ "node:" prefix recommended
  import * as path from "node:path";
  import * as http from "node:http";

  // process is globally available (no import needed with @types/node):
  console.log(process.env.NODE_ENV); // string | undefined
  console.log(process.argv);         // string[]
  process.exit(0);

3. FILE SYSTEM OPERATIONS (TYPED)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  import { readFile, writeFile, mkdir } from "node:fs/promises";
  import { existsSync } from "node:fs";

  async function readConfig(filePath: string): Promise<Record<string, unknown>> {
    const raw = await readFile(filePath, "utf-8"); // string
    return JSON.parse(raw) as Record<string, unknown>;
  }

  async function writeOutput(data: unknown, dest: string): Promise<void> {
    await mkdir(path.dirname(dest), { recursive: true });
    await writeFile(dest, JSON.stringify(data, null, 2), "utf-8");
  }

  // Typed Buffer operations:
  const buf: Buffer = Buffer.from("Hello TypeScript", "utf-8");
  const base64: string = buf.toString("base64");

4. HTTP SERVER WITH TYPESCRIPT
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  import http, { IncomingMessage, ServerResponse } from "node:http";

  const server = http.createServer(
    (req: IncomingMessage, res: ServerResponse): void => {
      if (req.method === "GET" && req.url === "/health") {
        res.writeHead(200, { "Content-Type": "application/json" });
        res.end(JSON.stringify({ status: "ok" }));
        return;
      }
      res.writeHead(404);
      res.end("Not Found");
    }
  );

  server.listen(3000, () => {
    console.log("Server running on http://localhost:3000");
  });

5. ENVIRONMENT VARIABLES — TYPED ACCESS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // process.env values are always string | undefined
  const port = process.env.PORT;  // string | undefined

  // Type-safe env helper:
  function requireEnv(key: string): string {
    const value = process.env[key];
    if (!value) throw new Error(`Missing required env var: ${key}`);
    return value;
  }

  const DB_URL = requireEnv("DATABASE_URL"); // string — guaranteed

  // Type the entire env with an interface:
  interface ProcessEnv {
    NODE_ENV: "development" | "production" | "test";
    PORT?: string;
    DATABASE_URL: string;
    JWT_SECRET: string;
  }

  declare global {
    namespace NodeJS {
      interface ProcessEnv extends ProcessEnv {}
    }
  }

  // Now process.env.NODE_ENV is typed as "development" | "production" | "test"
  if (process.env.NODE_ENV === "production") { /* ... */ }

6. RUNNING TYPESCRIPT DIRECTLY — ts-node / tsx
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  a) ts-node — original TS execution for Node:
     npm install ts-node --save-dev
     npx ts-node src/index.ts
     npx ts-node --esm src/index.ts   // ESM mode

  b) tsx — faster, Esbuild-based:
     npm install tsx --save-dev
     npx tsx src/index.ts
     npx tsx watch src/index.ts      // watch mode (restarts on change)

  c) node --loader ts-node/esm — experimental built-in:
     node --loader ts-node/esm src/index.ts

  d) Native Node.js support (Node 22.6+ / TS 5.8):
     node --experimental-strip-types src/index.ts
     // Strips types but does NOT type-check

7. TYPED EXPRESS.JS EXAMPLE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  import express, { Request, Response, NextFunction } from "express";

  // Typed Request params, body, query, and response:
  interface CreateUserBody {
    name: string;
    email: string;
  }

  interface UserParams {
    id: string;
  }

  const app = express();
  app.use(express.json());

  app.post(
    "/users",
    (req: Request<{}, {}, CreateUserBody>, res: Response) => {
      const { name, email } = req.body; // ✅ typed
      res.status(201).json({ id: "1", name, email });
    }
  );

  app.get(
    "/users/:id",
    (req: Request<UserParams>, res: Response) => {
      const { id } = req.params; // ✅ string
      res.json({ id });
    }
  );

  // Typed error middleware:
  app.use((err: Error, req: Request, res: Response, next: NextFunction) => {
    console.error(err.stack);
    res.status(500).json({ error: err.message });
  });

8. INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q1: What is @types/node and why do you need it?
    @types/node provides TypeScript type definitions for Node.js built-in
    modules (fs, path, http, etc.) and globals (process, Buffer, __dirname, etc.).
    Without it, TypeScript doesn't know the types of Node-specific APIs.

Q2: How do you run TypeScript files directly in Node.js?
    Use ts-node (`npx ts-node src/index.ts`) or tsx (`npx tsx src/index.ts`).
    Node 22.6+ supports `node --experimental-strip-types` to run TS directly
    (strips types, does not type-check). For production, always compile to JS
    first with `tsc`.

Q3: What is process.env's TypeScript type?
    NodeJS.ProcessEnv, where each property is `string | undefined`.
    You can augment it in a declaration file:
    declare namespace NodeJS { interface ProcessEnv { MY_VAR: string } }

Q4: What is the recommended module format for Node.js TypeScript projects?
    For new projects: "module": "NodeNext" with "moduleResolution": "NodeNext"
    for native ESM support. For legacy/simpler setups, "module": "CommonJS"
    with "moduleResolution": "Node" is still widely used.

Q5: How do you type Express request parameters and body?
    The Request type accepts 4 generics: Request<Params, ResBody, ReqBody, Query>
    e.g., Request<{ id: string }, {}, CreateUserBody> types params and body.

Q6: What is the "node:" import prefix?
    Prefixing built-in module imports with "node:" (e.g., import fs from "node:fs")
    explicitly marks them as Node.js built-ins rather than npm packages of the
    same name. Recommended to avoid ambiguity in NodeNext resolution mode.

Q7: How do you type a Node.js event emitter?
    Extend EventEmitter and use overloads for on/emit:
    class TypedEmitter extends EventEmitter {
      on(event: "data", listener: (data: Buffer) => void): this;
      on(event: string, listener: (...args: any[]) => void): this { ... }
    }
    Or use a library like `eventemitter3` with typed generics.

Q8: What does `esModuleInterop: true` do in a Node.js tsconfig?
    Enables default import interoperability with CommonJS modules that don't
    have a default export. Allows `import express from "express"` instead of
    `import * as express from "express"`.

Q9: How do you compile TypeScript for a Node.js project?
    Run `tsc` which reads tsconfig.json and outputs to `outDir`.
    Typical scripts in package.json:
    "build": "tsc", "start": "node dist/index.js",
    "dev": "tsx watch src/index.ts"

Q10: How do you handle unknown error types in TypeScript Node.js catch blocks?
     In TypeScript 4.0+, caught errors are typed as `unknown` with useUnknownInCatchVariables.
     Always narrow before use:
     try { ... }
     catch (err) {
       if (err instanceof Error) {
         console.error(err.message); // ✅ err is Error
       } else {
         console.error(String(err));
       }
     }
