1. THE FOUR ACCESS MODIFIERS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  ┌───────────────┬────────────┬───────────────┬────────────────┐
  │ Modifier      │ Same Class │ Subclass      │ Outside Class  │
  ├───────────────┼────────────┼───────────────┼────────────────┤
  │ public        │     ✅     │      ✅       │       ✅       │
  │ protected     │     ✅     │      ✅       │       ❌       │
  │ private       │     ✅     │      ❌       │       ❌       │
  │ readonly      │ read only  │  read only    │   read only    │
  └───────────────┴────────────┴───────────────┴────────────────┘

  Default (no modifier) = public.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
2. PUBLIC
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  class Person {
    public name: string; // Same as: name: string
    constructor(name: string) { this.name = name; }
    public greet(): string { return `Hello, ${this.name}`; }
  }

  const p = new Person("Alice");
  console.log(p.name);    // ✅ accessible anywhere
  console.log(p.greet()); // ✅

  // public is the default — explicit use is for documentation clarity only.

3. PRIVATE (TypeScript keyword)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  class BankAccount {
    private balance: number = 0;
    private transactionHistory: number[] = [];

    deposit(amount: number): void {
      if (amount <= 0) throw new Error("Invalid amount");
      this.balance += amount;               // ✅ accessible within class
      this.transactionHistory.push(amount);
    }

    getBalance(): number {
      return this.balance;                  // ✅
    }
  }

  const account = new BankAccount();
  account.deposit(100);
  console.log(account.getBalance()); // ✅ 100
  console.log(account.balance);      // ❌ Error: 'balance' is private

  // IMPORTANT: TypeScript `private` is COMPILE-TIME ONLY.
  // The compiled JS still exposes it: account.balance works in plain JS!

  // Access between instances of the same class is allowed:
  class Counter {
    private count = 0;
    isSameAs(other: Counter): boolean {
      return this.count === other.count; // ✅ accessing other's private is fine
    }
  }

4. ECMAScript PRIVATE FIELDS (#)  — TRUE RUNTIME PRIVACY
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  class BankAccount {
    #balance = 0; // ECMAScript private field

    deposit(amount: number): void { this.#balance += amount; }
    getBalance(): number { return this.#balance; }
  }

  const acct = new BankAccount();
  acct.deposit(100);
  console.log(acct.getBalance());  // ✅ 100
  console.log((acct as any).balance); // undefined — truly private at runtime!
  // acct.#balance — SyntaxError even in JS!

  // Differences between `private` and `#`:
  ┌──────────────────────────┬──────────────────────┬────────────────────────┐
  │ Feature                  │ TypeScript `private`  │ ECMAScript `#` field   │
  ├──────────────────────────┼──────────────────────┼────────────────────────┤
  │ Runtime enforcement      │ No (compile only)     │ Yes                    │
  │ Type-system enforcement  │ Yes (structural)      │ Yes (nominal/strict)   │
  │ Works with subtypes      │ Structural            │ Stricter               │
  │ Cross-instance access    │ Yes (same class)      │ Yes (same class)       │
  │ Polyfilled in older JS   │ N/A (erased)          │ WeakMap in older targets│
  └──────────────────────────┴──────────────────────┴────────────────────────┘

5. PROTECTED
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  class Animal {
    protected name: string;
    constructor(name: string) { this.name = name; }

    protected breathe(): void { console.log(`${this.name} breathes`); }
  }

  class Dog extends Animal {
    bark(): void {
      console.log(`${this.name} says woof!`); // ✅ protected accessible in subclass
      this.breathe(); // ✅
    }
  }

  const dog = new Dog("Rex");
  dog.bark();       // ✅
  dog.name;         // ❌ Error: 'name' is protected
  dog.breathe();    // ❌ Error: 'breathe' is protected

  // Protected constructor — only instantiable through subclass:
  class Singleton {
    private static instance: Singleton | null = null;
    protected constructor() {}
    static create(): Singleton {
      if (!Singleton.instance) Singleton.instance = new Singleton();
      return Singleton.instance;
    }
  }

6. READONLY
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  class Config {
    readonly apiUrl: string;
    readonly maxRetries: number = 3; // Can set a default

    constructor(url: string) {
      this.apiUrl = url; // ✅ can assign in constructor
    }

    update(): void {
      this.apiUrl = "other"; // ❌ Error: cannot assign to 'apiUrl' (readonly)
    }
  }

  const cfg = new Config("https://api.example.com");
  cfg.apiUrl = "other"; // ❌ Error: readonly

  // readonly CAN be combined with access modifiers:
  class Point {
    constructor(
      public readonly x: number,
      public readonly y: number
    ) {}
  }
  const p = new Point(1, 2);
  p.x; // ✅ readable
  p.x = 3; // ❌ readonly

  // readonly vs const:
  // const — for variables (doesn't exist on class properties)
  // readonly — for class properties (enforced at compile time)

7. ACCESS MODIFIERS ON CONSTRUCTOR (ABSTRACT FACTORY / DI PATTERN)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Private constructor — class cannot be instantiated from outside:
  class Immutable {
    private constructor(public value: number) {}
    static of(value: number) { return new Immutable(value); }
  }
  const x = new Immutable(5);       // ❌ Error
  const y = Immutable.of(5);        // ✅

  // Protected constructor — can only be instantiated via a subclass:
  abstract class Service {
    protected constructor(protected url: string) {}
  }

8. OVERRIDE KEYWORD (TypeScript 4.3+)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  class Base {
    greet() { return "Hello from Base"; }
  }

  class Derived extends Base {
    override greet() { return "Hello from Derived"; } // ✅ explicit override
    override typo() { ... }  // ❌ Error: 'typo' doesn't exist in Base
  }

  // With noImplicitOverride: true in tsconfig, ALL overrides must use `override`
  // This prevents accidental method shadowing and catches renamed base methods.

9. INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q1: What are the access modifiers in TypeScript?
    public (default, accessible everywhere), private (only within the declaring
    class), protected (within declaring class and subclasses), and readonly
    (can only be assigned in the constructor, not reassigned later).

Q2: What is the difference between TypeScript `private` and ECMAScript `#`?
    TypeScript `private` is compile-time enforcement only — the compiled JS
    exposes the property. ECMAScript `#` private fields are truly private at
    runtime — they cannot be accessed from outside the class even in JavaScript.
    # fields are nominal (can't be accessed on a structurally compatible type),
    while `private` uses structural typing.

Q3: Can a subclass access `protected` members?
    Yes. `protected` members are accessible in the declaring class AND all
    subclasses, but NOT from outside the class hierarchy.

Q4: When can a `readonly` property be assigned?
    Only during declaration (if given a default) or inside the constructor.
    After construction, any assignment to a readonly property is a compile error.

Q5: What happens when you access a TypeScript `private` property via `as any`?
    It compiles without error because `as any` bypasses type checking.
    The compiled JS also has no runtime guard, so accessing it works.
    ECMAScript `#` fields are the only truly runtime-enforced private fields.

Q6: What does `protected constructor` mean?
    The class cannot be instantiated directly from outside the class or its
    subclasses. It can only be constructed from within a subclass or from a
    static factory method in the same class.

Q7: How does readonly differ from const for class properties?
    `const` is for variables and cannot be used on class properties.
    `readonly` is the class-property equivalent — it allows assignment only
    in the constructor, not afterward.

Q8: What is the `override` keyword and what tsconfig option enables stricter use?
    `override` (TS 4.3+) marks a method as intentionally overriding a base
    class method. It produces an error if the method name doesn't exist in the
    base class. Setting `noImplicitOverride: true` in tsconfig REQUIRES all
    overriding methods to use the `override` keyword.

Q9: Can you combine access modifiers with readonly?
    Yes: `public readonly x: number` or as a parameter property in a constructor:
    constructor(public readonly id: number, private readonly name: string) {}

Q10: Does `private` make a member unavailable between two instances of the same class?
     No. Two instances of the same class can access each other's `private` members
     within the class body:
     class Box { private val = 0;
       equals(other: Box) { return this.val === other.val; } // ✅
     }
