1. TYPESCRIPT CLASSES — OVERVIEW
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  TypeScript extends JavaScript ES2015 classes with:
  - Type annotations on properties and parameters
  - Access modifiers (public, private, protected, readonly)
  - Parameter properties (shorthand constructor assignments)
  - Abstract classes and methods
  - Implements keyword for interface enforcement
  - Generics in classes
  - Override keyword (TS 4.3+)
  - Static blocks (TS 4.4+)

2. CLASS BASICS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  class User {
    // Property declarations (required in TS — JS doesn't require them)
    id: number;
    name: string;
    readonly createdAt: Date;

    constructor(id: number, name: string) {
      this.id = id;
      this.name = name;
      this.createdAt = new Date();
    }

    greet(): string {
      return `Hello, I am ${this.name}`;
    }

    get nameUppercase(): string {
      return this.name.toUpperCase();
    }

    set displayName(val: string) {
      this.name = val.trim();
    }
  }

  const user = new User(1, "Alice");
  console.log(user.greet());         // "Hello, I am Alice"
  console.log(user.nameUppercase);   // "ALICE"

3. PARAMETER PROPERTIES (SHORTHAND)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Parameter properties let you declare AND initialize class members in one step
  by adding an access modifier or `readonly` to the constructor parameter:

  // Long form:
  class PersonLong {
    public name: string;
    private age: number;
    readonly id: number;
    constructor(name: string, age: number, id: number) {
      this.name = name;
      this.age = age;
      this.id = id;
    }
  }

  // Short form (parameter properties):
  class Person {
    constructor(
      public name: string,
      private age: number,
      readonly id: number
    ) {}
    // Properties are declared and assigned automatically
  }

  const p = new Person("Bob", 30, 42);
  console.log(p.name); // "Bob" ✅
  console.log(p.age);  // ❌ Error: 'age' is private

4. INHERITANCE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  class Animal {
    constructor(public name: string) {}
    move(distance: number = 0): void {
      console.log(`${this.name} moved ${distance}m`);
    }
    toString(): string { return `Animal(${this.name})`; }
  }

  class Dog extends Animal {
    constructor(name: string, public breed: string) {
      super(name); // Must call super() before using `this`
    }

    bark(): void { console.log("Woof!"); }

    // Method override:
    override move(distance = 5): void {
      console.log("Running...");
      super.move(distance);
    }
  }

  const dog = new Dog("Rex", "Labrador");
  dog.bark();     // "Woof!"
  dog.move(10);   // "Running... Rex moved 10m"

  // TypeScript checks inheritance:
  // - super() MUST be called in derived constructors (before `this`)
  // - `override` keyword (TS 4.3): ensures you're overriding an existing method
  // - Without `noImplicitOverride: true` in tsconfig, override is optional

5. IMPLEMENTING INTERFACES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  interface Printable {
    print(): void;
  }

  interface Serializable {
    serialize(): string;
  }

  // A class can implement multiple interfaces:
  class Document implements Printable, Serializable {
    constructor(public content: string) {}
    print(): void { console.log(this.content); }
    serialize(): string { return JSON.stringify({ content: this.content }); }
  }

  // KEY POINT: `implements` only checks the shape — it doesn't provide
  // implementation. You still need to write every method/property.

  // IMPORTANT distinction: `implements` vs `extends`:
  // - `extends` inherits implementation from parent class
  // - `implements` only enforces a type contract (interface shape)

6. STATIC MEMBERS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  class MathUtils {
    static readonly PI = 3.14159265;

    static add(a: number, b: number): number { return a + b; }
    static square(x: number): number { return x * x; }

    // Static block for complex initialization (TS 4.4+):
    static {
      console.log("MathUtils loaded");
    }
  }

  console.log(MathUtils.PI);        // 3.14159265
  console.log(MathUtils.add(2, 3)); // 5

  // Singleton pattern using static:
  class Database {
    private static instance: Database | null = null;
    private constructor(private url: string) {}

    static getInstance(url: string): Database {
      if (!Database.instance) {
        Database.instance = new Database(url);
      }
      return Database.instance;
    }

    query(sql: string): void { /* ... */ }
  }

  const db = Database.getInstance("postgres://localhost/mydb");

7. GENERIC CLASSES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  class Stack<T> {
    private items: T[] = [];

    push(item: T): void { this.items.push(item); }
    pop(): T | undefined { return this.items.pop(); }
    peek(): T | undefined { return this.items[this.items.length - 1]; }
    get size(): number { return this.items.length; }
    isEmpty(): boolean { return this.items.length === 0; }
  }

  const stack = new Stack<number>();
  stack.push(1);
  stack.push(2);
  console.log(stack.pop()); // 2

  // Generic Repository pattern:
  interface Entity { id: number }

  class Repository<T extends Entity> {
    private store = new Map<number, T>();

    save(entity: T): T {
      this.store.set(entity.id, entity);
      return entity;
    }

    findById(id: number): T | undefined {
      return this.store.get(id);
    }

    findAll(): T[] {
      return [...this.store.values()];
    }

    delete(id: number): boolean {
      return this.store.delete(id);
    }
  }

8. CLASS AS A TYPE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  In TypeScript, a class declaration creates two things:
  1. A RUNTIME value (the class constructor function)
  2. A COMPILE-TIME type (the shape of instances)

  class Point { constructor(public x: number, public y: number) {} }

  // Using Point as a type (instance shape):
  function distance(p1: Point, p2: Point): number {
    return Math.sqrt((p2.x - p1.x) ** 2 + (p2.y - p1.y) ** 2);
  }

  // Using typeof Point as a type (the class/constructor itself):
  function createPoint(ctor: typeof Point, x: number, y: number): Point {
    return new ctor(x, y);
  }

  // Structural typing — a class is compatible with any type of the same shape:
  class AltPoint { constructor(public x: number, public y: number) {} }
  const p: Point = new AltPoint(1, 2); // ✅ structurally identical

9. MIXINS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  TypeScript supports a mixin pattern for composing class behavior without
  multiple inheritance:

  // Mixin type: a constructor that takes any args:
  type Constructor<T = {}> = new (...args: any[]) => T;

  // Mixin function:
  function Timestamped<TBase extends Constructor>(Base: TBase) {
    return class extends Base {
      createdAt = new Date();
      updatedAt = new Date();
    };
  }

  function Activatable<TBase extends Constructor>(Base: TBase) {
    return class extends Base {
      isActive = false;
      activate() { this.isActive = true; }
      deactivate() { this.isActive = false; }
    };
  }

  class UserBase { constructor(public name: string) {} }

  // Apply mixins:
  const TimestampedUser = Timestamped(UserBase);
  const FullUser = Activatable(Timestamped(UserBase));

  const user = new FullUser("Alice");
  user.activate();
  console.log(user.name, user.isActive, user.createdAt);

10. INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q1: What does TypeScript add to JavaScript classes?
    Type annotations for properties and parameters, access modifiers (public,
    private, protected, readonly), parameter properties, abstract classes,
    implements for interfaces, override keyword (TS 4.3), static blocks (TS 4.4),
    and generics on classes.

Q2: What are parameter properties?
    A TypeScript shorthand that declares and initializes class properties
    directly in the constructor signature using access modifiers:
    constructor(public name: string, private age: number) {}
    This automatically creates properties and assigns constructor arguments.

Q3: What is the difference between `implements` and `extends`?
    `extends` inherits from a parent class, gaining its implementation.
    `implements` declares that a class adheres to an interface's type contract
    but provides NO implementation — you must write everything yourself.

Q4: When do you need to call super()?
    In any derived class constructor, before you can access `this`. In TypeScript
    (with strictPropertyInitialization), failing to call super is a compile error.

Q5: What does a class declaration create in TypeScript?
    Both a runtime constructor function AND a compile-time type for instances.
    `typeof ClassName` refers to the constructor, while `ClassName` (as a type)
    refers to the instance shape.

Q6: How does TypeScript handle private class members?
    TypeScript has two kinds: `private` keyword (erased at runtime, structural
    in type system) and the # ECMAScript private field (truly private at runtime,
    nominal in type system). `private` is enforced only at compile time.

Q7: What is the override keyword and when should you use it?
    Added in TS 4.3, `override` on a method asserts it's overriding a method
    from the base class. With `noImplicitOverride: true` in tsconfig, every
    overridden method MUST use `override`. This prevents accidental overrides
    and catches renamed base methods.

Q8: Can a TypeScript class implement an interface and extend a class?
    Yes: class Dog extends Animal implements Serializable, Printable {}
    A class can extend exactly ONE other class but implement multiple interfaces.

Q9: How do generics work in TypeScript classes?
    Add a type parameter to the class: class Stack<T> { items: T[] = []; }
    The type parameter is resolved when the class is instantiated:
    const s = new Stack<number>(); // T = number throughout the class

Q10: What is the Mixin pattern in TypeScript?
     Mixins compose class behavior by applying higher-order functions that
     take a constructor and return a new class extending it with extra methods.
     type Constructor<T = {}> = new (...args: any[]) => T;
     function Timestamped<B extends Constructor>(Base: B) {
       return class extends Base { createdAt = new Date(); };
     }
     This avoids multiple inheritance while enabling code reuse.
