1. WHAT ARE TEMPLATE LITERAL TYPES? (TypeScript 4.1+)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Template literal types extend string literal types using the same backtick
  syntax as JavaScript template literals, but at the TYPE LEVEL.

  type Greeting = `Hello, ${string}`;
  type Name = "World" | "TypeScript";
  type Msg = `Hello, ${Name}`;  // "Hello, World" | "Hello, TypeScript"

  They allow string manipulation at the type system level, enabling precise
  string-based APIs, auto-generated key names, and strict validation.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
2. BASIC SYNTAX AND INTERPOLATION
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Interpolate any type that is a subtype of string | number | bigint | boolean:

  type EventName = `on${string}`;
  type CssUnit = `${number}px` | `${number}em` | `${number}rem`;

  // Concatenating multiple parts:
  type LangCode = "en" | "fr" | "de";
  type CountryCode = "US" | "FR" | "DE";
  type Locale = `${LangCode}-${CountryCode}`;
  // "en-US" | "en-FR" | "en-DE" | "fr-US" | "fr-FR" | ... (9 combinations)

  // Union distribution — automatically creates cross-product:
  type Sides = "top" | "bottom" | "left" | "right";
  type Margin = `margin-${Sides}`;
  // "margin-top" | "margin-bottom" | "margin-left" | "margin-right"

3. BUILT-IN STRING MANIPULATION TYPES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  TypeScript provides 4 intrinsic string manipulation types:

  type U = Uppercase<"hello">;      // "HELLO"
  type L = Lowercase<"HELLO">;      // "hello"
  type C = Capitalize<"hello">;     // "Hello"
  type UC = Uncapitalize<"Hello">; // "hello"

  // Used with type parameters:
  type CapKey<K extends string> = Capitalize<K>;
  type R = CapKey<"name" | "age">; // "Name" | "Age"

4. GENERATING GETTER/SETTER NAMES WITH MAPPED + TEMPLATE LITERAL TYPES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  type Getters<T> = {
    [K in keyof T as `get${Capitalize<string & K>}`]: () => T[K];
  };

  type Setters<T> = {
    [K in keyof T as `set${Capitalize<string & K>}`]: (val: T[K]) => void;
  };

  type User = { name: string; age: number; active: boolean };

  type UserGetters = Getters<User>;
  // { getName: () => string; getAge: () => number; getActive: () => boolean }

  type UserSetters = Setters<User>;
  // { setName: (val: string) => void; setAge: (val: number) => void; ... }

  // Combine both:
  type WithGettersSetters<T> = T & Getters<T> & Setters<T>;

5. EVENT SYSTEM TYPING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  type Events = {
    click: { x: number; y: number };
    focus: { target: HTMLElement };
    blur: {};
  };

  // Generate "on{Event}" handlers from event map:
  type EventHandlers = {
    [K in keyof Events as `on${Capitalize<string & K>}`]: (
      event: Events[K]
    ) => void;
  };
  // { onClick: (event: { x: number; y: number }) => void;
  //   onFocus: (event: { target: HTMLElement }) => void;
  //   onBlur: (event: {}) => void; }

  // Type-safe event emitter:
  type EventListener<T> = {
    on<K extends string & keyof T>(
      event: K,
      handler: (payload: T[K]) => void
    ): void;
    emit<K extends string & keyof T>(event: K, payload: T[K]): void;
  };

6. PARSING AND EXTRACTING PARTS FROM STRINGS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Combined with `infer` in conditional types, template literals can "parse"
  string patterns:

  // Extract part after a prefix:
  type RemovePrefix<Str extends string, Prefix extends string> =
    Str extends `${Prefix}${infer Rest}` ? Rest : Str;

  type R = RemovePrefix<"onFocus", "on">; // "Focus"

  // Extract part between separators:
  type ExtractMiddle<S extends string> =
    S extends `${infer _}:${infer Middle}:${infer _2}` ? Middle : never;
  type M = ExtractMiddle<"start:middle:end">; // "middle"

  // Split a string:
  type Split<S extends string, D extends string> =
    S extends `${infer L}${D}${infer R}` ? [L, ...Split<R, D>] : [S];
  type Parts = Split<"a,b,c", ",">; // ["a", "b", "c"]

  // Trim whitespace (left):
  type TrimLeft<S extends string> =
    S extends ` ${infer R}` ? TrimLeft<R> : S;

  // Trim whitespace (both):
  type Trim<S extends string> =
    S extends ` ${infer R}` | `${infer R} ` ? Trim<R> : S;
  type T1 = Trim<"  hello  ">; // "hello"

8. BUILDING TYPE-SAFE QUERY BUILDERS AND APIs
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // Type-safe route parameters:
  type ExtractRouteParams<T extends string> =
    string extends T
      ? Record<string, string>
      : T extends `${infer _Start}:${infer Param}/${infer Rest}`
        ? { [K in Param | keyof ExtractRouteParams<Rest>]: string }
        : T extends `${infer _}:${infer Param}`
          ? { [K in Param]: string }
          : {};

  type Params = ExtractRouteParams<"/users/:userId/posts/:postId">;
  // { userId: string; postId: string }

  // Type-safe i18n keys:
  type TranslationKeys = {
    "user.name": string;
    "user.email": string;
    "error.notFound": string;
  };
  type NSKey<NS extends string, K extends string> = `${NS}.${K}`;

9. LIMITATIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  1. Explosion risk: interpolating large unions can generate massive types
     and slow down the compiler.
     type A = "a" | "b" | "c"; // 3 members
     type B = "x" | "y" | "z"; // 3 members
     type C = `${A}_${B}`;      // 9 members — fine

     type Long = "a" | "b" | ... | "z"; // 26 members
     type Cross = `${Long}${Long}`;      // 676 members — can be slow

  2. Interpolated types must be string | number | bigint | boolean | null |
     undefined. Cannot interpolate objects directly.

  3. Pattern matching with `infer` is greedy — multiple infer sites can produce
     unexpected results with ambiguous patterns.

10. INTERVIEW QUESTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q1: What are template literal types in TypeScript?
    Template literal types (added in TS 4.1) allow you to build string types
    using backtick syntax at the type level, just like JavaScript template
    literals at runtime. They compose string literal types: `${"a"|"b"}_${"x"|"y"}`
    produces "a_x" | "a_y" | "b_x" | "b_y".

Q2: How do template literal types distribute over unions?
    TypeScript automatically creates a cross-product union when a union type
    is interpolated: `${A | B}_${C | D}` = `A_C` | `A_D` | `B_C` | `B_D`.

Q3: What are the 4 built-in string manipulation types?
    Uppercase<S>, Lowercase<S>, Capitalize<S>, Uncapitalize<S>.
    They operate on string literal types and are used to transform casing.

Q4: How do you use template literal types with mapped types?
    Use them in the `as` clause for key remapping:
    [K in keyof T as `get${Capitalize<string & K>}`]: () => T[K]
    This turns { name: string } into { getName: () => string }.

Q5: Can you extract information from a string type using template literals?
    Yes, by combining with `infer` in a conditional type:
    type RemoveOn<S> = S extends `on${infer Rest}` ? Rest : S;
    type R = RemoveOn<"onClick">; // "Click"

Q6: Why is `string & K` used when capitalizing keys?
    `keyof T` can yield string | number | symbol. Capitalize only accepts
    string, so `string & K` narrows K to the string subset.

Q7: What is a cross-product of unions in template literal types?
    When multiple positions in a template literal contain unions, TypeScript
    generates all possible combinations (Cartesian product).
    type T = `${"a" | "b"}_${"x" | "y"}` → "a_x" | "a_y" | "b_x" | "b_y"

Q8: Give a practical use case for template literal types.
    Typed event emitters: generate "on{Event}" handler names from event maps,
    CSS utility classes (margin-top, padding-left etc.),
    type-safe i18n key lookup, or parsing URL route parameters.

Q9: What are the performance concerns with template literal types?
    Interpolating large unions leads to an exponential explosion of combinations,
    causing type-checking to slow significantly. Use them with small, bounded
    unions for best performance.

Q10: What TypeScript version introduced template literal types?
     TypeScript 4.1 (November 2020) introduced template literal types.
     The built-in string manipulation types (Uppercase etc.) were also added
     in 4.1 as intrinsic types (implemented in the compiler, not user-land TS).
